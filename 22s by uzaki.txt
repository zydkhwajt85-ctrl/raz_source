local function stopIntroMusic()
    _introGen = (_introGen or 0) + 1
    if _introSoundEnded then
        pcall(function() _introSoundEnded:Disconnect() end)
        _introSoundEnded = nil
    end
    if _introSound then
        pcall(function()
            _introSound:Stop()
            _introSound:Destroy()
        end)
        _introSound = nil
    end
    pcall(function()
        for _, s in ipairs(SoundService:GetChildren()) do
            if s:IsA("Sound") and (s.Name == "22sIntroSong" or s.Name == "FiveHubIntroSong" or s.Name == "CleanIntroSong") then
                pcall(function() s:Stop(); s:Destroy() end)
            end
        end
    end)
end

local function customAssetFromFile(file)
    for _, fn in ipairs({getcustomasset, getsynasset, getasset}) do
        if typeof(fn) == "function" then
            local ok, asset = pcall(function() return fn(file) end)
            if ok and type(asset) == "string" and asset ~= "" then
                return asset
            end
        end
    end
    return nil
end

local function playIntroMusic(onEnded)
    if not IntroSongs or #IntroSongs == 0 then return end
    stopIntroMusic()
    local gen = _introGen
    IntroSongIndex = math.clamp(tonumber(IntroSongIndex) or 1, 1, #IntroSongs)
    local url = IntroSongs[IntroSongIndex]
    local rawId = tostring(url):match("catbox%.moe/([%w]+)") or ("idx" .. tostring(IntroSongIndex))
    local songFile = "22s_intro_" .. rawId .. ".mp3"

    task.spawn(function()
        local asset = nil
        if isfile and isfile(songFile) then
            asset = customAssetFromFile(songFile)
        end
        if not asset and writefile then
            local ok, data = pcall(function() return game:HttpGet(url) end)
            if ok and type(data) == "string" and #data > 100 then
                pcall(function() writefile(songFile, data) end)
                asset = customAssetFromFile(songFile)
            end
        end
        if not asset then
            asset = customAssetFromFile(songFile)
        end
        if gen ~= _introGen then return end
        if not asset then return end

        local snd = Instance.new("Sound")
        snd.Name = "22sIntroSong"
        snd.SoundId = asset
        snd.Volume = math.clamp(tonumber(introMusicVolume) or 0.7, 0, 1)
        snd.Looped = false
        snd.Parent = SoundService

        if gen ~= _introGen then
            pcall(function() snd:Destroy() end)
            return
        end

        _introSound = snd
        if onEnded then
            _introSoundEnded = snd.Ended:Connect(function()
                if gen == _introGen then
                    pcall(onEnded)
                end
            end)
        end
        pcall(function() snd:Play() end)
    end)
end

function previewIntroSong(index)
    if type(IntroSongs) ~= "table" or #IntroSongs == 0 then return end
    index = math.clamp(math.floor(tonumber(index) or IntroSongIndex or 1), 1, #IntroSongs)
    IntroSongIndex = index
    _introPreviewToken = _introPreviewToken + 1
    local myToken = _introPreviewToken
    playIntroMusic(nil)
    task.delay(10, function()
        if myToken == _introPreviewToken then
            stopIntroMusic()
        end
    end)
end

function playFullIntroSong(index)
    if type(IntroSongs) ~= "table" or #IntroSongs == 0 then return end
    index = math.clamp(math.floor(tonumber(index) or IntroSongIndex or 1), 1, #IntroSongs)
    IntroSongIndex = index
    _introPreviewToken = _introPreviewToken + 1
    playIntroMusic(nil)
end

local RagdollTracker = {
    active = false,
    startTime = 0,
}

local RagdollStealState = {
    suspended = false,
    wasOn = false,
}

local function isRagdolledState(hum)
    if not hum then return false end
    local st = hum:GetState()
    if hum.PlatformStand
        or st == Enum.HumanoidStateType.Physics
        or st == Enum.HumanoidStateType.Ragdoll
        or st == Enum.HumanoidStateType.FallingDown then
        return true
    end
    local endTime = Player:GetAttribute("RagdollEndTime")
    if endTime and (endTime - workspace:GetServerTimeNow()) > 0 then
        return true
    end
    return false
end

local function suspendAutoStealForRagdoll()
    if RagdollStealState.suspended then return end
    if not Enabled.AutoSteal then return end
    RagdollStealState.suspended = true
    RagdollStealState.wasOn = true
    Enabled.AutoSteal = false
    pcall(function()
        if stopNormalSteal then stopNormalSteal() end
        if stopSemiSteal then stopSemiSteal() end
    end)
    if VisualSetters and VisualSetters.AutoSteal then
        pcall(function() VisualSetters.AutoSteal(false, true) end)
    end
    isStealing = false
    if ResetProgressBar then pcall(ResetProgressBar) end
end

local function restoreAutoStealFromRagdoll()
    if not RagdollStealState.suspended then return end
    local shouldRestore = RagdollStealState.wasOn
    RagdollStealState.suspended = false
    RagdollStealState.wasOn = false
    if shouldRestore then
        Enabled.AutoSteal = true
        pcall(function()
            if syncAutoSteal then syncAutoSteal() end
        end)
        if VisualSetters and VisualSetters.AutoSteal then
            pcall(function() VisualSetters.AutoSteal(true, true) end)
        end
    end
end

_G._ragdollRestoreNow = restoreAutoStealFromRagdoll

RunService.Heartbeat:Connect(function()
    local char = Player.Character
    if not char then
        if RagdollTracker.active then
            RagdollTracker.active = false
        end
        if RagdollStealState.suspended then
            restoreAutoStealFromRagdoll()
        end
        if ragdollTimerLabel and ragdollTimerLabel.Text ~= "" then
            ragdollTimerLabel.Text = ""
        end
        return
    end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local nowRagdolled = isRagdolledState(hum)

    if nowRagdolled and not RagdollTracker.active then
        RagdollTracker.active = true
        RagdollTracker.startTime = tick()

        if Enabled.RagdollSteal then
            suspendAutoStealForRagdoll()
        end
    elseif not nowRagdolled and RagdollTracker.active then
        RagdollTracker.active = false
        if RagdollStealState.suspended then
            restoreAutoStealFromRagdoll()
        end
    end

    if ragdollTimerLabel then
        if RagdollTracker.active then
            local elapsed = tick() - RagdollTracker.startTime
            local remaining = math.max(0, RAGDOLL_DURATION - elapsed)
            ragdollTimerLabel.Text = string.format("RAGDOLL  %.1fs", remaining)
            ragdollTimerLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
        elseif ragdollTimerLabel.Text ~= "" then
            ragdollTimerLabel.Text = ""
        end
    end

    if RagdollTracker.active and RagdollStealState.suspended and RagdollStealState.wasOn then
        local elapsed = tick() - RagdollTracker.startTime
        local remaining = RAGDOLL_DURATION - elapsed
        local window = tonumber(Values.RagdollRestoreWindow) or 1.2
        if remaining <= window then
            restoreAutoStealFromRagdoll()
        end
    end
end)

Player.CharacterAdded:Connect(function()
    RagdollTracker.active = false
    if RagdollStealState.suspended then
        restoreAutoStealFromRagdoll()
    end
end)

local function assetId(v)
    if type(v) == "number" then return "rbxassetid://" .. tostring(v) end
    if type(v) == "string" and v:match("^%d+$") then return "rbxassetid://" .. v end
    return v
end

local function packFromPreset(preset)
    if not preset then return nil end
    local idle = preset.Idle or {}
    return {
        idle1 = assetId(preset.Animation1 or idle[1]),
        idle2 = assetId(preset.Animation2 or idle[2]),
        walk = assetId(preset.WalkAnim),
        run = assetId(preset.RunAnim),
        jump = assetId(preset.JumpAnim),
        fall = assetId(preset.FallAnim),
        climb = assetId(preset.ClimbAnim),
        swim = assetId(preset.Swim),
        swimidle = assetId(preset.SwimIdle),
    }
end

local animOriginals = setmetatable({}, { __mode = "k" })

local function captureAnimOriginals(character)
    local animate = character and character:FindFirstChild("Animate")
    if not animate then return end
    local targets = {
        idle1 = animate:FindFirstChild("idle") and animate.idle:FindFirstChild("Animation1"),
        idle2 = animate:FindFirstChild("idle") and animate.idle:FindFirstChild("Animation2"),
        walk = animate:FindFirstChild("walk") and animate.walk:FindFirstChild("WalkAnim"),
        run = animate:FindFirstChild("run") and animate.run:FindFirstChild("RunAnim"),
        jump = animate:FindFirstChild("jump") and animate.jump:FindFirstChild("JumpAnim"),
        fall = animate:FindFirstChild("fall") and animate.fall:FindFirstChild("FallAnim"),
        climb = animate:FindFirstChild("climb") and animate.climb:FindFirstChild("ClimbAnim"),
        swim = animate:FindFirstChild("swim") and animate.swim:FindFirstChild("Swim"),
        swimidle = animate:FindFirstChild("swimidle") and animate.swimidle:FindFirstChild("SwimIdle"),
    }
    local saved = animOriginals[character] or {}
    for k, anim in pairs(targets) do
        if anim and not saved[k] then saved[k] = anim.AnimationId end
    end
    animOriginals[character] = saved
    return targets, saved
end

function applySolarAnimPack(name)
    name = name or Values.AnimPackName or "Cartoon"
    Values.AnimPackName = name
    local preset = animationPresets[name]
    if not preset then return false end
    local pack = packFromPreset(preset)
    local char = Player.Character
    if not char then return false end
    local targets = select(1, captureAnimOriginals(char))
    if not targets then return false end
    for k, id in pairs(pack) do
        if id and targets[k] then
            pcall(function() targets[k].AnimationId = id end)
        end
    end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then
        for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
            pcall(function() track:Stop(0) end)
        end
    end
    return true
end

local function S2_setupVelChecked(char)
    S2_velChecked = {}
    if not char then return end
    local hrp = char:WaitForChild("HumanoidRootPart", 5)
    if hrp then S2_velChecked[hrp] = true end
    return hrp
end

local function S2_hookVelHRP(hrp)
    if not hrp or S2_hookedVelParts[hrp] then return end
    if S2_hookInstalled then return end
    S2_hookedVelParts[hrp] = true
    pcall(function()
        if type(getrawmetatable) ~= "function" then return end
        if type(setreadonly) ~= "function" then return end
        if type(newcclosure) ~= "function" then return end
        if type(checkcaller) ~= "function" then return end
        local mt = getrawmetatable(hrp)
        if not mt then return end
        setreadonly(mt, false)
        local originalVelIndex = rawget(mt, "__index")
        mt.__index = newcclosure(function(self, key)
            if not checkcaller() and S2_velChecked[self] and (key == "AssemblyLinearVelocity" or key == "Velocity") then
                local real
                if type(originalVelIndex) == "function" then
                    real = originalVelIndex(self, key)
                elseif type(originalVelIndex) == "table" then
                    real = originalVelIndex[key]
                end
                if typeof(real) == "Vector3" and real.Magnitude > 20 then
                    return real.Unit * 20
                end
                return real
            end
            if type(originalVelIndex) == "function" then
                return originalVelIndex(self, key)
            elseif type(originalVelIndex) == "table" then
                return originalVelIndex[key]
            end
        end)
        setreadonly(mt, true)
        S2_hookInstalled = true
    end)
end

function getActiveMoveSpeed()
    if S2_laggerCarryActive or (S2_laggerModeEnabled and S2_carrySpeedActive) then
        return tonumber(LAGGER_CARRY_SPEED) or 24.5
    elseif S2_laggerModeEnabled then
        return tonumber(LAGGER_SPEED) or 15
    elseif S2_carrySpeedActive then
        return tonumber(CS) or 30
    else
        return tonumber(NS) or 60
    end
end

local function S2_isRagdollState(hum)
    if not hum then return true end
    local st = hum:GetState()
    return hum.PlatformStand
        or st == Enum.HumanoidStateType.Physics
        or st == Enum.HumanoidStateType.Ragdoll
        or st == Enum.HumanoidStateType.FallingDown
end

function setSpeedMode(mode)
    if mode ~= "Normal" and mode ~= "Carry" and mode ~= "Lagger" and mode ~= "Lagger Carry" then
        mode = "Normal"
    end
    Values.SpeedMode = mode
    if mode == "Normal" then
        speedMode = false
        laggerToggled = false
        laggerPhase = 0
        _G.S2SpeedSetNormal()
    elseif mode == "Carry" then
        speedMode = true
        laggerToggled = false
        laggerPhase = 0
        _G.S2SpeedSetCarry()
    elseif mode == "Lagger" then
        speedMode = false
        laggerToggled = true
        laggerPhase = 1
        _G.S2SpeedSetLaggerNormal()
    elseif mode == "Lagger Carry" then
        speedMode = false
        laggerToggled = true
        laggerPhase = 2
        _G.S2SpeedSetLaggerCarry()
    end
    if refreshSpeedModeLabel then pcall(refreshSpeedModeLabel) end
end

function cycleNormalSpeed()
    -- ONE step only: Normal <-> Carry
    local mode = tostring(Values.SpeedMode or "Normal")
    if mode == "Carry" then
        setSpeedMode("Normal")
    else
        -- Normal, Lagger, Lagger Carry, anything else -> Carry
        setSpeedMode("Carry")
    end
    Enabled.CarryMode = (Values.SpeedMode == "Carry")
    Enabled.LaggerMode = false
    if VisualSetters then
        if VisualSetters.CarryMode then pcall(VisualSetters.CarryMode, Enabled.CarryMode, true) end
        if VisualSetters.LaggerMode then pcall(VisualSetters.LaggerMode, false, true) end
    end
    if refreshSpeedModeLabel then pcall(refreshSpeedModeLabel) end
    pcall(SaveConfig)
end

function cycleLaggerSpeed()
    -- ONE step only: Lagger <-> Lagger Carry (enter Lagger from other modes)
    local mode = tostring(Values.SpeedMode or "Normal")
    if mode == "Lagger" then
        setSpeedMode("Lagger Carry")
    elseif mode == "Lagger Carry" then
        setSpeedMode("Lagger")
    else
        setSpeedMode("Lagger")
    end
    Enabled.LaggerMode = true
    Enabled.CarryMode = false
    if VisualSetters then
        if VisualSetters.LaggerMode then pcall(VisualSetters.LaggerMode, true, true) end
        if VisualSetters.CarryMode then pcall(VisualSetters.CarryMode, false, true) end
    end
    if refreshSpeedModeLabel then pcall(refreshSpeedModeLabel) end
    pcall(SaveConfig)
end

function getNearestPlayer()
    local c = Player.Character
    if not c then return nil end
    local h = c:FindFirstChild("HumanoidRootPart")
    if not h then return nil end
    local pos = h.Position
    local nearest, dist = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= Player and p.Character then
            local oh = p.Character:FindFirstChild("HumanoidRootPart")
            if oh then
                local d = (pos - oh.Position).Magnitude
                if d < dist then dist = d nearest = p end
            end
        end
    end
    return nearest
end

do
local state={conn=nil,target=nil,lastAt=nil,lastPos=nil,stable=Vector3.zero,nextHit=0,hum=nil,rotate=true}
local function targetFor(root,range)
    local best,distance=nil,range or math.huge
    for _,player in ipairs(Players:GetPlayers()) do
        local char=player~=Player and player.Character
        local target=char and char:FindFirstChild("HumanoidRootPart")
        local hum=char and char:FindFirstChildOfClass("Humanoid")
        if target and hum and hum.Health>0 then
            local d=(target.Position-root.Position).Magnitude
            if d<distance then best,distance=target,d end
        end
    end
    return best,distance
end

    local bat=findAutoSwingBat()
    if bat and bat.Parent~=char then pcall(function() hum:EquipTool(bat) end) end
    return bat
end
local function swing(bat)
    if bat and os.clock()>=state.nextHit then state.nextHit=os.clock()+.08;pcall(function()bat:Activate()end) end
end
function stopBatAimbot()
    Enabled.BatAimbot=false
    if state.conn then state.conn:Disconnect();state.conn=nil end
    if state.hum and state.hum.Parent then state.hum.AutoRotate=state.rotate end
    state.target,state.lastAt,state.lastPos=nil,nil,nil
    state.stable=Vector3.zero
    local root=Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
    if root then root.AssemblyAngularVelocity=Vector3.zero;root.AssemblyLinearVelocity=Vector3.new(0,root.AssemblyLinearVelocity.Y,0) end
end
function startBatAimbot()
    stopBatAimbot()
    if Enabled.TPBat then Enabled.TPBat=false;stopTPBat() end
    if _G._22sAutoPlay then _G._22sAutoPlay.set(false) end
    Enabled.BatAimbot=true
    local signal=Values.AimbotMode=="Bypass" and RunService.Heartbeat or RunService.RenderStepped
    state.conn=signal:Connect(function()
        if not Enabled.BatAimbot then return end
        local char=Player.Character
        local root=char and char:FindFirstChild("HumanoidRootPart")
        local hum=char and char:FindFirstChildOfClass("Humanoid")
        if not root or not hum or hum.Health<=0 then return end
        if state.hum~=hum then state.hum=hum;state.rotate=hum.AutoRotate end
        hum.AutoRotate=false
        local bat=equip(char,hum)
        local target,distance=targetFor(root)
        if not target then root.AssemblyLinearVelocity=Vector3.new(0,root.AssemblyLinearVelocity.Y,0);return end
        if target~=state.target then state.target=target;state.lastAt=nil;state.lastPos=nil;state.stable=Vector3.zero end
        local tv=target.AssemblyLinearVelocity
        local pos,targetPos=root.Position,target.Position
        if Values.AimbotMode=="Bypass" then
            -- ZZZZ bounded prediction and teleport rejection.
            local now=os.clock()
            local dt=math.max(state.lastAt and now-state.lastAt or .016,.008)
            if state.lastPos and dt<.5 and (targetPos-state.lastPos).Magnitude>8+95*dt then
                state.lastPos,state.lastAt=targetPos,now;root.AssemblyLinearVelocity=Vector3.zero;return
            end
            state.lastPos,state.lastAt=targetPos,now
            if tv.Magnitude>95 then tv=tv.Unit*95 end
            local change=(tv-state.stable)/dt
            if change.Magnitude>140 then change=change.Unit*140 end
            state.stable+=change*dt;tv=state.stable
            local offset=tv*math.clamp(tv.Magnitude/150,.05,.2)
            if offset.Magnitude>8 then offset=offset.Unit*8 end
            local predicted=targetPos+offset
            local move=predicted+Vector3.new(0,5.75,0)-pos
            local look=Vector3.new(predicted.X,pos.Y,predicted.Z)
            if (look-pos).Magnitude>.1 then root.CFrame=CFrame.lookAt(pos,look) end
            hum:Move(Vector3.zero,false)
            root.AssemblyLinearVelocity=move.Magnitude>1 and move.Unit*55 or tv
            if distance<=9 then swing(bat) end
        else
            -- ZZZZ Normal V1 chase, vertical control and angular facing.
            local delta=targetPos+tv*.14+target.CFrame.LookVector*.3-pos
            local flat=Vector3.new(delta.X,0,delta.Z)
            local direction=flat.Magnitude>0 and flat.Unit or Vector3.zero
            local vertical=(targetPos.Y+3.7-pos.Y)*19.5+tv.Y*.8
            if hum.FloorMaterial~=Enum.Material.Air then vertical=math.max(vertical,13) end
            root.AssemblyLinearVelocity=root.AssemblyLinearVelocity:Lerp(Vector3.new(direction.X*Values.AUTO_BAT_SPEED,math.clamp(vertical,-70,110),direction.Z*Values.AUTO_BAT_SPEED),.8)
            local facing=targetPos+tv*math.clamp(tv.Magnitude/150,.05,.2)
            if (facing-pos).Magnitude>.1 then
                local x,y,z=(root.CFrame:Inverse()*CFrame.lookAt(pos,facing)):ToEulerAnglesXYZ()
                root.AssemblyAngularVelocity=root.CFrame:VectorToWorldSpace(Vector3.new(math.clamp(x,-2.5,2.5),math.clamp(y,-2.5,2.5),math.clamp(z,-2.5,2.5))*42)
            end
            if distance<=8 then swing(bat) end
        end
    end)
end

function setupGalaxyForce()
    pcall(function()
        local c = Player.Character
        if not c then return end
        local h = c:FindFirstChild("HumanoidRootPart")
        if not h then return end
        if galaxyVectorForce then galaxyVectorForce:Destroy() end
        if galaxyAttachment then galaxyAttachment:Destroy() end
        galaxyAttachment = Instance.new("Attachment")
        galaxyAttachment.Parent = h
        galaxyVectorForce = Instance.new("VectorForce")
        galaxyVectorForce.Attachment0 = galaxyAttachment
        galaxyVectorForce.ApplyAtCenterOfMass = true
        galaxyVectorForce.RelativeTo = Enum.ActuatorRelativeTo.World
        galaxyVectorForce.Force = Vector3.new(0, 0, 0)
        galaxyVectorForce.Parent = h
    end)
end

function updateGalaxyForce()
    if not galaxyEnabled or not galaxyVectorForce then return end
    local c = Player.Character
    if not c then return end
    local mass = 0
    for _, p in ipairs(c:GetDescendants()) do
        if p:IsA("BasePart") then mass = mass + p:GetMass() end
    end
    local tg = Values.DEFAULT_GRAVITY * (Values.GalaxyGravityPercent / 100)
    galaxyVectorForce.Force = Vector3.new(0, mass * (Values.DEFAULT_GRAVITY - tg) * 0.95, 0)
end

function adjustGalaxyJump()
    pcall(function()
        local c = Player.Character
        if not c then return end
        local hum = c:FindFirstChildOfClass("Humanoid")
        if not hum then return end
        if not galaxyEnabled then hum.JumpPower = originalJumpPower return end
        local ratio = math.sqrt((Values.DEFAULT_GRAVITY * (Values.GalaxyGravityPercent / 100)) / Values.DEFAULT_GRAVITY)
        hum.JumpPower = originalJumpPower * ratio
    end)
end

function doMiniHop()
    if not hopsEnabled then return end
    pcall(function()
        local c = Player.Character
        if not c then return end
        local h = c:FindFirstChild("HumanoidRootPart")
        local hum = c:FindFirstChildOfClass("Humanoid")
        if not h or not hum then return end
        if tick() - lastHopTime < Values.HOP_COOLDOWN then return end

        if hum.FloorMaterial == Enum.Material.Air then
            h.AssemblyLinearVelocity = Vector3.new(h.AssemblyLinearVelocity.X, Values.HOP_POWER, h.AssemblyLinearVelocity.Z)
        end
    end)
end

local jumpDropActive = false
local jumpDropConn = nil

function runJumpDrop()
    if jumpDropActive or VelocityBypass.jumpDrop then return end
    local char = Player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not char or not root then return end
    jumpDropActive = true
    VelocityBypass.jumpDrop = true
    local startTime = tick()
    local JUMP_DROP_ASCEND_DURATION = 0.2
    local JUMP_DROP_ASCEND_SPEED = 150
    if jumpDropConn then pcall(function() jumpDropConn:Disconnect() end) jumpDropConn = nil end
    jumpDropConn = RunService.Heartbeat:Connect(function()
        local currentChar = Player.Character
        local currentRoot = currentChar and currentChar:FindFirstChild("HumanoidRootPart")
        if not currentChar or not currentRoot then
            if jumpDropConn then pcall(function() jumpDropConn:Disconnect() end) jumpDropConn = nil end
            jumpDropActive = false
            VelocityBypass.jumpDrop = false
            return
        end
        if tick() - startTime >= JUMP_DROP_ASCEND_DURATION then
            if jumpDropConn then pcall(function() jumpDropConn:Disconnect() end) jumpDropConn = nil end
            local rayParams = RaycastParams.new()
            rayParams.FilterDescendantsInstances = {currentChar}
            pcall(function() rayParams.FilterType = Enum.RaycastFilterType.Exclude end)
            local rayResult = workspace:Raycast(currentRoot.Position, Vector3.new(0, -2000, 0), rayParams)
            if rayResult then
                local hum = currentChar:FindFirstChildOfClass("Humanoid")
                local offset = (hum and hum.HipHeight or 2) + (currentRoot.Size.Y / 2)
                currentRoot.CFrame = CFrame.new(currentRoot.Position.X, rayResult.Position.Y + offset, currentRoot.Position.Z)
                currentRoot.AssemblyLinearVelocity = Vector3.zero
                currentRoot.AssemblyAngularVelocity = Vector3.zero
            end
            jumpDropActive = false
            VelocityBypass.jumpDrop = false
            return
        end
        local cur = currentRoot.AssemblyLinearVelocity
        currentRoot.AssemblyLinearVelocity = Vector3.new(cur.X, JUMP_DROP_ASCEND_SPEED, cur.Z)
    end)
end

function waitBizarreBaseHold(stillActive)
    local hold = tonumber(Values.FreezeTime) or 1.8
    if hold < 1.2 then hold = 1.2 end
    if hold > 3.0 then hold = 3.0 end
    local t0 = tick()
    local minHold = 0.85
    while tick() - t0 < hold do
        if type(stillActive) == "function" and not stillActive() then return false end
        local carrying = false
        pcall(function()
            if Player:GetAttribute("Stealing") == true then carrying = true end
            local char = Player.Character
            if char then
                for _, t in ipairs(char:GetChildren()) do
                    if t:IsA("Tool") then
                        local n = string.lower(t.Name)
                        if string.find(n, "brain", 1, true) or string.find(n, "rot", 1, true) then
                            carrying = true
                            break
                        end
                    end
                end
            end
        end)
        if carrying and (tick() - t0) >= minHold then
            break
        end
        task.wait(0.05)
    end
    return true
end

function A.ownedPlot()
  if A.cachedPlot and A.cachedPlot.Parent then return A.cachedPlot end
  local plots = workspace:FindFirstChild("Plots")
  if not plots then return nil end
  local user = Player.Name:lower()
  local display = Player.DisplayName:lower()
  for _, plot in ipairs(plots:GetChildren()) do
    if plot:IsA("Model") then
      local sign = plot:FindFirstChild("PlotSign")
      if sign then
        local mine = false
        -- This is the same own-base indicator already used by Auto Steal.
        local yourBase = sign:FindFirstChild("YourBase")
        if yourBase and yourBase:IsA("BillboardGui") and yourBase.Enabled then
          mine = true
        end
        -- Fallback: the game's existing SurfaceGui.Frame.TextLabel owner.
        if not mine then
          local surface = sign:FindFirstChild("SurfaceGui")
          local frame = surface and surface:FindFirstChild("Frame")
          local label = frame and frame:FindFirstChild("TextLabel")
          if label and label:IsA("TextLabel") then
            local owner = label.Text:gsub("<.->", ""):gsub("'[sS]%s+[Bb]ase$", ""):gsub("^@", ""):gsub("^%s+", ""):gsub("%s+$", ""):lower()
            mine = owner == user or owner == display
          end
        end
        if mine then
          A.cachedPlot = plot
          A.plotUUID = plot.Name
          return plot
        end
      end
    end
  end
  return nil
end

function A.detectPlot()
  local plot = A.ownedPlot()
  if not plot then return nil end
  local sign = plot:FindFirstChild("PlotSign")
  local pos = sign and sign:IsA("BasePart") and sign.Position or plot:GetPivot().Position
  local dLeft = (Vector3.new(pos.X,0,pos.Z) - Vector3.new(-476.752,0,7.107)).Magnitude
  local dRight = (Vector3.new(pos.X,0,pos.Z) - Vector3.new(-476.752,0,114.107)).Magnitude
  A.side = (dLeft <= dRight) and "L" or "R"
  A.refresh()
  return A.side
end

function A.zoneDir()
  local c = Player.Character
  local hrp = c and c:FindFirstChild("HumanoidRootPart")
  if not hrp then return nil end
  local p = hrp.Position
  local function near(x,z)
    local dx, dz = p.X - x, p.Z - z
    return dx*dx + dz*dz <= A.cfg.RADIUS^2
  end
  if near(-466.48,6.65) then return "L" end
  if near(-466.90,113.56) then return "R" end
  return nil
end


function A.set(on)
  if not on then stopAutoWalk(); stopAutoRight(); return false end
  local side = A.detectPlot() or A.zoneDir()
  if side == "L" then startAutoWalk()
  elseif side == "R" then startAutoRight()
  else A.refresh(); return false end
  A.refresh()
  return true
end

-- ANTI-RAGDOLL
local antiRagdollMode = nil
local ragdollConnections = {}
local cachedCharData = {}
local isBoosting = false
local BOOST_SPEED = 400
local AR_DEFAULT_SPEED = 16

function arCacheCharacterData()
    local char = Player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return false end
    cachedCharData = { character = char, humanoid = hum, root = root }
    return true
end

function arDisconnectAll()
    for _, conn in ipairs(ragdollConnections) do pcall(function() conn:Disconnect() end) end
    ragdollConnections = {}
end

function arIsRagdolled()
    if not cachedCharData.humanoid then return false end
    local state = cachedCharData.humanoid:GetState()
    if state == Enum.HumanoidStateType.Physics
        or state == Enum.HumanoidStateType.Ragdoll
        or state == Enum.HumanoidStateType.FallingDown then return true end
    local endTime = Player:GetAttribute("RagdollEndTime")
    if endTime and (endTime - workspace:GetServerTimeNow()) > 0 then return true end
    return false
end

function arForceExitRagdoll()
    if not cachedCharData.humanoid or not cachedCharData.root then return end
    pcall(function() Player:SetAttribute("RagdollEndTime", workspace:GetServerTimeNow()) end)
    for _, descendant in ipairs(cachedCharData.character:GetDescendants()) do
        if descendant:IsA("BallSocketConstraint")
            or (descendant:IsA("Attachment") and descendant.Name:find("RagdollAttachment")) then
            descendant:Destroy()
        end
    end
    if not isBoosting then isBoosting = true cachedCharData.humanoid.WalkSpeed = BOOST_SPEED end
    if cachedCharData.humanoid.Health > 0 then cachedCharData.humanoid:ChangeState(Enum.HumanoidStateType.Running) end
    cachedCharData.root.Anchored = false
end

function arHeartbeatLoop()
    while antiRagdollMode == "v1" and Enabled.AntiRagdoll do
        task.wait()
        local currentlyRagdolled = arIsRagdolled()
        if currentlyRagdolled then arForceExitRagdoll()
        elseif isBoosting and not currentlyRagdolled then
            isBoosting = false
            if cachedCharData.humanoid then cachedCharData.humanoid.WalkSpeed = AR_DEFAULT_SPEED end
        end
    end
end

function startAntiRagdoll()
    if antiRagdollMode == "v1" then return end
    antiRagdollMode = nil
    if isBoosting and cachedCharData.humanoid then cachedCharData.humanoid.WalkSpeed = AR_DEFAULT_SPEED end
    isBoosting = false
    arDisconnectAll()
    cachedCharData = {}
    if not arCacheCharacterData() then return end
    antiRagdollMode = "v1"
    local camConn = RunService.RenderStepped:Connect(function()
        local cam = workspace.CurrentCamera
        if cam and cachedCharData.humanoid then cam.CameraSubject = cachedCharData.humanoid end
    end)
    table.insert(ragdollConnections, camConn)
    local respawnConn = Player.CharacterAdded:Connect(function()
        isBoosting = false
        task.wait(0.5)
        arCacheCharacterData()
    end)
    table.insert(ragdollConnections, respawnConn)
    task.spawn(arHeartbeatLoop)
end

function stopAntiRagdoll()
    antiRagdollMode = nil
    if isBoosting and cachedCharData.humanoid then cachedCharData.humanoid.WalkSpeed = AR_DEFAULT_SPEED end
    isBoosting = false
    arDisconnectAll()
    cachedCharData = {}
end

    if bat.Parent ~= char then
        if hum2 then pcall(function() hum2:EquipTool(bat) end) end
        task.wait(0.05)
    end
    local remote = bat:FindFirstChildOfClass("RemoteEvent") or bat:FindFirstChildOfClass("RemoteFunction")
    if remote and remote:IsA("RemoteEvent") then
        pcall(function() remote:FireServer() end)
        task.wait(0.15)
        pcall(function() remote:FireServer() end)
    else
        pcall(function() bat:Activate() end)
        task.wait(0.15)
        pcall(function() bat:Activate() end)
    end
end

function startBatCounter()
    if Connections.batCounter then return end
    Connections.batCounter = RunService.Heartbeat:Connect(function()
        if not Enabled.BatCounter then return end
        if batCounterDebounce then return end
        local char = Player.Character
        if not char then return end
        local hum2 = char:FindFirstChildOfClass("Humanoid")
        if not hum2 then return end
        local st = hum2:GetState()
        if st == Enum.HumanoidStateType.Physics
            or st == Enum.HumanoidStateType.Ragdoll
            or st == Enum.HumanoidStateType.FallingDown then
            batCounterDebounce = true
            task.spawn(function()
                local bat = findBatForCounter()
                if bat then swingBatForCounter(bat, char) end
                task.wait(0.5)
                batCounterDebounce = false
            end)
        end
    end)
end

function stopBatCounter()
    if Connections.batCounter then
        Connections.batCounter:Disconnect()
        Connections.batCounter = nil
    end
    batCounterDebounce = false
end

function findMedusa()
    local c = Player.Character
    if not c then return nil end
    for _, t in ipairs(c:GetChildren()) do
        if t:IsA("Tool") then
            local n = t.Name:lower()
            if n:find("medusa") or n:find("head") or n:find("stone") then return t end
        end
    end
    local bp = Player:FindFirstChild("Backpack")
    if bp then
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") then
                local n = t.Name:lower()
                if n:find("medusa") or n:find("head") or n:find("stone") then return t end
            end
        end
    end
    return nil
end

function useMedusaCounter()
    if medusaDebounce then return end
    if tick() - medusaLastUsed < MEDUSA_COOLDOWN then return end
    local c = Player.Character
    if not c then return end
    medusaDebounce = true
    local med = findMedusa()
    if not med then medusaDebounce = false return end
    if med.Parent ~= c then
        local hum2 = c:FindFirstChildOfClass("Humanoid")
        if hum2 then pcall(function() hum2:EquipTool(med) end) end
    end
    pcall(function() med:Activate() end)
    medusaLastUsed = tick()
    medusaDebounce = false
end

function onAnchorChanged(part)
    return part:GetPropertyChangedSignal("Anchored"):Connect(function()
        if part.Anchored and part.Transparency == 1 then
            if Enabled.MedusaCounter then useMedusaCounter() end
        end
    end)
end

function setupMedusa(char)
    for _, c in pairs(medusaAnchorConns) do pcall(function() c:Disconnect() end) end
    medusaAnchorConns = {}
    if not char then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            table.insert(medusaAnchorConns, onAnchorChanged(part))
        end
    end
    table.insert(medusaAnchorConns, char.DescendantAdded:Connect(function(part)
        if part:IsA("BasePart") then
            table.insert(medusaAnchorConns, onAnchorChanged(part))
        end
    end))
end

function stopMedusaCounter()
    for _, c in pairs(medusaAnchorConns) do pcall(function() c:Disconnect() end) end
    medusaAnchorConns = {}
end

local ProgressBarFill, ProgressLabel, ProgressPercentLabel
local StealData = {}
local DISCORD_TEXT = "discord.gg/22shub"

function getDiscordProgress(percent)
    local totalChars = #DISCORD_TEXT
    local adjustedPercent = math.min(percent * 1.5, 100)
    local charsToShow = math.floor((adjustedPercent / 100) * totalChars)
    return string.sub(DISCORD_TEXT, 1, charsToShow)
end

_stealProgress = 0
function setStealProgress(p, label)
    p = math.clamp(p or 0, 0, 1)
    _stealProgress = p
    if ProgressBarFill then ProgressBarFill.Size = UDim2.new(p, 0, 1, 0) end
    if ProgressLabel then ProgressLabel.Text = label or "STEALING..." end
    if ProgressPercentLabel then ProgressPercentLabel.Text = getDiscordProgress(math.floor(p * 100)) end
end

function ResetProgressBar()
    if ProgressLabel then ProgressLabel.Text = "READY" end
    if ProgressPercentLabel then ProgressPercentLabel.Text = "" end
    if ProgressBarFill then ProgressBarFill.Size = UDim2.new(0, 0, 1, 0) end
end

-- STEAL V1
local NormalSteal = { enabled = false, animals = {}, promptCache = {}, internalCache = {}, scannerStarted = false, stealConn = nil, lastSteal = 0, cooldown = 0.06 }

function normalGetRoot()
    local c = Player.Character
    if not c then return nil end
    return c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("UpperTorso")
end

function normalIsMyBase(plotName)
    local plots = workspace:FindFirstChild("Plots")
    local plot = plots and plots:FindFirstChild(plotName)
    if not plot then return false end
    local sign = plot:FindFirstChild("PlotSign")
    local yourBase = sign and sign:FindFirstChild("YourBase")
    return yourBase and yourBase:IsA("BillboardGui") and yourBase.Enabled == true
end

function normalScanPlots()
    NormalSteal.animals = {}
    local plots = workspace:FindFirstChild("Plots")
    if not plots then return end
    for _, plot in ipairs(plots:GetChildren()) do
        if plot:IsA("Model") and not normalIsMyBase(plot.Name) then
            local podiums = plot:FindFirstChild("AnimalPodiums")
            if podiums then
                for _, podium in ipairs(podiums:GetChildren()) do
                    if podium:IsA("Model") then
                        local base = podium:FindFirstChild("Base")
                        local spawn = base and base:FindFirstChild("Spawn")
                        if spawn then
                            table.insert(NormalSteal.animals, {
                                plot = plot.Name, slot = podium.Name,
                                worldPosition = spawn.Position,
                                uid = plot.Name .. "_" .. podium.Name,
                            })
                        end
                    end
                end
            end
        end
    end
end

function normalEnsureScanner()
    if NormalSteal.scannerStarted then return end
    NormalSteal.scannerStarted = true
    task.spawn(function()
        while true do
            if NormalSteal.enabled then pcall(normalScanPlots) end
            task.wait(3)
        end
    end)
end

function normalFindPrompt(data)
    if not data then return nil end
    local cached = NormalSteal.promptCache[data.uid]
    if cached and cached.Parent then return cached end
    local plots = workspace:FindFirstChild("Plots")
    local plot = plots and plots:FindFirstChild(data.plot)
    local pds = plot and plot:FindFirstChild("AnimalPodiums")
    local pod = pds and pds:FindFirstChild(data.slot)
    local base = pod and pod:FindFirstChild("Base")
    local spawn = base and base:FindFirstChild("Spawn")
    local attach = spawn and spawn:FindFirstChild("PromptAttachment")
    if not attach then return nil end
    for _, p in ipairs(attach:GetChildren()) do
        if p:IsA("ProximityPrompt") then
            NormalSteal.promptCache[data.uid] = p
            return p
        end
    end
    return nil
end

function normalCacheCallbacks(prompt)
    if NormalSteal.internalCache[prompt] then return end
    local data = { hold = {}, trigger = {}, ready = true }
    pcall(function()
        if getconnections then
            for _, c in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do
                if type(c.Function) == "function" then table.insert(data.hold, c.Function) end
            end
            for _, c in ipairs(getconnections(prompt.Triggered)) do
                if type(c.Function) == "function" then table.insert(data.trigger, c.Function) end
            end
        end
    end)
    if #data.hold > 0 or #data.trigger > 0 then NormalSteal.internalCache[prompt] = data end
end

function normalNearestAnimal()
    local root = normalGetRoot()
    if not root then return nil end
    local best, bestDist = nil, math.huge
    local radius = Values.NormalStealRadius or 63
    for _, data in ipairs(NormalSteal.animals) do
        if data.worldPosition and not normalIsMyBase(data.plot) then
            local dist = (root.Position - data.worldPosition).Magnitude
            if dist < bestDist then best = data bestDist = dist end
        end
    end
    if best and bestDist <= radius then return best end
    return nil
end

function normalDoSteal(prompt, animalData)
    if not prompt or not prompt.Parent or isStealing then return end
    if tick() - (NormalSteal.lastSteal or 0) < NormalSteal.cooldown then return end
    normalCacheCallbacks(prompt)
    local data = NormalSteal.internalCache[prompt]
    if not data or not data.ready then return end
    data.ready = false
    isStealing = true
    NormalSteal.lastSteal = tick()
    setStealProgress(0, "STEALING...")
    task.spawn(function()
        for _, fn in ipairs(data.hold) do task.spawn(function() pcall(fn) end) end
        local FILL_CAP = 0.8
        local HOLD_TIME = 2
        local COMPLETE_RADIUS = 9
        local dur = Values.STEAL_DURATION or 1.3
        local function inCompleteRange()
            local root = normalGetRoot()
            if not root or not animalData or not animalData.worldPosition then return false end
            return (root.Position - animalData.worldPosition).Magnitude <= COMPLETE_RADIUS
        end
        local t0 = tick()
        while Enabled.AutoSteal and selectedStealMode == "Normal" and tick() - t0 < dur do
            local p = math.min((tick() - t0) / dur, FILL_CAP)
            setStealProgress(p, "STEALING...")
            if p >= FILL_CAP then break end
            task.wait(0.02)
        end
        if not Enabled.AutoSteal or selectedStealMode ~= "Normal" then
            data.ready = true isStealing = false ResetProgressBar() return
        end
        local holdT0 = tick()
        local completed = false
        while Enabled.AutoSteal and selectedStealMode == "Normal" and tick() - holdT0 < HOLD_TIME do
            setStealProgress(FILL_CAP, "STEALING...")
            if inCompleteRange() then completed = true break end
            task.wait(0.02)
        end
        if not Enabled.AutoSteal or selectedStealMode ~= "Normal" or not completed then
            data.ready = true isStealing = false ResetProgressBar() return
        end
        local t1 = tick()
        local fillDur = dur * (1 - FILL_CAP)
        while Enabled.AutoSteal and selectedStealMode == "Normal" and tick() - t1 < fillDur do
            local p = FILL_CAP + math.min((tick() - t1) / fillDur, 1) * (1 - FILL_CAP)
            setStealProgress(p, "STEALING...")
            task.wait(0.02)
        end
        if not Enabled.AutoSteal or selectedStealMode ~= "Normal" then
            data.ready = true isStealing = false ResetProgressBar() return
        end
        setStealProgress(1, "STEALING...")
        for _, fn in ipairs(data.trigger) do task.spawn(function() pcall(fn) end) end
        task.wait(0.12)
        data.ready = true
        isStealing = false
        ResetProgressBar()
    end)
end

function startNormalSteal()
    NormalSteal.enabled = true
    normalEnsureScanner()
    pcall(normalScanPlots)
    if NormalSteal.stealConn then NormalSteal.stealConn:Disconnect() NormalSteal.stealConn = nil end
    NormalSteal.stealConn = RunService.Heartbeat:Connect(function()
        if not NormalSteal.enabled or selectedStealMode ~= "Normal" then return end
        if isStealing then return end
        local target = normalNearestAnimal()
        if not target then return end
        local prompt = normalFindPrompt(target)
        if prompt then normalDoSteal(prompt, target) end
    end)
end

function stopNormalSteal()
    NormalSteal.enabled = false
    isStealing = false
    if NormalSteal.stealConn then NormalSteal.stealConn:Disconnect() NormalSteal.stealConn = nil end
    ResetProgressBar()
end

-- SEMI STEAL
local SemiSteal = { enabled = false, conn = nil, lastSteal = 0, cooldown = 0.08, holdMin = 1.3, holdMax = 2.4 }

function semiGetRoot()
    local c = Player.Character
    return c and (c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("UpperTorso"))
end

function semiFindNearestPrompt()
    local root = semiGetRoot()
    if not root then return nil, nil end
    local plots = workspace:FindFirstChild("Plots")
    if not plots then return nil, nil end
    local bestPrompt, bestDist, bestName = nil, math.huge, nil
    local radius = Values.SemiStealRadius or 10
    for _, plot in ipairs(plots:GetChildren()) do
        if not normalIsMyBase(plot.Name) then
            local podiums = plot:FindFirstChild("AnimalPodiums")
            if podiums then
                for _, pod in ipairs(podiums:GetChildren()) do
                    local base = pod:FindFirstChild("Base")
                    local spawn = base and base:FindFirstChild("Spawn")
                    if spawn then
                        local dist = (spawn.Position - root.Position).Magnitude
                        if dist < bestDist and dist <= radius then
                            local att = spawn:FindFirstChild("PromptAttachment")
                            if att then
                                for _, ch in ipairs(att:GetChildren()) do
                                    if ch:IsA("ProximityPrompt") then
                                        bestPrompt = ch bestDist = dist bestName = pod.Name break
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return bestPrompt, bestName
end

function semiDoSteal(prompt, name)
    if not prompt or not prompt.Parent or isStealing then return end
    if tick() - (SemiSteal.lastSteal or 0) < SemiSteal.cooldown then return end
    if not StealData[prompt] then
        StealData[prompt] = { hold = {}, trigger = {}, ready = true }
        pcall(function()
            if getconnections then
                for _, c in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do
                    if c.Function then table.insert(StealData[prompt].hold, c.Function) end
                end
                for _, c in ipairs(getconnections(prompt.Triggered)) do
                    if c.Function then table.insert(StealData[prompt].trigger, c.Function) end
                end
            end
        end)
    end
    local data = StealData[prompt]
    if not data.ready then return end
    data.ready = false
    isStealing = true
    SemiSteal.lastSteal = tick()
    setStealProgress(0, name or "SEMI STEAL")
    task.spawn(function()
        for _, f in ipairs(data.hold) do task.spawn(f) end
        local t0 = tick()
        local maxHold = SemiSteal.holdMax
        while Enabled.AutoSteal and selectedStealMode == "Semi" and tick() - t0 < maxHold do
            local p = math.clamp((tick() - t0) / maxHold, 0, 1)
            setStealProgress(p, name or "SEMI STEAL")
            if tick() - t0 >= SemiSteal.holdMin then break end
            task.wait(0.03)
        end
        if not Enabled.AutoSteal or selectedStealMode ~= "Semi" then
            data.ready = true isStealing = false ResetProgressBar() return
        end
        setStealProgress(1, name or "SEMI STEAL")
        for _, f in ipairs(data.trigger) do task.spawn(f) end
        task.wait(0.15)
        data.ready = true
        isStealing = false
        ResetProgressBar()
    end)
end

function startSemiSteal()
    SemiSteal.enabled = true
    if SemiSteal.conn then SemiSteal.conn:Disconnect() SemiSteal.conn = nil end
    SemiSteal.conn = RunService.Heartbeat:Connect(function()
        if not SemiSteal.enabled or selectedStealMode ~= "Semi" then return end
        if isStealing then return end
        local prompt, name = semiFindNearestPrompt()
        if prompt then semiDoSteal(prompt, name) end
    end)
end

function stopSemiSteal()
    SemiSteal.enabled = false
    isStealing = false
    if SemiSteal.conn then SemiSteal.conn:Disconnect() SemiSteal.conn = nil end
    ResetProgressBar()
end

syncAutoSteal = function()
    if not Enabled.AutoSteal then
        stopNormalSteal()
        stopSemiSteal()
        return
    end
    if selectedStealMode == "V1" or selectedStealMode == "V2" then
        selectedStealMode = "Normal"
        Values.StealMode = "Normal"
    end
    stopNormalSteal()
    stopSemiSteal()
    if selectedStealMode == "Normal" then
        startNormalSteal()
    elseif selectedStealMode == "Semi" then
        startSemiSteal()
    else
        selectedStealMode = "Normal"
        Values.StealMode = "Normal"
        startNormalSteal()
    end
end

startAutoSteal = function()
    Enabled.AutoSteal = true
    syncAutoSteal()
end

stopAutoSteal = function()
    Enabled.AutoSteal = false
    stopNormalSteal()
    stopSemiSteal()
    isStealing = false
    ResetProgressBar()
end

    if TPBatV2.hittingCooldown then return end
    TPBatV2.hittingCooldown = true
    pcall(function()
        local bat = tpBatV2GetBat()
        if not bat then return end
        bat:Activate()
        local ev = bat:FindFirstChildWhichIsA("RemoteEvent")
        if ev then ev:FireServer() end
    end)
    task.delay(0.08, function() TPBatV2.hittingCooldown = false end)

function tpBatV2GetClosest()
    local hrp = TPBatV2.hrp
    if not hrp then return nil end
    local closest, bestDist = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= Player and p.Character then
            local root = p.Character:FindFirstChild("HumanoidRootPart")
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            if root and hum and hum.Health > 0 then
                local dist = (hrp.Position - root.Position).Magnitude
                if dist < bestDist then bestDist = dist closest = p end
            end
        end
    end
    return closest, bestDist
end

        local hum=char and char:FindFirstChildOfClass("Humanoid")
        if not root or not hum or hum.Health<=0 then return end

        hum.AutoRotate=false
        local target,distance=nil,mode==2 and 100 or math.huge
        for _,player in ipairs(Players:GetPlayers()) do
            local c=player~=Player and player.Character
            local r=c and c:FindFirstChild("HumanoidRootPart")
            local h=c and c:FindFirstChildOfClass("Humanoid")

        end

            local anchor=target.Position+Vector3.new(0,.9,0)+flat*.05

        end
        local cam=workspace.CurrentCamera
        if cam and (cam.CFrame.Position-target.Position).Magnitude>.01 then cam.CFrame=CFrame.lookAt(cam.CFrame.Position,target.Position) end
    end)

                        line.Rotation = math.deg(math.atan2(delta.Y, delta.X))
                        line.BackgroundColor3 = ESP_BLUE
                        line.Visible = true
                    end
                end
            end
        end
        for plr in pairs(TracerESP.drawings) do
            if not seen[plr] then cleanupTracer(plr) end
        end
    end)
end

stopTracerESP = function()
    if TracerESP.conn then TracerESP.conn:Disconnect() TracerESP.conn = nil end
    for plr in pairs(TracerESP.drawings) do cleanupTracer(plr) end
    if TracerESP.gui then TracerESP.gui:Destroy() TracerESP.gui = nil end
end

local AntiDie = { enabled = false, healthConn = nil, diedConn = nil, charConn = nil, humanoid = nil }

local function antiDieDisconnectHumanoid()
    if AntiDie.healthConn then AntiDie.healthConn:Disconnect() AntiDie.healthConn = nil end
    if AntiDie.diedConn then AntiDie.diedConn:Disconnect() AntiDie.diedConn = nil end
    AntiDie.humanoid = nil
end

local function activateAntiDie(char)
    if not AntiDie.enabled then return end
    char = char or Player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid") or char:WaitForChild("Humanoid", 3)
    if not hum or not AntiDie.enabled then return end
    antiDieDisconnectHumanoid()
    AntiDie.humanoid = hum
    hum.BreakJointsOnDeath = false
    hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
    AntiDie.healthConn = hum:GetPropertyChangedSignal("Health"):Connect(function()
        if not AntiDie.enabled or not hum.Parent then return end
        if hum.Health <= 0 then hum.Health = hum.MaxHealth end
    end)
    AntiDie.diedConn = hum.Died:Connect(function()
        if not AntiDie.enabled or not char.Parent then return end
        task.wait()
        if not AntiDie.enabled or not char.Parent then return end
        local newHum = Instance.new("Humanoid")
        newHum.Name = "ReplacedHumanoid"
        newHum.BreakJointsOnDeath = false
        newHum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
        newHum.Parent = char
        if workspace.CurrentCamera then workspace.CurrentCamera.CameraSubject = newHum end
        if hum.Parent then hum:Destroy() end
        if AntiDie.enabled then activateAntiDie(char) end
    end)
end

startAntiDie = function()
    AntiDie.enabled = true
    if AntiDie.charConn then AntiDie.charConn:Disconnect() end
    AntiDie.charConn = Player.CharacterAdded:Connect(function(char)
        if AntiDie.enabled then task.spawn(function() activateAntiDie(char) end) end
    end)
    task.spawn(function() activateAntiDie(Player.Character) end)
end

stopAntiDie = function()
    AntiDie.enabled = false
    if AntiDie.charConn then AntiDie.charConn:Disconnect() AntiDie.charConn = nil end
    local hum = AntiDie.humanoid
    antiDieDisconnectHumanoid()
    if hum and hum.Parent then
        pcall(function()
            hum.BreakJointsOnDeath = true
            hum:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
        end)
    end
end

local AntiFling = { enabled = false, loop = nil, velocityThreshold = 80 }

local function stabilizeRoot(root)
    if not root or not root.Parent then return end
    local velocity
    local ok = pcall(function() velocity = root.AssemblyLinearVelocity end)
    if not ok or typeof(velocity) ~= "Vector3" then
        local legacyOk
        legacyOk, velocity = pcall(function() return root.Velocity end)
        if not legacyOk or typeof(velocity) ~= "Vector3" then return end
    end
    if velocity.Magnitude <= AntiFling.velocityThreshold then return end
    local stabilized = Vector3.new(0, velocity.Y, 0)
    pcall(function() root.AssemblyLinearVelocity = stabilized end)
    pcall(function() root.AssemblyAngularVelocity = Vector3.zero end)
    pcall(function() root.Velocity = stabilized end)
    pcall(function() root.RotVelocity = Vector3.zero end)
end

startAntiFling = function()
    AntiFling.enabled = true
    if AntiFling.loop then AntiFling.loop:Disconnect() end
    AntiFling.loop = RunService.Heartbeat:Connect(function()
        if not AntiFling.enabled then return end
        local char = Player.Character
        if char then stabilizeRoot(char:FindFirstChild("HumanoidRootPart")) end
    end)
end

stopAntiFling = function()
    AntiFling.enabled = false
    if AntiFling.loop then AntiFling.loop:Disconnect() AntiFling.loop = nil end
end

local waitingForKeybind = nil
local function finishKeybind(key)
    if not waitingForKeybind then return end
    local name = waitingForKeybind
    if key and key ~= Enum.KeyCode.Unknown and key ~= Enum.KeyCode.Escape then
        KEYBINDS[name] = key
    end
    if KeyButtons[name] then
        KeyButtons[name].Text = (KEYBINDS[name] and KEYBINDS[name].Name) or "None"
        KeyButtons[name].BackgroundColor3 = Color3.fromRGB(40, 100, 220)
    end
    waitingForKeybind = nil
    SaveConfig()
end
local function beginKeybind(name)
    if waitingForKeybind == name then
        finishKeybind(nil)
        return
    end
    if waitingForKeybind then finishKeybind(nil) end
    waitingForKeybind = name
    if KeyButtons[name] then
        KeyButtons[name].Text = "..."
        KeyButtons[name].BackgroundColor3 = Color3.fromRGB(255, 160, 40)
    end
    -- auto-cancel so keybinds cannot get stuck forever
    local token = name
    task.delay(6, function()
        if waitingForKeybind == token then
            finishKeybind(nil)
        end
    end)
end

end

local backgroundCache={}
local backgroundBindings=setmetatable({}, {__mode="k"})
local backgroundWaiters={}
local function bind22sBackground(image,asset)
    backgroundBindings[image]=asset
    image.Image=backgroundCache[asset] or asset
    image.ScaleType=Enum.ScaleType.Crop
    image.ResampleMode=Enum.ResamplerMode.Default
    if backgroundCache[asset] then return end
    local function apply(uri) if image.Parent and backgroundBindings[image]==asset then image.Image=uri end end
    if backgroundWaiters[asset] then table.insert(backgroundWaiters[asset],apply);return end
    backgroundWaiters[asset]={apply}
    task.spawn(function()
        local uri=asset
        local ok,objects=pcall(function()return game:GetObjects(asset)end)
        if ok and type(objects)=="table" then
            for _,object in ipairs(objects) do
                local texture=object:IsA("Decal") and object or object:FindFirstChildWhichIsA("Decal",true)
                if texture and texture.Texture~="" then uri=texture.Texture end
                object:Destroy()
            end
        end
        backgroundCache[asset]=uri
        for _,fn in ipairs(backgroundWaiters[asset]) do fn(uri) end
        backgroundWaiters[asset]=nil
    end)
end

                elseif text:find("speed") or text:find("carry") or text:find("lagger") or text:find("auto play") or text:find("autoplay") or text:find("destination") or text:find("jump") or text:find("tp down") or text:find("tp height") or text:find("unwalk") then category="Movement"
                elseif text:find("galaxy") or text:find("fov") or text:find("esp") or text:find("tracer") or text:find("spin") then category="Visuals" end
                table.insert(groups[category],{row=row,y=row.Position.Y.Offset})
            end
        end
    end
    local columnHeights={[leftSide]=8,[rightSide]=8}
    for _,name in ipairs({"Movement","Combat","Steal","Visuals","Settings"}) do
        local column=(name=="Combat" or name=="Steal") and rightSide or leftSide
        local y=columnHeights[column]
        local heading=Instance.new("TextLabel",column)
        heading.Position,heading.Size=UDim2.fromOffset(10,y),UDim2.new(1,-20,0,28)
        heading.Text=name:upper();heading.TextSize=13;heading.Font=Enum.Font.GothamBold
        heading.TextColor3=C.accentLight;heading.TextXAlignment=Enum.TextXAlignment.Left
        heading.BackgroundTransparency=1;heading.ZIndex=12;y+=34
        table.sort(groups[name],function(a,b)return a.y<b.y end)
        for _,entry in ipairs(groups[name]) do
            local row=entry.row;row.Parent=column
            row.Position=UDim2.fromOffset(6,y)
            local minH = isMobile and 34 or 44
            local gap = isMobile and 5 or 8
            row.Size=UDim2.new(1,-12,0,math.max(minH,row.Size.Y.Offset))
            y+=row.Size.Y.Offset+gap
        end
        columnHeights[column]=y+12
    end
    for _,obj in ipairs(main:GetDescendants()) do
        if obj:IsA("TextLabel") or obj:IsA("TextButton") or obj:IsA("TextBox") then
            local minTs = isMobile and 9 or 12
            if isMobile and obj.TextSize > 14 then
                obj.TextSize = math.min(obj.TextSize, 13)
            end
            obj.TextSize = math.max(minTs, obj.TextSize)
            obj.TextStrokeTransparency = 1
        end
    end
    local fit=Instance.new("UIScale",main)
    local function fitLayout()
        local vp = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(800, 600)
        local narrow = vp.X < 650
        local width, scale, height
        if narrow then
            -- Compact phone panel: ~72% width, ~52% height of screen
            width = math.floor(math.clamp(vp.X * 0.78, 250, 310))
            height = math.floor(math.clamp(vp.Y * 0.52, 340, 480))
            scale = 1
        else
            width = 560
            scale = math.min(1, (vp.X - 20) / width)
            height = math.min(680, (vp.Y - 32) / math.max(scale, 0.01))
        end
        fit.Scale = scale
        main.Size = UDim2.fromOffset(width, height)
        main:SetAttribute("FullSize", main.Size)
        main.AnchorPoint = Vector2.new(0.5, 0.5)
        main.Position = UDim2.fromScale(0.5, 0.5)
        leftSide.Size = UDim2.new(narrow and 1 or .5, 0, 0, columnHeights[leftSide])
        rightSide.Size = UDim2.new(narrow and 1 or .5, 0, 0, columnHeights[rightSide])
        leftSide.Position = UDim2.new()
        rightSide.Position = narrow and UDim2.fromOffset(0, columnHeights[leftSide] + 8) or UDim2.fromScale(.5, 0)
        contentFrame.CanvasSize = UDim2.fromOffset(0, narrow and (columnHeights[leftSide] + columnHeights[rightSide] + 16) or math.max(columnHeights[leftSide], columnHeights[rightSide]))
    end
    fitLayout()
    main.Position=UDim2.fromScale(.5,.5)
    main.AnchorPoint=Vector2.new(.5,.5)
    local cameraConn
    local function watchCamera()
        if cameraConn then cameraConn:Disconnect() end
        if workspace.CurrentCamera then cameraConn=workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(fitLayout) end
        fitLayout()
    end
    workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(watchCamera)
    watchCamera()
end

local function tryRestore(name, fn)
    local ok, err = pcall(fn)

end

UserInputService.InputBegan:Connect(function(input, gpe)
    if waitingForKeybind then
        if input.UserInputType == Enum.UserInputType.Keyboard or input.UserInputType == Enum.UserInputType.Gamepad1 then
            if input.KeyCode ~= Enum.KeyCode.Unknown then
                if input.KeyCode == Enum.KeyCode.Escape then
                    finishKeybind(nil)
                else
                    finishKeybind(input.KeyCode)
                end
            end
        end
        return
    end
    if UserInputService:GetFocusedTextBox() then return end
    -- Allow keybinds even if gameProcessed (GUI open); still skip chat/textbox above
    if KEYBINDS.JUMPDROP and input.KeyCode == KEYBINDS.JUMPDROP then pcall(runJumpDrop) return end
    if KEYBINDS.INSTANTRESET and input.KeyCode == KEYBINDS.INSTANTRESET then pcall(runInstantReset) return end
    if input.KeyCode == Enum.KeyCode.Space then
        spaceHeld = true
        if Enabled.InfJump then infJumpHeld = true end
        return
    end
    local function keyIs(bind)
        if bind == nil or input.KeyCode == Enum.KeyCode.Unknown then return false end
        if input.KeyCode == bind then return true end
        local pressed = input.KeyCode.Name
        if type(bind) == "EnumItem" and bind.Name == pressed then return true end
        if type(bind) == "string" and bind == pressed then return true end
        if type(bind) == "string" then
            local ok, kc = pcall(function() return Enum.KeyCode[bind] end)
            if ok and kc and input.KeyCode == kc then return true end
        end
        return false
    end
    if keyIs(KEYBINDS.CARRY) then
        cycleNormalSpeed()
        pcall(playSound, "rbxassetid://6895079813", 0.3, 1.2)
        return
    elseif keyIs(KEYBINDS.LAGGER) then
        cycleLaggerSpeed()
        pcall(playSound, "rbxassetid://6895079813", 0.3, 1.2)
        return
    elseif keyIs(KEYBINDS.SPIN) then
        Enabled.SpinBot = not Enabled.SpinBot
        if VisualSetters.SpinBot then VisualSetters.SpinBot(Enabled.SpinBot) end
        if Enabled.SpinBot then startSpinBot() else stopSpinBot() end
        pcall(SaveConfig)
    elseif keyIs(KEYBINDS.GALAXY) then
        Enabled.Galaxy = not Enabled.Galaxy
        if VisualSetters.Galaxy then VisualSetters.Galaxy(Enabled.Galaxy) end
        if Enabled.Galaxy then startGalaxy() else stopGalaxy() end
        pcall(SaveConfig)
    elseif keyIs(KEYBINDS.BATAIMBOT) then
        Enabled.BatAimbot = not Enabled.BatAimbot
        if VisualSetters.BatAimbot then VisualSetters.BatAimbot(Enabled.BatAimbot) end
        if Enabled.BatAimbot then startBatAimbot() else stopBatAimbot() end
        pcall(SaveConfig)
    elseif keyIs(KEYBINDS.NUKE) then
        local n = getNearestPlayer()
        if n then INSTANT_NUKE(n) end
    elseif keyIs(KEYBINDS.AUTOPLAY) then
        toggleAutoPlay()
        pcall(SaveConfig)
    elseif keyIs(KEYBINDS.TPDOWN) then
        runTPDown()
    elseif keyIs(KEYBINDS.TPBAT) then
        Enabled.TPBat = not Enabled.TPBat
        if VisualSetters.TPBat then VisualSetters.TPBat(Enabled.TPBat) end
        if Enabled.TPBat then startTPBat() else stopTPBat() end
        pcall(SaveConfig)
    end
end)

function buildMobileButtons()
    if not isMobileDevice then return end
    local UIS = UserInputService
    local HS = HttpService
    local BTN_SIZE = 58
    local BTN_GAP = 12
    local COLS = 2
    local POS_FILE = "22s_MobileBtnPos.json"

    local OFF = Color3.fromRGB(8, 12, 24)
    local ON = Color3.fromRGB(60, 130, 255)
    local OFF_TEXT = Color3.fromRGB(180, 210, 245)
    local ON_TEXT = Color3.fromRGB(255, 255, 255)
    local BORDER = Color3.fromRGB(50, 80, 140)

    local function parentGui(gui)
        pcall(function() if gethui then gui.Parent = gethui() return end end)
        if not gui.Parent then pcall(function() gui.Parent = game:GetService("CoreGui") end) end
        if not gui.Parent then pcall(function() gui.Parent = Player:WaitForChild("PlayerGui") end) end
    end

    local function loadPos()
        local data = {}
        pcall(function()
            if readfile and isfile and isfile(POS_FILE) then
                local raw = readfile(POS_FILE)
                if raw and #raw > 2 then
                    local d = HS:JSONDecode(raw)
                    if type(d) == "table" then data = d end
                end
            end
        end)
        return data
    end

    local function savePos(gui)
        if not gui or not writefile then return end
        local out = {}
        for _, child in ipairs(gui:GetDescendants()) do
            if child:IsA("Frame") and child:GetAttribute("BtnKey") then
                local key = child:GetAttribute("BtnKey")
                out[key] = { xo = child.Position.X.Offset, yo = child.Position.Y.Offset }
            end
        end
        pcall(function() writefile(POS_FILE, HS:JSONEncode(out)) end)
    end

    pcall(function()
        for _, n in ipairs({"NOXMobileButtons", "K7MobileButtons", "HateVSMobileButtons", "22s_MobileButtons"}) do
            local cg = game:GetService("CoreGui"):FindFirstChild(n)
            if cg then cg:Destroy() end
            local pg = Player:FindFirstChild("PlayerGui")
            if pg then local o = pg:FindFirstChild(n) if o then o:Destroy() end end
            if gethui then
                local h = gethui()
                local o = h and h:FindFirstChild(n)
                if o then o:Destroy() end
            end
        end
    end)

    local buttons = {
        { key = "jumpDrop",     label = "DROP\nBR",       color = Color3.fromRGB(140, 180, 255) },
        { key = "autoPlay",     label = "AUTO\nPLAY",     color = Color3.fromRGB(100, 200, 255) },
        { key = "batAimbot",    label = "BAT\nAIMBOT",    color = Color3.fromRGB(255, 80, 100) },
        { key = "tpDown",       label = "TP\nDOWN",       color = Color3.fromRGB(255, 180, 100) },
        { key = "carry",        label = "CARRY\nSPD",     color = Color3.fromRGB(80, 180, 255) },
        { key = "laggerCarry",  label = "LAGGER\nCARRY",  color = Color3.fromRGB(255, 100, 150) },
        { key = "lagger",       label = "LAGGER\nMODE",   color = Color3.fromRGB(255, 140, 80) },
        { key = "tpBat",        label = "TP\nAIMBOT",     color = Color3.fromRGB(255, 130, 220) },
        { key = "instantReset", label = "RESET",          color = Color3.fromRGB(200, 130, 255) },
    }

    local saved = loadPos()
    local totalW = (COLS * (BTN_SIZE + BTN_GAP)) - BTN_GAP
    local rows = math.ceil(#buttons / COLS)
    local totalH = (rows * (BTN_SIZE + BTN_GAP)) - BTN_GAP
    local vp = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(800, 600)
    local startX = vp.X - totalW - 22
    local startY = math.max(18, vp.Y / 2 - totalH / 2 - 60)

    local gui = Instance.new("ScreenGui")
    gui.Name = "22s_MobileButtons"
    gui.ResetOnSpawn = false
    gui.IgnoreGuiInset = true
    gui.DisplayOrder = 60
    parentGui(gui)

    gui.Enabled = not Enabled.HideMobileButtons
    _G._22sMobileGui = gui
    _G._22sMobileSetHidden = function(hidden)
        if gui and gui.Parent then gui.Enabled = not hidden end
    end

    local states = {}
    local refs = {}

    local function setVisual(key, on)
        states[key] = on == true
        local r = refs[key]
        if not r then return end
        local col = r.color or ON
        if on then
            r.bg.BackgroundColor3 = col
            r.lbl.TextColor3 = ON_TEXT
            r.stroke.Color = col
            r.stroke.Transparency = 0
            r.grad.Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, Color3.fromRGB(math.max(0, col.R * 255 - 50), math.max(0, col.G * 255 - 50), math.max(0, col.B * 255 - 50))),
                ColorSequenceKeypoint.new(0.5, col),
                ColorSequenceKeypoint.new(1, Color3.fromRGB(math.max(0, col.R * 255 - 50), math.max(0, col.G * 255 - 50), math.max(0, col.B * 255 - 50))),
            })
        else
            r.bg.BackgroundColor3 = OFF
            r.lbl.TextColor3 = OFF_TEXT
            r.stroke.Color = BORDER
            r.stroke.Transparency = 0.3
            r.grad.Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, Color3.fromRGB(8, 12, 24)),
                ColorSequenceKeypoint.new(0.5, Color3.fromRGB(4, 6, 14)),
                ColorSequenceKeypoint.new(1, Color3.fromRGB(8, 12, 24)),
            })
        end
    end

    local function syncVisuals()
        setVisual("carry", Values.SpeedMode == "Carry")
        setVisual("lagger", Values.SpeedMode == "Lagger")
        setVisual("laggerCarry", Values.SpeedMode == "Lagger Carry")
        setVisual("autoPlay", AutoWalkEnabled == true or AutoRightEnabled == true)
        setVisual("batAimbot", Enabled.BatAimbot == true)
        setVisual("tpBat", Enabled.TPBat == true)
    end

    local function fire(key)
        if key == "carry" then
            if Values.SpeedMode == "Carry" then setSpeedMode("Normal") else setSpeedMode("Carry") end
        elseif key == "lagger" then
            if Values.SpeedMode == "Lagger" then setSpeedMode("Normal") else setSpeedMode("Lagger") end
        elseif key == "laggerCarry" then
            if Values.SpeedMode == "Lagger Carry" then setSpeedMode("Normal") else setSpeedMode("Lagger Carry") end
        elseif key == "autoPlay" then
            toggleAutoPlay()
        elseif key == "batAimbot" then
            Enabled.BatAimbot = not Enabled.BatAimbot
            if VisualSetters.BatAimbot then VisualSetters.BatAimbot(Enabled.BatAimbot) end
            if Enabled.BatAimbot then startBatAimbot() else stopBatAimbot() end
        elseif key == "tpBat" then
            Enabled.TPBat = not Enabled.TPBat
            if VisualSetters.TPBat then VisualSetters.TPBat(Enabled.TPBat) end
            if Enabled.TPBat then startTPBat() else stopTPBat() end
        elseif key == "jumpDrop" then pcall(runJumpDrop)
        elseif key == "tpDown" then pcall(runTPDown)
        elseif key == "instantReset" then pcall(runInstantReset) end
        task.defer(syncVisuals)
        pcall(SaveConfig)
    end

    local function playToggleOnEffect(frame, col)
        col = col or ON
        local origSize = frame.Size
        TweenService:Create(frame, TweenInfo.new(0.12, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            Size = UDim2.new(0, BTN_SIZE * 1.25, 0, BTN_SIZE * 1.25),
        }):Play()
        task.delay(0.12, function()
            if frame and frame.Parent then
                TweenService:Create(frame, TweenInfo.new(0.22, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                    Size = origSize,
                }):Play()
            end
        end)

    end

    for i, def in ipairs(buttons) do
        local col = ((i - 1) % COLS)
        local row = math.floor((i - 1) / COLS)
        local defX = startX + col * (BTN_SIZE + BTN_GAP)
        local defY = startY + row * (BTN_SIZE + BTN_GAP)
        local sp = saved[def.key]
        local xo = (sp and sp.xo) or defX
        local yo = (sp and sp.yo) or defY

        local frame = Instance.new("Frame")
        frame.Name = "Btn_" .. def.key
        frame.Size = UDim2.new(0, BTN_SIZE, 0, BTN_SIZE)
        frame.Position = UDim2.new(0, xo, 0, yo)
        frame.BackgroundColor3 = OFF
        frame.BorderSizePixel = 0
        frame.Active = true
        frame.ZIndex = 10
        frame:SetAttribute("BtnKey", def.key)
        frame.Parent = gui
        Instance.new("UICorner", frame).CornerRadius = UDim.new(1, 0)

        local bgGrad = Instance.new("UIGradient", frame)
        bgGrad.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(8, 12, 24)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(4, 6, 14)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(8, 12, 24)),
        })
        bgGrad.Rotation = 45

        local stroke = Instance.new("UIStroke", frame)
        stroke.Color = BORDER
        stroke.Thickness = 1.8
        stroke.Transparency = 0.3

        local lbl = Instance.new("TextLabel")
        lbl.Size = UDim2.new(1, -6, 1, -6)
        lbl.Position = UDim2.new(0, 3, 0, 3)
        lbl.BackgroundTransparency = 1
        lbl.Text = def.label
        lbl.TextColor3 = OFF_TEXT
        lbl.Font = Enum.Font.GothamBold
        lbl.TextSize = 11
        lbl.TextWrapped = true
        lbl.ZIndex = 11
        lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        lbl.TextStrokeTransparency = 0.3
        lbl.Parent = frame

        local hit = Instance.new("TextButton")
        hit.Size = UDim2.new(1, 0, 1, 0)
        hit.BackgroundTransparency = 1
        hit.Text = ""
        hit.ZIndex = 12
        hit.Parent = frame

        refs[def.key] = { bg = frame, lbl = lbl, stroke = stroke, grad = bgGrad, color = def.color }

        local dragging, dragStart, startPos = false, nil, nil
        local groupStarts = nil
        hit.InputBegan:Connect(function(input)
            if _G._22s_mobileLocked then return end
            if input.UserInputType == Enum.UserInputType.MouseButton1
                or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = frame.Position
                if _G._22s_groupDrag then
                    groupStarts = {}
                    for k, r in pairs(refs) do
                        groupStarts[k] = r.bg.Position
                    end
                end
                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        if dragging then
                            dragging = false
                            savePos(gui)
                            groupStarts = nil
                        end
                    end
                end)
            end
        end)

        UIS.InputChanged:Connect(function(input)
            if _G._22s_mobileLocked then return end
            if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
                or input.UserInputType == Enum.UserInputType.Touch) then
                local d = input.Position - dragStart
                if _G._22s_groupDrag and groupStarts then
                    for k, startP in pairs(groupStarts) do
                        local r = refs[k]
                        if r then
                            r.bg.Position = UDim2.new(0, startP.X.Offset + d.X, 0, startP.Y.Offset + d.Y)
                        end
                    end
                else
                    frame.Position = UDim2.new(0, startPos.X.Offset + d.X, 0, startPos.Y.Offset + d.Y)
                end
            end
        end)

        hit.Activated:Connect(function()
            local wasOn = states[def.key] == true
            local pressCol = Color3.fromRGB(
                math.min(255, (def.color or ON).R * 255 + 40),
                math.min(255, (def.color or ON).G * 255 + 40),
                math.min(255, (def.color or ON).B * 255 + 40)
            )
            TweenService:Create(frame, TweenInfo.new(0.07), { BackgroundColor3 = pressCol }):Play()
            TweenService:Create(stroke, TweenInfo.new(0.07), { Transparency = 0 }):Play()
            task.delay(0.12, function()
                setVisual(def.key, states[def.key] == true)
            end)
            fire(def.key)
            task.delay(0.16, function()
                local nowOn = states[def.key] == true
                if nowOn and not wasOn and frame and frame.Parent then
                    playToggleOnEffect(frame, def.color or ON)
                end
            end)
        end)
    end

    -- Expose reset function for main GUI button
    _G._22s_resetMobilePositions = function()
        pcall(function()
            if delfile and isfile and isfile(POS_FILE) then delfile(POS_FILE) end
        end)
        local vp2 = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(800, 600)
        local sx2 = vp2.X - totalW - 22
        local sy2 = math.max(18, vp2.Y / 2 - totalH / 2 - 60)
        for i, def in ipairs(buttons) do
            local r = refs[def.key]
            if r and r.bg then
                local col = ((i - 1) % COLS)
                local row = math.floor((i - 1) / COLS)
                local x = sx2 + col * (BTN_SIZE + BTN_GAP)
                local y = sy2 + row * (BTN_SIZE + BTN_GAP)
                TweenService:Create(r.bg, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
                    Position = UDim2.new(0, x, 0, y),
                }):Play()
            end
        end
        task.delay(0.4, function() pcall(savePos, gui) end)
    end

    task.spawn(function()
        while gui and gui.Parent do
            task.wait(0.3)
            pcall(syncVisuals)
        end
    end)

    task.spawn(function()
        while gui and gui.Parent do
            task.wait(3)
            pcall(savePos, gui)
        end
    end)

    task.spawn(function()
        while gui and gui.Parent do
            local t = tick()
            for k, r in pairs(refs) do
                if states[k] and r.stroke then
                    r.stroke.Transparency = 0.15 + math.sin(t * 3) * 0.15
                    r.grad.Offset = Vector2.new(math.sin(t * 1.5) * 0.3, 0)
                end
            end
            task.wait(0.05)
        end
    end)

    task.defer(function()
        if Values.SpeedMode ~= "Carry" and Values.SpeedMode ~= "Lagger" and Values.SpeedMode ~= "Lagger Carry" then
            setSpeedMode("Normal")
        end
        syncVisuals()
    end)

    print("[22s Hub] Mobile buttons loaded")
end
