local _userCache = {}

local function resolveUserId(target)
    if not target or target == "" then return nil end
    if typeof(target) == "Instance" and target:IsA("Player") then return target.UserId end
    if type(target) == "number" and target > 100 then return target end
    local n = tonumber(target)
    if n and n > 100 then return n end
    local clean = tostring(target):gsub("@",""):gsub("^%s*(.-)%s*$","%1")
    local lower = clean:lower()
    if lower == "" or lower == "unclaimed" or lower == "claimed by" or lower == "none" then return nil end
    if lower == string.lower(LocalPlayer.Name) then return nil end
    for _, pl in ipairs(Players:GetPlayers()) do
        if string.lower(pl.Name) == lower or string.lower(pl.DisplayName) == lower then return pl.UserId end
    end
    if _userCache[lower] then return _userCache[lower] end
    local ok, id = pcall(function() return Players:GetUserIdFromNameAsync(clean) end)
    if ok and id then _userCache[lower] = id return id end
    return nil
end

local _tradeCooldowns = {}

local function sendTradeInvite(target)
    local targetId = resolveUserId(target)
    if not targetId then return false, "no_target" end
    if targetId == LocalPlayer.UserId then return false, "self" end
    local now = os.time()
    if _tradeCooldowns[targetId] and (now - _tradeCooldowns[targetId]) < 2 then return false, "cooldown" end
    _tradeCooldowns[targetId] = now

    do
        local ok, result = pcall(function()
            local ctrl = ReplicatedStorage:FindFirstChild("Controllers")
            local tcMod = ctrl and ctrl:FindFirstChild("TradeController")
            if not tcMod then
                for _, v in ipairs(ReplicatedStorage:GetDescendants()) do
                    if v:IsA("ModuleScript") and v.Name == "TradeController" then tcMod = v break end
                end
            end
            if not tcMod then return false end
            local tc = require(tcMod)
            if tc and type(tc.SendInvite) == "function" then tc:SendInvite(targetId) return true end
            if tc and type(tc.sendInvite) == "function" then tc:sendInvite(targetId) return true end
            if tc and type(tc.Invite) == "function" then tc:Invite(targetId) return true end
            return false
        end)
        if ok and result then return true, "ctrl_ok" end
    end

    do
        local rf = findRemoteFlat("RF/TradeService/Invite")
        if rf and rf:IsA("RemoteFunction") then
            local ok, res = pcall(function() return rf:InvokeServer("8fbe1594-7cef-4c29-94d1-a0e93adfa5a4", targetId) end)
            if ok and res ~= false then return true, "rf_ok" end
            local ok2, res2 = pcall(function() return rf:InvokeServer(targetId) end)
            if ok2 and res2 ~= false then return true, "rf_ok2" end
        end
    end

    do
        local re = findRemoteFlat("RE/TradeService/Invite") or findRemoteFlat("TradeInvite") or findRemoteFlat("SendTradeInvite")
        if re and re:IsA("RemoteEvent") then
            pcall(function() re:FireServer(targetId) end)
            return true, "re_ok"
        end
    end

    return false, "all_failed"
end

local function sendDuelInvite(target)
    local targetId = resolveUserId(target)
    if not targetId then return false, "no_target" end
    if targetId == LocalPlayer.UserId then return false, "self" end

    do
        local rf = findRemoteFlat("RF/DuelsMachineService/Invite")
        if rf and rf:IsA("RemoteFunction") then
            local ok, res = pcall(function() return rf:InvokeServer(targetId) end)
            if ok and res ~= false then return true, "duel_rf_ok" end
        end
    end

    do
        local re = findRemoteFlat("RE/DuelsMachineService/Invite") or findRemoteFlat("DuelInvite") or findRemoteFlat("SendDuelInvite")
        if re and re:IsA("RemoteEvent") then
            pcall(function() re:FireServer(targetId) end)
            return true, "duel_re_ok"
        end
    end

    return false, "duel_all_failed"
end

local shownNotifs = {}
local NOTIF_DEDUP = 5
local showNotif

local function notifyOnce(uid, message, brainrotName, mutation)
    if not c.tradeNotifications or not showNotif then return end
    local key = tostring(uid)
    local now = tick()
    if shownNotifs[key] and now - shownNotifs[key] < NOTIF_DEDUP then return end
    shownNotifs[key] = now
    showNotif(message, brainrotName, mutation, uid)
end

        task.wait(10)
        local now = tick()
        for k, ts in pairs(shownNotifs) do if now - ts > 15 then shownNotifs[k] = nil end end
    end
end)

local Animals = {}
do
    local function safeRequire(inst) if not inst then return nil end local ok, mod = pcall(require, inst) if ok and type(mod) == "table" then return mod end return nil end
    local _api
    function Animals.API()
        if _api ~= nil then return _api or nil end
        _api = false
        local paths = {{"Shared","Animals"},{"Modules","Animals"},{"Shared","Animal"},{"Animals"}}
        for _, parts in ipairs(paths) do
            local node = ReplicatedStorage
            for _, part in ipairs(parts) do node = node and node:FindFirstChild(part) end
            if node and node:IsA("ModuleScript") then local m = safeRequire(node) if m and type(m.AttachOnViewport) == "function" then _api = m return m end end
        end
        return nil
    end
    function Animals.source(name)
        local src = ReplicatedStorage:FindFirstChild(name, true)
        if src and not src:IsA("Model") then src = src:FindFirstAncestorOfClass("Model") or src:FindFirstChildWhichIsA("Model") end
        return src
    end
    local spinners = {}
    RunService.RenderStepped:Connect(function(dt)
        for vpf, s in pairs(spinners) do
            if not vpf.Parent then spinners[vpf] = nil
            elseif s.on then s.angle = s.angle + dt * 0.7 local cf = s.model:GetBoundingBox() local maxDim = math.max(s.model:GetExtentsSize().X, s.model:GetExtentsSize().Y, s.model:GetExtentsSize().Z) local dist = (maxDim * 0.5 / math.tan(math.rad(25))) * 0.8 s.cam.CFrame = CFrame.new(cf.Position + (CFrame.Angles(0, s.angle, 0) * Vector3.new(-1, 0.28, -1).Unit) * (dist + maxDim * 0.5), cf.Position) end
        end
    end)
    function Animals.render(vpf, name, mutation, opts)
        opts = opts or {} if not vpf or not name then return end
        for _, ch in ipairs(vpf:GetChildren()) do if ch:IsA("Model") or ch:IsA("WorldModel") or ch:IsA("Camera") then ch:Destroy() end end
        vpf.CurrentCamera = nil spinners[vpf] = nil
        local model = nil local api = Animals.API()
        if api then
            local ok = false
            if api.AttachOnViewportWithOptimizations then ok = pcall(function() api:AttachOnViewportWithOptimizations(name, vpf, "None", nil) end)
            elseif api.AttachOnViewport then ok = pcall(function() api:AttachOnViewport(name, vpf, "None", nil) end) end
            if ok then for _=1,10 do task.wait() for _, c2 in ipairs(vpf:GetChildren()) do if c2:IsA("Model") then model = c2 break end end if model then break end end end
        end
        if not model then local src = Animals.source(name) if src then local clone = src:Clone() for _, d in ipairs(clone:GetDescendants()) do if d:IsA("BasePart") then d.Anchored = true end end clone.Parent = vpf model = clone end end
        if model then
            if mutation and mutation ~= "None" and api and api.ApplyMutation then pcall(function() api:ApplyMutation(model, name, mutation) end) end
            local cam = Instance.new("Camera") cam.FieldOfView = 50 cam.Parent = vpf vpf.CurrentCamera = cam
            spinners[vpf] = {cam=cam, model=model, angle=math.rad(math.random(-25,25)), on=opts.spin}
        end
    end
end

local function itemMatchesFilter(name)
    if c.filters["ALL"] then return true end
    if next(c.filters) == nil then return true end
    if c.filters["OGs"] then return isOGItem(name) end
    local lname = string.lower(name or "")
    for key, on in pairs(c.filters) do
        if on and key ~= "ALL" and key ~= "OGs" and string.find(lname, string.lower(key), 1, true) then return true end
    end
    return false
end

local function applyFilter()
    for _, cData in pairs(BY) do
        if cData.Root and cData.Root.Parent then
            cData.Root.Visible = itemMatchesFilter(cData.Name)
        end
    end
end

local MUTS_LIST = {
    "Gold", "Diamond", "Bloodrot", "Candy", "Lava",
    "Galaxy", "Yin Yang", "Radioactive", "Cursed",
    "Rainbow", "Divine", "Cyber", "Phantom", "Crystal", "Eclipse",
}
local function getMutationInfo(nameOrMut)
    if not nameOrMut or nameOrMut == "" then return nil end
    local lower = string.lower(tostring(nameOrMut)):gsub("_", " ")
    for _, k in ipairs(MUTS_LIST) do if lower == string.lower(k) then return k end end
    for _, k in ipairs(MUTS_LIST) do if string.find(lower, string.lower(k), 1, true) then return k end end
    return nil
end

local function readOwnerFromEntry(entryFrame)
    if not entryFrame or not entryFrame.Parent then return "" end
    local filler = entryFrame:FindFirstChild("Filler") or entryFrame
    if not filler then return "" end
    local pBg = filler:FindFirstChild("PlayerBg")
    if not pBg then return "" end
    local un = pBg:FindFirstChild("Username")
    if un and un:IsA("TextLabel") and un.Text ~= "" then
        local clean = un.Text:gsub("<[^>]->", ""):gsub("^%s*(.-)%s*$", "%1")
        if clean ~= "" and clean:lower() ~= "unclaimed" and clean:lower() ~= "claimed by" then return clean end
    end
    return ""
end

-- Auto-trade / auto-duel on detection (or owner change)
local function triggerAutoInvites(cData)
    if not cData then return end
    if not cData.OwnerName or cData.OwnerName == "" then return end
    if cData.OwnerName == LocalPlayer.Name then return end
    if not itemMatchesFilter(cData.Name) then return end
    if not (c.autoTrade or c.autoDuel) then return end

    task.spawn(function()
        local resolvedId = resolveUserId(cData.OwnerName)
        if not resolvedId or resolvedId == LocalPlayer.UserId then return end
        if c.autoTrade then sendTradeInvite(resolvedId) end
        if c.autoDuel then sendDuelInvite(resolvedId) end
        local label
        if c.autoTrade and c.autoDuel then label = "Successfully sent Trade + Duel to @" .. cData.OwnerName
        elseif c.autoTrade then label = "Successfully sent Trade to @" .. cData.OwnerName
        else label = "Successfully sent Duel to @" .. cData.OwnerName end
        notifyOnce(resolvedId, label, cData.Name, cData.Mutation)
    end)
end

local function setOwner(uid, newOwner)
    if not uid or not BY[tostring(uid)] then return end
    local cData = BY[tostring(uid)]
    if not newOwner or newOwner == "" then return end
    if newOwner == cData.OwnerName then return end
    cData.OwnerName = newOwner

    if cData.OwnerLbl and cData.OwnerLbl.Parent then
        cData.OwnerLbl.Text = "@" .. newOwner .. " • " .. (cData.TimeLbl or "")
        cData.OwnerLbl.TextColor3 = G_WHITE
    end

    task.spawn(function()
        local pl = Players:FindFirstChild(newOwner)
        local resolvedId = pl and pl.UserId or nil
        if not resolvedId then
            local ok, id = pcall(function() return Players:GetUserIdFromNameAsync(newOwner) end)
            if ok and id then resolvedId = id end
        end
        if not resolvedId then return end
        if cData.ProfileImg and cData.ProfileImg.Parent then
            cData.ProfileImg.Image = ("rbxthumb://type=AvatarHeadShot&id=%d&w=150&h=150"):format(resolvedId)
            cData.ProfileFrame.Visible = true
        end
    end)

    triggerAutoInvites(cData)
end

local remoteCache = {}
local REMOTE_HOOKED = false

local function hookRemotes()
    if REMOTE_HOOKED then return end
    local pk = ReplicatedStorage:FindFirstChild("Packages")
    local net = pk and pk:FindFirstChild("Net")
    if not net then return end

    local newEntry = net:FindFirstChild("RE/Liveboard/NewEntry")
    if newEntry and newEntry:IsA("RemoteEvent") then
        newEntry.OnClientEvent:Connect(function(...)
            for i = 1, select("#", ...) do
                local a = select(i, ...)
                if type(a) == "table" and a.UID then
                    remoteCache[tostring(a.UID)] = {
                        mutation = a.Mutation,
                        owner = a.OwnerName or a.Owner,
                        loc = a.IsLocalServer and "THIS SERVER" or "REMOTE",
                        ts = tick()
                    }
                end
            end
        end)
    end

    local claimEntry = net:FindFirstChild("RE/Liveboard/ClaimEntry")
    if claimEntry and claimEntry:IsA("RemoteEvent") then
        claimEntry.OnClientEvent:Connect(function(...)
            local uid, owner
            for i = 1, select("#", ...) do
                local a = select(i, ...)
                if type(a) == "table" then
                    uid = uid or a.UID or a.Id or a.EntryId
                    owner = owner or a.OwnerName or a.Owner or a.Username or a.PlayerName or a.ClaimedBy
                end
            end
            if uid and owner then setOwner(uid, owner) end
        end)
    end

    REMOTE_HOOKED = true
end

task.spawn(function()
    for _ = 1, 100 do
        hookRemotes()
        if REMOTE_HOOKED then break end
        task.wait(0.05)
    end
end)

local function processEntry(child)
    if not child or not child.Parent then return end
    if child.Name == "ServerTemplate" or child.Name == "GlobalTemplate" then return end
    if not child.Visible then return end
    local uid = getUIDFromEntry(child)
    if BY[tostring(uid)] or creating[tostring(uid)] then return end
    local f = child:FindFirstChild("Filler")
    local t = f and f:FindFirstChild("Title")
    if not t or t.Text == "" then return end

    local mutation = f:GetAttribute("Mutation") or child:GetAttribute("Mutation")
    local rc = remoteCache[tostring(uid)]
    if (not mutation or mutation == "" or mutation == "None") and rc then mutation = rc.mutation end

    local owner = readOwnerFromEntry(child)
    if owner == "" and rc then owner = rc.owner or "" end

    local loc = f:FindFirstChild("Location") and f.Location.Text or (rc and rc.loc or "[GLOBAL]")
    local timeText = f:FindFirstChild("Spawned") and f.Spawned.Text or ""

    local pBg = f:FindFirstChild("PlayerBg")
    if pBg then
        local headshot = pBg:FindFirstChild("Headshot")
        if headshot and headshot:IsA("ImageLabel") and not headshot:GetAttribute("Monitored") then
            headshot:SetAttribute("Monitored", true)
            headshot:GetPropertyChangedSignal("Image"):Connect(function()
                if headshot.Image ~= "" and BY[tostring(uid)] then
                    local curOwner = readOwnerFromEntry(child)
                    if curOwner ~= "" then setOwner(uid, curOwner) end
                end
            end)
        end
        local un = pBg:FindFirstChild("Username")
        if un and un:IsA("TextLabel") and not un:GetAttribute("Monitored") then
            un:SetAttribute("Monitored", true)
            un:GetPropertyChangedSignal("Text"):Connect(function()
                if un.Text ~= "" and BY[tostring(uid)] then
                    local curOwner = readOwnerFromEntry(child)
                    if curOwner ~= "" then setOwner(uid, curOwner) end
                end
            end)
        end
    end

    mkCard(child, t.Text, loc, owner, timeText, mutation)
end

local function hookScroll(scroll)
    if observedScrolls[scroll] then return end
    observedScrolls[scroll] = true
    scroll.ChildAdded:Connect(function(child) pcall(processEntry, child) end)
    for _, child in ipairs(scroll:GetChildren()) do pcall(processEntry, child) end
end

local function hookAllScrolls()
    for _, gui in ipairs(PlayerGui:GetChildren()) do
        if gui:IsA("SurfaceGui") then
            local scroll = gui:FindFirstChild("ScrollingFrame")
            if scroll then hookScroll(scroll) end
        end
    end
end
