-- =============================================
--                Connections

SpeedToggleBtn.MouseButton1Click:Connect(function()
    if SpeedEnabled then
        SpeedEnabled.Value = false
        SpeedToggleBtn.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
        SpeedToggleLabel.Text = "OFF"
    end
end)

UnwalkToggleBtn.MouseButton1Click:Connect(function()
    if UnwalkEnabled then
        UnwalkEnabled.Value = false
        UnwalkToggleBtn.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
        UnwalkToggleLabel.Text = "OFF"
    end
end)

SpeedTextBox.FocusLost:Connect(function(enterPressed)
    if SavedSpeed and enterPressed then
        local num = tonumber(SpeedTextBox.Text)
        if num then
            SavedSpeed.Value = num
        end
    end
end)

JumpTextBox.FocusLost:Connect(function(enterPressed)
    if SavedJump and enterPressed then
        local num = tonumber(JumpTextBox.Text)
        if num then
            SavedJump.Value = num
        end
    end
end)
