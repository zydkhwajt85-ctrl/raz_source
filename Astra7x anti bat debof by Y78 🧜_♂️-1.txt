local antiBatEnabled = true

local function updateAntiBatVisual(state)

        G.StatusDot.BackgroundColor3 = Color3.fromRGB(90, 215, 255)

        TweenService:Create(G.AntiBatToggleBg, TweenInfo.new(0.18), {BackgroundColor3 = Color3.fromRGB(20, 40, 75)}):Play()
        TweenService:Create(G.AntiBatToggleKnob, TweenInfo.new(0.18), {Position = UDim2.new(0, 4, 0.5, -7)}):Play()
        G.StatusText.Text = "INACTIVE"

local function setAntiBat(state)
    antiBatEnabled = state
    updateAntiBatVisual(antiBatEnabled)
    if antiBatConn then
        antiBatConn:Disconnect()
        antiBatConn = nil
    end
    if antiBatEnabled then
        antiBatConn = RunService.Heartbeat:Connect(function()
            if not antiBatEnabled then
                return
            end
            local char = LocalPlayer.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if not root or not root.Parent then
                return
            end
            local vel = root.AssemblyLinearVelocity
            local flat = Vector3.new(vel.X, 0, vel.Z)
            root.AssemblyLinearVelocity = Vector3.new(currentPower, vel.Y, currentPower)
            RunService.RenderStepped:Wait()
            if root and root.Parent then
                local y2 = root.AssemblyLinearVelocity.Y
                root.AssemblyLinearVelocity = Vector3.new(flat.X, y2, flat.Z)
            end
        end)
    end
end
G.AntiBatBtn.MouseButton1Click:Connect(function()
    setAntiBat(not antiBatEnabled)
end)
local InfiniteJump = {
    enabled = true,
    jumpPower = 55,
    minVelocity = 35,
    fallClamp = -120,
    jumpConn = nil,
    heartbeatConn = nil
}
local function applyJump(root)
    if not root then
        return
    end
    pcall(function()
        root.AssemblyLinearVelocity = Vector3.new(root.AssemblyLinearVelocity.X, InfiniteJump.jumpPower, root.AssemblyLinearVelocity.Z)
    end)
end
local function onJumpRequest()
    if not InfiniteJump.enabled then
        return
    end
    local char = LocalPlayer.Character
    if not char then
        return
    end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root then
        applyJump(root)
    end
end
local function onHeartbeat()
    if not InfiniteJump.enabled then
        return
    end
    local char = LocalPlayer.Character
    if not char then
        return
    end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then
        return
    end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then
        return
    end
    local jumpHeld = UserInputService:IsKeyDown(Enum.KeyCode.Space) or hum.Jump == true
    if jumpHeld and root.AssemblyLinearVelocity.Y < InfiniteJump.minVelocity then
        applyJump(root)
    end
    if root.AssemblyLinearVelocity.Y < InfiniteJump.fallClamp then
        root.AssemblyLinearVelocity = Vector3.new(root.AssemblyLinearVelocity.X, InfiniteJump.fallClamp, root.AssemblyLinearVelocity.Z)
    end
end
local function connectJumpEvents()
    if InfiniteJump.jumpConn then
        InfiniteJump.jumpConn:Disconnect()
    end
    if InfiniteJump.heartbeatConn then
        InfiniteJump.heartbeatConn:Disconnect()
    end
    InfiniteJump.jumpConn = UserInputService.JumpRequest:Connect(onJumpRequest)
    InfiniteJump.heartbeatConn = RunService.Heartbeat:Connect(onHeartbeat)
end
local function setInfJump(state)
    InfiniteJump.enabled = state
    if InfiniteJump.enabled then
        TweenService:Create(G.InfJumpToggleBg, TweenInfo.new(0.18), {BackgroundColor3 = Color3.fromRGB(40, 160, 255)}):Play()
        TweenService:Create(G.InfJumpToggleKnob, TweenInfo.new(0.18), {Position = UDim2.new(1, -18, 0.5, -7)}):Play()
        connectJumpEvents()
    else
        TweenService:Create(G.InfJumpToggleBg, TweenInfo.new(0.18), {BackgroundColor3 = Color3.fromRGB(20, 40, 75)}):Play()
        TweenService:Create(G.InfJumpToggleKnob, TweenInfo.new(0.18), {Position = UDim2.new(0, 4, 0.5, -7)}):Play()
        if InfiniteJump.jumpConn then
            InfiniteJump.jumpConn:Disconnect()
            InfiniteJump.jumpConn = nil
        end
        if InfiniteJump.heartbeatConn then
            InfiniteJump.heartbeatConn:Disconnect()
            InfiniteJump.heartbeatConn = nil
        end
    end
end
G.InfJumpBtn.MouseButton1Click:Connect(function()
    setInfJump(not InfiniteJump.enabled)
end)
local sliderActive = false
local function updatePowerFromInput(input)
    local trackAbsPos = G.SliderTrack.AbsolutePosition
    local trackAbsSize = G.SliderTrack.AbsoluteSize
    local pct = math.clamp((input.Position.X - trackAbsPos.X) / trackAbsSize.X, 0, 1)
    currentPower = math.floor(pct * 300000)
    G.SliderFill.Size = UDim2.new(pct, 0, 1, 0)
    G.SliderKnob.Position = UDim2.new(pct, 0, 0.5, 0)
    G.PowerValText.Text = tostring(math.floor(pct * 100)) .. "%"
end
G.SliderBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        sliderActive = true
        updatePowerFromInput(input)
    end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        sliderActive = false
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if sliderActive and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        updatePowerFromInput(input)
    end
end)
