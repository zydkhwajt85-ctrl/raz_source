local holdingJump = false
local infJumpConnection = nil

local function startInfJump()
    if infJumpConnection then infJumpConnection:Disconnect(); infJumpConnection = nil end
    infJumpConnection = RunService.RenderStepped:Connect(function()
        if not infJumpEnabled then return end
        if not holdingJump then return end

        local char = player.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not hum or not root or hum.Health <= 0 then return end

        local state = hum:GetState()
        if state ~= Enum.HumanoidStateType.Jumping and state ~= Enum.HumanoidStateType.Freefall then
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
        end

        local vel = root.AssemblyLinearVelocity
        if vel.Y < 35 then
            root.AssemblyLinearVelocity = Vector3.new(vel.X, 48, vel.Z)
        end
    end)
end

local function stopInfJump()
    if infJumpConnection then
        infJumpConnection:Disconnect()
        infJumpConnection = nil
    end
    holdingJump = false
end

-- Toggle functions (used by GUI)
function startHoldInfJump()
    startInfJump()
end

function stopHoldInfJump()
    stopInfJump()
end

local function setupSpeedVelocity(root)
    if speedLinearVelocity then speedLinearVelocity:Destroy(); speedLinearVelocity = nil end
    if speedAttachment then speedAttachment:Destroy(); speedAttachment = nil end
    if not root then return nil end
    local attachment = Instance.new("Attachment")
    attachment.Name = "S2SpeedAttachment"
    attachment.Parent = root
    local linearVelocity = Instance.new("LinearVelocity")
    linearVelocity.Name = "S2SpeedVelocity"
    linearVelocity.Attachment0 = attachment
    linearVelocity.RelativeTo = Enum.ActuatorRelativeTo.World
    linearVelocity.ForceLimitMode = Enum.ForceLimitMode.PerAxis
    linearVelocity.MaxAxesForce = Vector3.new(math.huge, 0, math.huge)
    linearVelocity.VectorVelocity = Vector3.zero
    linearVelocity.Enabled = true
    linearVelocity.Parent = attachment
    speedAttachment = attachment
    speedLinearVelocity = linearVelocity
    return linearVelocity
end

local function destroySpeedVelocity()
    if speedLinearVelocity then pcall(function() speedLinearVelocity:Destroy() end); speedLinearVelocity = nil end
    if speedAttachment then pcall(function() speedAttachment:Destroy() end); speedAttachment = nil end
end

player.CharacterAdded:Connect(function() destroySpeedVelocity() end)

local function isRagdollState(hum)
    if not hum then return true end
    local st = hum:GetState()
    return hum.PlatformStand or st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll or st == Enum.HumanoidStateType.FallingDown
end

RunService.RenderStepped:Connect(function()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    if isRagdollState(hum) then
        lastMoveDir = Vector3.new(0,0,0)
        destroySpeedVelocity()
        return
    end

    if autoBatEnabled or autoLeftEnabled or autoRightEnabled then
        destroySpeedVelocity()
    else
        local linearVelocity = speedLinearVelocity
        if not linearVelocity or not speedAttachment or speedAttachment.Parent ~= hrp then
            linearVelocity = setupSpeedVelocity(hrp)
        end

        local md = hum.MoveDirection
        local spd = getActiveMoveSpeed()
        local dir = nil

        if md.Magnitude > 0 then
            lastMoveDir = md
            dir = md
        elseif lastMoveDir.Magnitude > 0 then
            local moveKeys = {Enum.KeyCode.W, Enum.KeyCode.A, Enum.KeyCode.S, Enum.KeyCode.D}
            for _, key in ipairs(moveKeys) do
                if UIS:IsKeyDown(key) then
                    dir = lastMoveDir
                    break
                end
            end
        end

        if linearVelocity then
            if dir then
                local flat = Vector3.new(dir.X, 0, dir.Z)
                if flat.Magnitude > 0 then
                    flat = flat.Unit
                    linearVelocity.VectorVelocity = Vector3.new(flat.X * spd, 0, flat.Z * spd)
                else
                    linearVelocity.VectorVelocity = Vector3.zero
                end
            else
                linearVelocity.VectorVelocity = Vector3.zero
            end
        end
    end

    if speedLabel then
        speedLabel.Text = string.format("%.1f", Vector3.new(hrp.AssemblyLinearVelocity.X, 0, hrp.AssemblyLinearVelocity.Z).Magnitude)
    end
end)

function runDrop()
    if dropActive then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    dropActive = true
    local t0 = tick()
    local dc
    dc = RunService.Heartbeat:Connect(function()
        local r = char and char:FindFirstChild("HumanoidRootPart")
        if not r then dc:Disconnect(); dropActive = false; return end
        if tick() - t0 >= DROP_ASCEND_DURATION then
            dc:Disconnect()
            local rp = RaycastParams.new()
            rp.FilterDescendantsInstances = {char}
            rp.FilterType = Enum.RaycastFilterType.Exclude
            local rr = workspace:Raycast(r.Position, Vector3.new(0, -2000, 0), rp)
            if rr then
                local hum2 = char:FindFirstChildOfClass("Humanoid")
                local off = (hum2 and hum2.HipHeight or 2) + (r.Size.Y / 2)
                r.CFrame = CFrame.new(r.Position.X, rr.Position.Y + off, r.Position.Z)
                r.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            end
            dropActive = false
            return
        end
        r.AssemblyLinearVelocity = Vector3.new(r.AssemblyLinearVelocity.X, DROP_ASCEND_SPEED, r.AssemblyLinearVelocity.Z)
    end)
end

local function isMyPlotByName(plotName)
    local plots = workspace:FindFirstChild("Plots")
    if not plots then return false end
    local plot = plots:FindFirstChild(plotName)
    if not plot then return false end
    local sign = plot:FindFirstChild("PlotSign")
    if sign then
        local yb = sign:FindFirstChild("YourBase")
        if yb and yb:IsA("BillboardGui") then return yb.Enabled == true end
    end
    return false
end

local function isNearPodiumWithPrompt()
    local char = player.Character
    local hrpL = char and char:FindFirstChild("HumanoidRootPart")
    if not hrpL then return false end
    local plots = workspace:FindFirstChild("Plots")
    if not plots then return false end
    for _, plot in ipairs(plots:GetChildren()) do
        if isMyPlotByName(plot.Name) then continue end
        local podiums = plot:FindFirstChild("AnimalPodiums")
        if not podiums then continue end
        for _, podium in ipairs(podiums:GetChildren()) do
            local base = podium:FindFirstChild("Base")
            if not base then continue end
            local sp = base:FindFirstChild("Spawn")
            if not sp then continue end
            local d = (hrpL.Position - sp.Position).Magnitude
            if d > Steal.StealRadius then continue end
            local att = sp:FindFirstChild("PromptAttachment")
            if not att then continue end
            for _, obj in ipairs(att:GetChildren()) do
                if obj:IsA("ProximityPrompt") and obj.Enabled then return true, d end
            end
        end
    end
    return false, math.huge
end

local function findNearestPrompt()
    local char = player.Character
    if not char then return nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local plots = workspace:FindFirstChild("Plots")
    if not plots then return nil end
    local nearest, dist = nil, math.huge
    for _, plot in ipairs(plots:GetChildren()) do
        if isMyPlotByName(plot.Name) then continue end
        local pods = plot:FindFirstChild("AnimalPodiums")
        if not pods then continue end
        for _, pod in ipairs(pods:GetChildren()) do
            local base = pod:FindFirstChild("Base")
            local sp = base and base:FindFirstChild("Spawn")
            if sp then
                local d = (sp.Position - root.Position).Magnitude
                if d <= Steal.StealRadius and dist > d then
                    local att = sp:FindFirstChild("PromptAttachment")
                    if att then
                        for _, prompt in ipairs(att:GetChildren()) do
                            if prompt:IsA("ProximityPrompt") and prompt.ActionText:find("Steal") then
                                nearest, dist = prompt, d
                            end
                        end
                    end
                end
            end
        end
    end
    return nearest
end

local function executeSteal(prompt)
    if isStealing then return end
    if not Steal.Data[prompt] then
        Steal.Data[prompt] = {hold = {}, trigger = {}, ready = true}
        if getconnections then
            for _, c in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do
                if c.Function then table.insert(Steal.Data[prompt].hold, c.Function) end
            end
            for _, c in ipairs(getconnections(prompt.Triggered)) do
                if c.Function then table.insert(Steal.Data[prompt].trigger, c.Function) end
            end
        end
    end
    local data = Steal.Data[prompt]
    if not data.ready then return end
    data.ready = false
    isStealing = true
    stealStartTime = tick()
    task.spawn(function()
        for _, fn in ipairs(data.hold) do task.spawn(fn) end
        task.wait(Steal.StealDuration)
        for _, fn in ipairs(data.trigger) do task.spawn(fn) end
        data.ready = true
        isStealing = false
        stealStartTime = nil
    end)
end

local stealConn = nil
function startAutoSteal()
    if stealConn then return end
    stealConn = RunService.Heartbeat:Connect(function()
        if not Steal.AutoStealEnabled or isStealing then return end
        local p = findNearestPrompt()
        if p then executeSteal(p) end
    end)
end

function stopAutoSteal()
    if stealConn then stealConn:Disconnect(); stealConn = nil end
    isStealing = false
    stealStartTime = nil
end

            local now = tick()
            if Steal.AutoStealEnabled then
                local inRadius = isNearPodiumWithPrompt()
                local pct = 0
                if isStealing and stealStartTime then
                    pct = math.clamp((now - stealStartTime) / Steal.StealDuration, 0, 1)
                    fillLine.Size = UDim2.new(pct, 0, 1, 0)
                    fillGrad.Offset = Vector2.new(math.sin(now * 0.8) * 0.3, 0)
                elseif inRadius then
                    local cyclePos = (now % Steal.StealDuration) / Steal.StealDuration
                    pct = cyclePos * cyclePos * (3 - 2 * cyclePos)
                    fillLine.Size = UDim2.new(pct * 0.6, 0, 1, 0)
                    fillGrad.Offset = Vector2.new(math.sin(now * 0.5) * 0.4, 0)
                else
                    fillLine.Size = UDim2.new(0, 0, 1, 0)
                    fillGrad.Offset = Vector2.new(0, 0)
                end
                pctLbl.Text = math.floor(pct * 100) .. "%"
                local r = 255
                local g = math.floor(230 - (pct * 140))
                local b = math.floor(50 - (pct * 30))
                pctLbl.TextColor3 = Color3.fromRGB(r, math.max(g, 30), math.max(b, 10))
            else
                fillLine.Size = UDim2.new(0, 0, 1, 0)
                pctLbl.Text = "0%"
                pctLbl.TextColor3 = Color3.fromRGB(255, 220, 100)
                fillGrad.Offset = Vector2.new(0, 0)
            end
            task.wait(0.016)
        end
    end)

    local dragStart2, dragStartPos2, dragging2 = nil, nil, false
    stealBarFrame.InputBegan:Connect(function(input)
        if uiLocked then return end  -- Lock dragging when locked
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging2 = true
            dragStart2 = input.Position
            dragStartPos2 = stealBarFrame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging2 = false
                    saveAllConfig()
                end
            end)
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if uiLocked then dragging2 = false; return end
        if dragging2 and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart2
            stealBarFrame.Position = UDim2.new(dragStartPos2.X.Scale, dragStartPos2.X.Offset + delta.X, dragStartPos2.Y.Scale, dragStartPos2.Y.Offset + delta.Y)
        end
    end)
end

local function saveAllConfig()
    if not writefile then return end

    local mobilePositions = {}
    local gui = player.PlayerGui:FindFirstChild("DivineMobileVS")
    if gui then
        for _, child in ipairs(gui:GetChildren()) do
            if child:IsA("Frame") and child:GetAttribute("BtnKey") then
                local key = child:GetAttribute("BtnKey")
                mobilePositions[key] = {
                    xo = child.Position.X.Offset,
                    yo = child.Position.Y.Offset
                }
            end
        end
    end

    local stealBarPos = nil
    if stealBarFrame then
        stealBarPos = {
            x = stealBarFrame.Position.X.Offset,
            y = stealBarFrame.Position.Y.Offset
        }
    end

    local cfg = {
        normalSpeed = NS, carrySpeed = CS, laggerSpeed = LAGGER_SPEED, laggerCarrySpeed = LAGGER_CARRY_SPEED,
        carrySpeedActive = carrySpeedActive, laggerModeEnabled = laggerModeEnabled, laggerCarryActive = laggerCarryActive,
        uiLocked = uiLocked, autoLeftEnabled = autoLeftEnabled, autoRightEnabled = autoRightEnabled,
        autoBatEnabled = autoBatEnabled, autoSwingEnabled = autoSwingEnabled,
        autoStealEnabled = Steal.AutoStealEnabled, antiRagdollEnabled = antiRagdollEnabled,
        infJumpEnabled = infJumpEnabled, infJumpMode = infJumpMode,
        autoTPEnabled = autoTPEnabled, autoTPHeight = autoTPHeight,
        antiLagEnabled = antiLagEnabled, stretchRezEnabled = stretchRezEnabled,
        batCounterEnabled = batCounterEnabled, medusaCounterEnabled = medusaCounterEnabled,
        medusaResetEnabled = medusaResetEnabled, unwalkEnabled = unwalkEnabled,
        antiKickEnabled = antiKickEnabled, ragdollGuiEnabled = ragdollGuiEnabled,
        fovValue = fovValue, skyTheme = currentSkyTheme, introSoundEnabled = introSoundEnabled,
        mobileButtonsEnabled = mobileButtonsEnabled, mobileButtonsSize = mobileButtonsSize,
        circleButtonsEnabled = circleButtonsEnabled, guiTransparencyEnabled = guiTransparencyEnabled,
        aimbotSpeed = aimbotSpeedValue,
        stealRadius = Steal.StealRadius, stealDuration = Steal.StealDuration,
        mobilePositions = mobilePositions,
        guiOpen = guiOpen,
        stealBarPosition = stealBarPos,
    }
    pcall(function() writefile(CONFIG_FILE, HS:JSONEncode(cfg)) end)
end

local function loadAllConfig()
    if not isfile or not isfile(CONFIG_FILE) then return end
    local ok, data = pcall(function() return HS:JSONDecode(readfile(CONFIG_FILE)) end)
    if not ok or not data then return end

    if data.normalSpeed then NS = tonumber(data.normalSpeed) or 58 end
    if data.carrySpeed then CS = tonumber(data.carrySpeed) or 29 end
    if data.laggerSpeed then LAGGER_SPEED = tonumber(data.laggerSpeed) or 15 end
    if data.laggerCarrySpeed then LAGGER_CARRY_SPEED = tonumber(data.laggerCarrySpeed) or 20 end
    if data.carrySpeedActive ~= nil then carrySpeedActive = data.carrySpeedActive == true end
    if data.laggerModeEnabled ~= nil then laggerModeEnabled = data.laggerModeEnabled == true end
    if data.laggerCarryActive ~= nil then laggerCarryActive = data.laggerCarryActive == true end
    if data.uiLocked ~= nil then uiLocked = data.uiLocked == true end
    if data.autoLeftEnabled ~= nil then autoLeftEnabled = data.autoLeftEnabled == true end
    if data.autoRightEnabled ~= nil then autoRightEnabled = data.autoRightEnabled == true end
    if data.autoBatEnabled ~= nil then autoBatEnabled = data.autoBatEnabled == true end
    if data.autoSwingEnabled ~= nil then autoSwingEnabled = data.autoSwingEnabled == true end
    if data.autoStealEnabled ~= nil then Steal.AutoStealEnabled = data.autoStealEnabled == true end
    if data.antiRagdollEnabled ~= nil then antiRagdollEnabled = data.antiRagdollEnabled == true end
    if data.infJumpEnabled ~= nil then infJumpEnabled = data.infJumpEnabled == true end
    if data.infJumpMode then infJumpMode = data.infJumpMode end
    if data.autoTPEnabled ~= nil then autoTPEnabled = data.autoTPEnabled == true end
    if data.autoTPHeight then autoTPHeight = data.autoTPHeight end
    if data.antiLagEnabled ~= nil then antiLagEnabled = data.antiLagEnabled == true end
    if data.stretchRezEnabled ~= nil then stretchRezEnabled = data.stretchRezEnabled == true end
    if data.batCounterEnabled ~= nil then batCounterEnabled = data.batCounterEnabled == true end
    if data.medusaCounterEnabled ~= nil then medusaCounterEnabled = data.medusaCounterEnabled == true end
    if data.medusaResetEnabled ~= nil then medusaResetEnabled = data.medusaResetEnabled == true end
    if data.unwalkEnabled ~= nil then unwalkEnabled = data.unwalkEnabled == true end
    if data.antiKickEnabled ~= nil then antiKickEnabled = data.antiKickEnabled == true end
    if data.ragdollGuiEnabled ~= nil then ragdollGuiEnabled = data.ragdollGuiEnabled == true end
    if data.fovValue then fovValue = data.fovValue end
    if data.skyTheme then currentSkyTheme = data.skyTheme end
    if data.introSoundEnabled ~= nil then introSoundEnabled = data.introSoundEnabled == true end
    if data.mobileButtonsEnabled ~= nil then mobileButtonsEnabled = data.mobileButtonsEnabled == true end
    if data.mobileButtonsSize then mobileButtonsSize = data.mobileButtonsSize end
    if data.circleButtonsEnabled ~= nil then circleButtonsEnabled = data.circleButtonsEnabled == true end
    if data.guiTransparencyEnabled ~= nil then guiTransparencyEnabled = data.guiTransparencyEnabled == true end
    if data.aimbotSpeed then aimbotSpeedValue = data.aimbotSpeed end
    if data.stealRadius then Steal.StealRadius = data.stealRadius end
    if data.stealDuration then Steal.StealDuration = data.stealDuration end
    if data.guiOpen ~= nil then guiOpen = data.guiOpen == true end

    if data.stealBarPosition then
        savedStealBarPos = data.stealBarPosition
    end

    if data.mobilePositions then
        mobilePositions = data.mobilePositions
    end
end

loadAllConfig()

-- ============================================================
-- TOGGLE GUI
-- ============================================================
function toggleGui()
    guiOpen = not guiOpen
    if mainGui then
        mainGui.Enabled = guiOpen
    end
    saveAllConfig()
end

function buildMobileButtons()
    destroyMobileButtons()
    if not mobileButtonsEnabled then return end

    local gui = Instance.new("ScreenGui")
    gui.Name = "DivineMobileVS"
    gui.ResetOnSpawn = false
    gui.DisplayOrder = 15
    gui.Parent = player:WaitForChild("PlayerGui")

    local buttons = {
        {key = "menu", label = "MENU"},
        {key = "drop", label = "DROP\nBR"},
        {key = "autoLeft", label = "AUTO\nLEFT"},
        {key = "autoBat", label = "BAT\nAIMBOT"},
        {key = "autoRight", label = "AUTO\nRIGHT"},
        {key = "tpDown", label = "TP\nDOWN"},
        {key = "carrySpeed", label = "CARRY\nSPD"},
        {key = "lagger", label = "LAGGER\nMODE"},
        -- No TP Bat button
    }

    local size = mobileButtonsSize or 50
    local gap = 8
    local vp = workspace.CurrentCamera and workspace.CurrentCamera.ViewportSize or Vector2.new(800, 600)
    local totalW = size * 2 + gap
    local totalH = math.ceil(#buttons / 2) * (size + gap) - gap
    local startX = vp.X - totalW - 20
    local startY = vp.Y / 2 - totalH / 2

    for i, def in ipairs(buttons) do
        local col = (i - 1) % 2
        local row = math.floor((i - 1) / 2)
        local localX = col * (size + gap)
        local localY = row * (size + gap)

        local sp = mobilePositions[def.key]
        local initXO = sp and sp.xo or (startX + localX)
        local initYO = sp and sp.yo or (startY + localY)

        local frame = Instance.new("Frame", gui)
        frame.Size = UDim2.new(0, size, 0, size)
        frame.Position = UDim2.new(0, initXO, 0, initYO)
        frame.BackgroundColor3 = def.key == "menu" and Color3.fromRGB(0, 200, 0) or Color3.fromRGB(255, 215, 0)
        frame.BorderSizePixel = 0
        frame.Active = true
        frame:SetAttribute("BtnKey", def.key)

        local cornerRadius = circleButtonsEnabled and size or math.max(8, math.floor(size * 0.2))
        local uic = Instance.new("UICorner", frame)
        uic.CornerRadius = circleButtonsEnabled and UDim.new(1, 0) or UDim.new(0, cornerRadius)

        local btn = Instance.new("TextButton", frame)
        btn.Size = UDim2.new(1, 0, 1, 0)
        btn.BackgroundTransparency = 1
        btn.Text = def.label
        btn.TextColor3 = def.key == "menu" and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(180, 30, 30)
        btn.Font = Enum.Font.GothamBlack
        btn.TextSize = def.key == "menu" and math.max(12, math.floor(size * 0.25)) or math.max(8, math.floor(size * 0.18))
        btn.TextWrapped = true
        btn.AutoButtonColor = false

        local state = false
        local function setOn(v)
            state = v
            if def.key ~= "menu" then
                frame.BackgroundColor3 = v and Color3.fromRGB(255, 10, 10) or Color3.fromRGB(255, 215, 0)
                btn.TextColor3 = v and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(180, 30, 30)
            end
        end

        mobBtnRefs[def.key] = setOn

        local dragStart2, dragStartPos2, dragMoved, dragDown = nil, nil, false, false
        btn.InputBegan:Connect(function(input)
            if uiLocked then return end  -- Lock dragging when locked
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragMoved = false
                dragDown = true
                dragStart2 = input.Position
                dragStartPos2 = frame.Position
                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragDown = false
                        if dragMoved then
                            saveAllConfig()
                        end
                    end
                end)
            end
        end)

        btn.InputChanged:Connect(function(input)
            if uiLocked then return end
            if dragDown and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                local delta = input.Position - dragStart2
                if delta.Magnitude > 4 then dragMoved = true end
                if dragMoved then
                    frame.Position = UDim2.new(dragStartPos2.X.Scale, dragStartPos2.X.Offset + delta.X, dragStartPos2.Y.Scale, dragStartPos2.Y.Offset + delta.Y)
                end
            end
        end)

        btn.MouseButton1Click:Connect(function()
            -- NO uiLocked check here – buttons always work
            if dragMoved then return end

            if def.key == "menu" then
                toggleGui()
                return
            end

            if def.key == "drop" then runDrop(); return end
            if def.key == "tpDown" then runTPFloor(); return end

            if def.key == "carrySpeed" then
                carrySpeedActive = not carrySpeedActive
                if carrySpeedActive then laggerCarryActive = false end
                setOn(carrySpeedActive)
                updateSpeedDisplay()
                if mobBtnRefs.lagger then mobBtnRefs.lagger(laggerModeEnabled) end
                saveAllConfig()
                return
            end

            if def.key == "lagger" then
                laggerModeEnabled = not laggerModeEnabled
                if laggerModeEnabled then laggerCarryActive = false end
                setOn(laggerModeEnabled)
                updateSpeedDisplay()
                if mobBtnRefs.carrySpeed then mobBtnRefs.carrySpeed(carrySpeedActive) end
                saveAllConfig()
                return
            end

            if def.key == "autoLeft" then
                if autoLeftEnabled then
                    stopAutoLeft()
                    setOn(false)
                else
                    if autoRightEnabled then stopAutoRight() end
                    if autoBatEnabled then stopBatAimbot() end
                    startAutoLeft()
                    setOn(true)
                end
                saveAllConfig()
                return
            end

            if def.key == "autoRight" then
                if autoRightEnabled then
                    stopAutoRight()
                    setOn(false)
                else
                    if autoLeftEnabled then stopAutoLeft() end
                    if autoBatEnabled then stopBatAimbot() end
                    startAutoRight()
                    setOn(true)
                end
                saveAllConfig()
                return
            end

            if def.key == "autoBat" then
                if autoBatEnabled then
                    stopBatAimbot()
                    setOn(false)
                else
                    if autoLeftEnabled then stopAutoLeft() end
                    if autoRightEnabled then stopAutoRight() end
                    queueAutoBatStart()
                    setOn(true)
                end
                saveAllConfig()
                return
            end
        end)

        if def.key == "carrySpeed" and carrySpeedActive then setOn(true) end
        if def.key == "lagger" and laggerModeEnabled then setOn(true) end
        if def.key == "autoLeft" and autoLeftEnabled then setOn(true) end
        if def.key == "autoRight" and autoRightEnabled then setOn(true) end
        if def.key == "autoBat" and autoBatEnabled then setOn(true) end
    end
end

local aimbotConn = nil
local _predBall = nil

function findBat()
    local char = player.Character
    if not char then return nil end
    for _, tool in ipairs(char:GetChildren()) do
        if tool:IsA("Tool") and (tool.Name:lower():find("bat") or tool.Name:lower():find("slap")) then return tool end
    end
    local bp = player:FindFirstChild("Backpack")
    if bp then
        for _, tool in ipairs(bp:GetChildren()) do
            if tool:IsA("Tool") and (tool.Name:lower():find("bat") or tool.Name:lower():find("slap")) then return tool end
        end
    end
    return nil
end

local function getClosestTarget()
    local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    local closest, minDist = nil, math.huge
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= player and plr.Character then
            local tRoot = plr.Character:FindFirstChild("HumanoidRootPart")
            local hum = plr.Character:FindFirstChildOfClass("Humanoid")
            if tRoot and hum and hum.Health > 0 then
                local dist = (tRoot.Position - root.Position).Magnitude
                if dist < minDist then minDist = dist; closest = tRoot end
            end
        end
    end
    return closest
end

local function swingCurrentBat()
    if not autoSwingEnabled then return end
    local bat = findBat()
    if bat and bat.Parent == player.Character and bat:IsA("Tool") then
        pcall(function() bat:Activate() end)
    end
end

function startBatAimbot()
    if aimbotConn then aimbotConn:Disconnect() end
    autoBatEnabled = true
    if autoLeftEnabled then autoLeftEnabled = false; stopAutoLeft() end
    if autoRightEnabled then autoRightEnabled = false; stopAutoRight() end

    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum0 = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum0 then return end
    hum0.AutoRotate = false

    local MELEE_OFFSET = 3
    local velHistory = {}
    local MAX_HIST = 6
    local lastTargetPos = nil

    aimbotConn = RunService.RenderStepped:Connect(function()
        if not autoBatEnabled then return end
        local c = player.Character
        if not c then return end
        local hrp = c:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local hum = c:FindFirstChildOfClass("Humanoid")
        if not hum then return end
        local bat = findBat()
        if bat and bat.Parent ~= c then pcall(function() hum:EquipTool(bat) end) end

        local target = getClosestTarget()
        if not target then
            swingCurrentBat()
            if _predBall then _predBall:Destroy(); _predBall = nil end
            velHistory = {}
            lastTargetPos = nil
            return
        end

        local targetVel = target.AssemblyLinearVelocity
        local targetPos = target.Position
        if lastTargetPos and (targetPos - lastTargetPos).Magnitude > 20 then velHistory = {} end
        lastTargetPos = targetPos
        table.insert(velHistory, targetVel)
        if #velHistory > MAX_HIST then table.remove(velHistory, 1) end

        local smoothVel = Vector3.zero
        local totalW = 0
        for i, v in ipairs(velHistory) do
            local w = i / #velHistory
            smoothVel = smoothVel + v * w
            totalW = totalW + w
        end
        if totalW > 0 then smoothVel = smoothVel / totalW end
        local speed = smoothVel.Magnitude
        local predictTime = #velHistory < 2 and 0 or math.clamp(speed / 150, 0.05, 0.2)
        local predictedPos = targetPos + smoothVel * predictTime

        pcall(function()
            if not _predBall then
                _predBall = Instance.new("Part")
                _predBall.Name = "PredictionBall"
                _predBall.Shape = Enum.PartType.Ball
                _predBall.Size = Vector3.new(3, 3, 3)
                _predBall.Color = Color3.fromRGB(255, 0, 0)
                _predBall.Material = Enum.Material.Neon
                _predBall.CanCollide = false
                _predBall.CastShadow = false
                _predBall.Parent = workspace
            end
            if _predBall and _predBall.Parent then
                _predBall.CFrame = CFrame.new(predictedPos)
                _predBall.Transparency = 0.1
            end
        end)

        local toTarget = predictedPos - hrp.Position
        if toTarget.Magnitude > 0.5 then
            local goalCF = CFrame.lookAt(hrp.Position, hrp.Position + toTarget)
            local diffCF = hrp.CFrame:Inverse() * goalCF
            local rx, ry, rz = diffCF:ToEulerAnglesXYZ()
            rx = math.clamp(rx, -3, 3); ry = math.clamp(ry, -3, 3); rz = math.clamp(rz, -3, 3)
            hrp.AssemblyAngularVelocity = hrp.CFrame:VectorToWorldSpace(Vector3.new(rx * 50, ry * 50, rz * 50))
        end

        local dirToTarget = predictedPos - hrp.Position
        local targetStandPos = predictedPos
        if dirToTarget.Magnitude > 0 then targetStandPos = predictedPos - (dirToTarget.Unit * MELEE_OFFSET) end
        local moveDir = targetStandPos - hrp.Position
        if moveDir.Magnitude > 1 then
            hrp.AssemblyLinearVelocity = moveDir.Unit * aimbotSpeedValue
        else
            hrp.AssemblyLinearVelocity = smoothVel
        end
        swingCurrentBat()
    end)

    if mobBtnRefs.autoBat then mobBtnRefs.autoBat(true) end
end

function stopBatAimbot()
    if aimbotConn then aimbotConn:Disconnect(); aimbotConn = nil end
    autoBatEnabled = false
    if _predBall then _predBall:Destroy(); _predBall = nil end
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if root then
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end
    local hum2 = char and char:FindFirstChildOfClass("Humanoid")
    if hum2 then hum2.AutoRotate = true end
    if mobBtnRefs.autoBat then mobBtnRefs.autoBat(false) end
end

function queueAutoBatStart()
    if antiKickEnabled and brainrotDetected then return end
    if autoLeftEnabled then autoLeftEnabled = false; stopAutoLeft() end
    if autoRightEnabled then autoRightEnabled = false; stopAutoRight() end
    startBatAimbot()
end

function startAutoLeft()

    alPhase = 1
    autoLeftEnabled = true

    alConn = RunService.Heartbeat:Connect(function()
        if not autoLeftEnabled then return end
        local char = player.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hrp or not hum then return end
        if isRagdollState(hum) then hum:Move(Vector3.zero, false); return end

        local spd = getAutoPathSpeed()
        hum.AutoRotate = false

        if alPhase == 1 then
            local tgt = Vector3.new(AP_L1.X, hrp.Position.Y, AP_L1.Z)
            if (tgt - hrp.Position).Magnitude < 1 then
                alPhase = 2
                local d = AP_L2 - hrp.Position
                local mv = Vector3.new(d.X, 0, d.Z).Unit
                hum:Move(mv, false)
                hrp.AssemblyLinearVelocity = Vector3.new(mv.X * spd, hrp.AssemblyLinearVelocity.Y, mv.Z * spd)
                return
            end
            local d = AP_L1 - hrp.Position
            local mv = Vector3.new(d.X, 0, d.Z).Unit
            hum:Move(mv, false)
            hrp.AssemblyLinearVelocity = Vector3.new(mv.X * spd, hrp.AssemblyLinearVelocity.Y, mv.Z * spd)
        elseif alPhase == 2 then
            local tgt = Vector3.new(AP_L2.X, hrp.Position.Y, AP_L2.Z)
            if (tgt - hrp.Position).Magnitude < 1 then
                hum:Move(Vector3.zero, false)
                hrp.AssemblyLinearVelocity = Vector3.zero
                autoLeftEnabled = false
                if alConn then alConn:Disconnect(); alConn = nil end
                alPhase = 1
                hum.AutoRotate = true
                if mobBtnRefs.autoLeft then mobBtnRefs.autoLeft(false) end
                saveAllConfig()
                return
            end
            local d = AP_L2 - hrp.Position
            local mv = Vector3.new(d.X, 0, d.Z).Unit
            hum:Move(mv, false)
            hrp.AssemblyLinearVelocity = Vector3.new(mv.X * spd, hrp.AssemblyLinearVelocity.Y, mv.Z * spd)
        end
    end)
end

local function findMedusa()
    local c = player.Character
    if not c then return nil end
    for _, t in ipairs(c:GetChildren()) do
        if t:IsA("Tool") then
            local n = t.Name:lower()
            if n:find("medusa") or n:find("head") or n:find("stone") then return t end
        end
    end
    local bp = player:FindFirstChild("Backpack")
    if bp then
        for _, t in ipairs(bp:GetChildren()) do
            if t:IsA("Tool") then
                local n = t.Name:lower()
                if n:find("medusa") or n:find("head") or n:find("stone") then return t end
            end
        end
    end
    return nil
end

local function useMedusaCounter()
    if medusaDebounce then return end
    if MEDUSA_COOLDOWN > (tick() - medusaLastUsed) then return end
    local c = player.Character
    if not c then return end
    medusaDebounce = true
    local med = findMedusa()
    if not med then medusaDebounce = false; return end
    if med.Parent ~= c then
        local hum2 = c:FindFirstChildOfClass("Humanoid")
        if hum2 then hum2:EquipTool(med) end
    end
    pcall(function() med:Activate() end)
    medusaLastUsed = tick()
    medusaDebounce = false
end

local function onAnchorChanged(part)
    return part:GetPropertyChangedSignal("Anchored"):Connect(function()
        if part.Anchored and part.Transparency == 1 then
            if medusaResetEnabled then
                cursedInstaReset()
            elseif medusaCounterEnabled then
                useMedusaCounter()
            end
        end
    end)
end

function setupMedusa(char)

        if part:IsA("BasePart") then
            table.insert(Conns.anchor, onAnchorChanged(part))
        end

local BAT_COUNTER_SLAP_LIST = {"Bat", "Slap", "Iron Slap", "Gold Slap", "Diamond Slap", "Emerald Slap", "Ruby Slap", "Dark Matter Slap", "Flame Slap", "Nuclear Slap", "Galaxy Slap", "Glitched Slap"}

local function findBatForCounter()
    local c = player.Character
    if not c then return nil end
    local bp = player:FindFirstChildOfClass("Backpack")
    for _, name in ipairs(BAT_COUNTER_SLAP_LIST) do
        local t = c:FindFirstChild(name) or (bp and bp:FindFirstChild(name))
        if t then return t end
    end
    for _, ch in ipairs(c:GetChildren()) do
        if ch:IsA("Tool") and ch.Name:lower():find("bat") then return ch end
    end
    if bp then
        for _, ch in ipairs(bp:GetChildren()) do
            if ch:IsA("Tool") and ch.Name:lower():find("bat") then return ch end
        end
    end
    return nil
end

local function swingBatForCounter(bat, char)
    local hum2 = char:FindFirstChildOfClass("Humanoid")
    if bat.Parent ~= char then
        if hum2 then pcall(function() hum2:EquipTool(bat) end) end
        task.wait(0.05)
    end
    local remote = bat:FindFirstChildOfClass("RemoteEvent") or bat:FindFirstChildOfClass("RemoteFunction")
    if remote and remote:IsA("RemoteEvent") then
        pcall(function() remote:FireServer() end)
        task.wait(0.15)
        pcall(function() remote:FireServer() end)
    else
        pcall(function() bat:Activate() end)
        task.wait(0.15)
        pcall(function() bat:Activate() end)
    end
end

function startBatCounter()
    if batCounterConn then return end
    batCounterConn = RunService.Heartbeat:Connect(function()
        if not batCounterEnabled or batCounterDebounce then return end
        local char = player.Character
        if not char then return end
        local hum2 = char:FindFirstChildOfClass("Humanoid")
        if not hum2 then return end
        local st = hum2:GetState()
        if st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll or st == Enum.HumanoidStateType.FallingDown then
            batCounterDebounce = true
            task.spawn(function()
                local bat = findBatForCounter()
                if bat then swingBatForCounter(bat, char) end
                task.wait(0.5)
                batCounterDebounce = false
            end)
        end
    end)
end

function stopBatCounter()
    if batCounterConn then batCounterConn:Disconnect(); batCounterConn = nil end
    batCounterDebounce = false
end

function startUnwalk()
    local c = player.Character
    if not c then return end
    local hum = c:FindFirstChildOfClass("Humanoid")
    if hum then
        for _, t in ipairs(hum:GetPlayingAnimationTracks()) do
            t:Stop()
        end
    end
    local anim = c:FindFirstChild("Animate")
    if anim then
        unwalkSavedAnimate = anim:Clone()
        anim:Destroy()
    end
end

function stopUnwalk()
    local c = player.Character
    if c and unwalkSavedAnimate then
        unwalkSavedAnimate:Clone().Parent = c
        unwalkSavedAnimate = nil
    end
end

function enableAntiLag()
    removeAccessoriesEnabled = true
    antiLagEnabled = true
    defLightBrightness = defLightBrightness or Lighting.Brightness
    defLightClock = defLightClock or Lighting.ClockTime
    defLightAmbient = defLightAmbient or Lighting.OutdoorAmbient
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 1e10
    Lighting.Brightness = 1
    Lighting.EnvironmentDiffuseScale = 0
    Lighting.EnvironmentSpecularScale = 0
    for _, e in pairs(Lighting:GetChildren()) do
        pcall(function()
            if e:IsA("BlurEffect") or e:IsA("SunRaysEffect") or e:IsA("ColorCorrectionEffect") or e:IsA("BloomEffect") or e:IsA("DepthOfFieldEffect") then
                e.Enabled = false
            end
        end)
    end
    for _, obj in ipairs(workspace:GetDescendants()) do
        pcall(function()
            if obj:IsA("Accessory") or obj:IsA("Hat") then obj:Destroy()
            elseif obj:IsA("BasePart") then obj.Material = Enum.Material.Plastic; obj.Reflectance = 0; obj.CastShadow = false
            elseif obj:IsA("Decal") or obj:IsA("Texture") then obj.Transparency = 1
            elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then obj.Enabled = false end
        end)
    end
    if antiLagDescConn then antiLagDescConn:Disconnect() end
    antiLagDescConn = workspace.DescendantAdded:Connect(function(obj)
        if removeAccessoriesEnabled then
            pcall(function()
                if obj:IsA("Accessory") or obj:IsA("Hat") then obj:Destroy()
                elseif obj:IsA("BasePart") then obj.Material = Enum.Material.Plastic; obj.Reflectance = 0; obj.CastShadow = false
                elseif obj:IsA("Decal") or obj:IsA("Texture") then obj.Transparency = 1
                elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Beam") or obj:IsA("Fire") or obj:IsA("Smoke") or obj:IsA("Sparkles") then obj.Enabled = false end
            end)
        end
    end)
    saveAllConfig()
end

function disableAntiLag()
    removeAccessoriesEnabled = false
    antiLagEnabled = false
    if antiLagDescConn then antiLagDescConn:Disconnect(); antiLagDescConn = nil end
    pcall(function()
        if defLightBrightness then Lighting.Brightness = defLightBrightness end
        if defLightClock then Lighting.ClockTime = defLightClock end
        if defLightAmbient then Lighting.OutdoorAmbient = defLightAmbient end
        Lighting.ExposureCompensation = 0
    end)
    saveAllConfig()
end

local antiRagdollMode = nil
local ragdollConnections = {}
local cachedCharData = {}
local isBoosting = false
local BOOST_SPEED = 400
local AR_DEFAULT_SPEED = 16
local antiRagCoroutine = nil

local function arCacheCharacterData()
    local char = player.Character
    if not char then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return false end
    cachedCharData = { character = char, humanoid = hum, root = root }
    return true
end

local function arDisconnectAll()
    for _, conn in ipairs(ragdollConnections) do
        pcall(function() conn:Disconnect() end)
    end
    ragdollConnections = {}
end

local function arIsRagdolled()
    if not cachedCharData.humanoid then return false end
    local state = cachedCharData.humanoid:GetState()
    local ragdollStates = {
        [Enum.HumanoidStateType.Physics] = true,
        [Enum.HumanoidStateType.Ragdoll] = true,
        [Enum.HumanoidStateType.FallingDown] = true,
    }
    if ragdollStates[state] then return true end
    local endTime = player:GetAttribute("RagdollEndTime")
    if endTime and (endTime - workspace:GetServerTimeNow()) > 0 then return true end
    return false
end

local function arForceExitRagdoll()
    if not cachedCharData.humanoid or not cachedCharData.root then return end
    pcall(function()
        player:SetAttribute("RagdollEndTime", workspace:GetServerTimeNow())
    end)
    for _, descendant in ipairs(cachedCharData.character:GetDescendants()) do
        if descendant:IsA("BallSocketConstraint") or
           (descendant:IsA("Attachment") and descendant.Name:find("RagdollAttachment")) then
            descendant:Destroy()
        end
    end
    if not isBoosting then
        isBoosting = true
        cachedCharData.humanoid.WalkSpeed = BOOST_SPEED
    end
    if cachedCharData.humanoid.Health > 0 then
        cachedCharData.humanoid:ChangeState(Enum.HumanoidStateType.Running)
    end
    cachedCharData.root.Anchored = false
end

local function arHeartbeatLoop()
    while antiRagdollMode == "v1" do
        task.wait()
        local currentlyRagdolled = arIsRagdolled()
        if currentlyRagdolled then
            arForceExitRagdoll()
        elseif isBoosting and not currentlyRagdolled then
            isBoosting = false
            if cachedCharData.humanoid then
                cachedCharData.humanoid.WalkSpeed = AR_DEFAULT_SPEED
            end
        end
    end
end

function startAntiRagdoll()
    if antiRagdollMode == "v1" then return end
    if not arCacheCharacterData() then return end
    antiRagdollMode = "v1"
    antiRagdollEnabled = true

    local camConn = RunService.RenderStepped:Connect(function()
        local cam = workspace.CurrentCamera
        if cam and cachedCharData.humanoid then
            cam.CameraSubject = cachedCharData.humanoid
        end
    end)
    table.insert(ragdollConnections, camConn)

    local respawnConn = player.CharacterAdded:Connect(function()
        isBoosting = false
        task.wait(0.5)
        arCacheCharacterData()
    end)
    table.insert(ragdollConnections, respawnConn)

    if antiRagCoroutine then task.cancel(antiRagCoroutine) end
    antiRagCoroutine = task.spawn(arHeartbeatLoop)
end

function stopAntiRagdoll()
    antiRagdollMode = nil
    antiRagdollEnabled = false
    if isBoosting and cachedCharData.humanoid then
        cachedCharData.humanoid.WalkSpeed = AR_DEFAULT_SPEED
    end
    isBoosting = false
    arDisconnectAll()
    if antiRagCoroutine then
        task.cancel(antiRagCoroutine)
        antiRagCoroutine = nil
    end
    cachedCharData = {}
end
