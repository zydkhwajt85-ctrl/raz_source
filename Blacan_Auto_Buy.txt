--- DRAGGING
local isMobile = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled
local dragging = false
local dragStartPos = nil
local frameStartPos = nil

if isMobile then
	title.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStartPos = input.Position
			frameStartPos = borderFrame.Position
		end
	end)

	title.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.Touch then
			local delta = input.Position - dragStartPos
			borderFrame.Position = UDim2.new(
				frameStartPos.X.Scale,
				frameStartPos.X.Offset + delta.X,
				frameStartPos.Y.Scale,
				frameStartPos.Y.Offset + delta.Y
			)
		end
	end)

	title.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
else
	title.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			dragStartPos = input.Position
			frameStartPos = borderFrame.Position
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and dragStartPos and input.UserInputType == Enum.UserInputType.MouseMovement then
			local delta = input.Position - dragStartPos
			borderFrame.Position = UDim2.new(
				frameStartPos.X.Scale,
				frameStartPos.X.Offset + delta.X,
				frameStartPos.Y.Scale,
				frameStartPos.Y.Offset + delta.Y
			)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = false
		end
	end)
end

-- Función para hacer clic en la mesa de ping pong
local function clickPingPong()
	for _, obj in pairs(workspace:GetDescendants()) do
		if obj:IsA("Part") and (obj.Name:lower():find("ping") or obj.Name:lower():find("pong") or obj.Name:lower():find("table")) then
			local clickable = obj.Parent:FindFirstChild("ClickDetector")
			if clickable then
				clickable:Click()
				return true
			end
		end
	end
	return false
end

-- Auto Click loop
task.spawn(function()
	while true do
		task.wait(0.1)
		if autoClickActive then
			if not clickPingPong() then
				for _, prompt in pairs(workspace:GetDescendants()) do
					if prompt:IsA("ProximityPrompt") and (prompt.Name:lower():find("ping") or prompt.Name:lower():find("pong")) then
						prompt.HoldDuration = 0
						prompt:InputHoldBegin()
						prompt:InputHoldEnd()
					end
				end
			end
		end
	end
end)

-- FUNCIÓN PARA OBTENER DINERO REAL
local function updateMoney()
	local leaderstats = player:FindFirstChild("leaderstats")
	if leaderstats then
		local cash = leaderstats:FindFirstChild("Cash") or leaderstats:FindFirstChild("Money") or leaderstats:FindFirstChild("Brainrot") or leaderstats:FindFirstChild("Points")
		if cash then
			local amount = cash.Value
			moneyLabel.Text = "$" .. amount
		end
	end
end

updateMoney()
task.spawn(function()
	while true do
		task.wait(1)
		updateMoney()
	end
end)

-- EFECTO DE ONDA PARA EL TÍTULO (Blanco a Negro y viceversa)
task.spawn(function()
	local time = 0
	while true do
		task.wait(0.05)
		time = time + 0.05
		-- Calcular valor entre 0 y 1 usando seno para efecto de onda
		local wave = (math.sin(time * 2) + 1) / 2
		-- Interpolar entre blanco (255) y negro (0)
		local colorValue = 255 * wave
		title.TextColor3 = Color3.fromRGB(colorValue, colorValue, colorValue)
	end
end)
