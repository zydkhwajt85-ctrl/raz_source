local toggleKey = Enum.KeyCode.X

local function saveConfig()
    pcall(function()
        if writefile then
            writefile(CONFIG_FILE, HttpService:JSONEncode({
                toggleKey    = toggleKey.Name,
                activeEngine = State.activeEngine,
                v1Speed      = State.v1Speed,
                v2Speed      = State.v2Speed,
                v3Speed      = State.v3Speed,
                autoNormal   = State.autoNormal,
                lockGui      = State.lockGui,
                antiDie      = State.antiDie,
                guiPos       = State.guiPos,
            }))
        end
    end)
end

local function loadConfig()
    if not (isfile and isfile(CONFIG_FILE)) then return end
    local ok, cfg = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FILE)) end)
    if not ok or type(cfg) ~= "table" then return end
    if cfg.toggleKey and Enum.KeyCode[cfg.toggleKey] then toggleKey = Enum.KeyCode[cfg.toggleKey] end
    if cfg.activeEngine == "V1" or cfg.activeEngine == "V2" or cfg.activeEngine == "V3" then State.activeEngine = cfg.activeEngine end
    if cfg.v1Speed then State.v1Speed = math.max(0, tonumber(cfg.v1Speed) or 25) end
    if cfg.v2Speed then State.v2Speed = math.max(0, tonumber(cfg.v2Speed) or 25) end
    if cfg.v3Speed then State.v3Speed = math.max(0, tonumber(cfg.v3Speed) or 25) end
    if cfg.autoNormal ~= nil then State.autoNormal = cfg.autoNormal end
    if cfg.lockGui    ~= nil then State.lockGui    = cfg.lockGui end
    if cfg.antiDie    ~= nil then State.antiDie    = cfg.antiDie end
    if cfg.guiPos and type(cfg.guiPos) == "table" then State.guiPos = cfg.guiPos end
end
loadConfig()

local Refs = { char = nil, root = nil, hum = nil }
local function refreshRefs()
    local c = LP.Character
    if not c then Refs.char, Refs.root, Refs.hum = nil, nil, nil; return false end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    if not r or not h then Refs.char, Refs.root, Refs.hum = nil, nil, nil; return false end
    Refs.char, Refs.root, Refs.hum = c, r, h
    return true
end
LP.CharacterAdded:Connect(function()
    Refs.char, Refs.root, Refs.hum = nil, nil, nil
    task.wait(0.5); refreshRefs()
end)
refreshRefs()

local function attachPlatform(char)
    if not char then return end
    if char:FindFirstChild("synapseSpinPlatform") then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local plat = Instance.new("Part")
    plat.Name = "synapseSpinPlatform"
    plat.Size = Vector3.new(4, 0.5, 4)
    plat.Transparency = 1
    plat.CanCollide = true
    plat.Anchored = false
    plat.Massless = true
    plat.CanQuery = false
    plat.CanTouch = false
    plat.Parent = char
    plat.CFrame = root.CFrame * CFrame.new(0, -3, 0)
    local weld = Instance.new("WeldConstraint")
    weld.Part0 = root
    weld.Part1 = plat
    weld.Parent = plat
end

local function detachPlatform(char)
    if not char then return end
    local plat = char:FindFirstChild("synapseSpinPlatform")
    if plat then plat:Destroy() end
end

local AntiVoid = { lastSafePos = nil, conn = nil }
do
    local lastCheck = 0
    AntiVoid.conn = RunService.Heartbeat:Connect(function()
        local char = LP.Character
        if not char then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum  = char:FindFirstChildOfClass("Humanoid")
        if not root or not hum then return end
        local now = os.clock()
        if now - lastCheck > 0.25 then
            lastCheck = now
            if root.Position.Y > -25 and hum.FloorMaterial ~= Enum.Material.Air then
                AntiVoid.lastSafePos = root.Position + Vector3.new(0, 5, 0)
            end
        end
        if root.Position.Y < -50 then
            if AntiVoid.lastSafePos then
                root.CFrame = CFrame.new(AntiVoid.lastSafePos)
                root.AssemblyLinearVelocity = Vector3.zero
            else
                local spawn = workspace:FindFirstChildOfClass("SpawnLocation")
                if spawn then
                    root.CFrame = spawn.CFrame + Vector3.new(0, 5, 0)
                    root.AssemblyLinearVelocity = Vector3.zero
                end
            end
        end
    end)
end

local Spin1 = { enabled = false, conn = nil, angle = 0 }
local function startSpin1()
    if Spin1.conn then return end
    Spin1.angle = 0
    refreshRefs()
    local hum = Refs.hum
    if hum then pcall(function() hum.AutoRotate = false end) end
    Spin1.conn = RunService.Heartbeat:Connect(function(dt)
        if not Spin1.enabled then return end
        local root = Refs.root
        if not root or not root.Parent then
            if not refreshRefs() then return end
            root = Refs.root
            if not root then return end
            local h = Refs.hum
            if h then pcall(function() h.AutoRotate = false end) end
        end
        Spin1.angle = (Spin1.angle + State.v1Speed * dt * 60) % 360
        local pos = root.CFrame.Position
        local vel = root.AssemblyLinearVelocity
        root.CFrame = CFrame.new(pos) * CFrame.Angles(0, math.rad(Spin1.angle), 0)
        root.AssemblyLinearVelocity = vel
        root.AssemblyAngularVelocity = Vector3.zero
    end)
end
local function stopSpin1()
    if Spin1.conn then Spin1.conn:Disconnect(); Spin1.conn = nil end
    local hum = Refs.hum
    if hum then pcall(function() hum.AutoRotate = true end) end
    local root = Refs.root
    if root then pcall(function() root.AssemblyAngularVelocity = Vector3.zero end) end
    detachPlatform(LP.Character)
end
local function setV1(on)
    Spin1.enabled = on
    if on then startSpin1() else stopSpin1() end
end

local Spin2 = { enabled = false, conn = nil, yAngle = 0, xAngle = 0 }
local function startSpin2()
    if Spin2.conn then return end
    Spin2.yAngle = 0; Spin2.xAngle = 0
    attachPlatform(LP.Character)
    Spin2.conn = RunService.Heartbeat:Connect(function(dt)
        if not Spin2.enabled then return end
        local char = LP.Character
        if not char then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        attachPlatform(char)
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then pcall(function() hum.AutoRotate = false end) end
        Spin2.yAngle = (Spin2.yAngle + State.v2Speed * dt * 60) % 360
        Spin2.xAngle = (Spin2.xAngle + State.v2Speed * dt * 60) % 360
        local pos = root.CFrame.Position
        local vel = root.AssemblyLinearVelocity
        root.CFrame = CFrame.new(pos)
            * CFrame.Angles(0, math.rad(Spin2.yAngle), 0)
            * CFrame.Angles(math.rad(Spin2.xAngle), 0, 0)
        root.AssemblyLinearVelocity = vel
        root.AssemblyAngularVelocity = Vector3.zero
    end)
end

        local root = char:FindFirstChild("HumanoidRootPart")
        if root then pcall(function() root.AssemblyAngularVelocity = Vector3.zero end) end
        detachPlatform(char)
    end
end
local function setV2(on)

local function stopAllEngines()
    if Spin1.enabled then setV1(false) end
    if Spin2.enabled then setV2(false) end
    if Spin3.enabled then setV3(false) end
end

local function setEngine(on)
    State.on = on
    -- Stop every version first, then enable only the selected version.
    stopAllEngines()
    if on then
        if State.activeEngine == "V1" then
            setV1(true)
        elseif State.activeEngine == "V2" then
            setV2(true)
        elseif State.activeEngine == "V3" then
            setV3(true)
        end
    else
        detachPlatform(LP.Character)
    end
    if updateUI then updateUI() end
end

-- Auto Enable (same pattern as before, just no hitbox companion)
RunService.Heartbeat:Connect(function()
    if not State.autoNormal then return end
    local hum
    if State.activeEngine == "V1" then
        hum = Refs.hum
        if not hum or not hum.Parent then refreshRefs(); hum = Refs.hum end
    else
        local char = LP.Character
        if char then hum = char:FindFirstChildOfClass("Humanoid") end
    end
    if not hum then return end
    local hasBrainrot = hum.WalkSpeed < 25
    if hasBrainrot and not State.on then setEngine(true)
    elseif (not hasBrainrot) and State.on then setEngine(false) end
end)

local AntiDie = { conns = {}, hum = nil, char = nil, started = false }
local function onAntiDieHealthChanged(newHealth)
    if not State.antiDie then return end
    local hum = AntiDie.hum
    if not hum or not hum.Parent then return end
    if newHealth <= 0 then hum.Health = hum.MaxHealth end
end
local function bindAntiDieToCharacter(char)
    if not char then return end
    AntiDie.char = char
    local hum = char:WaitForChild("Humanoid", 5)
    if not hum then return end
    AntiDie.hum = hum
    table.insert(AntiDie.conns, hum.HealthChanged:Connect(onAntiDieHealthChanged))
end
local function startAntiDie()
    if AntiDie.started then return end
    AntiDie.started = true
    bindAntiDieToCharacter(LP.Character)
    table.insert(AntiDie.conns, LP.CharacterAdded:Connect(function(newChar) bindAntiDieToCharacter(newChar) end))
end
local function stopAntiDie()
    for _, c in ipairs(AntiDie.conns) do pcall(function() c:Disconnect() end) end
    AntiDie.conns = {}
    AntiDie.started = false
    AntiDie.hum = nil
    AntiDie.char = nil
end
local function setAntiDie(on)
    State.antiDie = on
    if on then startAntiDie() else stopAntiDie() end
end

-- UI
function updateUI()
    if State.on then
        activateBtn.Text = "ON (" .. State.activeEngine .. ")"
        activateBtn.TextColor3 = COLORS.toggleOn
    else
        activateBtn.Text = "ACTIVATE"
        activateBtn.TextColor3 = COLORS.text
    end
    v1Select.BackgroundColor3 = State.activeEngine == "V1" and COLORS.toggleOn or COLORS.toggleOff
    v2Select.BackgroundColor3 = State.activeEngine == "V2" and COLORS.toggleOn or COLORS.toggleOff
    v3Select.BackgroundColor3 = State.activeEngine == "V3" and COLORS.toggleOn or COLORS.toggleOff
    autoSettBtn.BackgroundColor3 = State.autoNormal and COLORS.toggleOn or COLORS.toggleOff
    autoSettBtn.Text = State.autoNormal and "ON" or "OFF"
    lockBtn.Text = State.lockGui and "🔒" or "🔓"
    minimizeBtn.Visible = not isMinimized
    maximizeBtn.Visible = isMinimized
    activateBtn.Visible = true
    antiDieBtn.BackgroundColor3 = State.antiDie and COLORS.toggleOn or COLORS.toggleOff
    antiDieBtn.Text = State.antiDie and "ON" or "OFF"
    keybindBtnBody.Text = "Keybind: " .. toggleKey.Name
end

local function currentSpeed()
    if State.activeEngine == "V1" then return State.v1Speed
    elseif State.activeEngine == "V2" then return State.v2Speed
    else return State.v3Speed end
end

local function setCurrentSpeed(n)
    if State.activeEngine == "V1" then State.v1Speed = n
    elseif State.activeEngine == "V2" then State.v2Speed = n
    else State.v3Speed = n end
end

-- Keybind listen
local listeningForKey = false
local listenConn = nil
local function beginKeybindListen(displayBtn)
    if listeningForKey then
        listeningForKey = false
        if listenConn then listenConn:Disconnect(); listenConn = nil end
        displayBtn.Text = "Keybind: " .. toggleKey.Name
        return
    end
    listeningForKey = true
    displayBtn.Text = "..."
    listenConn = UserInputService.InputBegan:Connect(function(input, gp)
        if gp or not listeningForKey then return end
        local ut = input.UserInputType
        if ut ~= Enum.UserInputType.Keyboard
           and ut ~= Enum.UserInputType.Gamepad1
           and ut ~= Enum.UserInputType.Gamepad2 then
            return
        end
        local newKey = input.KeyCode
        if newKey == Enum.KeyCode.Unknown then return end
        if newKey == Enum.KeyCode.Escape then
            listeningForKey = false
            displayBtn.Text = "Keybind: " .. toggleKey.Name
            if listenConn then listenConn:Disconnect(); listenConn = nil end
            return
        end
        toggleKey = newKey
        displayBtn.Text = "Keybind: " .. newKey.Name
        listeningForKey = false
        if listenConn then listenConn:Disconnect(); listenConn = nil end
        saveConfig()
    end)
end

keybindBtnBody.MouseButton1Click:Connect(function()
    beginKeybindListen(keybindBtnBody)
end)

-- Global keybind
UserInputService.InputBegan:Connect(function(input, gp)
    if gp or listeningForKey then return end
    local ut = input.UserInputType
    if ut ~= Enum.UserInputType.Keyboard
       and ut ~= Enum.UserInputType.Gamepad1
       and ut ~= Enum.UserInputType.Gamepad2 then
        return
    end
    if input.KeyCode == toggleKey then
        setEngine(not State.on)
        saveConfig()
    end
end)

-- Drag
local isDragging = false
local dragInput, mousePos, framePos
mainFrame.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) and not State.lockGui then
        isDragging = true
        mousePos = input.Position
        framePos = mainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                isDragging = false
                local pos = mainFrame.Position
                State.guiPos = {XScale = pos.X.Scale, XOffset = pos.X.Offset, YScale = pos.Y.Scale, YOffset = pos.Y.Offset}
                saveConfig()
            end
        end)
    end
end)
mainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and isDragging then
        local delta = input.Position - mousePos
        mainFrame.Position = UDim2.new(framePos.X.Scale, framePos.X.Offset + delta.X, framePos.Y.Scale, framePos.Y.Offset + delta.Y)
    end
end)

updateUI()
