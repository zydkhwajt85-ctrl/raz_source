local function persistConfigJSON(json)
	if not json then return end
	_G[G_KEY] = json
	pcall(function()
		if writefile then
			writefile(configFileName, json)
			pcall(function() writefile(configFileName .. ".bak", json) end)
			pcall(function() writefile(configLegacy, json) end)
		end
	end)
end

local function readConfigJSON()
	local out = nil
	pcall(function()
		if readfile and isfile then
			if isfile(configFileName) then out = readfile(configFileName)
			elseif isfile(configFileName .. ".bak") then out = readfile(configFileName .. ".bak")
			elseif isfile(configLegacy) then out = readfile(configLegacy)
			end
		end
	end)
	if (not out or out == "") and type(_G[G_KEY]) == "string" then
		out = _G[G_KEY]
	end
	return out
end

local function saveConfig()
	local data = buildConfigData()
	local okE, json = pcall(function() return HttpService:JSONEncode(data) end)
	if okE and json then
		persistConfigJSON(json)
	end

	if not RT.suppressHistory then
		if okE and json then
			for i = #RT.configHistory, RT.configHistoryIx + 1, -1 do
				RT.configHistory[i] = nil
			end
			if RT.configHistory[RT.configHistoryIx] ~= json then
				table.insert(RT.configHistory, json)
				RT.configHistoryIx = #RT.configHistory
				if #RT.configHistory > 40 then
					table.remove(RT.configHistory, 1)
					RT.configHistoryIx = #RT.configHistory
				end
			end
		end
	end
end

local function applyPurpleGradient(text)
	if not text or text == "" then return text end
	local len = #text
	if len == 1 then return '<font color="rgb(200,100,200)">'..text..'</font>' end
	local result = ""
	for i = 1, len do
		local char = text:sub(i, i)
		local pos = (i - 1) / (len - 1)
		local distance_from_center = math.abs(pos - 0.5) / 0.5
		local intensity = 0.55 + (distance_from_center * 0.45)
		local r = math.floor(255 * intensity)
		result = result .. '<font color="rgb('..r..',0,'..r..')">'..char..'</font>'
	end
	return result
end

local PURPLE_PALETTES = {
	{ Color3.fromRGB(118,  36, 205), Color3.fromRGB(255, 140, 255), Color3.fromRGB(118,  36, 205) },
	{ Color3.fromRGB( 74,  16, 158), Color3.fromRGB(205,  92, 255), Color3.fromRGB( 74,  16, 158) },
	{ Color3.fromRGB(163,  64, 255), Color3.fromRGB(255, 205, 255), Color3.fromRGB(163,  64, 255) },
	{ Color3.fromRGB( 96,  24, 180), Color3.fromRGB(236, 120, 255), Color3.fromRGB( 96,  24, 180) },
}

local function paletteSequence(a, b, t)
	return ColorSequence.new({
		ColorSequenceKeypoint.new(0,   a[1]:Lerp(b[1], t)),
		ColorSequenceKeypoint.new(0.5, a[2]:Lerp(b[2], t)),
		ColorSequenceKeypoint.new(1,   a[3]:Lerp(b[3], t)),
	})
end

local function animatedPurple(parent, shimmerTime, cycleTime)
	shimmerTime = shimmerTime or 2.4
	cycleTime   = cycleTime   or 2.0

	local grad = Instance.new("UIGradient", parent)
	grad.Color = paletteSequence(PURPLE_PALETTES[1], PURPLE_PALETTES[1], 0)

	task.spawn(function()
		while grad.Parent do
			grad.Offset = Vector2.new(-1, 0)
			local tw = TweenService:Create(grad, TweenInfo.new(shimmerTime, Enum.EasingStyle.Linear), { Offset = Vector2.new(1, 0) })
			tw:Play()
			tw.Completed:Wait()
		end
	end)

	task.spawn(function()
		local i = 1
		while grad.Parent do
			local a = PURPLE_PALETTES[i]
			local b = PURPLE_PALETTES[(i % #PURPLE_PALETTES) + 1]
			local t0 = os.clock()
			while grad.Parent do
				local t = (os.clock() - t0) / cycleTime
				if t >= 1 then break end
				grad.Color = paletteSequence(a, b, t)
				task.wait(0.03)
			end
			i = (i % #PURPLE_PALETTES) + 1
		end
	end)

	return grad
end

local function reapplyConfigRuntime()
	pcall(function() if applyLoadedConfigStates then applyLoadedConfigStates() end end)
	pcall(function() if applyMobileStyleSafe then applyMobileStyleSafe() end end)
	pcall(function() if applyUIScale then applyUIScale() end end)

local function applySnapshot(json)
	if not json then return end
	local ok, data = pcall(function() return HttpService:JSONDecode(json) end)
	if not ok or not data then return end
	RT.suppressHistory = true
	applyConfigData(data)
	persistConfigJSON(json)
	RT.suppressHistory = false
	reapplyConfigRuntime()
end

local function undoConfig()
	if RT.configHistoryIx > 1 then
		RT.configHistoryIx = RT.configHistoryIx - 1
		applySnapshot(RT.configHistory[RT.configHistoryIx])
		return true
	end
	return false
end

local function redoConfig()
	if RT.configHistoryIx < #RT.configHistory then
		RT.configHistoryIx = RT.configHistoryIx + 1
		applySnapshot(RT.configHistory[RT.configHistoryIx])
		return true
	end
	return false
end

local function resetAllConfig()
	if not defaultConfigJSON then return end
	RT.suppressHistory = true
	if stopEverythingRef then pcall(stopEverythingRef) end
	applyConfigData(HttpService:JSONDecode(defaultConfigJSON))
	persistConfigJSON(defaultConfigJSON)
	RT.suppressHistory = false
	for i = #RT.configHistory, RT.configHistoryIx + 1, -1 do RT.configHistory[i] = nil end
	table.insert(RT.configHistory, defaultConfigJSON)
	RT.configHistoryIx = #RT.configHistory
	reapplyConfigRuntime()
end

pcall(function()
	RT.configHistory  = { HttpService:JSONEncode(buildConfigData()) }
	RT.configHistoryIx = 1
end)

-- ─── DRAGGABLE ───
local function makeDraggable(frame, getLocked, handle)
	local grab = handle or frame
	local dragging, dragStart, startPos = false, nil, nil

	grab.InputBegan:Connect(function(inp)
		if getLocked and getLocked() then return end
		if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = inp.Position
			startPos = frame.Position
			inp.Changed:Connect(function()
				if inp.UserInputState == Enum.UserInputState.End then dragging = false end
			end)
		end
	end)
	UIS.InputChanged:Connect(function(inp)
		if not dragging then return end
		if getLocked and getLocked() then dragging = false; return end
		if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then
			if dragStart and startPos then
				local d = inp.Position - dragStart
				frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset+d.X, startPos.Y.Scale, startPos.Y.Offset+d.Y)
			end
		end
	end)
	UIS.InputEnded:Connect(function(inp)
		if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end

		RT.uiPositions[key] = { pos.X.Scale, pos.X.Offset, pos.Y.Scale, pos.Y.Offset }
		saveConfigSoon()
	end)
end

mainUIScale = Instance.new("UIScale", RT.main)

local BASE_W, BASE_H = W, H

local function getViewport()
	local cam = workspace.CurrentCamera
	if not cam then return nil end
	local vp = cam.ViewportSize
	if vp.X < 1 or vp.Y < 1 then return nil end
	return vp
end

local function getFitScale()
	if not RT.autoFitEnabled then return 1 end
	local vp = getViewport()
	if not vp then return 1 end
	local fit = math.min((vp.X * 0.92) / BASE_W, (vp.Y * 0.86) / BASE_H, 1)
	return math.clamp(fit, 0.4, 1)
end

local function clampToScreen(frame, w, h)
	local vp = getViewport()
	if not vp then return end
	w = w or frame.AbsoluteSize.X
	h = h or frame.AbsoluteSize.Y
	local px = math.clamp(frame.Position.X.Offset, 0, math.max(vp.X - w, 0))
	local py = math.clamp(frame.Position.Y.Offset, 0, math.max(vp.Y - h, 0))
	frame.Position = UDim2.new(0, px, 0, py)
end

applyUIScale = function()
	local s = (uiScaleValue / 100) * getFitScale()
	mainUIScale.Scale = s
	if miniUIScale then miniUIScale.Scale = math.clamp(s, 0.6, 1.5) end
	if mobileUIScale then mobileUIScale.Scale = math.clamp(s * RT.mobileBtnScale, 0.5, 2.0) end
	task.defer(function()
		pcall(clampToScreen, RT.main, BASE_W * s, BASE_H * s)
	end)
end

applyUIScale()

do
	local function hookCamera()
		local cam = workspace.CurrentCamera
		if cam then
			cam:GetPropertyChangedSignal("ViewportSize"):Connect(applyUIScale)
		end
	end
	hookCamera()
	workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
		hookCamera()
		applyUIScale()
	end)
end

-- ─── CONTROLLO KEYBIND DOPPIE ───
-- field = "kb" (tastiera) o "gp" (pad). Restituisce il nome del comando
-- che usa gia quel tasto, oppure nil.
local function findBindConflict(myEntry, field, key)
	if not key then return nil end
	for name, entry in pairs(KB) do
		if entry ~= myEntry and entry[field] == key then
			return name
		end
	end
	return nil
end

-- ─── COMPONENTE KEYBIND: tastiera + pad + tasto per toglierlo ───
local KBROW_W = 108   -- larghezza totale del gruppo

local function makeKB(parent, kbEntry)
	local box = Instance.new("Frame", parent)
	box.Size                   = UDim2.new(0, KBROW_W, 0, 20)
	box.BackgroundTransparency = 1
	box.BorderSizePixel        = 0
	box.ZIndex                 = 11

	-- tasto TASTIERA
	local bKb = Instance.new("TextButton", box)
	bKb.Size             = UDim2.new(0, 44, 1, 0)
	bKb.Position         = UDim2.new(0, 0, 0, 0)
	bKb.BackgroundColor3 = KB_BG
	bKb.BorderSizePixel  = 0
	bKb.TextColor3       = WHITE
	bKb.Font             = Enum.Font.GothamBold
	bKb.TextSize         = 8
	bKb.ZIndex           = 12
	Instance.new("UICorner", bKb).CornerRadius = UDim.new(0, 5)
	local sKb = Instance.new("UIStroke", bKb); sKb.Color = BORDER2; sKb.Thickness = 1

	-- tasto PAD
	local bPad = Instance.new("TextButton", box)
	bPad.Size             = UDim2.new(0, 38, 1, 0)
	bPad.Position         = UDim2.new(0, 46, 0, 0)
	bPad.BackgroundColor3 = KB_BG
	bPad.BorderSizePixel  = 0
	bPad.TextColor3       = WHITE
	bPad.Font             = Enum.Font.GothamBold
	bPad.TextSize         = 8
	bPad.ZIndex           = 12
	Instance.new("UICorner", bPad).CornerRadius = UDim.new(0, 5)
	local sPad = Instance.new("UIStroke", bPad); sPad.Color = BORDER2; sPad.Thickness = 1

	-- tasto NO KEYBIND (toglie sia tastiera che pad)
	local bNone = Instance.new("TextButton", box)
	bNone.Size             = UDim2.new(0, 20, 1, 0)
	bNone.Position         = UDim2.new(0, 88, 0, 0)
	bNone.BackgroundColor3 = Color3.fromRGB(60, 15, 15)
	bNone.BorderSizePixel  = 0
	bNone.Text             = "X"
	bNone.TextColor3       = Color3.fromRGB(255, 170, 170)
	bNone.Font             = Enum.Font.GothamBold
	bNone.TextSize         = 9
	bNone.ZIndex           = 12
	Instance.new("UICorner", bNone).CornerRadius = UDim.new(0, 5)
	local sNone = Instance.new("UIStroke", bNone); sNone.Color = Color3.fromRGB(150, 60, 60); sNone.Thickness = 1

	if type(kbEntry) ~= "table" then
		bKb.Text = tostring(kbEntry)
		bPad.Text = "-"
		return box
	end

	local function redraw()
		bKb.Text  = kbEntry.kb and kbEntry.kb.Name or "None"
		bPad.Text = kbEntry.gp and padShortName(kbEntry.gp) or "-"
		-- col pad collegato il bottoncino si illumina
		if RT.padActive then
			bPad.BackgroundColor3 = kbEntry.gp and Color3.fromRGB(90, 30, 130) or KB_BG
			sPad.Color = Color3.fromRGB(180, 100, 240)
		else
			bPad.BackgroundColor3 = KB_BG
			sPad.Color = BORDER2
		end
	end
	table.insert(RT.padRefreshers, redraw)
	redraw()

	-- ascolta il prossimo tasto premuto: mode = "kb" o "pad"
	local listening = false
	local function listen(mode, btn)
		if listening then return end
		listening = true
		btn.Text = "..."
		btn.TextColor3 = Color3.fromRGB(255, 100, 100)
		local conn

		-- assegna davvero il tasto e salva
		local function commit(field, key)
			kbEntry[field] = key
			if field == "gp" and key then setPadActive(true) end
			redraw()
			refreshPadVisuals()   -- ridisegna anche le altre righe
			saveConfig()          -- salvataggio automatico, anche da pad
		end

		local function finish()
			btn.TextColor3 = WHITE
			if conn then conn:Disconnect() end
			listening = false
			redraw()
		end

		conn = UIS.InputBegan:Connect(function(input)
			local ut = input.UserInputType
			local key = input.KeyCode
			local field

			if mode == "kb" and ut == Enum.UserInputType.Keyboard then
				if key == Enum.KeyCode.Unknown or key == Enum.KeyCode.Escape then
					finish()
					commit("kb", nil)
					return
				end
				field = "kb"
			elseif mode == "pad" and isGamepadInput(ut) then
				if key == Enum.KeyCode.Unknown then return end
				field = "gp"
			else
				return
			end

			finish()

			-- stesso tasto gia usato da un altro comando: chiedo conferma
			local clash = findBindConflict(kbEntry, field, key)
			if clash then
				local what = (field == "gp") and padShortName(key) or key.Name
				showConfirm(
					"Il tasto " .. what .. " e gia usato da \"" .. clash .. "\".\n" ..
					"Se continui, quel comando resta senza questo tasto.",
					function()
						if KB[clash] then KB[clash][field] = nil end
						commit(field, key)
					end,
					function() redraw() end
				)
				return
			end

			commit(field, key)
		end)
	end

	bKb.Activated:Connect(function()  listen("kb",  bKb)  end)
	bPad.Activated:Connect(function() listen("pad", bPad) end)
	bNone.Activated:Connect(function()
		kbEntry.kb = nil
		kbEntry.gp = nil
		redraw()
		saveConfig()
	end)

	return box
end

-- ─── PERFORMANCE ───
local function applyAntiLag(instance)
	if instance:IsA("ParticleEmitter") then
		instance.Enabled = false
	elseif instance:IsA("Decal") then
		instance.Transparency = 1
	elseif instance:IsA("BasePart") then
		instance.Material = Enum.Material.Plastic
		instance.Reflectance = 0
		instance.CastShadow = false
	end
end

local function optimizeLighting()
	local Lighting = game:GetService("Lighting")
	Lighting.GlobalShadows = false
	Lighting.FogEnd = 9e9
	Lighting.Brightness = 1
	Lighting.EnvironmentDiffuseScale = 0
	Lighting.EnvironmentSpecularScale = 0
	for _, child in ipairs(Lighting:GetChildren()) do
		if child:IsA("BloomEffect") or child:IsA("BlurEffect") or child:IsA("SunRaysEffect") then
			child.Enabled = false
		end
	end
end

local function enableAntiLag()
	pcall(optimizeLighting)
	for _, desc in ipairs(workspace:GetDescendants()) do
		pcall(applyAntiLag, desc)
		if desc:IsA("Accessory") then pcall(function() desc:Destroy() end) end
	end
	if RT._antiLagConn then RT._antiLagConn:Disconnect() end
	RT._antiLagConn = workspace.DescendantAdded:Connect(function(desc)
		pcall(applyAntiLag, desc)
		if desc:IsA("Accessory") then pcall(function() desc:Destroy() end) end
	end)
end

local function disableAntiLag()
	if RT._antiLagConn then RT._antiLagConn:Disconnect(); RT._antiLagConn = nil end
end

local function makePlastic(obj)
	if obj:IsA("BasePart") then
		pcall(function()
			obj.Material = Enum.Material.Plastic
			obj.TopSurface = Enum.SurfaceType.Smooth
			obj.BottomSurface = Enum.SurfaceType.Smooth
			obj.FrontSurface = Enum.SurfaceType.Smooth
			obj.BackSurface = Enum.SurfaceType.Smooth
			obj.LeftSurface = Enum.SurfaceType.Smooth
			obj.RightSurface = Enum.SurfaceType.Smooth
			obj.Reflectance = 0
		end)
	end
end

local function enableAllPlastic()
	for _, desc in ipairs(workspace:GetDescendants()) do
		makePlastic(desc)
	end
	if RT._plasticConn then RT._plasticConn:Disconnect() end
	RT._plasticConn = workspace.DescendantAdded:Connect(function(desc)
		task.wait(0.01)
		if State.allPlasticEnabled then makePlastic(desc) end
	end)
end

local function disableAllPlastic()
	if RT._plasticConn then RT._plasticConn:Disconnect(); RT._plasticConn = nil end
end

local function isCarryingBrainrot(char)
	if not char then return false end
	local ok, stealing = pcall(function() return LP:GetAttribute("Stealing") end)
	if ok and stealing == true then return true end

	local ok2, anticarry = pcall(function() return LP:GetAttribute("AntiKick") end)

		local v = char:FindFirstChild(name, true)
		if v then
			if v:IsA("BoolValue") and v.Value then return true end
			if v:IsA("ObjectValue") and v.Value then return true end

		end
	end

	for _, child in ipairs(char:GetChildren()) do
		if child:IsA("Model") and child:FindFirstChildWhichIsA("BasePart", true) then
			local n = child.Name:lower()
			if n:find("brainrot") or n:find("animal") or n:find("carry") or
			   n:find("grab") or n:find("steal") or n:find("hold") then
				return true
			end
		elseif child:IsA("Tool") then
			local n = child.Name:lower()
			if not (n:find("bat") or n:find("slap") or n:find("medusa") or n:find("head") or n:find("stone") or n:find("carpet")) then
				return true
			end
		end
	end
	return false
end

-- ─── RILEVAMENTO BRAINROT DA GRABBARE ───
-- Distanza reale dal bordo della parte, non dal suo centro: cosi 0.9 studs
-- significa davvero "ci sono attaccato".
local function partSurfaceDistance(part, pos)
	local ok, d = pcall(function()
		local rel = part.CFrame:PointToObjectSpace(pos)
		local half = part.Size * 0.5
		local clamped = Vector3.new(
			math.clamp(rel.X, -half.X, half.X),
			math.clamp(rel.Y, -half.Y, half.Y),
			math.clamp(rel.Z, -half.Z, half.Z)
		)
		return (rel - clamped).Magnitude
	end)
	if ok and d then return d end
	return (part.Position - pos).Magnitude
end

-- Un podio conta solo se ha davvero il prompt "Steal" attivo
local function podiumHasSteal(sp)
	local att = sp:FindFirstChild("PromptAttachment")
	if att then
		for _, pr in ipairs(att:GetChildren()) do
			if pr:IsA("ProximityPrompt") and pr.Enabled ~= false and pr.ActionText and pr.ActionText:find("Steal") then
				return true
			end
		end
	end
	for _, pr in ipairs(sp:GetDescendants()) do
		if pr:IsA("ProximityPrompt") and pr.Enabled ~= false and pr.ActionText and pr.ActionText:find("Steal") then
			return true
		end
	end
	return false
end

-- Scansione completa della mappa: podi rubabili nei plot altrui + brainrot
-- lasciati a terra nel workspace. Restituisce distanza e posizione del piu vicino.
local function scanNearestBrainrot()
	local char = LP.Character; if not char then return nil end
	local root = char:FindFirstChild("HumanoidRootPart")
		or char:FindFirstChild("UpperTorso")
		or char:FindFirstChild("Torso")
	if not root then return nil end
	local myPos = root.Position

	local minDist, minPos = math.huge, nil

	local plots = workspace:FindFirstChild("Plots")
	if plots then
		for _, plot in ipairs(plots:GetChildren()) do
			if not isMyPlotByName(plot.Name) then
				local pods = plot:FindFirstChild("AnimalPodiums")
				if pods then
					for _, pod in ipairs(pods:GetChildren()) do
						local base  = pod:FindFirstChild("Base")
						local spawn = base and base:FindFirstChild("Spawn")
						if spawn and spawn:IsA("BasePart") and podiumHasSteal(spawn) then
							-- se sul podio c'e il modello del brainrot, misuro su quello
							local ref = spawn
							for _, m in ipairs(pod:GetDescendants()) do
								if m:IsA("BasePart") and m ~= spawn and m.Transparency < 1 then
									local n = m.Name:lower()
									if n:find("brainrot") or n:find("animal") or (m.Parent and m.Parent:IsA("Model") and m.Parent ~= pod) then
										ref = m
										break
									end
								end
							end
							local d = partSurfaceDistance(ref, myPos)
							if d < minDist then minDist, minPos = d, ref.Position end
						end
					end
				end
			end
		end
	end

	for _, child in ipairs(workspace:GetChildren()) do
		if child:IsA("Model") or child:IsA("Tool") or child:IsA("BasePart") then
			local n = child.Name:lower()
			if n:find("brainrot") or n:find("animal") then
				local pt = child:IsA("BasePart") and child or child:FindFirstChildWhichIsA("BasePart", true)
				if pt then
					local d = partSurfaceDistance(pt, myPos)
					if d < minDist then minDist, minPos = d, pt.Position end
				end
			end
		end
	end

	if minDist < math.huge then return minDist, minPos end
	return nil
end

-- La scansione e pesante e viene interrogata a ogni frame: risultato in cache
RT.brainrotCacheAt   = 0
RT.brainrotCacheDist = nil
RT.brainrotCachePos  = nil

local function getNearestBrainrotDist()
	local now = os.clock()
	if now - RT.brainrotCacheAt < 0.08 then
		return RT.brainrotCacheDist, RT.brainrotCachePos
	end
	RT.brainrotCacheAt = now
	local d, pos = scanNearestBrainrot()
	RT.brainrotCacheDist, RT.brainrotCachePos = d, pos
	return d, pos
end

local function isNearOrCarryingBrainrot(char)
	if not char then return false end
	if isCarryingBrainrot(char) then return true end
	local dist = getNearestBrainrotDist()
	return (dist ~= nil and dist <= RT.carryProximity)
end

local function isAutoCarryActive()
	if not State.autoCarryEnabled then return false end
	local char = LP.Character
	if not char then return false end
	return isNearOrCarryingBrainrot(char)
end

local function AceUseMedusaCounter()

	if RT.medDebounce then return end

	local c = LP.Character
	if not c then return end
	RT.medDebounce = true

		local hum = c:FindFirstChildOfClass("Humanoid")

	RT.medDebounce = false

local INF_CLAMP_FALL  = 80
local JUMP_COOLDOWN   = 0.08
local JUMP_PULSE_TIME = 0.07

RT.spaceHeld      = false
RT.lastJumpTime   = 0
RT.jumpPulseUntil = 0

RT.jumpLVAtt, RT.jumpLV = nil, nil

local function destroyJumpLV()
	if RT.jumpLV and RT.jumpLV.Parent then pcall(function() RT.jumpLV:Destroy() end) end
	if RT.jumpLVAtt and RT.jumpLVAtt.Parent then pcall(function() RT.jumpLVAtt:Destroy() end) end
	RT.jumpLV, RT.jumpLVAtt = nil, nil
end

local function setupJumpLV(hrp)
	destroyJumpLV()

	RT.jumpLVAtt = Instance.new("Attachment")
	RT.jumpLVAtt.Name   = "EdofaxJumpAtt"
	RT.jumpLVAtt.Parent = hrp

	RT.jumpLV = Instance.new("LinearVelocity")
	RT.jumpLV.Name        = "EdofaxJumpLV"
	RT.jumpLV.Attachment0 = RT.jumpLVAtt

	RT.jumpLV.VelocityConstraintMode = Enum.VelocityConstraintMode.Line
	RT.jumpLV.LineDirection          = Vector3.new(0, 1, 0)
	RT.jumpLV.LineVelocity           = 0
	RT.jumpLV.MaxForce               = 0
	RT.jumpLV.RelativeTo             = Enum.ActuatorRelativeTo.World
	RT.jumpLV.Parent                 = hrp
end

local function ensureJumpLV(hrp)
	if not RT.jumpLV or RT.jumpLV.Parent ~= hrp then setupJumpLV(hrp) end
	return RT.jumpLV
end

local function infJumpForce()
	return State.jumpBoostEnabled and RT.jumpPowerValue or 50
end

local function pulseJump(hrp, force)
	local lv = ensureJumpLV(hrp)
	if not lv then return end
	lv.LineVelocity = force
	lv.MaxForce     = math.huge
	RT.jumpPulseUntil  = os.clock() + JUMP_PULSE_TIME
end

table.insert(RT.infJumpInputConns, UIS.InputBegan:Connect(function(input, gpe)
	if gpe then return end
	if input.KeyCode == Enum.KeyCode.Space then RT.spaceHeld = true end
end))

table.insert(RT.infJumpInputConns, UIS.InputEnded:Connect(function(input)
	if input.KeyCode == Enum.KeyCode.Space then RT.spaceHeld = false end
end))

RT.infJumpConn = RunService.Heartbeat:Connect(function()
	if not State.infJumpEnabled and not State.jumpBoostEnabled then
		if RT.jumpLV then destroyJumpLV() end
		return
	end

	local char = LP.Character
	if not char then
		if RT.jumpLV then destroyJumpLV() end
		return
	end
	local hrp = char:FindFirstChild("HumanoidRootPart"); if not hrp then return end
	local lv = ensureJumpLV(hrp); if not lv then return end

	if State.infJumpEnabled and RT.spaceHeld then
		local now = tick()
		if now - RT.lastJumpTime >= JUMP_COOLDOWN then
			RT.lastJumpTime = now
			pulseJump(hrp, infJumpForce())
			return
		end
	end

	if os.clock() < RT.jumpPulseUntil then return end

	if State.infJumpEnabled and hrp.AssemblyLinearVelocity.Y < -INF_CLAMP_FALL then
		lv.LineVelocity = -INF_CLAMP_FALL
		lv.MaxForce     = math.huge
	else
		lv.MaxForce = 0
	end
end)

-- ─── DROP ───
RT.dropConn = nil
local function runDrop()
	if RT.dropActive then return end
	local char = LP.Character; if not char then return end
	local root = char:FindFirstChild("HumanoidRootPart"); if not root then return end

	RT.dropActive = true
	if RT.dropConn then pcall(function() RT.dropConn:Disconnect() end); RT.dropConn = nil end

	-- TP Down automatico poco dopo il drop (opzione del hub)
	if RT.dropTPDown then
		task.delay(RT.dropTPDownDelay, function()
			pcall(doAutoTPDown, true)
		end)
	end

	local function finishDrop()
		if RT.dropConn then pcall(function() RT.dropConn:Disconnect() end); RT.dropConn = nil end
		RT.dropActive = false
		local c = LP.Character
		if not c then return end
		local r = c:FindFirstChild("HumanoidRootPart")
		local hum = c:FindFirstChildOfClass("Humanoid")
		if r then
			r.AssemblyLinearVelocity = Vector3.zero
			r.AssemblyAngularVelocity = Vector3.zero
			local rp = RaycastParams.new()
			rp.FilterDescendantsInstances = {c}
			rp.FilterType = Enum.RaycastFilterType.Exclude
			local rr = workspace:Raycast(r.Position, Vector3.new(0, -3000, 0), rp)
			if rr then
				local off = ((hum and hum.HipHeight) or 2) + (r.Size.Y / 2)
				r.CFrame = CFrame.new(r.Position.X, rr.Position.Y + off, r.Position.Z)
				r.AssemblyLinearVelocity = Vector3.zero
			end
			if hum and hum.Health > 0 then hum:ChangeState(Enum.HumanoidStateType.Running) end
		end
	end

	local t0 = tick()
	RT.dropConn = RunService.Heartbeat:Connect(function()
		local c = LP.Character
		local r = c and c:FindFirstChild("HumanoidRootPart")
		if not r or not RT.dropActive then
			if RT.dropConn then pcall(function() RT.dropConn:Disconnect() end); RT.dropConn = nil end
			RT.dropActive = false
			return
		end
		if tick() - t0 >= 0.22 then
			finishDrop()
			return
		end
		local lv = r.AssemblyLinearVelocity
		r.AssemblyLinearVelocity = Vector3.new(lv.X, 160, lv.Z)
	end)

	task.delay(1.5, function()
		if RT.dropActive then finishDrop() end
	end)
end

-- ─── INSTANT RESET ───
local RESET_MAX_DURATION = 0.05
local resetCooldown, resetThread = false, nil
local resetSuccessful, stopResetSequence = false, false

local function StopResetSequence()
	stopResetSequence = true
	if resetThread then pcall(function() task.cancel(resetThread) end); resetThread = nil end
	resetCooldown = false
	local character = LP.Character
	if character then
		local humanoid = character:FindFirstChildOfClass("Humanoid")
		if humanoid then
			pcall(function()
				humanoid.HipHeight = 2
				local rootPart = character:FindFirstChild("HumanoidRootPart")
				if rootPart then rootPart.CanCollide = true end
				for _, part in ipairs(character:GetChildren()) do
					if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then part.CanCollide = true end
				end
			end)
		end
	end
end

local function doJWInstaReset()
	if resetCooldown then return end
	resetCooldown = true
	resetSuccessful = false
	stopResetSequence = false

	local character = LP.Character
	if not character then resetCooldown = false; return end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then resetCooldown = false; return end

	local isRespawning = false
	resetThread = task.spawn(function()
		local attempts = 0
		local maxAttempts = 40
		local originalHipHeight = humanoid.HipHeight

		while character and character.Parent and humanoid and humanoid.Health > 0
			and not isRespawning and not stopResetSequence do
			if LP.Character ~= character then isRespawning = true; break end

			pcall(function()
				humanoid.HipHeight = 1e30
				humanoid.AutoRotate = true
				local rootPart = character:FindFirstChild("HumanoidRootPart")
				if rootPart then rootPart.CanCollide = false end
				for _, part in ipairs(character:GetChildren()) do
					if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then part.CanCollide = false end
				end
			end)

			if not character or not character.Parent or not humanoid
				or humanoid.Health <= 0 or LP.Character ~= character then
				resetSuccessful = true
				break
			end

			attempts = attempts + 1
			if attempts >= maxAttempts then break end
			task.wait(RESET_MAX_DURATION)
		end

		if not resetSuccessful then
			if character and character.Parent and humanoid and humanoid.Health > 0 and not isRespawning then
				pcall(function() humanoid.Health = 0 end)
				task.wait(0.1)
				if not character.Parent or humanoid.Health <= 0 then resetSuccessful = true end
			end
		end

		if not resetSuccessful and character and character.Parent and humanoid then
			pcall(function()
				humanoid.HipHeight = originalHipHeight
				local rootPart = character:FindFirstChild("HumanoidRootPart")
				if rootPart then rootPart.CanCollide = true end
				for _, part in ipairs(character:GetChildren()) do
					if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then part.CanCollide = true end
				end
			end)
		end

		resetCooldown = false
		resetThread = nil
		stopResetSequence = false
	end)
end

LP.CharacterAdded:Connect(function()
	StopResetSequence()
	resetCooldown = false
	resetSuccessful = false
	stopResetSequence = false
end)

local function AceStartAutoLeft()
	AceStopAutoLeft()
	AceStopAutoRight()
	RT.leftPhase = 1
	RT.leftConn = RunService.Heartbeat:Connect(function()
		if not State.autoLeftEnabled then return end
		local char = LP.Character; if not char then return end
		local hrp = char:FindFirstChild("HumanoidRootPart")
		local hum = char:FindFirstChildOfClass("Humanoid")
		if not hrp or not hum then return end
		local st = hum:GetState()
		if hum.PlatformStand or st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll or st == Enum.HumanoidStateType.FallingDown then
			hum:Move(Vector3.zero, false)
			return
		end
		local P = AceAutoPathPoints
		local okS, spd = pcall(getCurrentSpeed)
		if not okS or type(spd) ~= "number" then spd = NS end

		local function step(target)
			local d = target - hrp.Position
			local flat = Vector3.new(d.X, 0, d.Z)
			if flat.Magnitude < 0.001 then return end
			local mv = flat.Unit
			hum:Move(mv, false)
			hrp.AssemblyLinearVelocity = Vector3.new(mv.X * spd, hrp.AssemblyLinearVelocity.Y, mv.Z * spd)
		end
		local function reached(target)
			local tgt = Vector3.new(target.X, hrp.Position.Y, target.Z)
			return (tgt - hrp.Position).Magnitude < 1
		end

		if RT.leftPhase == 1 then
			if reached(P.L1) then RT.leftPhase = 2; return end
			step(P.L1)

		elseif RT.leftPhase == 2 then
			if reached(P.L2) then
				hum:Move(Vector3.zero, false)
				hrp.AssemblyLinearVelocity = Vector3.zero
				RT.leftPhase = 0
				task.delay(0.2, function()
					if State.autoLeftEnabled and RT.leftPhase == 0 then RT.leftPhase = 3 end
				end)
				return
			end
			step(P.L2)

		elseif RT.leftPhase == 0 then
			return

		elseif RT.leftPhase == 3 then
			if reached(P.L1) then RT.leftPhase = 4; return end
			step(P.L1)

		elseif RT.leftPhase == 4 then
			if reached(P.LFinal) then
				hum:Move(Vector3.zero, false)
				hrp.AssemblyLinearVelocity = Vector3.zero
				State.autoLeftEnabled = false
				if RT.leftConn then RT.leftConn:Disconnect(); RT.leftConn = nil end
				RT.leftPhase = 1
				if RT.autoPlayToggle then RT.autoPlayToggle(false) end
				setMobileButtonVisual("AutoPlay", false)
				saveConfig()
				return
			end
			step(P.LFinal)
		end
	end)
end

local function autoPlayIsLocalPlot(plot)
	local sign = plot:FindFirstChild("PlotSign")
	if sign then
		local surface = sign:FindFirstChildWhichIsA("SurfaceGui", true)
		local label = surface and surface:FindFirstChildWhichIsA("TextLabel", true)
		if label and string.find(string.lower(label.Text), string.lower(LP.Name)) then
			return true
		end
	end

end

local function detectStealSide()
	local plots = workspace:FindFirstChild("Plots")
	if not plots then return nil end
	local opponentPlot
	for _, plot in ipairs(plots:GetChildren()) do
		if not autoPlayIsLocalPlot(plot) then opponentPlot = plot; break end
	end
	if not opponentPlot then return nil end
	local part = opponentPlot:FindFirstChild("PlotSign")
		or opponentPlot.PrimaryPart
		or opponentPlot:FindFirstChildWhichIsA("BasePart", true)
	if not part then return nil end
	local posX = part.Position.X
	if posX > -476 then return "Left" else return "Right" end
end

-- ─── RILEVAMENTO AUTOMATICO DEL LATO (dove sta il brainrot da grabbare) ───
-- I due percorsi si distinguono sulla Z: LEFT ~ 93, RIGHT ~ 25.
-- Punto di taglio a meta strada fra i due.
local SIDE_SPLIT_Z = (AceAutoPathPoints.L1.Z + AceAutoPathPoints.R1.Z) / 2

local function sideFromPosition(pos)
	if not pos then return nil end
	if pos.Z > SIDE_SPLIT_Z then return "Left" else return "Right" end
end

-- Cerca il brainrot rubabile piu vicino nei plot non miei e restituisce
-- il lato in cui si trova (+ la distanza). nil se non trova nulla.
local function detectBrainrotSide()
	local char = LP.Character; if not char then return nil end
	local root = char:FindFirstChild("HumanoidRootPart")
		or char:FindFirstChild("UpperTorso")
		or char:FindFirstChild("Torso")
	if not root then return nil end

	local plots = workspace:FindFirstChild("Plots"); if not plots then return nil end
	local bestPos, bestDist = nil, math.huge

	for _, plot in ipairs(plots:GetChildren()) do
		if not isMyPlotByName(plot.Name) then
			local pods = plot:FindFirstChild("AnimalPodiums")
			if pods then
				for _, pod in ipairs(pods:GetChildren()) do
					local base = pod:FindFirstChild("Base")
					local sp   = base and base:FindFirstChild("Spawn")
					if sp and sp:IsA("BasePart") then
						local hasSteal = false
						local att = sp:FindFirstChild("PromptAttachment")
						if att then
							for _, pr in ipairs(att:GetChildren()) do
								if pr:IsA("ProximityPrompt") and pr.ActionText and pr.ActionText:find("Steal") then
									hasSteal = true
									break
								end
							end
						end
						if not hasSteal then
							for _, pr in ipairs(sp:GetDescendants()) do
								if pr:IsA("ProximityPrompt") and pr.ActionText and pr.ActionText:find("Steal") then
									hasSteal = true
									break
								end
							end
						end
						if hasSteal then
							local d = (sp.Position - root.Position).Magnitude
							if d < bestDist then bestDist, bestPos = d, sp.Position end
						end
					end
				end
			end
		end
	end

	if bestPos then
		return sideFromPosition(bestPos), bestDist, bestPos
	end
	return nil
end

-- Decide il lato da usare: se l'auto e attivo prova prima il brainrot,
-- poi il plot avversario, altrimenti resta sulla scelta manuale.
resolveAutoSide = function()
	if RT.autoSideAuto then
		local ok, side = pcall(detectBrainrotSide)
		if ok and side then
			if side ~= RT.autoPlaySide then
				RT.autoPlaySide = side
				for _, fn in ipairs(autoPlaySideRefreshers) do pcall(fn) end
			end
			return side
		end
		local ok2, side2 = pcall(detectStealSide)
		if ok2 and side2 then
			if side2 ~= RT.autoPlaySide then
				RT.autoPlaySide = side2
				for _, fn in ipairs(autoPlaySideRefreshers) do pcall(fn) end
			end
			return side2
		end
	end
	return RT.autoPlaySide
end

-- ─── AUTO PLAY IN ATTESA (parte appena l'anti-kick ti lascia libero) ───
-- Se clicchi Auto Play mentre hai il brainrot in mano, invece di annullare
-- l'attivazione la mette in coda: appena sei al sicuro parte subito.
-- Vale SOLO per l'Auto Play, le altre funzioni restano bloccate come prima.
RT.autoPlayPending = false
RT.autoPlayPendingThread = nil

local function startAutoPlayNow()
	local side = resolveAutoSide()
	if side == "Right" then
		State.autoRightEnabled = true
		State.autoLeftEnabled  = false
		pcall(AceStopAutoLeft)
		pcall(AceStartAutoRight)

local TP_PATHS = {

local TP_REACH_DIST   = 4      -- quanto vicino deve stare per passare al waypoint dopo
local TP_STEP_TIMEOUT = 2.5    -- se si blocca su uno step, passa avanti
local TP_TOTAL_TIMEOUT = 12    -- limite totale del percorso

local function runEmergencyTP()
	if RT.tpConn then return end

	local char = LP.Character; if not char then return end
	local hrp = char:FindFirstChild("HumanoidRootPart"); if not hrp then return end

	local side = resolveAutoSide and resolveAutoSide() or RT.autoPlaySide
	local path = TP_PATHS[(side == "Right") and "Right" or "Left"]
	if not path or #path == 0 then return end

	local P = AceAutoPathPoints
	local face = (side == "Right") and P.RFace or P.LFace

	RT.tpPhase = 1                 -- indice del waypoint corrente
	local stepStart = os.clock()

	setMobileButtonVisual("TP", true)
	setupTPLV(hrp)

	RT.tpConn = RunService.Heartbeat:Connect(function()
		local c = LP.Character; if not c then stopEmergencyTP(); return end
		local h = c:FindFirstChild("HumanoidRootPart")
		if not h then return end
		if not RT.tpLV or RT.tpLV.Parent ~= h then setupTPLV(h) end

		local target = path[RT.tpPhase]
		if not target then stopEmergencyTP(); return end

		-- distanza sul piano: il movimento e orizzontale, la Y la lascia alla gravita
		local flatTgt = Vector3.new(target.X, h.Position.Y, target.Z)
		local dist = (flatTgt - h.Position).Magnitude

		-- waypoint raggiunto (o step bloccato troppo a lungo): si passa al prossimo
		if dist < TP_REACH_DIST or (os.clock() - stepStart) > TP_STEP_TIMEOUT then
			if RT.tpPhase >= #path then
				if RT.tpLV then RT.tpLV.PlaneVelocity = Vector2.zero end
				h.AssemblyLinearVelocity = Vector3.new(0, h.AssemblyLinearVelocity.Y, 0)
				if face then
					pcall(function()
						h.CFrame = CFrame.new(h.Position, Vector3.new(face.X, h.Position.Y, face.Z))
					end)
				end
				stopEmergencyTP()
				return
			end
			RT.tpPhase = RT.tpPhase + 1
			stepStart  = os.clock()
			return
		end

		local d = target - h.Position
		local mv = Vector3.new(d.X, 0, d.Z)
		if mv.Magnitude > 0 then mv = mv.Unit end
		RT.tpLV.PlaneVelocity = Vector2.new(mv.X * TP_SPEED, mv.Z * TP_SPEED)
	end)

	task.delay(TP_TOTAL_TIMEOUT, function()
		if RT.tpConn then stopEmergencyTP() end
	end)
end

-- ─── BAT AIMBOT ───
local function AceFindAimbotBat()
	local char = LP.Character
	if not char then return nil end
	for _, tool in ipairs(char:GetChildren()) do
		if tool:IsA("Tool") and (tool.Name:lower():find("bat") or tool.Name:lower():find("slap")) then
			return tool
		end
	end
	local bp = LP:FindFirstChild("Backpack")
	if bp then
		for _, tool in ipairs(bp:GetChildren()) do
			if tool:IsA("Tool") and (tool.Name:lower():find("bat") or tool.Name:lower():find("slap")) then
				return tool
			end
		end
	end
	return nil
end

local function AceGetClosestAimbotTarget()
	local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	if not root then return nil end
	local closest, minDist = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP and plr.Character then
			local tRoot = plr.Character:FindFirstChild("HumanoidRootPart")
			local hum = plr.Character:FindFirstChildOfClass("Humanoid")
			if tRoot and hum and hum.Health > 0 then
				local dist = (tRoot.Position - root.Position).Magnitude
				if dist < minDist then
					minDist = dist
					closest = tRoot
				end
			end
		end
	end
	return closest
end

-- Parametri dell'inseguimento TP Down del Bat Aimbot
local BATLOCK_DROP_MIN     = 12    -- caduta minima (studs) per considerarlo un TP Down
local BATLOCK_MIN_GAP      = 8     -- quanto devo restare sopra perche convenga scendere
local BATLOCK_TP_COOLDOWN  = 0.6   -- pausa fra un inseguimento e l'altro

RT.batLockTrackRoot = nil
RT.batLockLastY     = nil
RT.batLockTPAt      = 0

local function AceStartNormalAimbot()
	if RT.aimbotConn then RT.aimbotConn:Disconnect(); RT.aimbotConn = nil end
	local hum0 = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
	if hum0 then hum0.AutoRotate = false end

	RT.aimbotConn = RunService.RenderStepped:Connect(function()
		if not State.autoBatEnabled then return end
		local char = LP.Character
		if not char then return end
		local root = char:FindFirstChild("HumanoidRootPart")
		local hum = char:FindFirstChildOfClass("Humanoid")
		if not root or not hum then return end

		local bat = char:FindFirstChildOfClass("Tool") or AceFindAimbotBat()
		if bat and bat.Parent ~= char then
			pcall(function() hum:EquipTool(bat) end)
		end

		local target = AceGetClosestAimbotTarget()
		if not target then
			RT.batLockTrackRoot = nil
			return
		end

		-- ── INSEGUIMENTO TP DOWN ──────────────────────────────────────────
		-- Se il bersaglio sparisce di colpo verso il basso (ha usato il TP Down)
		-- e io sono rimasto in alto, scendo anch'io con un TP Down.
		-- Solo in discesa: se sale, non faccio nulla.
		if State.batFollowTPDown then
			local ty = target.Position.Y
			if RT.batLockTrackRoot ~= target then
				RT.batLockTrackRoot = target
				RT.batLockLastY     = ty
			else
				local dy = ty - (RT.batLockLastY or ty)
				RT.batLockLastY = ty
				if dy <= -BATLOCK_DROP_MIN then
					local gap = root.Position.Y - ty
					if gap >= BATLOCK_MIN_GAP and (os.clock() - RT.batLockTPAt) > BATLOCK_TP_COOLDOWN then
						RT.batLockTPAt = os.clock()
						pcall(doAutoTPDown, true)
						return
					end
				end
			end
		end

		local targetVel = target.AssemblyLinearVelocity
		local myPos = root.Position
		local targetPos = target.Position
		local predictPos = targetPos + targetVel * 0.14 + target.CFrame.LookVector * 0.3
		local direction = predictPos - myPos
		if direction.Magnitude < 0.01 then return end
		local flatDir = Vector3.new(direction.X, 0, direction.Z)
		if flatDir.Magnitude < 0.01 then return end
		flatDir = flatDir.Unit

		local chaseSpeed = 58
		local desiredHeight = targetPos.Y + 3.7
		local yVel = (desiredHeight - myPos.Y) * 19.5 + targetVel.Y * 0.8
		if hum.FloorMaterial ~= Enum.Material.Air then
			yVel = math.max(yVel, 13)
		end
		yVel = math.clamp(yVel, -70, 110)

		local desiredVel = Vector3.new(flatDir.X * chaseSpeed, yVel, flatDir.Z * chaseSpeed)
		root.AssemblyLinearVelocity = root.AssemblyLinearVelocity:Lerp(desiredVel, 0.8)

		local speed3 = targetVel.Magnitude
		local predictTime = math.clamp(speed3 / 150, 0.05, 0.2)
		local predictedPos = targetPos + targetVel * predictTime
		local toPredict = predictedPos - myPos
		if toPredict.Magnitude > 0.1 then
			local goalCF = CFrame.lookAt(myPos, predictedPos)
			local diffCF = root.CFrame:Inverse() * goalCF
			local rx, ry, rz = diffCF:ToEulerAnglesXYZ()
			rx = math.clamp(rx, -2.5, 2.5)
			ry = math.clamp(ry, -2.5, 2.5)
			rz = math.clamp(rz, -2.5, 2.5)
			root.AssemblyAngularVelocity = root.CFrame:VectorToWorldSpace(Vector3.new(rx * 42, ry * 42, rz * 42))
		end

		if bat and not RT.aimbotSwingCooldown then
			RT.aimbotSwingCooldown = true
			pcall(function() bat:Activate() end)
			task.delay(0.08, function()
				RT.aimbotSwingCooldown = false
			end)
		end
	end)
end

local function startTPBat()
	RT.tpBatEnabled = true
	if RT.tpBatConn then pcall(function() RT.tpBatConn:Disconnect() end) end
	setMobileButtonVisual("TPBat", true)
	if syncAutoTPDown then syncAutoTPDown() end

		if not RT.tpBatEnabled then return end
		local char = LP.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if not hrp then return end

		if hum and hum.AutoRotate then pcall(function() hum.AutoRotate = false end) end

		local target = tpBatClosestPlayer()
		if not target or not target.Character then return end
		local tr = target.Character:FindFirstChild("HumanoidRootPart")
		if not tr then return end

		if sethiddenproperty then
			pcall(function() sethiddenproperty(hrp, "PhysicsRepRootPart", tr) end)
		end

		local targetVel = tr.AssemblyLinearVelocity
		local aimPos    = tr.Position + Vector3.new(0, RT.tpBatHeight, 0) + targetVel * 0.06

		if RT.tpBatUltra then
			-- ULTRA: piu teletrasporti nello stesso frame, girando attorno al
			-- bersaglio, con un colpo a ogni riposizionamento.
			local n = math.clamp(math.floor(RT.tpBatUltraCount), 1, 12)
			local ring = RT.tpBatUltraRing
			for i = 1, n do
				local a = (os.clock() * 14 + (i / n) * math.pi * 2) % (math.pi * 2)
				local off = Vector3.new(math.cos(a) * ring, 0, math.sin(a) * ring)
				hrp.CFrame = CFrame.new(aimPos + off, tr.Position)
				hrp.AssemblyLinearVelocity = Vector3.zero
				tpBatHitNow()
			end
		else
			if (hrp.Position - aimPos).Magnitude > 5 then
				hrp.CFrame = CFrame.new(aimPos)
			end
		end

		local myPos = hrp.Position
		local speed3 = targetVel.Magnitude
		local predictTime = math.clamp(speed3 / 150, 0.05, 0.2)
		local predictedPos = tr.Position + targetVel * predictTime
		local toPredict = predictedPos - myPos
		if toPredict.Magnitude > 0.1 then
			local goalCF = CFrame.lookAt(myPos, predictedPos)
			local diffCF = hrp.CFrame:Inverse() * goalCF
			local rx, ry, rz = diffCF:ToEulerAnglesXYZ()
			rx = math.clamp(rx, -2.5, 2.5)
			ry = math.clamp(ry, -2.5, 2.5)
			rz = math.clamp(rz, -2.5, 2.5)
			hrp.AssemblyAngularVelocity = hrp.CFrame:VectorToWorldSpace(Vector3.new(rx * 42, ry * 42, rz * 42))
		end

		pcall(function()
			local cam = workspace.CurrentCamera
			cam.CFrame = CFrame.lookAt(cam.CFrame.Position, predictedPos)
		end)

		if not RT.tpBatUltra then
			tpBatTryHit()
		end
	end)
end

-- ─── AUTO STEAL ───
local function findNearestPrompt()
	local char = LP.Character; if not char then return nil end
	local root = char:FindFirstChild("HumanoidRootPart")
		or char:FindFirstChild("UpperTorso")
		or char:FindFirstChild("Torso")
	if not root then return nil end

	local plots = workspace:FindFirstChild("Plots"); if not plots then return nil end
	local nearest, dist = nil, math.huge

	for _, plot in ipairs(plots:GetChildren()) do
		if plot:IsA("Model") and not isMyPlotByName(plot.Name) then
			local pods = plot:FindFirstChild("AnimalPodiums")
			if pods then
				for _, pod in ipairs(pods:GetChildren()) do
					local base = pod:FindFirstChild("Base")
					local sp   = base and base:FindFirstChild("Spawn")
					if sp then
						local d = (sp.Position - root.Position).Magnitude
						if d <= Steal.StealRadius and d < dist then
							local found = nil
							local att = sp:FindFirstChild("PromptAttachment")
							if att then
								for _, pr in ipairs(att:GetChildren()) do
									if pr:IsA("ProximityPrompt") and pr.ActionText and pr.ActionText:find("Steal") then
										found = pr
									end
								end
							end
							if not found then
								for _, pr in ipairs(sp:GetDescendants()) do
									if pr:IsA("ProximityPrompt") and pr.ActionText and pr.ActionText:find("Steal") then
										found = pr
									end
								end
							end
							if found then nearest, dist = found, d end
						end
					end
				end
			end
		end
	end

	return nearest
end

local function promptDistance(prompt)
	local char = LP.Character; if not char then return math.huge end
	local root = char:FindFirstChild("HumanoidRootPart")
		or char:FindFirstChild("UpperTorso")
		or char:FindFirstChild("Torso")
	if not root then return math.huge end

	local part = prompt.Parent
	if part and part:IsA("Attachment") then part = part.Parent end
	if part and part:IsA("BasePart") then
		return (part.Position - root.Position).Magnitude
	end

	local ok, wp = pcall(function() return prompt.Parent and prompt.Parent.WorldPosition end)
	if ok and wp then return (wp - root.Position).Magnitude end
	return math.huge
end

local function executeSteal(prompt)
	if RT.isStealing then return end

	if not Steal.Data[prompt] then
		Steal.Data[prompt] = { hold = {}, trigger = {}, ready = true }
		if getconnections then
			pcall(function()
				for _, c in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do
					if c.Function then table.insert(Steal.Data[prompt].hold, c.Function) end
				end
				for _, c in ipairs(getconnections(prompt.Triggered)) do
					if c.Function then table.insert(Steal.Data[prompt].trigger, c.Function) end
				end
			end)
		end
	end

	local data = Steal.Data[prompt]
	if not data.ready then return end

	data.ready     = false
	RT.isStealing     = true
	RT.stealStartTime = tick()

	if Steal.Mode == "half" then
		task.spawn(function()
			for _, fn in ipairs(data.hold) do task.spawn(fn) end
			task.wait(Steal.HalfHoldMin)

			local inRange = promptDistance(prompt) <= Steal.HalfFireRange
			while true do
				local elapsed = tick() - RT.stealStartTime
				if elapsed > Steal.HalfHoldMax or not prompt.Parent then break end
				if promptDistance(prompt) <= Steal.HalfFireRange then
					if not inRange then task.wait(Steal.HalfEntryDelay) end
					for _, fn in ipairs(data.trigger) do task.spawn(fn) end
					break
				end
				task.wait()
			end

			task.wait(0.05)
			data.ready = true
			RT.isStealing = false
		end)
	else
		task.spawn(function()
			for _, fn in ipairs(data.hold) do task.spawn(fn) end
			if #data.hold == 0 then
				pcall(function() prompt:InputHoldBegin() end)
			end

			local elapsed = 0
			while elapsed < Steal.StealDuration do elapsed = elapsed + task.wait() end

			for _, fn in ipairs(data.trigger) do task.spawn(fn) end
			if #data.trigger == 0 then
				pcall(function() prompt:InputHoldEnd() end)
				if fireproximityprompt then pcall(fireproximityprompt, prompt) end
			end

			task.wait(0.05)
			data.ready = true
			RT.isStealing = false
		end)
	end
end

local function startAutoSteal()
	if RT.Conns.autoSteal then return end
	RT.Conns.autoSteal = RunService.Heartbeat:Connect(function()
		if not State.autoStealEnabled or RT.isStealing then return end
		local p = findNearestPrompt()
		if p then executeSteal(p) end
	end)
end

			if type(dur) ~= "number" or dur <= 0 then dur = 1 end
			pct = math.clamp((tick() - RT.stealStartTime) / dur, 0, 1)
		elseif State.autoStealEnabled then
			local now = os.clock()
			if now - _lastScan > 0.12 then
				_lastScan = now
				_cachedHit = findNearestPrompt() ~= nil
			end
			pct = _cachedHit and 1 or 0

-- ─── COPIA / INCOLLA CONFIG (condivisibile con altri) ───
local function clipboardWrite(txt)
	local fn = setclipboard or toclipboard or (Clipboard and Clipboard.set) or writeclipboard
	if fn then
		local ok = pcall(fn, txt)
		if ok then return true end
	end
	return false
end

local function clipboardRead()
	local fn = getclipboard or (Clipboard and Clipboard.get) or readclipboard
	if fn then
		local ok, v = pcall(fn)
		if ok and type(v) == "string" and v ~= "" then return v end
	end
	return nil
end

-- carica una config presa da fuori (stringa JSON)
local function importConfigJSON(txt)
	if type(txt) ~= "string" then return false, "niente da caricare" end
	txt = txt:gsub("^%s+", ""):gsub("%s+$", "")
	if txt == "" then return false, "niente da caricare" end
	local ok, data = pcall(function() return HttpService:JSONDecode(txt) end)
	if not ok or type(data) ~= "table" then return false, "config non valida" end
	applySnapshot(txt)
	-- registra il caricamento come passo di cronologia (cosi puoi tornare indietro)
	for i = #RT.configHistory, RT.configHistoryIx + 1, -1 do RT.configHistory[i] = nil end
	table.insert(RT.configHistory, txt)
	RT.configHistoryIx = #RT.configHistory
	return true
end

RT.mainRenderStepped = RunService.RenderStepped:Connect(function()
	local char = LP.Character
	if not char then
		if RT.linearVelocity then destroyLV() end
		return
	end
	local hrp = char:FindFirstChild("HumanoidRootPart")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum then return end

	local state = hum:GetState()
	if hum.PlatformStand or state == Enum.HumanoidStateType.Physics or
	   state == Enum.HumanoidStateType.Ragdoll or state == Enum.HumanoidStateType.FallingDown then
		if RT.linearVelocity and RT.linearVelocity.Parent then
			RT.linearVelocity.PlaneVelocity = Vector2.zero
		end
		return
	end

	local drivenByOther = State.autoBatEnabled or State.autoLeftEnabled or State.autoRightEnabled or RT.tpBatEnabled

	if State.speedBoostEnabled and not drivenByOther then
		if not RT.linearVelocity or RT.linearVelocity.Parent ~= hrp then
			setupLV(hrp)
		end

		local md = hum.MoveDirection
		if md.Magnitude > 0.1 then
			local ok, speed = pcall(getCurrentSpeed)
			if ok and speed then
				local flat = Vector3.new(md.X, 0, md.Z).Unit
				RT.linearVelocity.PlaneVelocity = Vector2.new(flat.X * speed, flat.Z * speed)
			end
		else
			RT.linearVelocity.PlaneVelocity = Vector2.zero
		end
	elseif RT.linearVelocity then
		destroyLV()
	end

	if State.jumpBoostEnabled and not State.infJumpEnabled then
		local pressing = UIS:IsKeyDown(Enum.KeyCode.Space) or hum.Jump
		if pressing and hum.FloorMaterial ~= Enum.Material.Air then
			if hrp.AssemblyLinearVelocity.Y < RT.jumpPowerValue then
				pulseJump(hrp, RT.jumpPowerValue)
			end
		end
	end

	if RT.speedLbl and RT.speedLbl.Parent then
		local hspd = Vector3.new(hrp.AssemblyLinearVelocity.X, 0, hrp.AssemblyLinearVelocity.Z).Magnitude
		RT.speedLbl.Text = string.format("%.1f", hspd)
	end
end)

RT.mainHeartbeat = RunService.Heartbeat:Connect(function()
	local char = LP.Character

	if char and isCarryingBrainrot(char) then
		local disabledAny = false
		if State.autoBatEnabled then
			State.autoBatEnabled = false
			if RT.autoBatToggle then RT.autoBatToggle(false) end
			setMobileButtonVisual("AutoBat", false)
			pcall(AceStopNormalAimbot)
			disabledAny = true
		end
		if RT.tpBatEnabled then
			pcall(stopTPBat)
			if RT.tpBatToggleRef then RT.tpBatToggleRef(false) end
			setMobileButtonVisual("TPBat", false)
			disabledAny = true
		end
		if State.autoLeftEnabled or State.autoRightEnabled then
			State.autoLeftEnabled = false
			State.autoRightEnabled = false
			if RT.autoPlayToggle then RT.autoPlayToggle(false) end
			setMobileButtonVisual("AutoPlay", false)
			pcall(AceStopAutoLeft)
			pcall(AceStopAutoRight)
			disabledAny = true
		end
		if disabledAny then
			showProtectionNotification("Auto-Disattivazione: Rilevato Brainrot in mano per proteggerti dal Kick!")
			saveConfig()
		end
	end

	if not RT.modeValLbl then return end
	local mode = "Normal"
	local ok, autoCarry = pcall(isAutoCarryActive)
	autoCarry = ok and autoCarry or false
	if State.laggerToggled then
		mode = autoCarry and "Carry 1" or "Normal 1"
	elseif State.speedToggled or autoCarry then
		mode = "Carry"
	end
	RT.modeValLbl.Text = mode
end)

RT.inputBeganConn = UIS.InputBegan:Connect(function(inp, gp)
	local fromPad = isGamepadInput(inp.UserInputType)
	if gp and not fromPad then return end
	if fromPad then setPadActive(true) end
	local kc = inp.KeyCode; if kc == Enum.KeyCode.Unknown then return end

	if kbMatch(KB.TPDown, kc) then
		task.spawn(runTPDown)

	elseif kbMatch(KB.Drop, kc) then
		task.spawn(runDrop)

	elseif kbMatch(KB.InstaReset, kc) then
		task.spawn(doJWInstaReset)

	elseif kbMatch(KB.EmergencyTP, kc) then
		task.spawn(runEmergencyTP)

	elseif kbMatch(KB.TPBat, kc) then
		local newOn = not RT.tpBatEnabled
		if newOn and checkBrainrotKickPrevention("TP Bat") then return end
		if newOn then startTPBat() else stopTPBat() end
		if RT.tpBatToggleRef then RT.tpBatToggleRef(RT.tpBatEnabled) end
		saveConfig()

	elseif kbMatch(KB.Aimbot, kc) then
		local newOn = not State.autoBatEnabled
		if newOn and checkBrainrotKickPrevention("Bat Aimbot") then return end
		State.autoBatEnabled = newOn
		if RT.autoBatToggle then RT.autoBatToggle(newOn) end
		setMobileButtonVisual("AutoBat", newOn)
		if newOn then pcall(AceStartNormalAimbot) else pcall(AceStopNormalAimbot) end
		saveConfig()

	elseif kbMatch(KB.AutoPlay, kc) then
		-- se e gia in coda, un secondo tocco annulla l'attesa
		if RT.autoPlayPending then
			cancelAutoPlayPending()
			if RT.autoPlayToggle then pcall(RT.autoPlayToggle, false) end
			setMobileButtonVisual("AutoPlay", false)
			return
		end
		local newOn = not (State.autoLeftEnabled or State.autoRightEnabled)
		if newOn and checkBrainrotKickPrevention("Auto Play") then
			queueAutoPlayStart()
			return
		end
		if newOn then
			local side = resolveAutoSide()
			if side == "Right" then
				State.autoRightEnabled = true
				State.autoLeftEnabled = false
				pcall(AceStopAutoLeft)
				pcall(AceStartAutoRight)
			else
				State.autoLeftEnabled = true
				State.autoRightEnabled = false
				pcall(AceStopAutoRight)
				pcall(AceStartAutoLeft)
			end
			if RT.autoPlayToggle then RT.autoPlayToggle(true) end
			setMobileButtonVisual("AutoPlay", true)
		else
			State.autoLeftEnabled = false
			State.autoRightEnabled = false
			if RT.autoPlayToggle then RT.autoPlayToggle(false) end
			setMobileButtonVisual("AutoPlay", false)
			pcall(AceStopAutoLeft)
			pcall(AceStopAutoRight)
		end
		saveConfig()

	elseif kbMatch(KB.Speed, kc) then
		State.laggerToggled = false
		State.speedToggled = not State.speedToggled
		setMobileButtonVisual("Speed", State.speedToggled)
		setMobileButtonVisual("Lagger", false)
		refreshAllTogglesRef()
		saveConfig()

	elseif kbMatch(KB.Lagger, kc) then
		State.speedToggled = false
		State.laggerToggled = not State.laggerToggled
		setMobileButtonVisual("Lagger", State.laggerToggled)
		setMobileButtonVisual("Speed", false)
		refreshAllTogglesRef()
		saveConfig()
	end
end)

task.spawn(function()

	local seen, dupes = {}, {}
	for _, field in ipairs({"kb", "gp"}) do
		local used = {}
		for name, entry in pairs(KB) do
			local k = entry[field]
			if k then
				if used[k] and not seen[k.Name] then
					seen[k.Name] = true
					table.insert(dupes, k.Name .. " (" .. used[k] .. " / " .. name .. ")")
				end
				used[k] = name
			end
		end
	end
	if #dupes > 0 then
		showProtectionNotification("Tasti doppi in config: " .. table.concat(dupes, ", "))
	end
end)
