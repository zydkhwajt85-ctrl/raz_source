--==================================================
-- DRAG SYSTEM
--==================================================

local Dragging = false
local DragStart
local StartPosition

Title.InputBegan:Connect(function(Input)

    if Input.UserInputType == Enum.UserInputType.MouseButton1
        or Input.UserInputType == Enum.UserInputType.Touch then

        Dragging = true
        DragStart = Input.Position
        StartPosition = MainFrame.Position

        Input.Changed:Connect(function()

            if Input.UserInputState == Enum.UserInputState.End then
                Dragging = false
            end

        end)
    end
end)

Title.InputChanged:Connect(function(Input)

    if Input.UserInputType == Enum.UserInputType.MouseMovement
        or Input.UserInputType == Enum.UserInputType.Touch then

        Input.Changed:Connect(function()

            if Dragging then

                local Delta = Input.Position - DragStart

                MainFrame.Position = UDim2.new(
                    StartPosition.X.Scale,
                    StartPosition.X.Offset + Delta.X,
                    StartPosition.Y.Scale,
                    StartPosition.Y.Offset + Delta.Y
                )

            end

        end)
    end
end)

--==================================================
-- SELECT PLAYER
--==================================================

local function SelectPlayer(Player)

    if Player == LocalPlayer then
        return
    end

    TargetPlayer = Player

    TargetLabel.Text = "Target: " .. Player.Name

    FollowEnabled = true

    FollowButton.Text = "FOLLOW : ON"
end

--==================================================
-- REFRESH PLAYER LIST
--==================================================

local function RefreshPlayerList()

    for _, Child in ipairs(PlayerList:GetChildren()) do

        if Child:IsA("TextButton") then
            Child:Destroy()
        end

    end

    for _, Player in ipairs(Players:GetPlayers()) do

        if Player ~= LocalPlayer then

            local PlayerButton = Instance.new("TextButton")

            PlayerButton.Size = UDim2.new(1, -8, 0, 30)
            PlayerButton.BackgroundColor3 =
                Color3.fromRGB(35, 40, 52)

            PlayerButton.Text =
                Player.DisplayName .. "  @" .. Player.Name

            PlayerButton.TextColor3 =
                Color3.new(1, 1, 1)

            PlayerButton.TextSize = 12
            PlayerButton.Font = Enum.Font.Gotham
            PlayerButton.BorderSizePixel = 0
            PlayerButton.Parent = PlayerList

            local Corner = Instance.new("UICorner")
            Corner.CornerRadius = UDim.new(0, 6)
            Corner.Parent = PlayerButton

            PlayerButton.MouseButton1Click:Connect(function()
                SelectPlayer(Player)
            end)

        end
    end

    task.wait()

    PlayerList.CanvasSize = UDim2.new(
        0,
        0,
        0,
        ListLayout.AbsoluteContentSize.Y + 5
    )
end

--==================================================
-- FOLLOW + SPIN ENGINE
--==================================================

local function StartFollowSpin(Character)

    if FollowSpinConnection then

        FollowSpinConnection:Disconnect()
        FollowSpinConnection = nil

    end

    local Root =
        Character:WaitForChild("HumanoidRootPart")

    SpinAngle = 0

    FollowSpinConnection =
        RunService.RenderStepped:Connect(function()

            if not Character.Parent then
                return
            end

            if not Root.Parent then
                return
            end

            --==========================================
            -- SPIN ANGLE
            --==========================================

            if SpinEnabled then

                SpinAngle =
                    SpinAngle + math.rad(SPIN_SPEED)

                if SpinAngle > math.pi * 2 then

                    SpinAngle =
                        SpinAngle % (math.pi * 2)

                end
            end

            --==========================================
            -- FOLLOW TARGET
            --==========================================

            if FollowEnabled and TargetPlayer then

                local TargetCharacter =
                    TargetPlayer.Character

                if TargetCharacter then

                    local TargetRoot =
                        TargetCharacter:FindFirstChild(
                            "HumanoidRootPart"
                        )

                    if TargetRoot then

                        -- Posisi menempel target
                        local TargetCFrame =
                            TargetRoot.CFrame * OFFSET

                        local TargetPosition =
                            TargetCFrame.Position

                        -- Rotasi target
                        local TargetRotation =
                            TargetRoot.CFrame.Rotation

                        -- Posisi + rotasi target
                        -- + rotasi spin
                        Root.CFrame =
                            CFrame.new(TargetPosition)
                            * TargetRotation
                            * CFrame.Angles(
                                0,
                                SpinAngle,
                                0
                            )

                        Root.AssemblyLinearVelocity =
                            Vector3.zero

                        Root.AssemblyAngularVelocity =
                            Vector3.zero

                        return

                    end
                end
            end

            --==========================================
            -- SPIN TANPA FOLLOW
            --==========================================

            if SpinEnabled then

                local Position = Root.Position

                Root.CFrame =
                    CFrame.new(Position)
                    * CFrame.Angles(
                        0,
                        SpinAngle,
                        0
                    )

            end

        end)
end

--==================================================
-- START CHARACTER
--==================================================

if LocalPlayer.Character then

    StartFollowSpin(
        LocalPlayer.Character
    )

end

--==================================================
-- RESPAWN
--==================================================

LocalPlayer.CharacterAdded:Connect(function(Character)

    Character:WaitForChild(
        "HumanoidRootPart"
    )

    task.wait(0.2)

    StartFollowSpin(Character)

end)

--==================================================
-- ANIMATION LOOP
--
-- OUTLINE:
-- DARK RED <-> PINK
--
-- ON BUTTON:
-- DARK GREEN <-> LIGHT GREEN
--==================================================

RunService.RenderStepped:Connect(function()

    --==============================================
    -- OUTLINE ANIMATION
    --==============================================

    local OutlineAlpha =
        (math.sin(
            os.clock() * OutlineSpeed
        ) + 1) / 2

    local AnimatedOutline =
        DarkRed:Lerp(
            Pink,
            OutlineAlpha
        )

    MainStroke.Color =
        AnimatedOutline

    OpenStroke.Color =
        AnimatedOutline

    --==============================================
    -- BUTTON COLOR ANIMATION
    --==============================================

    local GreenAlpha =
        (math.sin(
            os.clock() * ButtonAnimationSpeed
        ) + 1) / 2

    local AnimatedGreen =
        DarkGreen:Lerp(
            LightGreen,
            GreenAlpha
        )

    -- FOLLOW
    if FollowEnabled then

        FollowButton.BackgroundColor3 =
            AnimatedGreen

    else

        FollowButton.BackgroundColor3 =
            Color3.fromRGB(
                170,
                50,
                50
            )

    end

    -- SPIN
    if SpinEnabled then

        SpinButton.BackgroundColor3 =
            AnimatedGreen

    else

        SpinButton.BackgroundColor3 =
            Color3.fromRGB(
                170,
                50,
                50
            )

    end

end)
