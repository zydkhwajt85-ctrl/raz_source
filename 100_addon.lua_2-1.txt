local desyncActive = false
local normalActive = false
local spoofConnection = nil
local root = nil
local hum = nil
 
btnDesync.MouseButton1Click:Connect(function()
    if desyncActive then return end
    desyncActive = true
    btnDesync.Text = "Desync ATIVO"
    local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    root = char:WaitForChild("HumanoidRootPart")
    hum = char:WaitForChild("Humanoid")
    pcall(function() root:SetNetworkOwner(nil) end)
    spoofConnection = RunService.Heartbeat:Connect(function()
        if not root or not root.Parent then if spoofConnection then spoofConnection:Disconnect() end return end
        root.AssemblyLinearVelocity = Vector3.new(0,0,0)
        root.AssemblyAngularVelocity = Vector3.new(0,0,0)
    end)
end)
 
btnNormal.MouseButton1Click:Connect(function()
    if not desyncActive then return end
    normalActive = true
    btnNormal.Text = "Normal ATIVO"
    if spoofConnection then spoofConnection:Disconnect() spoofConnection = nil end
    if root then pcall(function() root:SetNetworkOwner(LocalPlayer) end) end
    if hum then
        hum:ChangeState(Enum.HumanoidStateType.None)
        hum.PlatformStand = false
        hum.AutoRotate = true
        hum.WalkSpeed = 16
        hum.JumpPower = 50
        hum.UseJumpPower = true
    end
end)
 
local unwalkConn
btnUnwalk.MouseButton1Click:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    local h = char:FindFirstChildOfClass("Humanoid")
    if not h then return end
    local anim = h:FindFirstChildWhichIsA("Animator")
    if not anim then return end
    if unwalkConn then unwalkConn:Disconnect() end
    unwalkConn = RunService.Heartbeat:Connect(function()
        for _, track in anim:GetPlayingAnimationTracks() do track:Stop() end
    end)
end)
 
local autoLaserEnabled = false
local delayBetweenShots = 0.6
local laserRange = 300
local Event = ReplicatedStorage:WaitForChild("Packages"):WaitForChild("Net"):WaitForChild("RE/UseItem")
 
local function getNearestPlayer()
    if not LocalPlayer.Character or not LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then return nil end
    local myPos = LocalPlayer.Character.HumanoidRootPart.Position
    local nearest = nil
    local shortest = laserRange
    for _, pl in Players:GetPlayers() do
        if pl ~= LocalPlayer and pl.Character and pl.Character:FindFirstChild("HumanoidRootPart") then
            local dist = (pl.Character.HumanoidRootPart.Position - myPos).Magnitude
            if dist < shortest then shortest = dist nearest = pl end
        end
    end
    return nearest
end
 
local function equipOnlyLaserCape()
    local char = LocalPlayer.Character
    if not char then return false end
    for _, tool in char:GetChildren() do if tool:IsA("Tool") then tool.Parent = LocalPlayer.Backpack end end
    local tool = LocalPlayer.Backpack:FindFirstChild("Laser Cape")
    if tool then tool.Parent = char return true end
    return false
end
 
local function fireLaserCape(part)
    if not part then return end
    pcall(function() Event:FireServer(part.Position, part) end)
end
 
task.spawn(function()
    while task.wait(delayBetweenShots) do
        if autoLaserEnabled then
            local t = getNearestPlayer()
            if t and t.Character then
                if equipOnlyLaserCape() then
                    local hand = t.Character:FindFirstChild("RightHand") or t.Character:FindFirstChild("LeftHand")
                    if hand then fireLaserCape(hand) end
                end
            end
        end
    end
end)
 
btnLaser.MouseButton1Click:Connect(function()
    autoLaserEnabled = not autoLaserEnabled
    btnLaser.Text = autoLaserEnabled and "Laser Cape: ON" or "Laser Cape: OFF"
end)
 
LocalPlayer.CharacterAdded:Connect(function()
    desyncActive = false
    normalActive = false
    autoLaserEnabled = false

end)
