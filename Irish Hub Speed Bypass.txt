local dragging, dragInput, dragStart, startPos

Header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
	or input.UserInputType == Enum.UserInputType.Touch then
		dragging  = true
		dragStart = input.Position
		startPos  = Main.Position

		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

Header.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement
	or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if dragging and input == dragInput then
		local delta = input.Position - dragStart
		Main.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)

	for _, entry in ipairs(modeButtons) do
		local btn = entry.Button

		btn.BackgroundColor3 = isActive and ACTIVE_BG or INACTIVE_BG
		btn.TextColor3       = isActive and ACTIVE_TEXT or INACTIVE_TEXT

		entry.Stroke.Color        = isActive and ACTIVE_STROKE or INACTIVE_STROKE
		entry.Stroke.Transparency = isActive and 0.1 or 0.6
	end

	currentMode = selected
	print("[Irish Bypass] Mode ->", selected.Name)
end

	or input.UserInputType == Enum.UserInputType.Touch then
		setAutoEnable(not autoEnable)
	end
end)

setAutoEnable(false)

local running    = false
local elapsed    = 0
local startClock = 0

local function formatTime(seconds)
	local m = math.floor(seconds / 60)
	local s = math.floor(seconds % 60)
	return string.format("%02d:%02d", m, s)
end

TextButton_4.MouseButton1Click:Connect(function()
	running = not running

	if running then
		startClock = os.clock() - elapsed
		TextButton_4.Text             = "STOP"
		TextButton_4.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
		UIStroke_13.Color             = Color3.fromRGB(200, 50, 50)
		Frame_7.BackgroundColor3      = Color3.fromRGB(30, 200, 80)
		TextLabel_9.Text              = "ACTIVE"
	else
		TextButton_4.Text             = "START"
		TextButton_4.BackgroundColor3 = Color3.fromRGB(30, 200, 80)
		UIStroke_13.Color             = Color3.fromRGB(30, 200, 80)
		Frame_7.BackgroundColor3      = Color3.fromRGB(90, 90, 90)
		TextLabel_9.Text              = "IDLE"
	end
end)

RunService.RenderStepped:Connect(function()
	if running then
		elapsed = os.clock() - startClock
		TextLabel_10.Text = formatTime(elapsed)
	end
end)

local scale = 0.7

local function applyScale()
	UIScale.Scale = scale
	TextLabel_5.Text = math.floor(scale * 100 + 0.5) .. "%"
end

TextButton.MouseButton1Click:Connect(function()
	scale = math.clamp(scale - 0.05, 0.4, 1.5)
	applyScale()
end)

TextButton_2.MouseButton1Click:Connect(function()
	scale = math.clamp(scale + 0.05, 0.4, 1.5)
	applyScale()
end)

applyScale()

TextBox.FocusLost:Connect(function()
	local value = tonumber(TextBox.Text)

	if value then
		value = math.clamp(math.floor(value), 0, 1000000)
		TextBox.Text = tostring(value)

		if value >= 1000000 then
			TextLabel_8.Text = string.format("%.1fM", value / 1000000)
		elseif value >= 1000 then
			TextLabel_8.Text = string.format("%.1fK", value / 1000)
		else
			TextLabel_8.Text = tostring(value)
		end

		local ratio = math.clamp(value / 1000000, 0, 1)
		Frame_4.Size     = UDim2.new(ratio, 0, 1, 0)
		Frame_5.Position = UDim2.new(ratio, 0, 0.5, 0)
	else
		TextBox.Text = "100000"
	end
end)
