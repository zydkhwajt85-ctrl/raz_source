-- FUNGSI DRAG
local dragging, dragStart, startPos
TitleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

-- LOGIKA UTAMA AUTO REDEEM
_G.AutoRedeem = false
local teksTerakhir = {}

local function dapatkanTextBoxGame()
    for _, gui in pairs(PlayerGui:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Enabled and gui.Name ~= "AutoRedeemMobileUI" then
            for _, descendant in pairs(gui:GetDescendants()) do
                if descendant:IsA("TextBox") and (descendant.PlaceholderText == "Code Here..." or string.find(string.lower(descendant.PlaceholderText), "code")) then
                    return descendant
                end
            end
        end
    end
    return nil
end

local function isBlacklisted(text)
    local blacklist = {"starting", "in:", "h ", "m ", "s ", "live", "now", "event", "start", "play", "claim", "code", "redeem", "health", "mana", "level", "lvl", "submit", "friend boost", "codes", "global", "transparency", "namelabel", "stroke", "expired", "invalid", "success", "here", "purchase", "buy", "price", "robux", "cost"}
    for _, word in ipairs(blacklist) do
        if string.find(string.lower(text), word) then return true end
    end
    return false
end

local function masukkanDanSubmitKode(codeText)
    if not _G.AutoRedeem or teksTerakhir[codeText] then return end
    teksTerakhir[codeText] = true
    
    local foundTextBox = dapatkanTextBoxGame()
    if foundTextBox then
        -- DI SINI PERUBAHANNYA: Menggunakan operator `..` untuk menggabungkan teks
        foundTextBox.Text = foundTextBox.Text .. codeText
        StatusLabel.Text = "Gabung kode: " .. foundTextBox.Text
        
        -- Beri jeda sedikit agar teks gabungan selesai diproses sebelum di-submit otomatis
        task.wait(0.5)
        foundTextBox:ReleaseFocus(true)
    end
    
    task.delay(2, function() teksTerakhir[codeText] = nil end)
end

ToggleBtn.MouseButton1Click:Connect(function()
    _G.AutoRedeem = not _G.AutoRedeem
    ToggleBtn.BackgroundColor3 = _G.AutoRedeem and Color3.fromRGB(50, 180, 50) or Color3.fromRGB(180, 50, 50)
    ToggleBtn.Text = _G.AutoRedeem and "STATUS: AKTIF" or "STATUS: MATI"
    StatusLabel.Text = _G.AutoRedeem and "Mencari kode..." or "Auto Redeem dinonaktifkan."
end)

PlayerGui.DescendantAdded:Connect(function(d)
    if _G.AutoRedeem and d:IsA("TextLabel") and d.Visible and not d:IsDescendantOf(ScreenGui) then
        task.wait(0.1)
        local t = d.Text
        if t ~= "" and string.len(t) <= 15 and not isBlacklisted(t) then
            masukkanDanSubmitKode(t)
        end
    end
end)
