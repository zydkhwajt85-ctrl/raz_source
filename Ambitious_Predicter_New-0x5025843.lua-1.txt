local function setDiceColor(i, colorName)
    local chosenColor = COLORS[colorName]
    local dice = diceButtons[i]
    if not chosenColor or not dice then return end

    diceColors[i] = colorName
    for _, child in ipairs(dice.btn:GetChildren()) do
        if child:IsA("UIGradient") then child:Destroy() end
    end
    dice.btn.BackgroundColor3 = chosenColor
    dice.btn.BackgroundTransparency = 0
    dice.label.Text = colorName
end

local function updateHitRate()
	local total = hits + misses
	if total == 0 then
		HR_Value.Text = "—"
		HR_Sub.Text = "No results yet"
		HR_BarFill.Size = UDim2.new(0.5, 0, 1, 0)
		return
	end
	local rate = hits / total
	HR_Value.Text = string.format("%.1f%%", rate * 100)
	HR_Sub.Text = string.format("%d / %d hits • streak %d", hits, total, streak)
	TweenService:Create(HR_BarFill, TweenInfo.new(0.4), {Size = UDim2.new(rate, 0, 1, 0)}):Play()
	local col = Color3.fromRGB(110, 220, 145)
	if rate < 0.4 then col = Color3.fromRGB(220, 110, 110)
	elseif rate < 0.55 then col = Color3.fromRGB(220, 200, 100) end
	HR_BarFill.BackgroundColor3 = col
	HR_Value.TextColor3 = col
end

end

local function pickBestColor()
	if #history == 0 then
		return COLOR_ORDER[math.random(#COLOR_ORDER)], math.random(50, 63)
	end
	local counts, streaks = {}, {}
	for _, c in ipairs(COLOR_ORDER) do counts[c] = 0; streaks[c] = 0 end
	for _, entry in ipairs(history) do
		for i = 1, 4 do if entry.picks[i] then counts[entry.picks[i]] = counts[entry.picks[i]] + 1 end end
	end
	for _, c in ipairs(COLOR_ORDER) do
		for i = #history, 1, -1 do
			local present = false
			for j = 1, 4 do if history[i].picks[j] == c then present = true break end end
			if not present then streaks[c] = streaks[c] + 1 else break end
		end
	end
	
	local bestScore = -math.huge
	local candidates = {}
	for _, c in ipairs(COLOR_ORDER) do
		local score = streaks[c] + 1 / (1 + counts[c]) + math.random()
		if score > bestScore then
			bestScore = score
			candidates = {c}
		elseif math.abs(score - bestScore) < 0.001 then 
			table.insert(candidates, c)
		end
	end
	local best = candidates[math.random(1, #candidates)]
	local conf = math.random(50, 63)
	return best, conf
end

local function doPredict()
	if isCooldownActive then return end

	local picks = {}
	for i = 1, 4 do picks[i] = diceColors[i] end

	local entry = {picks = picks, result = "PENDING", confidence = 0}
	local chosen, conf = pickBestColor()
	table.insert(history, entry)
	rollsObserved = rollsObserved + 1

	entry.prediction = chosen
	entry.confidence = conf
	setPrediction(chosen, conf, string.format("Based on %d rolls", #history))

	local hit = false
	for i = 1, 4 do if picks[i] == chosen then hit = true break end end
	if #history == 1 then hit = true end
	if hit then
		entry.result = "HIT"
		hits = hits + 1
		streak = streak + 1
	else
		entry.result = "MISS"
		misses = misses + 1
		streak = 0
	end

	updateHitRate()
	Footer.Text = string.format("Rolls %d • Next #%d • Bet %d", rollsObserved, rollsObserved + 1, betAmount)
	if HistoryOverlay.Visible then refreshHistoryUI() end

	if not removeExpireCooldown then
		isCooldownActive = true
		CooldownBarBg.Visible = true
		CooldownBarFill.Size = UDim2.new(1, 0, 1, 0)
		CooldownBarFill.BackgroundColor3 = ACCENT_START
		
		TweenService:Create(CooldownBarFill, TweenInfo.new(3, Enum.EasingStyle.Linear), {Size = UDim2.new(0, 0, 1, 0)}):Play()
		
		task.spawn(function()
			for i = 3, 1, -1 do
				PP_Status.Text = string.format("Expires in %ds", i)
				PP_Status.TextColor3 = Color3.fromRGB(220, 200, 100)
				task.wait(1)
			end
			isCooldownActive = false
			CooldownBarBg.Visible = false
			resetPrediction()
		end)
	else
		CooldownBarBg.Visible = false
	end
end

PredictBtn.MouseButton1Click:Connect(doPredict)

local draggingSize = false
local function setSizeFromMouse(mouseX)
	local rel = math.clamp((mouseX - SizeSlider.AbsolutePosition.X) / SizeSlider.AbsoluteSize.X, 0, 1)
	local newScale = 0.7 + rel * 0.6
	uiScaleValue = math.floor(newScale * 100) / 100
	UIScale.Scale = uiScaleValue
	SizeValue.Text = string.format("%.2fx", uiScaleValue)
	SizeFill.Size = UDim2.new(rel, 0, 1, 0)
	SizeKnob.Position = UDim2.new(rel, 0, 0.5, 0)
end

SizeSlider.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		draggingSize = true
		setSizeFromMouse(input.Position.X)
	end
end)
SizeKnob.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		draggingSize = true
	end
end)
UserInputService.InputChanged:Connect(function(input)
	if draggingSize and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		setSizeFromMouse(input.Position.X)
	end
end)
UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		draggingSize = false
	end
end)

local dragging = false
local dragStart, startPos
Header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		local mouse = UserInputService:GetMouseLocation()
		for _, btn in ipairs({MinBtn, MenuBtn, CloseBtn}) do
			local abspos, abssize = btn.AbsolutePosition, btn.AbsoluteSize
			if mouse.X >= abspos.X and mouse.X <= abspos.X + abssize.X and mouse.Y >= abspos.Y and mouse.Y <= abspos.Y + abssize.Y then return end
		end
		dragging = true
		dragStart = input.Position
		startPos = Frame.Position
	end
end)
UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		Frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)
UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)
