-- // [4] FUNCTIONAL LOGIC & SIGNAL CONNECTIONS // --
local function clearAntiDieConnections()
	for _, conn in ipairs(antiDieConnections) do
		if typeof(conn) == "RBXScriptConnection" then
			conn:Disconnect()
		end
	end
	antiDieConnections = {}
end

local function applyAntiDieToCharacter(char)
	if not featureStates.AntiDieActive then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum then return end

	hum.BreakJointsOnDeath = false
	hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)

	local healthConn = hum:GetPropertyChangedSignal("Health"):Connect(function()
		if not featureStates.AntiDieActive then return end
		if hum.Health <= 0 then
			hum.Health = hum.MaxHealth
		end
	end)
	table.insert(antiDieConnections, healthConn)

	local diedConn = hum.Died:Connect(function()
		if not featureStates.AntiDieActive then return end
		task.wait()
		local newHum = Instance.new("Humanoid")
		newHum.Name = "ReplacedHumanoid"
		newHum.Parent = char
		workspace.CurrentCamera.CameraSubject = newHum
		hum:Destroy()
	end)
	table.insert(antiDieConnections, diedConn)
end

local function toggleAntiDie(isEnabled)
	featureStates.AntiDieActive = isEnabled
	clearAntiDieConnections()

	if isEnabled then
		local char = LocalPlayer.Character
		if char then
			applyAntiDieToCharacter(char)
		end
	end
end

LocalPlayer.CharacterAdded:Connect(function(char)
	if featureStates.AntiDieActive then
		task.wait(0.1)
		applyAntiDieToCharacter(char)
	end
end)

local function toggleFreeze(isEnabled)
	if isEnabled then
		featureStates.FreezePlayer = true
		local connection = RunService.Stepped:Connect(function()
			if not featureStates.FreezePlayer then
				connection:Disconnect()
				return
			end
			for _, player in ipairs(Players:GetPlayers()) do
				if player ~= LocalPlayer then
					local character = player.Character
					if character then
						local humanoid = character:FindFirstChildWhichIsA("Humanoid")
						if humanoid then
							humanoid.WalkSpeed = 0
							humanoid.JumpPower = 0
							humanoid.AutoRotate = false
						end
						for _, object in ipairs(character:GetDescendants()) do
							if object:IsA("BasePart") then
								object.CanCollide = false
							end
						end
					end
				end
			end
		end)
		table.insert(connections, connection)
	else
		featureStates.FreezePlayer = false
		for _, connection in ipairs(connections) do
			if typeof(connection) == "RBXScriptConnection" then
				connection:Disconnect()
			end
		end
		connections = {}
		for _, player in ipairs(Players:GetPlayers()) do
			if player ~= LocalPlayer then
				local character = player.Character
				if character then
					local humanoid = character:FindFirstChildWhichIsA("Humanoid")
					if humanoid then
						humanoid.WalkSpeed = 16
						humanoid.JumpPower = 50
						humanoid.AutoRotate = true
					end
				end
			end
		end
	end
end

-- Dragging
local isDragging = false
local dragOffset = Vector2.new()
local startPosition = UDim2.new()

mainFrame.InputBegan:Connect(function(input)
	if featureStates.isUiLocked then return end
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		isDragging = true
		dragOffset = input.Position
		startPosition = mainFrame.Position
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = Vector2.new(input.Position.X - dragOffset.X, input.Position.Y - dragOffset.Y)
		mainFrame.Position = UDim2.new(startPosition.X.Scale, startPosition.X.Offset + delta.X, startPosition.Y.Scale, startPosition.Y.Offset + delta.Y)
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		isDragging = false
	end
end)

-- Core Heartbeat Logic (Shield + Velocity Desync)
local function getChar()
	return LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
end

RunService.Heartbeat:Connect(function()
	if not (featureStates.ShieldActive or featureStates.DesyncActive) then return end

	local char = getChar()
	local hrp = char:FindFirstChild("HumanoidRootPart")
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hrp or not hum or hum.Health <= 0 then return end

	if featureStates.DesyncActive then
		local oldVelocity = hrp.Velocity
		hrp.Velocity = Vector3.new(999, 0, 999)
		RunService.RenderStepped:Wait()
		hrp.Velocity = oldVelocity
	end

	if featureStates.ShieldActive then
		for _, player in pairs(Players:GetPlayers()) do
			if player ~= LocalPlayer and player.Character then
				local enemyHrp = player.Character:FindFirstChild("HumanoidRootPart")
				local enemyTool = player.Character:FindFirstChildWhichIsA("Tool")
				if enemyHrp then
					local distance = (hrp.Position - enemyHrp.Position).Magnitude
					if distance < SHIELD_RANGE and enemyTool and enemyTool.Name == "Bat" then
						local angle = math.rad(tick() * 500)
						hrp.CFrame = hrp.CFrame * CFrame.new(math.sin(angle) * 3, 0, math.cos(angle) * 3)
					end
				end
			end
		end
	end
end)
