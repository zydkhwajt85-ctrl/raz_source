local function applyReadableTextOutline(textElement)
    if not textElement then
        return
    end

    if
        textElement:IsA("TextLabel")
        or textElement:IsA("TextButton")
        or textElement:IsA("TextBox")
    then
        textElement.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)

        local TextSize = textElement.TextSize

        if textElement.TextScaled then
            TextSize = math.floor(textElement.AbsoluteSize.Y / 24 * 18)
        end

        if TextSize <= 12 then
            textElement.TextStrokeTransparency = 0.5
        elseif TextSize <= 16 then
            textElement.TextStrokeTransparency = 0.3
        else
            textElement.TextStrokeTransparency = 0.2
        end
    end
end

local function loadConfiguration()
    if not isfile or not isfile(configurationFilePath) then
        return
    end

    local ok, result = pcall(readfile, configurationFilePath)

    if not ok then
        return
    end

    local ok2, result2 = pcall(services.HttpService.JSONDecode, services.HttpService, result)

    if ok2 and result2 then
        if result2.CONFIG_VERSION ~= configVersion then
            pcall(function()
                delfile(configurationFilePath)
            end)

            return
        end

        for key2, value4 in pairs(result2) do
            if key2 ~= "CONFIG_VERSION" then
                DUEL_CONFIG[key2] = value4
            end
        end
    end

    getgenv().DUEL_CONFIG = DUEL_CONFIG
end

local function cachePromptCallbacks(prompt)
    if promptCallbackCache[prompt] then
        return
    end

    local promptCallbacks = {
        holdCallbacks = {},
        triggerCallbacks = {},
    }
    local ok, result = pcall(getconnections, prompt.PromptButtonHoldBegan)

    if ok and type(result) == "table" then
        for _, value5 in ipairs(result) do
            if type(value5.Function) == "function" then
                table.insert(promptCallbacks.holdCallbacks, value5.Function)
            end
        end
    end

    local ok2, result2 = pcall(getconnections, prompt.Triggered)

    if ok2 and type(result2) == "table" then
        for _, value6 in ipairs(result2) do
            if type(value6.Function) == "function" then
                table.insert(promptCallbacks.triggerCallbacks, value6.Function)
            end
        end
    end

    if #promptCallbacks.holdCallbacks > 0 or #promptCallbacks.triggerCallbacks > 0 then
        promptCallbackCache[prompt] = promptCallbacks
    end
end

local function findNearestStealPrompt(grabRange)
    local localRootPart = getLocalRootPart()

    if not localRootPart then
        return nil
    end

    local Position = localRootPart.Position
    local grabRangeSquared = grabRange * grabRange
    local nearestStealPrompt
    local huge = math.huge
    local Plots = workspace:FindFirstChild("Plots")

    if not Plots then
        return nil
    end

    for _, child in ipairs(Plots:GetChildren()) do
        if not isLocalPlayerPlot(child) then
            local AnimalPodiums = child:FindFirstChild("AnimalPodiums")

            if AnimalPodiums then
                for _, child2 in ipairs(AnimalPodiums:GetChildren()) do
                    local Base = child2:FindFirstChild("Base")

                    if Base then
                        local Spawn = Base:FindFirstChild("Spawn")

                        if Spawn then
                            local PromptAttachment = Spawn:FindFirstChild("PromptAttachment")

                            if PromptAttachment then
                                local Position2 = Spawn.Position
                                local promptDeltaX = Position2.X - Position.X
                                local promptDeltaY = Position2.Y - Position.Y
                                local promptDeltaZ = Position2.Z - Position.Z
                                local promptDistanceSquared = promptDeltaX * promptDeltaX
                                    + promptDeltaY * promptDeltaY
                                    + promptDeltaZ * promptDeltaZ

                                if not (grabRangeSquared < promptDistanceSquared) then
                                    for _, child3 in ipairs(PromptAttachment:GetChildren()) do
                                        if
                                            child3:IsA("ProximityPrompt")
                                            and promptDistanceSquared < huge
                                        then
                                            huge = promptDistanceSquared
                                            nearestStealPrompt = child3
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    return nearestStealPrompt
end
local function startInstantGrab()
    if signalConnection2 then
        signalConnection2:Disconnect()
    end

    isGrabInProgress = false
    signalConnection2 = services.RunService.Heartbeat:Connect(function()
        if not DUEL_CONFIG.INSTANT_GRAB_ENABLED then
            signalConnection2:Disconnect()
            signalConnection2 = nil
            grabProgress.active = false

            return
        end

        if isGrabInProgress then
            return
        end

        local nearestStealPrompt2 = findNearestStealPrompt(DUEL_CONFIG.INSTANT_GRAB_RANGE or 16)

        if not nearestStealPrompt2 then
            return
        end

        cachePromptCallbacks(nearestStealPrompt2)

        local result3 = promptCallbackCache[nearestStealPrompt2]

        if not result3 then
            return
        end

        isGrabInProgress = true
        grabProgress.active = true
        grabProgress.startTime = tick()
        task.spawn(function()
            for _, holdCallback in ipairs(result3.holdCallbacks) do
                pcall(holdCallback)
            end

            task.wait(0.2)

            for _, triggerCallback in ipairs(result3.triggerCallbacks) do
                pcall(triggerCallback)
            end

            grabProgress.active = false
            isGrabInProgress = false
        end)
    end)
end
local function stopInstantGrab()
    if signalConnection2 then
        signalConnection2:Disconnect()
        signalConnection2 = nil
    end

    isGrabInProgress = false
    grabProgress.active = false
end

local ragdollController

do
    local ragdollRecoveryState = {
        active = false,
        connections = {},
        initConnections = {},
    }
    local Character
    local Humanoid
    local HumanoidRootPart
    local ragdollHumanoidStates = {
        [Enum.HumanoidStateType.Ragdoll] = true,
        [Enum.HumanoidStateType.FallingDown] = true,
        [Enum.HumanoidStateType.Physics] = true,
        [Enum.HumanoidStateType.Dead] = true,
    }

    local function refreshCharacterReferences()
        Character = services.LocalPlayer.Character

        if Character then
            Humanoid = Character:FindFirstChildOfClass("Humanoid")
            HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")
                or Character:FindFirstChild("RootPart")
                or Character:FindFirstChild("HumanoidRoot")
        else
            Humanoid = nil
            HumanoidRootPart = nil
        end
    end
    local function isHumanoidAvailable()
        return Humanoid and Humanoid.Parent ~= nil
    end
    local function isCharacterRagdolled()
        if not Humanoid then
            return false
        end

        local state = Humanoid:GetState()

        if
            state == Enum.HumanoidStateType.Jumping
            or state == Enum.HumanoidStateType.Freefall
            or state == Enum.HumanoidStateType.Landed
        then
            return false
        end

        if
            state == Enum.HumanoidStateType.Physics
            or state == Enum.HumanoidStateType.Ragdoll
            or state == Enum.HumanoidStateType.FallingDown
            or state == Enum.HumanoidStateType.GettingUp
        then
            return true
        end

        return false
    end
    local function restorePlayerControls()
        pcall(function()
            local PlayerModule =
                services.LocalPlayer:WaitForChild("PlayerScripts"):WaitForChild("PlayerModule")

            require(PlayerModule):GetControls():Enable()
        end)
    end
    local function removeActiveRagdollConstraints()
        if not isCharacterRagdolled() then
            return
        end

        if not Character then
            return
        end

        for _, descendant in pairs(Character:GetDescendants()) do
            if
                descendant:IsA("BallSocketConstraint")
                or descendant:IsA("NoCollisionConstraint")
                or descendant:IsA("HingeConstraint")
                or descendant:IsA("Attachment")
                    and (descendant.Name == "A" or descendant.Name == "B")
            then
                pcall(function()
                    descendant:Destroy()
                end)
            elseif
                descendant:IsA("BodyVelocity")
                or descendant:IsA("BodyPosition")
                or descendant:IsA("BodyGyro")
            then
                pcall(function()
                    descendant:Destroy()
                end)
            elseif descendant:IsA("Motor6D") then
                pcall(function()
                    descendant.Enabled = true
                end)
            end
        end
    end
    local function findCharacterRootPart(character3)
        if character3 then
            return character3:FindFirstChild("HumanoidRootPart")
                or character3:FindFirstChild("RootPart")
                or character3:FindFirstChild("HumanoidRoot")
        end

        return nil
    end
    local function removeRagdollAttachments(character4)
        if not character4 then
            return
        end

        for _, descendant in ipairs(character4:GetDescendants()) do
            if descendant:IsA("BallSocketConstraint") then
                pcall(function()
                    descendant:Destroy()
                end)
            elseif descendant:IsA("Attachment") then
                if descendant.Name:find("RagdollAttachment") then
                    pcall(function()
                        descendant:Destroy()
                    end)
                end
            end
        end
    end
    local function recoverCharacterFromRagdoll(character5)
        if not character5 or character5.Parent == nil then
            return
        end

        local Humanoid2 = character5:FindFirstChildOfClass("Humanoid")
        local characterRootPart = findCharacterRootPart(character5)

        if not Humanoid2 or not characterRootPart then
            return
        end

        if Humanoid2.Health < Humanoid2.MaxHealth then
            Humanoid2.Health = Humanoid2.MaxHealth
        end

        pcall(function()
            local serverTimeNow = workspace:GetServerTimeNow()

            services.LocalPlayer:SetAttribute("RagdollEndTime", serverTimeNow)
        end)
        removeRagdollAttachments(character5)

        if Humanoid2.Health > 0 then
            pcall(function()
                Humanoid2:ChangeState(Enum.HumanoidStateType.Running)
            end)
        end

        characterRootPart.Anchored = false
        characterRootPart.AssemblyLinearVelocity = Vector3.zero
        characterRootPart.AssemblyAngularVelocity = Vector3.zero
    end
    local function unanchorCharacterRoot(character6)
        if not character6 then
            return
        end

        local characterRootPart2 = findCharacterRootPart(character6)

        if characterRootPart2 then
            characterRootPart2.Anchored = false
        end
    end
    local function restoreCharacterJoints(character7)
        for _, descendant in ipairs(character7:GetDescendants()) do
            if descendant:IsA("Motor6D") then
                descendant.Enabled = true
            elseif descendant:IsA("Constraint") then
                descendant.Enabled = false
            end
        end
    end
    local function connectRagdollRecovery(character8)
        if not character8 or character8.Parent == nil then
            return
        end

        local Humanoid2 = character8:WaitForChild("Humanoid", 10)

        if not Humanoid2 or Humanoid2.Parent == nil then
            return
        end

        for key4 in pairs(ragdollHumanoidStates) do
            Humanoid2:SetStateEnabled(key4, false)
        end

        local signalConnection5 = Humanoid2.StateChanged:Connect(function(old, new)
            if DUEL_CONFIG.ANTI_RAGDOLL_ENABLED then
                if ragdollHumanoidStates[new] then
                    recoverCharacterFromRagdoll(character8)
                    restoreCharacterJoints(character8)
                end
            end
        end)

        table.insert(ragdollRecoveryState.initConnections, signalConnection5)

        local signalConnection6 = services.RunService.Stepped:Connect(function()
            if character8 then
                if character8.Parent ~= nil and Humanoid2.Parent then
                    if not DUEL_CONFIG.ANTI_RAGDOLL_ENABLED then
                        unanchorCharacterRoot(character8)

                        return
                    end

                    if ragdollHumanoidStates[Humanoid2:GetState()] then
                        recoverCharacterFromRagdoll(character8)
                    else
                        unanchorCharacterRoot(character8)
                    end

                    return
                end
            end
        end)

        table.insert(ragdollRecoveryState.initConnections, signalConnection6)
    end
    local function enableRagdollRecovery()
        if ragdollRecoveryState.active then
            return
        end

        ragdollRecoveryState.active = true
        refreshCharacterReferences()

        if not isHumanoidAvailable() then
            return
        end

        if Humanoid then
            table.insert(
                ragdollRecoveryState.connections,
                Humanoid.StateChanged:Connect(function()
                    if isCharacterRagdolled() then
                        pcall(function()
                            Humanoid:ChangeState(Enum.HumanoidStateType.Running)
                        end)
                        removeActiveRagdollConstraints()
                        pcall(function()
                            workspace.CurrentCamera.CameraSubject = Humanoid
                        end)
                        restorePlayerControls()
                    end
                end)
            )
        end

        local Packages = services.ReplicatedStorage:FindFirstChild("Packages")

        if Packages then
            local Net = Packages:FindFirstChild("Net")

            if Net then
                local RECombatServiceApplyImpulse = Net:FindFirstChild(
                    "RE/CombatService/ApplyImpulse"
                ) or Net:FindFirstChild("ApplyImpulse")

                if
                    RECombatServiceApplyImpulse and RECombatServiceApplyImpulse:IsA("RemoteEvent")
                then
                    table.insert(
                        ragdollRecoveryState.connections,
                        RECombatServiceApplyImpulse.OnClientEvent:Connect(function()
                            if isCharacterRagdolled() and HumanoidRootPart then
                                pcall(function()
                                    HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                                end)
                            end
                        end)
                    )
                end
            end
        end

        if Character then
            table.insert(
                ragdollRecoveryState.connections,
                Character.DescendantAdded:Connect(function()
                    if isCharacterRagdolled() then
                        removeActiveRagdollConstraints()
                    end
                end)
            )
        end

        local vector = Vector3.new(0, 0, 0)

        table.insert(
            ragdollRecoveryState.connections,
            services.RunService.Heartbeat:Connect(function()
                if isCharacterRagdolled() and HumanoidRootPart then
                    removeActiveRagdollConstraints()

                    if not isMovementControlledByAnotherFeature(1) then
                        local AssemblyLinearVelocity = HumanoidRootPart.AssemblyLinearVelocity

                        if (AssemblyLinearVelocity - vector).Magnitude > 40 then
                            if AssemblyLinearVelocity.Magnitude > 25 then
                                pcall(function()
                                    HumanoidRootPart.AssemblyLinearVelocity = AssemblyLinearVelocity.Unit
                                        * math.min(AssemblyLinearVelocity.Magnitude, 15)
                                end)
                            end
                        end

                        vector = AssemblyLinearVelocity
                    end
                elseif HumanoidRootPart then
                    vector = HumanoidRootPart.AssemblyLinearVelocity
                end
            end)
        )
        table.insert(
            ragdollRecoveryState.connections,
            services.LocalPlayer.CharacterAdded:Connect(function()
                task.wait(0.5)
                refreshCharacterReferences()
                restorePlayerControls()
                removeActiveRagdollConstraints()
            end)
        )
        restorePlayerControls()
        removeActiveRagdollConstraints()
    end

    ragdollController = {
        Enable = function()
            if ragdollRecoveryState.active then
                return
            end

            DUEL_CONFIG.ANTI_RAGDOLL_ENABLED = true
            enableRagdollRecovery()

            if services.LocalPlayer.Character then
                task.spawn(function()
                    task.wait(0.4)

                    local Character2 = services.LocalPlayer.Character

                    if Character2 then
                        if Character2.Parent then
                            connectRagdollRecovery(Character2)
                        end
                    end
                end)
            end

            local signalConnection7 = services.LocalPlayer.CharacterAdded:Connect(
                function(character)
                    local initConnections = ragdollRecoveryState.initConnections or {}

                    for _, initConnection in ipairs(initConnections) do
                        if initConnection then
                            pcall(function()
                                initConnection:Disconnect()
                            end)
                        end
                    end

                    ragdollRecoveryState.initConnections = {}

                    local characterRootPart3 = findCharacterRootPart(character)

                    if characterRootPart3 then
                        characterRootPart3.Anchored = false
                    end

                    task.wait(0.4)
                    connectRagdollRecovery(character)
                end
            )

            table.insert(ragdollRecoveryState.connections, signalConnection7)
        end,
        Disable = function()
            if not ragdollRecoveryState.active then
                return
            end

            DUEL_CONFIG.ANTI_RAGDOLL_ENABLED = false
            ragdollRecoveryState.active = false

            for _, signalConnection8 in ipairs(ragdollRecoveryState.connections) do
                if signalConnection8 then
                    pcall(function()
                        signalConnection8:Disconnect()
                    end)
                end
            end

            local initConnections = ragdollRecoveryState.initConnections or {}

            for _, initConnection in ipairs(initConnections) do
                if initConnection then
                    pcall(function()
                        initConnection:Disconnect()
                    end)
                end
            end

            ragdollRecoveryState.connections = {}
            ragdollRecoveryState.initConnections = {}

            if Character and HumanoidRootPart then
                HumanoidRootPart.Anchored = false
            end

            Character = nil
            Humanoid = nil
            HumanoidRootPart = nil
        end,
    }
end

local function findNearestMedusaTarget()
    local Character = services.LocalPlayer.Character

    if not Character or not Character:FindFirstChild("HumanoidRootPart") then
        return nil
    end

    local HumanoidRootPart = Character.HumanoidRootPart
    local player22
    local huge = math.huge
    local AUTO_MEDUSA_RANGE = DUEL_CONFIG.AUTO_MEDUSA_RANGE or 10

    for _, player2 in ipairs(otherPlayers) do
        local Character2 = player2.Character

        if Character2 and Character2:FindFirstChild("HumanoidRootPart") then
            local Humanoid = Character2:FindFirstChild("Humanoid")

            if Humanoid and Humanoid.Health > 0 then
                local Magnitude = (HumanoidRootPart.Position - Character2.HumanoidRootPart.Position).Magnitude

                if Magnitude < huge and Magnitude <= AUTO_MEDUSA_RANGE then
                    huge = Magnitude
                    player22 = player2
                end
            end
        end
    end

    return player22
end

                return

            return
        end
    end

local medusaCounterInterval = 2

local function updateMedusaHitCounter()
    if not medusaState.enabled then
        return
    end

    local now = tick()

    if now - medusaState.lastCounterTime < medusaCounterInterval then
        return
    end

    local Character = services.LocalPlayer.Character

    if not Character then
        return
    end

    local Head = Character:FindFirstChild("Head")

    if not Head or not Head:IsA("BasePart") or Head.Transparency < 0.9 then
        return
    end

    local Humanoid = Character:FindFirstChildOfClass("Humanoid")

    if not Humanoid or Humanoid.Health <= 0 then
        return
    end

    local MedusaSHead = Character:FindFirstChild("Medusa's Head")

    if not MedusaSHead or not MedusaSHead:IsA("Tool") then
        MedusaSHead = services.LocalPlayer.Backpack
            and services.LocalPlayer.Backpack:FindFirstChild("Medusa's Head")

        if not MedusaSHead or not MedusaSHead:IsA("Tool") then
            return
        end

        Humanoid:EquipTool(MedusaSHead)
    end

    medusaState.lastCounterTime = now
    MedusaSHead:Activate()
end

    local MAX_TARGET_DISTANCE = DUEL_CONFIG.MAX_TARGET_DISTANCE or 100

local function startTargetLock()
    if targetLockState.isLocking then
        return
    end

    if featureControllers.AutoPlay and featureControllers.AutoPlay.StopAction then
        featureControllers.AutoPlay.StopAction()
    end

    local Character = services.LocalPlayer.Character

    if not Character then
        return
    end

    local Humanoid = Character:FindFirstChildOfClass("Humanoid")
    local localRootPart2 = getLocalRootPart()

    if not Humanoid or not localRootPart2 then
        return
    end

    local lockedPlayer = findNearestLockTarget()

    if not lockedPlayer then
        warn("Lock Target: No opponent found")

        return
    end

    targetLockState.lockedPlayer = lockedPlayer
    targetLockState.targetHRP = lockedPlayer.Character
        and lockedPlayer.Character:FindFirstChild("HumanoidRootPart")
    targetLockState.isLocking = true
    updateTargetLockButton(true)

    if targetLockState.connection then
        targetLockState.connection:Disconnect()
    end

    targetLockState.connection = services.RunService.PreSimulation:Connect(function()
        if not targetLockState.isLocking then
            return
        end

        if flingController and flingController.IsActive and flingController.IsActive() then
            return
        end

        local Character2 = services.LocalPlayer.Character

        if not Character2 then
            return
        end

        local Humanoid2 = Character2:FindFirstChildOfClass("Humanoid")
        local localRootPart3 = getLocalRootPart()

        if not Humanoid2 or not localRootPart3 then
            return
        end

        if Humanoid2.Health <= 0 then
            return
        end

        local state = Humanoid2:GetState()

        if
            state == Enum.HumanoidStateType.Dead
            or state == Enum.HumanoidStateType.Physics
            or state == Enum.HumanoidStateType.Ragdoll
        then
            return
        end

        local now = tick()
        local ok = not (
            not (now - targetLockState.lastTargetCheck > 0.1)
            and not not targetLockState.lockedPlayer
            and not not targetLockState.lockedPlayer.Character
            and not not targetLockState.targetHRP
            and not not targetLockState.targetHRP.Parent
        )

        if not ok then
            ok = not targetLockState.lockedPlayer.Character:FindFirstChild("Humanoid")
        end

        local ok2 = ok

        if not ok2 then
            ok2 = not not (
                targetLockState.lockedPlayer.Character:FindFirstChild("Humanoid").Health <= 0
            )
        end

        if ok2 then
            local lockedPlayer2 = findNearestLockTarget()

            if not lockedPlayer2 then
                stopTargetLock()

                return
            end

            targetLockState.lockedPlayer = lockedPlayer2
            targetLockState.targetHRP = lockedPlayer2.Character
                and lockedPlayer2.Character:FindFirstChild("HumanoidRootPart")
            targetLockState.lastTargetCheck = now
        end

        local targetHRP = targetLockState.targetHRP

        if not targetHRP or not targetHRP.Parent then
            stopTargetLock()

            return
        end

        local offset = targetHRP.Position - localRootPart3.Position

        if DUEL_CONFIG.MAX_TARGET_DISTANCE < offset.Magnitude then
            local lockedPlayer3 = findNearestLockTarget()

            if not lockedPlayer3 then
                stopTargetLock()

                return
            end

            targetLockState.lockedPlayer = lockedPlayer3
            targetLockState.targetHRP = lockedPlayer3.Character
                and lockedPlayer3.Character:FindFirstChild("HumanoidRootPart")
            targetLockState.lastTargetCheck = now
            offset = targetLockState.targetHRP.Position - localRootPart3.Position
        end

        localRootPart3.AssemblyLinearVelocity = offset.Unit * getNormalMovementSpeed()
    end)
end

end

local function setCharacterCollision(character9, canCollide)
    if not character9 then
        return
    end

    local Humanoid = character9:FindFirstChildOfClass("Humanoid")

    if Humanoid then
        Humanoid.HipHeight = canCollide

local function startFloating()
    if floatState.isFloating then
        return
    end

    floatState.isFloating = true

    local Character = services.LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")

    HipHeight = Humanoid and Humanoid.HipHeight or defaultHipHeight

    if floatState.connection then
        floatState.connection:Disconnect()
    end

    floatState.connection = services.RunService.Heartbeat:Connect(function(dt)
        local Character2 = services.LocalPlayer.Character

        if not Character2 then
            return
        end

        local FLOAT_HEIGHT = DUEL_CONFIG.FLOAT_HEIGHT or 7
        local FLOAT_SPEED = DUEL_CONFIG.FLOAT_SPEED or 35

        if HipHeight < FLOAT_HEIGHT then
            HipHeight = math.min(HipHeight + FLOAT_SPEED * dt, FLOAT_HEIGHT)
        elseif FLOAT_HEIGHT < HipHeight then
            HipHeight = math.max(HipHeight - FLOAT_SPEED * dt, FLOAT_HEIGHT)
        end

        setCharacterCollision(Character2, HipHeight)
    end)

    if floatState.button then
        floatState.button.BackgroundColor3 = Color3.fromRGB(50, 180, 100)
        floatState.button.Text = "FLOAT ON"
    end

local superJumpState = {
    enabled = false,
    charConn = nil,
    currentHeight = 2,
    jumpHoldBuffer = 0,
}
local currentHeight = 2
local jumpHoldBuffer = 0.15
local text2 = "SuperJumpControl"
local superJumpController = {
    Enable = function()
        if superJumpState.enabled then
            return
        end

        superJumpState.enabled = true
        superJumpState.currentHeight = currentHeight
        superJumpState.jumpHoldBuffer = 0

        if superJumpState.charConn then
            superJumpState.charConn:Disconnect()
        end

        superJumpState.charConn = services.LocalPlayer.CharacterAdded:Connect(function()
            task.wait(0.3)

            if superJumpState.enabled then
                superJumpState.currentHeight = currentHeight
                superJumpState.jumpHoldBuffer = 0
            end
        end)
        pcall(function()
            services.RunService:UnbindFromRenderStep(text2)
        end)
        services.RunService:BindToRenderStep(
            text2,
            Enum.RenderPriority.Input.Value + 20,
            function(value26)
                local Character = services.LocalPlayer.Character

                if not Character then
                    return
                end

                local Humanoid = Character:FindFirstChildOfClass("Humanoid")

                if not Humanoid or Humanoid.Health <= 0 then
                    return
                end

                if Humanoid.Jump then
                    superJumpState.jumpHoldBuffer = jumpHoldBuffer
                    Humanoid.Jump = false
                else
                    superJumpState.jumpHoldBuffer =
                        math.max(0, superJumpState.jumpHoldBuffer - value26)
                end

                local FLOAT_HEIGHT = currentHeight

                if superJumpState.jumpHoldBuffer > 0 then
                    FLOAT_HEIGHT = DUEL_CONFIG.FLOAT_HEIGHT or 7
                end

                local FLOAT_SPEED = DUEL_CONFIG.FLOAT_SPEED or 35

                if superJumpState.currentHeight < FLOAT_HEIGHT then
                    superJumpState.currentHeight =
                        math.min(superJumpState.currentHeight + FLOAT_SPEED * value26, FLOAT_HEIGHT)
                elseif FLOAT_HEIGHT < superJumpState.currentHeight then
                    superJumpState.currentHeight = currentHeight
                    Humanoid.HipHeight = currentHeight

                    local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

                    if HumanoidRootPart then
                        local raycastParams = RaycastParams.new()

                        raycastParams.FilterType = Enum.RaycastFilterType.Exclude
                        raycastParams.FilterDescendantsInstances = { Character }

                        local hit = workspace:Raycast(
                            HumanoidRootPart.Position,
                            Vector3.new(0, -500, 0),
                            raycastParams
                        )

                        if hit then
                            HumanoidRootPart.CFrame = CFrame.new(
                                hit.Position + Vector3.new(0, 3, 0)
                            ) * HumanoidRootPart.CFrame.Rotation
                        end

                        HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(
                            HumanoidRootPart.AssemblyLinearVelocity.X,
                            0,
                            HumanoidRootPart.AssemblyLinearVelocity.Z
                        )
                    end

                    return
                end

                Humanoid.HipHeight = superJumpState.currentHeight
            end
        )
    end,
    Disable = function()
        if not superJumpState.enabled then
            return
        end

        superJumpState.enabled = false
        pcall(function()
            services.RunService:UnbindFromRenderStep(text2)
        end)

        if superJumpState.charConn then
            superJumpState.charConn:Disconnect()
            superJumpState.charConn = nil
        end

        superJumpState.currentHeight = currentHeight
        superJumpState.jumpHoldBuffer = 0

        local Character = services.LocalPlayer.Character

        if Character then
            local Humanoid = Character:FindFirstChildOfClass("Humanoid")

            if Humanoid then
                Humanoid.HipHeight = currentHeight
            end
        end
    end,
}

    }
    local flingVelocityMultiplier = 999999
    local flingVerticalVelocity = 999999
    local flingVelocityNudge = 0.1

    local function findNearbyFlingTarget()
        local Character = services.LocalPlayer.Character

        if Character then
            if Character:FindFirstChild("HumanoidRootPart") then
                local HumanoidRootPart = Character.HumanoidRootPart
                local player42
                local huge = math.huge
                local FLING_RANGE = DUEL_CONFIG.FLING_RANGE or 8

                for _, player4 in ipairs(otherPlayers) do
                    local Character2 = player4.Character

                    if Character2 and Character2:FindFirstChild("HumanoidRootPart") then
                        local Humanoid = Character2:FindFirstChildOfClass("Humanoid")

                        if Humanoid and Humanoid.Health > 0 then
                            local Magnitude = (
                                HumanoidRootPart.Position - Character2.HumanoidRootPart.Position
                            ).Magnitude

                            if Magnitude < huge and Magnitude <= FLING_RANGE then
                                huge = Magnitude
                                player42 = player4
                            end
                        end
                    end
                end

                return player42, huge
            end
        end

        return nil, math.huge
    end
    local function disableLocalCharacterCollision()
        local Character = services.LocalPlayer.Character

        if not Character then
            return
        end

        for _, descendant in ipairs(Character:GetDescendants()) do
            if descendant:IsA("BasePart") then
                descendant.CanCollide = false
            end
        end
    end
    local function restoreLocalCharacterCollision()
        local Character = services.LocalPlayer.Character

        if not Character then
            return
        end

        for _, descendant in ipairs(Character:GetDescendants()) do
            if descendant:IsA("BasePart") then
                descendant.CanCollide = true
            end
        end
    end
    local function updateFlingButton(isFlinging)
        if not flingState.button or not flingState.button.Parent then
            return
        end

        if isFlinging then
            flingState.button.BackgroundColor3 = Color3.fromRGB(255, 50, 50)
            flingState.button.Text = "FLING ON"
        else
            flingState.button.BackgroundColor3 = interfaceTheme.Blue
            flingState.button.Text = "FLING"
        end
    end
    local function runFlingLoop()
        if flingState.loopRunning then
            return
        end

        flingState.loopRunning = true

        while flingState.isFlingActive do
            local Character = services.LocalPlayer.Character

            if Character then
                local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

                if HumanoidRootPart then
                    local Humanoid = Character:FindFirstChildOfClass("Humanoid")

                    if Humanoid and not (Humanoid.Health <= 0) then
                        if lockTarget.IsLocking() then
                            local result4 = lockTarget.GetTargetHRP()

                            if not result4 or not result4.Parent then
                                services.RunService.Heartbeat:Wait()

                                continue
                            end

                            if not flingState.isInRange then
                                flingState.isInRange = true
                            end

                            disableLocalCharacterCollision()

                            local offset7 = result4.Position - HumanoidRootPart.Position

                            if offset7.Magnitude > 0.1 then
                                HumanoidRootPart.AssemblyLinearVelocity = offset7.Unit
                                    * getNormalMovementSpeed()
                            end
                        else
                            local nearbyFlingTarget, result5 = findNearbyFlingTarget()

                            if not nearbyFlingTarget then
                                if flingState.isInRange then
                                    restoreLocalCharacterCollision()
                                    flingState.isInRange = false
                                end

                                services.RunService.Heartbeat:Wait()

                                continue
                            end

                            if (DUEL_CONFIG.FLING_RANGE or 8) < result5 then
                                if flingState.isInRange then
                                    restoreLocalCharacterCollision()
                                    flingState.isInRange = false
                                end

                                services.RunService.Heartbeat:Wait()

                                continue
                            end

                            if not flingState.isInRange then
                                flingState.isInRange = true
                            end

                            disableLocalCharacterCollision()
                        end

                        services.RunService.Heartbeat:Wait()

                        local AssemblyLinearVelocity = HumanoidRootPart.AssemblyLinearVelocity

                        HumanoidRootPart.AssemblyLinearVelocity = AssemblyLinearVelocity
                                * flingVelocityMultiplier
                            + Vector3.new(0, flingVerticalVelocity, 0)
                        services.RunService.RenderStepped:Wait()
                        HumanoidRootPart.AssemblyLinearVelocity = AssemblyLinearVelocity
                        services.RunService.Stepped:Wait()
                        HumanoidRootPart.AssemblyLinearVelocity = AssemblyLinearVelocity
                            + Vector3.new(0, flingVelocityNudge, 0)
                        flingVelocityNudge = -flingVelocityNudge
                    else
                        services.RunService.Heartbeat:Wait()
                    end
                else
                    services.RunService.Heartbeat:Wait()
                end
            else
                services.RunService.Heartbeat:Wait()
            end
        end

        if flingState.isInRange then
            restoreLocalCharacterCollision()
            flingState.isInRange = false
        end

        flingState.loopRunning = false
    end
    local function startFlinging()
        if flingState.isFlingActive then
            return
        end

        flingState.isFlingActive = true
        updateFlingButton(true)
        task.spawn(runFlingLoop)
    end
    local function stopFlinging()
        if not flingState.isFlingActive then
            return
        end

        flingState.isFlingActive = false
        updateFlingButton(false)
    end

            end
        end
    end)
    task.spawn(function()
        while Frame and Frame.Parent do

            end

                    end

            end

        desyncState.positionConnection:Disconnect()
        desyncState.positionConnection = nil

local function findNearestMeleeTarget()
    local Character = services.LocalPlayer.Character

    if not Character or not Character:FindFirstChild("HumanoidRootPart") then
        return nil
    end

    local HumanoidRootPart = Character.HumanoidRootPart
    local player52
    local huge = math.huge
    local MELEE_AIMBOT_RANGE = DUEL_CONFIG.MELEE_AIMBOT_RANGE or 20

    for _, player5 in ipairs(otherPlayers) do
        local Character2 = player5.Character

        if Character2 and Character2:FindFirstChild("HumanoidRootPart") then
            local Humanoid = Character2:FindFirstChild("Humanoid")

            if Humanoid and Humanoid.Health > 0 then
                local Magnitude = (HumanoidRootPart.Position - Character2.HumanoidRootPart.Position).Magnitude

                if Magnitude < huge and Magnitude <= MELEE_AIMBOT_RANGE then
                    huge = Magnitude
                    player52 = player5
                end
            end
        end
    end

    return player52
end

local function updateMeleeAiming(targetRoot, deltaTime)
    if not targetRoot then
        return
    end

    local Humanoid = targetRoot:FindFirstChildOfClass("Humanoid")

    if not Humanoid then
        return
    end

    if deltaTime then
        meleeAimState.originalAutoRotate = Humanoid.AutoRotate
        Humanoid.AutoRotate = false
    else
        if
            featureControllers.AutoPlay
            and featureControllers.AutoPlay.IsWalking
            and featureControllers.AutoPlay.IsWalking()
        then
            return
        end

        Humanoid.AutoRotate = meleeAimState.originalAutoRotate ~= nil
                and meleeAimState.originalAutoRotate
            or true
    end
end

    IsEnabled = function()
        return meleeAimState.enabled
    end,
    Enable = function()
        if meleeAimState.enabled then
            return
        end

        if
            DUEL_CONFIG.SPIN_BODY_ENABLED
            and bodySpinController
            and bodySpinController.IsSpinning
        then
            if bodySpinController.IsSpinning() then
                bodySpinController.StopSpinning()
            end
        end

        meleeAimState.enabled = true

        if meleeAimState.connection then

        end

        if meleeAimState.charConn then
            meleeAimState.charConn:Disconnect()

        end

        if Character then
            local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")
            local Humanoid = Character:FindFirstChildOfClass("Humanoid")

            if HumanoidRootPart then
                if Humanoid then
                    meleeAimState.originalAutoRotate = Humanoid.AutoRotate
                end

                createAimAlignment(HumanoidRootPart)
            end
        end

        meleeAimState.charConn = services.LocalPlayer.CharacterAdded:Connect(function(character)
            task.wait(0.5)

            if meleeAimState.enabled then
                meleeTargetCache.cachedTarget = nil
                meleeTargetCache.lastCheck = 0

                local HumanoidRootPart = character:FindFirstChild("HumanoidRootPart")
                local Humanoid = character:FindFirstChildOfClass("Humanoid")

                if HumanoidRootPart then
                    if Humanoid then
                        meleeAimState.originalAutoRotate = Humanoid.AutoRotate
                    end

                    createAimAlignment(HumanoidRootPart)
                end
            end
        end)
        meleeAimState.connection = services.RunService.RenderStepped:Connect(function()
            if not meleeAimState.enabled then
                return
            end

            if not aimAlignmentState.alignOrient or not aimAlignmentState.alignOrient.Parent then
                return
            end

            local Character2 = services.LocalPlayer.Character

            if not Character2 then
                return
            end

            local HumanoidRootPart = Character2:FindFirstChild("HumanoidRootPart")
            local Humanoid = Character2:FindFirstChildOfClass("Humanoid")

            if not HumanoidRootPart or not Humanoid then
                return
            end

            if
                featureControllers.AutoPlay
                and featureControllers.AutoPlay.IsWalking
                and featureControllers.AutoPlay.IsWalking()
            then
                aimAlignmentState.alignOrient.Enabled = false

                return
            end

            local now = tick()
            local ok = not (
                not (meleeTargetCache.recheckInterval < now - meleeTargetCache.lastCheck)
                and not not meleeTargetCache.cachedTarget
                and not not meleeTargetCache.cachedTarget.Character
            )

            if not ok then
                ok = not meleeTargetCache.cachedTarget.Character:FindFirstChild("HumanoidRootPart")
            end

            local ok2 = ok

            if not ok2 then
                ok2 = not meleeTargetCache.cachedTarget.Character:FindFirstChild("Humanoid")
            end

            local ok3 = ok2

            if not ok3 then
                ok3 = not not (
                    meleeTargetCache.cachedTarget.Character:FindFirstChild("Humanoid").Health <= 0
                )
            end

            if ok3 then
                meleeTargetCache.cachedTarget = findNearestMeleeTarget()
                meleeTargetCache.lastCheck = now
            end

            local cachedTarget = meleeTargetCache.cachedTarget

            if not cachedTarget then
                if not Humanoid.AutoRotate then
                    Humanoid.AutoRotate = true
                end

                aimAlignmentState.alignOrient.Enabled = false

                return
            end

            local Character3 = cachedTarget.Character

            if not Character3 then
                if not Humanoid.AutoRotate then
                    Humanoid.AutoRotate = true
                end

                aimAlignmentState.alignOrient.Enabled = false

                return
            end

            local HumanoidRootPart2 = Character3:FindFirstChild("HumanoidRootPart")

            if not HumanoidRootPart2 then
                if not Humanoid.AutoRotate then
                    Humanoid.AutoRotate = true
                end

                aimAlignmentState.alignOrient.Enabled = false

                return
            end

            local Position = HumanoidRootPart.Position
            local vector2 =
                Vector3.new(HumanoidRootPart2.Position.X, Position.Y, HumanoidRootPart2.Position.Z)

            if (vector2 - Position).Magnitude < 0.5 then
                if not Humanoid.AutoRotate then
                    Humanoid.AutoRotate = true
                end

                aimAlignmentState.alignOrient.Enabled = false

                return
            end

            if Humanoid.AutoRotate then
                Humanoid.AutoRotate = false
            end

            aimAlignmentState.alignOrient.CFrame = CFrame.lookAt(Position, vector2)
            aimAlignmentState.alignOrient.Enabled = true
        end)
    end,
    Disable = function()
        meleeAimState.enabled = false

        if meleeAimState.connection then
            meleeAimState.connection:Disconnect()
            meleeAimState.connection = nil
        end

        if meleeAimState.charConn then

            meleeAimState.charConn = nil
        end

        meleeTargetCache.lastCheck = 0

            updateMeleeAiming(Character, false)
        end

    end,

    local playerCollisionState = {
        enabled = false,
        originalCanCollide = {},
        connections = {},
    }

    local function disableCharacterCollision(character11, shouldCollide)
        if not character11 or character11.Parent == nil then
            return
        end

        for _, descendant in ipairs(character11:GetDescendants()) do
            if descendant:IsA("BasePart") then
                if shouldCollide and playerCollisionState.originalCanCollide[descendant] == nil then
                    playerCollisionState.originalCanCollide[descendant] = descendant.CanCollide
                end

                if descendant.CanCollide then
                    descendant.CanCollide = false
                end
            end
        end
    end
    local function rememberCharacterCollision(character12)
        if not character12 then
            return
        end

        for _, descendant in ipairs(character12:GetDescendants()) do
            if
                descendant:IsA("BasePart")
                and playerCollisionState.originalCanCollide[descendant] ~= nil
            then
                pcall(function()
                    descendant.CanCollide = playerCollisionState.originalCanCollide[descendant]
                end)
                playerCollisionState.originalCanCollide[descendant] = nil
            end
        end
    end
    local function restorePlayerCollisions()
        for key5, canCollide2 in pairs(playerCollisionState.originalCanCollide) do
            if key5 and key5.Parent and key5:IsA("BasePart") then
                pcall(function()
                    key5.CanCollide = canCollide2
                end)
            end
        end

        playerCollisionState.originalCanCollide = {}
    end
    local function updatePlayerCollisions(shouldCollide2)
        local isMatching4 = shouldCollide2 == nil or shouldCollide2

        for _, player6 in ipairs(otherPlayers) do
            if player6 and player6.Character then
                disableCharacterCollision(player6.Character, isMatching4)
            end
        end
    end
    local function connectCharacterCollisionUpdates(character13)
        if not character13 or not playerCollisionState.enabled then
            return
        end

        disableCharacterCollision(character13, true)
        character13.DescendantAdded:Connect(function(descendant)
            if playerCollisionState.enabled and descendant:IsA("BasePart") then
                if playerCollisionState.originalCanCollide[descendant] == nil then
                    playerCollisionState.originalCanCollide[descendant] = descendant.CanCollide
                end

                if descendant.CanCollide then
                    descendant.CanCollide = false
                end
            end
        end)
    end

    local playerCollisionController = {
        Enable = function()
            if playerCollisionState.enabled then
                return
            end

            playerCollisionState.enabled = true
            updatePlayerCollisions(true)
            playerCollisionState.connections.frameLoop = services.RunService.PreSimulation:Connect(
                function()
                    if playerCollisionState.enabled then
                        updatePlayerCollisions(false)

                        return
                    end
                end
            )
            playerCollisionState.connections.playerAdded = services.Players.PlayerAdded:Connect(
                function(player)
                    local function applyCollisionAfterRespawn(character)
                        task.wait(0.2)

                        if playerCollisionState.enabled then
                            connectCharacterCollisionUpdates(character, player)
                        end
                    end

                    player.CharacterAdded:Connect(applyCollisionAfterRespawn)

                    if player.Character then
                        task.defer(function()
                            task.wait(0.2)

                            if playerCollisionState.enabled then
                                connectCharacterCollisionUpdates(player.Character, player)
                            end
                        end)
                    end
                end
            )
            playerCollisionState.connections.playerRemoving = services.Players.PlayerRemoving:Connect(
                function(player)
                    if player.Character then
                        rememberCharacterCollision(player.Character)
                    end
                end
            )

            for _, player7 in ipairs(otherPlayers) do
                player7.CharacterAdded:Connect(function(character)
                    task.wait(0.2)

                    if playerCollisionState.enabled then
                        connectCharacterCollisionUpdates(character, player7)
                    end
                end)

                if player7.Character then
                    connectCharacterCollisionUpdates(player7.Character, player7)
                end
            end
        end,
        Disable = function()
            if not playerCollisionState.enabled then
                return
            end

            playerCollisionState.enabled = false

            if playerCollisionState.connections.frameLoop then
                playerCollisionState.connections.frameLoop:Disconnect()
                playerCollisionState.connections.frameLoop = nil
            end

            restorePlayerCollisions()

            for _, signalConnection14 in pairs(playerCollisionState.connections) do
                if signalConnection14 and signalConnection14.Disconnect then
                    signalConnection14:Disconnect()
                end
            end

        end,

    local function captureWalkingAnimations()
        local Character = services.LocalPlayer.Character

        if not Character then
            return false
        end

        walkingAnimationState.script = Character:FindFirstChild("Animate")

        if not walkingAnimationState.script then
            return false
        end

        walkingAnimationState.originalIds = {}
        walkingAnimationState.cachedObjects = {}

        for _, type_ in ipairs(walkingAnimationState.types) do
            local result6 = walkingAnimationState.script:FindFirstChild(type_)

            if result6 then
                walkingAnimationState.originalIds[type_] = {}
                walkingAnimationState.cachedObjects[type_] = {}

                for _, child in ipairs(result6:GetChildren()) do
                    if child:IsA("Animation") then
                        walkingAnimationState.originalIds[type_][child.Name] = child.AnimationId
                        walkingAnimationState.cachedObjects[type_][child.Name] = child
                    end
                end
            end
        end

        return true
    end
    local function suppressWalkingAnimations()
        if not walkingAnimationState.script then
            return
        end

        for _, cachedObject in pairs(walkingAnimationState.cachedObjects) do
            for _, value9 in pairs(cachedObject) do
                if value9 and value9.Parent then
                    value9.AnimationId = ""
                end
            end
        end

        local Character = services.LocalPlayer.Character

        if Character then
            local Humanoid = Character:FindFirstChildOfClass("Humanoid")

            if Humanoid then
                for _, value10 in ipairs(Humanoid:GetPlayingAnimationTracks()) do
                    value10:Stop(0)
                end
            end
        end
    end
    local function restoreWalkingAnimations()
        if not walkingAnimationState.script or not walkingAnimationState.originalIds then
            return
        end

        for key6, originalId in pairs(walkingAnimationState.originalIds) do
            local result7 = walkingAnimationState.cachedObjects[key6]

            if result7 then
                for k2, animationId in pairs(originalId) do
                    local result8 = result7[k2]

                    if result8 and result8.Parent then
                        result8.AnimationId = animationId
                    end
                end
            end
        end
    end

    local function isBaseGeometry(worldPart)

            return false

    local function enableBaseXray()
        if workspace:FindFirstChild("Plots") then
            if not isBaseXrayActive then
                isBaseXrayActive = true

                for _, descendant in ipairs(workspace:GetDescendants()) do
                    if
                        descendant:IsA("BasePart")
                        and descendant.Anchored
                        and descendant.CanCollide
                        and isBaseGeometry(descendant)
                    then
                        originalBaseTransparency[descendant] = descendant.LocalTransparencyModifier
                        descendant.LocalTransparencyModifier = 0.85
                    end
                end

                signalConnection = workspace.DescendantAdded:Connect(function(descendant)
                    if DUEL_CONFIG.XRAY_BASE_ENABLED then
                        if isBaseGeometry(descendant) then
                            originalBaseTransparency[descendant] =
                                descendant.LocalTransparencyModifier
                            descendant.LocalTransparencyModifier = 0.85
                        end
                    end
                end)

                return
            end
        end
    end
    local function disableBaseXray()
        isBaseXrayActive = false

        if signalConnection then
            signalConnection:Disconnect()
            signalConnection = nil
        end

        for key7, localTransparencyModifier in pairs(originalBaseTransparency) do
            if key7 then
                key7.LocalTransparencyModifier = localTransparencyModifier
            end
        end

        originalBaseTransparency = {}
    end

        local function clearDuplicateKeybinding(configurationKey2, keyCode)
            for _, value16 in ipairs({
                "KEYBIND_SPEED_CUSTOMIZER",
                "KEYBIND_AUTO_PLAY",
                "KEYBIND_LOCK_TARGET",
                "KEYBIND_SPIN_BODY",
                "KEYBIND_FLOAT",
                "KEYBIND_FLING",
                "KEYBIND_DROP_BRAINROT",
                "KEYBIND_TP_DOWN",
                "KEYBIND_DESYNC",
                "KEYBIND_SPEED_PRESET",
            }) do
                if value16 == keyCode then
                    continue
                end

                if DUEL_CONFIG[value16] == configurationKey2 then
                    return true, value16
                end
            end

            return false
        end
