local function callHook(name, ...)
    local callback = hooks[name]
    if type(callback) ~= "function" then
        warn(("[Ambitious reconstruction] `%s` was not recoverable from the trace."):format(name))
        return nil
    end

    local ok, first, second = pcall(callback, ...)
    if not ok then
        warn(("[Ambitious reconstruction] `%s` failed: %s"):format(name, tostring(first)))
        return nil
    end

    return first, second
end

local function create(className, properties)
    local object = Instance.new(className)
    local parent = properties.Parent

    for property, value in pairs(properties) do
        if property ~= "Parent" then
            object[property] = value
        end
    end

    object.Parent = parent
    return object
end

local function makeNumberRow(labelText, order, initialValue, onChanged)
    local row = makeBaseRow(labelText, order)
    local box = makeTextBox(row, initialValue, 102)
    local value = initialValue

    box.FocusLost:Connect(function()
        local parsed = tonumber(box.Text)
        if parsed and parsed >= 0 then
            value = parsed
            box.Text = tostring(parsed)
            onChanged(parsed)
        else
            box.Text = tostring(value)
        end
    end)

    return box
end

        enabled = nextValue == true
        TweenService:Create(button, TweenInfo.new(0.18), {

        }):Play()
        TweenService:Create(knob, TweenInfo.new(0.18, Enum.EasingStyle.Quad), {
            Position = enabled and UDim2.fromOffset(29, 3) or UDim2.fromOffset(3, 3),
        }):Play()
        onChanged(enabled)
    end

    button.MouseButton1Click:Connect(function()
        setEnabled(not enabled)
    end)

    return {
        Button = button,
        Get = function()
            return enabled
        end,
        Set = setEnabled,
    }
end

local function makeKeybindRow(labelText, order, initialKey, onChanged)
    local row = makeBaseRow(labelText, order)
    local box = makeTextBox(row, initialKey, 56)
    box.PlaceholderText = "..."

    box.FocusLost:Connect(function()
        local key = box.Text:upper():gsub("%s+", "")
        if #key > 0 and Enum.KeyCode[key] then
            box.Text = key
            onChanged(key)
        else
            box.Text = initialKey
        end
    end)

    return box
end

local function refreshCharacter(nextCharacter)
    character = nextCharacter or localPlayer.Character
    if not character then
        humanoid = nil
        rootPart = nil
        speedBooster = nil
        return
    end

    humanoid = character:WaitForChild("Humanoid", 10)
    rootPart = character:WaitForChild("HumanoidRootPart", 10)

    local oldBooster = character:FindFirstChild("AmbitiousSpeedBooster")
    if oldBooster then
        oldBooster:Destroy()
    end

    if not rootPart then
        return
    end

    speedBooster = create("Part", {
        Name = "AmbitiousSpeedBooster",
        Size = Vector3.new(1, 1, 1),
        CFrame = rootPart.CFrame,
        Transparency = 1,
        CanCollide = false,
        CanTouch = false,
        CanQuery = false,
        Massless = true,
        Parent = character,
    })

    create("Weld", {
        Part0 = rootPart,
        Part1 = speedBooster,
        C0 = CFrame.new(),
        Parent = speedBooster,
    })

    if antiRagdollEnabled and humanoid then
        for _, state in ipairs({
            Enum.HumanoidStateType.FallingDown,
            Enum.HumanoidStateType.Ragdoll,
            Enum.HumanoidStateType.Physics,
        }) do
            pcall(humanoid.SetStateEnabled, humanoid, state, false)
        end
    end

    if antiDieEnabled and humanoid then
        pcall(humanoid.SetStateEnabled, humanoid, Enum.HumanoidStateType.Dead, false)
    end
end

local function applyAntiRagdoll(enabled)
    antiRagdollEnabled = enabled
    if not humanoid then
        return
    end

    for _, state in ipairs({
        Enum.HumanoidStateType.FallingDown,
        Enum.HumanoidStateType.Ragdoll,
        Enum.HumanoidStateType.Physics,
    }) do
        pcall(humanoid.SetStateEnabled, humanoid, state, not enabled)
    end

    if enabled then
        humanoid.PlatformStand = false
    end
end

local function applyUnwalk(enabled)
    if not character or not humanoid then
        return
    end

    local animate = character:FindFirstChild("Animate")
    if enabled then
        unwalkRestore = {
            Animate = animate,
            AnimateWasDisabled = animate and animate.Disabled or false,
        }

        local animator = humanoid:FindFirstChildOfClass("Animator")
        if animator then
            for _, animationTrack in ipairs(animator:GetPlayingAnimationTracks()) do
                animationTrack:Stop()
            end
        end

        if animate then
            animate.Disabled = true
        end
    elseif unwalkRestore then
        if unwalkRestore.Animate and unwalkRestore.Animate.Parent then
            unwalkRestore.Animate.Disabled = unwalkRestore.AnimateWasDisabled
        end
        unwalkRestore = nil
    end
end

local function applyAntiDie(enabled)
    antiDieEnabled = enabled
    if humanoid then
        -- SetStateEnabled() and a Died connection were observed; the exact state
        -- argument was not logged. Dead is the closest semantic reconstruction.
        pcall(humanoid.SetStateEnabled, humanoid, Enum.HumanoidStateType.Dead, not enabled)
    end
end

local function optimiseDescendant(descendant)
    if descendant:IsA("BasePart") then
        if not originalPartVisuals[descendant] then
            originalPartVisuals[descendant] = {
                Material = descendant.Material,
                Reflectance = descendant.Reflectance,
                CastShadow = descendant.CastShadow,
            }
        end

        descendant.Material = Enum.Material.Plastic
        descendant.Reflectance = 0
        descendant.CastShadow = false
    elseif CONFIG.destructiveAntiLag and descendant:IsA("Accessory") then
        descendant:Destroy()
    end
end

local function applyAntiLag(enabled)
    antiLagEnabled = enabled

    if enabled then
        originalLighting = {
            GlobalShadows = Lighting.GlobalShadows,
            FogEnd = Lighting.FogEnd,
            FogStart = Lighting.FogStart,
            Brightness = Lighting.Brightness,
            EnvironmentDiffuseScale = Lighting.EnvironmentDiffuseScale,
            EnvironmentSpecularScale = Lighting.EnvironmentSpecularScale,
            ShadowSoftness = Lighting.ShadowSoftness,
            ExposureCompensation = Lighting.ExposureCompensation,
        }

        local terrain = workspace:FindFirstChildOfClass("Terrain")
        if terrain then
            originalTerrain = {
                WaterWaveSize = terrain.WaterWaveSize,
                WaterWaveSpeed = terrain.WaterWaveSpeed,
                WaterReflectance = terrain.WaterReflectance,
                WaterTransparency = terrain.WaterTransparency,
            }
            terrain.WaterWaveSize = 0
            terrain.WaterWaveSpeed = 0
            terrain.WaterReflectance = 0
            terrain.WaterTransparency = 0
        end

        Lighting.GlobalShadows = false
        Lighting.FogEnd = 1e10
        Lighting.FogStart = 0
        Lighting.Brightness = 1
        Lighting.EnvironmentDiffuseScale = 0
        Lighting.EnvironmentSpecularScale = 0
        Lighting.ShadowSoftness = 0
        Lighting.ExposureCompensation = 0

        for _, descendant in ipairs(workspace:GetDescendants()) do
            optimiseDescendant(descendant)
        end

        antiLagDescendantConnection = workspace.DescendantAdded:Connect(function(descendant)
            if antiLagEnabled then
                optimiseDescendant(descendant)
            end
        end)
    else
        if antiLagDescendantConnection then
            antiLagDescendantConnection:Disconnect()
            antiLagDescendantConnection = nil
        end

        for part, values in pairs(originalPartVisuals) do
            if part.Parent then
                part.Material = values.Material
                part.Reflectance = values.Reflectance
                part.CastShadow = values.CastShadow
            end
        end

        for property, value in pairs(originalLighting) do
            pcall(function()
                Lighting[property] = value
            end)
        end

        local terrain = workspace:FindFirstChildOfClass("Terrain")
        if terrain then
            for property, value in pairs(originalTerrain) do
                terrain[property] = value
            end
        end
    end
end

local function applyNoPlayerCollision(enabled)
    noCollisionEnabled = enabled

    if noCollisionConnection then
        noCollisionConnection:Disconnect()
        noCollisionConnection = nil
    end

    if not enabled then
        for part, canCollide in pairs(originalCollision) do
            if part.Parent then
                part.CanCollide = canCollide
            end
        end
        return
    end

    noCollisionConnection = RunService.Stepped:Connect(function()
        if not noCollisionEnabled or not character then
            return
        end

        for _, descendant in ipairs(character:GetDescendants()) do
            if descendant:IsA("BasePart") then
                if originalCollision[descendant] == nil then
                    originalCollision[descendant] = descendant.CanCollide
                end
                descendant.CanCollide = false
            end
        end
    end)
end

local lastStealComplete = false
local function setStealProgress(percent)
    percent = math.clamp(tonumber(percent) or 0, 0, 100)
    stealing = percent > 0
    progressOuter.Visible = stealing
    progressText.Text = ("%d%%"):format(math.floor(percent + 0.5))

    TweenService:Create(progressFill, TweenInfo.new(0.08, Enum.EasingStyle.Linear), {
        Size = UDim2.new(percent / 100, 0, 1, 0),
    }):Play()

    local complete = percent >= 100
    if complete and not lastStealComplete and instantKickAfterSteal then
        localPlayer:Kick("Steal completed")
    end
    lastStealComplete = complete
end

local function selectedSpeed()
    if carpetModeEnabled then
        return stealing and CONFIG.lowCarpetSpeed or CONFIG.carpetSpeed
    end
    return stealing and CONFIG.stealSpeed or CONFIG.normalSpeed
end

-- Observed movement engine: write Humanoid.MoveDirection * selected speed to the
-- welded part while preserving the assembly's vertical velocity.
track(RunService.RenderStepped:Connect(function()
    if not alive then
        return
    end

    local camera = workspace.CurrentCamera
    if camera then
        local viewport = camera.ViewportSize
        windowScale.Scale = math.min(1, (viewport.X - 24) / 430, (viewport.Y - 24) / 540)
    end

    local absolutePosition = window.AbsolutePosition
    local absoluteSize = window.AbsoluteSize
    progressOuter.Position = UDim2.fromOffset(
        absolutePosition.X + absoluteSize.X / 2,
        absolutePosition.Y + absoluteSize.Y + 8
    )
    progressOuter.Size = UDim2.fromOffset(absoluteSize.X, 36 * windowScale.Scale)

    local progressGetter = hooks.getStealProgress
    if type(progressGetter) == "function" then
        local ok, progress = pcall(progressGetter)
        if ok and progress ~= nil then
            setStealProgress(progress)
        end
    end

    if speedBooster and speedBooster.Parent and humanoid then
        local direction = humanoid.MoveDirection
        local vertical = speedBooster.AssemblyLinearVelocity.Y
        local speed = selectedSpeed()
        speedBooster.AssemblyLinearVelocity = Vector3.new(
            direction.X * speed,
            vertical,
            direction.Z * speed
        )
    end
end))

-- Reconstructed infinite-jump trigger. The trace created a welded
-- AmbitiousJumpProxy and set its vertical assembly velocity to exactly 50.
track(UserInputService.JumpRequest:Connect(function()
    if not infiniteJumpEnabled or not character or not rootPart then
        return
    end

    local oldProxy = character:FindFirstChild("AmbitiousJumpProxy")
    if oldProxy then
        oldProxy:Destroy()
    end

    local proxy = create("Part", {
        Name = "AmbitiousJumpProxy",
        Size = Vector3.new(1, 1, 1),
        CFrame = rootPart.CFrame,
        Transparency = 1,
        CanCollide = false,
        CanTouch = false,
        CanQuery = false,
        Massless = true,
        Parent = character,
    })
    create("Weld", {
        Part0 = rootPart,
        Part1 = proxy,
        C0 = CFrame.new(),
        Parent = proxy,
    })

    local velocity = proxy.AssemblyLinearVelocity
    proxy.AssemblyLinearVelocity = Vector3.new(velocity.X, 50, velocity.Z)

    task.delay(0.2, function()
        if proxy.Parent then
            proxy:Destroy()
        end
    end)
end))

        end

-- Window dragging.
do
    local dragging = false
    local dragInput
    local dragStart
    local startPosition

    track(header.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = window.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end))

    track(header.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end))

    track(UserInputService.InputChanged:Connect(function(input)
        if dragging and input == dragInput then
            local delta = input.Position - dragStart
            window.Position = UDim2.new(
                startPosition.X.Scale,
                startPosition.X.Offset + delta.X,
                startPosition.Y.Scale,
                startPosition.Y.Offset + delta.Y
            )
        end
    end))
end

local function cleanUp()
    if not alive then
        return
    end
    alive = false

    applyNoPlayerCollision(false)
    if antiLagEnabled then
        applyAntiLag(false)
    end
    applyUnwalk(false)
    applyAntiRagdoll(false)
    applyAntiDie(false)

    for _, connection in ipairs(connections) do
        if connection.Connected then
            connection:Disconnect()
        end
    end

    if speedBooster and speedBooster.Parent then
        speedBooster:Destroy()
    end
end

    cleanUp()
    screenGui:Destroy()
end)
