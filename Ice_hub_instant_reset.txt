-- // [4] UTILITY FUNCTIONS //
local function instaRespawn()
	if not LocalPlayer.Character then return end
	local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
	if humanoid then
		humanoid.Health = 0
	end
end

local function toggleBalloonReset()
	ResetOnBalloon = not ResetOnBalloon
	BalloonToggle.Text = "Reset on Balloon: " .. (ResetOnBalloon and "ON" or "OFF")
	BalloonToggle.TextColor3 = ResetOnBalloon and CONFIG.Colors.Text or CONFIG.Colors.ToggleOff
end

local function createDraggable(frame)
	frame.Active = true
	frame.Draggable = true
end

-- // [6] FUNCTIONALITY //
local function setupFunctionality()
	-- Insta Respawn Button
	Buttons.Respawn.MouseButton1Click:Connect(function()
		instaRespawn()
	end)

	-- Balloon Toggle
	Buttons.Balloon.MouseButton1Click:Connect(function()
		toggleBalloonReset()
	end)

	-- Close Button (hides GUI - easy to re-open via code if needed)
	Buttons.Close.MouseButton1Click:Connect(function()
		ScreenGui.Enabled = false
	end)

	-- Global Keybind R
	UserInputService.InputBegan:Connect(function(input, gameProcessed)
		if gameProcessed then return end
		if input.KeyCode == Enum.KeyCode.R then
			instaRespawn()
		end
	end)
