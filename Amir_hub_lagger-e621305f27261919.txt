local function updateGradientRotation()
    gradientRotation = (gradientRotation + 2) % 360
    if Gradient then
        Gradient.Rotation = gradientRotation
    end
end

local function createSlider(yPos, prefix, initialPercent, isPercentage, callback)
    local container = Instance.new("Frame")
    container.Size = CONFIG.Size.FillBar
    container.Position = UDim2.new(0.5, -87, 0, yPos)
    container.BackgroundTransparency = 1
    container.Parent = MainFrame
    local TextLabel = Instance.new("TextLabel")
    TextLabel.Size = UDim2.new(1, 0, 0, 12)
    TextLabel.Position = UDim2.new(0, 0, 0, 0)
    TextLabel.BackgroundTransparency = 1
    TextLabel.TextSize = CONFIG.Settings.LabelSize
    TextLabel.TextColor3 = CONFIG.Colors.TextDim
    TextLabel.Font = CONFIG.Settings.Font
    TextLabel.TextXAlignment = Enum.TextXAlignment.Left
    TextLabel.Parent = container
    local BarBg = Instance.new("Frame")
    BarBg.Size = CONFIG.Size.FillBar
    BarBg.Position = UDim2.new(0, 0, 0, 18)
    BarBg.BackgroundColor3 = CONFIG.Colors.BarBackground
    BarBg.BorderSizePixel = 0
    BarBg.Parent = container
    local BarCorner = Instance.new("UICorner")
    BarCorner.CornerRadius = UDim.new(1, 0)
    BarCorner.Parent = BarBg
    local Fill = Instance.new("Frame")
    Fill.Size = UDim2.new(initialPercent, 0, 1, 0)
    Fill.BackgroundColor3 = CONFIG.Colors.Fill
    Fill.BorderSizePixel = 0
    Fill.Parent = BarBg
    local FillCorner = Instance.new("UICorner")
    FillCorner.CornerRadius = UDim.new(1, 0)
    FillCorner.Parent = Fill
    local Knob = Instance.new("Frame")
    Knob.Size = CONFIG.Size.Knob
    Knob.Position = UDim2.new(1, 0, 0.5, 0)
    Knob.AnchorPoint = Vector2.new(0.5, 0.5)
    Knob.BackgroundColor3 = CONFIG.Colors.Knob
    Knob.BorderSizePixel = 0
    Knob.Parent = Fill
    local KnobCorner = Instance.new("UICorner")
    KnobCorner.CornerRadius = UDim.new(1, 0)
    KnobCorner.Parent = Knob
    local function updateVisuals(percent)
        Fill.Size = UDim2.new(percent, 0, 1, 0)
        local displayValue = ""
        if isPercentage then
            displayValue = math.floor(percent * 100) .. "%"
        else
            displayValue = string.format("%.1f", CONFIG.Settings.MinSpeed + (percent * (CONFIG.Settings.MaxSpeed - CONFIG.Settings.MinSpeed)))
        end
        TextLabel.Text = prefix .. displayValue
    end
    updateVisuals(initialPercent)
    local dragging = false
    local function updateSlider(input)
        local pos = math.clamp((input.Position.X - BarBg.AbsolutePosition.X) / BarBg.AbsoluteSize.X, 0, 1)
        updateVisuals(pos)
        callback(pos)
    end
    BarBg.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            updateSlider(input)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            updateSlider(input)
        end
    end)
end
createSlider(32, CONFIG.Text.LagPrefix, 0, true, function(val)
    currentLagAmount = val * CONFIG.Settings.MaxLag
end)
createSlider(68, CONFIG.Text.SpeedPrefix, 0, false, function(val)
    currentSpeedAmount = CONFIG.Settings.MinSpeed + (val * (CONFIG.Settings.MaxSpeed - CONFIG.Settings.MinSpeed))
end)

RunService.RenderStepped:Connect(function()
    updateGradientRotation()
    if not enabled then return end
    -- Velocity Speed Bypass logic
    local char = LocalPlayer.Character
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hrp and hum and hum.MoveDirection.Magnitude > 0 then
            local dir = hum.MoveDirection.Unit
            hrp.AssemblyLinearVelocity = Vector3.new(
                dir.X * currentSpeedAmount,
                hrp.AssemblyLinearVelocity.Y,
                dir.Z * currentSpeedAmount
            )
        end
    end
    -- Thread Freezing Lag logic
    if currentLagAmount > 0 then
        local t = tick()
        while tick() - t < currentLagAmount do end
    end
end)
