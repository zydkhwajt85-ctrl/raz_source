local antiLagActive = false
local function optimizePart(part)
	if part:IsA("BasePart") and not part:IsA("Accessory") then
		part.Material = Enum.Material.Plastic
		part.Reflectance = 0
		part.CastShadow = false
		part.LocalTransparencyModifier = 0
	elseif part:IsA("Decal") or part:IsA("Texture") then
		part.Transparency = 1
	elseif part:IsA("Fire") or part:IsA("Smoke") or part:IsA("Sparkles") then
		part.Enabled = false
	elseif part:IsA("ParticleEmitter") then
		part.Enabled = false
	elseif part:IsA("Sound") then
		part.Volume = 0
	end
end
local function applyOptimizations()
	Lighting.GlobalShadows = false
	Lighting.FogEnd = 9000000000
	Lighting.Brightness = 1
	Lighting.EnvironmentDiffuseScale = 0
	Lighting.EnvironmentSpecularScale = 0
	for _, effect in pairs(Lighting:GetChildren()) do
		if effect:IsA("BloomEffect") or effect:IsA("BlurEffect") or effect:IsA("ColorCorrectionEffect") then
			effect.Enabled = false
		end
	end
	for _, descendant in pairs(Workspace:GetDescendants()) do
		optimizePart(descendant)
	end
	Workspace.DescendantAdded:Connect(optimizePart)
end

local rotatingGradients = {borderGradient, discordGradient}
local rotatingFrames = {}
RunService.Heartbeat:Connect(function(deltaTime)
	-- Rotation des gradients
	for _, grad in pairs(rotatingGradients) do
		if grad then
			grad.Rotation = (grad.Rotation + deltaTime * 12) % 360
		end
	end
	-- Rotation du gradient Discord
	if discordGradient then
		discordGradient.Rotation = (discordGradient.Rotation + deltaTime * 120) % 360
	end
	-- Animation des particules du panneau
	for _, p in ipairs(animatedParticles) do
		p.y = p.y + p.speedY * deltaTime
		p.x = p.x + p.speedX * deltaTime
		if p.y > 1.05 then
			p.y = -0.05
			p.x = math.random()
		end
		if p.x < -0.02 then
			p.x = 1.02
		elseif p.x > 1.02 then
			p.x = -0.02
		end
		if p.frame then
			p.frame.BackgroundTransparency = math.clamp(p.baseT + math.sin(tick() * 2.2 + p.phase) * 0.15, 0, 0.88)
			p.frame.Position = UDim2.new(p.x, 0, p.y, 0)
		end
	end
	-- Animation des particules d'en-tête
	for _, p in ipairs(headerParticles) do
		p.y = p.y + p.speedY * deltaTime
		p.x = p.x + p.speedX * deltaTime
		if p.y > 1.1 then
			p.y = -0.1
			p.x = math.random()
		end
		if p.x < -0.02 then
			p.x = 1.02
		elseif p.x > 1.02 then
			p.x = -0.02
		end
		if p.frame then
			p.frame.BackgroundTransparency = math.clamp(p.baseT + math.sin(tick() * 4.5 + p.phase) * 0.18, 0, 0.85)
			p.frame.Position = UDim2.new(p.x, 0, p.y, 0)
		end
	end
end)
