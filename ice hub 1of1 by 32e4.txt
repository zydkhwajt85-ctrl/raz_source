		do
			local originalObserveOneOfOne = nil

			originalObserveOneOfOne = hookfunction(brainrotCards.ObserveOneOfOne, function(brainrot, onChanged)
				return originalObserveOneOfOne(brainrot, function(isOneOfOne)
					if appliedNames[brainrot.Index] then
						onChanged(true)
					else
						onChanged(isOneOfOne)
					end
				end)
			end)
		end

		local originalApplyOneOfOne = nil

		originalApplyOneOfOne = hookfunction(brainrotCards.ApplyOneOfOne, function(card, isOneOfOne, ...)
			local spacer = card:FindFirstChild("Spacer")

			if spacer and spacer:FindFirstChild("Title") then
				if appliedNames[spacer.Title.Text] then
					isOneOfOne = true
				end
			end

			return originalApplyOneOfOne(card, isOneOfOne, ...)
		end)
	end

	do
		local originalPopulate = nil

		originalPopulate = hookfunction(overheadController.Populate, function(controller, animal)
			originalPopulate(controller, animal)

			if animal.Overhead then
				local oneOfOneBanner = animal.Overhead:FindFirstChild("1OF1Banner")

				if oneOfOneBanner then
					local displayName = animal.Overhead:FindFirstChild("DisplayName")

					if displayName and displayName:IsA("TextLabel") then
						oneOfOneBanner.Visible = appliedNames[displayName.Text] or false
					end
				end

	refreshBanners = function()
		do
			for _, descendant in ipairs(world:GetDescendants()) do
				if descendant.Name == "1OF1Banner" and descendant:IsA("GuiObject") then
					local parent = descendant.Parent
					parent = parent and parent:FindFirstChild("DisplayName")

					if parent and parent:IsA("TextLabel") then
						descendant.Visible = appliedNames[parent.Text] or false
					end
				end
			end

				if child:IsA("Part") and child.Name == "FastOverheadTemplate" then

							oneOfOneBanner.Visible = appliedNames[displayName.Text] or false
						end
					end
				end
			end

		end

tween = function(instance, duration, properties)
	do
		tweenService
			:Create(instance, TweenInfo.new(duration, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), properties)
			:Play()
		return
	end

						row.toggle()
					end
				end
			end)

			selectNoneButton.MouseButton1Click:Connect(function()
				for _, row in ipairs(selectionRows) do
					if row.selected then
						row.toggle()
					end
				end
			end)

			applyButton.MouseButton1Click:Connect(function()
				appliedNames = {}

				for _, row in ipairs(selectionRows) do
					if row.selected then
						appliedNames[row.model.Name] = true
					end
				end

				refreshBanners()
				local appliedCount = 0

				for k in pairs(appliedNames) do
					appliedCount += 1
				end

				applyButton.Text = "✓ APPLIED (" .. tostring(appliedCount) .. ")"
				tween(applyPanel, 0.12, { BackgroundColor3 = Color3.fromRGB(8, 55, 52) })

				task.delay(0.85, function()
					do
						if applyButton and applyButton.Parent then
							applyButton.Text = "✓ APPLY 1 OF 1"
							tween(applyPanel, 0.12, { BackgroundColor3 = Color3.fromRGB(10, 30, 60) })
						end

						return
					end
				end)
			end)
		end

		updateSelectionCount()

		return
	end
end
