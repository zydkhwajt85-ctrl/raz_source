                        local child32
                        local huge2 = math.huge

                        for _, child in ipairs(Plots:GetChildren()) do
                            if isLocalPlayerPlot(child.Name) then
                                continue
                            end

                            local AnimalPodiums = child:FindFirstChild("AnimalPodiums")

                            if not AnimalPodiums then
                                continue
                            end

                            for _, child2 in ipairs(AnimalPodiums:GetChildren()) do
                                pcall(function()
                                    local Base = child2:FindFirstChild("Base")
                                    local Spawn = Base and Base:FindFirstChild("Spawn")

                                    if Spawn then
                                        local PromptAttachment =
                                            Spawn:FindFirstChild("PromptAttachment")

                                        if PromptAttachment then
                                            for _, child3 in ipairs(PromptAttachment:GetChildren()) do
                                                if child3:IsA("ProximityPrompt") then
                                                    local Magnitude = (
                                                        Spawn.Position - HumanoidRootPart.Position
                                                    ).Magnitude

                                                    table.insert(
                                                        scriptState.stealingSettings.cachedPrompts,
                                                        {
                                                            prompt = child3,
                                                            spawn = Spawn,
                                                        }
                                                    )

                                                    if
                                                        Magnitude <= StealRadius
                                                        and Magnitude < huge2
                                                    then
                                                        child32 = child3
                                                        huge2 = Magnitude
                                                    end

                                                    break
                                                end
                                            end
                                        end
                                    end
                                end)
                            end
                        end

                        return child32, huge2
                    end
                end

                do
                    local function triggerStealPrompt(prompt2)
                        local now = tick()

                        if now - lastGrabTime < grabCooldown then
                            return
                        end

                        if scriptState.isStealInProgress then
                            return
                        end

                        if not prompt2 or not prompt2.Parent then
                            return
                        end

                        if not scriptState.stealingSettings.Data[prompt2] then
                            scriptState.stealingSettings.Data[prompt2] = {
                                hold = {},
                                trigger = {},
                                ready = true,
                            }
                            pcall(function()
                                if getconnections then
                                    for _, callback2 in
                                        ipairs(getconnections(prompt2.PromptButtonHoldBegan))
                                    do
                                        if not callback2.Function then
                                            continue
                                        end

                                        do
                                            table.insert(
                                                scriptState.stealingSettings.Data[prompt2].hold,
                                                callback2.Function
                                            )

                                            continue
                                        end
                                    end

                                    for _, callback3 in ipairs(getconnections(prompt2.Triggered)) do
                                        if callback3.Function then
                                            table.insert(
                                                scriptState.stealingSettings.Data[prompt2].trigger,
                                                callback3.Function
                                            )
                                        end
                                    end
                                else
                                    scriptState.stealingSettings.Data[prompt2].useFallback = true
                                end
                            end)
                        end

                        local result9 = scriptState.stealingSettings.Data[prompt2]

                        if not result9.ready then
                            return
                        end

                        result9.ready = false
                        scriptState.isStealInProgress = true
                        stealStartTime = now
                        lastGrabTime = now

                        if scriptState.featureConnections.progress then
                            scriptState.featureConnections.progress:Disconnect()
                        end

                        scriptState.featureConnections.progress = scriptState.RunService.Heartbeat:Connect(
                            function()
                                if not scriptState.isStealInProgress then
                                    scriptState.featureConnections.progress:Disconnect()
                                    scriptState.featureConnections.progress = nil

                                    return
                                end

                                local clampedValue = math.clamp(
                                    (tick() - stealStartTime)
                                        / scriptState.stealingSettings.StealDuration,
                                    0,
                                    1
                                )

                                if scriptState.stealProgressFill then
                                    scriptState.stealProgressFill.Size =
                                        UDim2.new(clampedValue, 0, 1, 0)
                                end

                                if scriptState.stealProgressLabel then
                                    scriptState.stealProgressLabel.Text = math.floor(
                                        clampedValue * 100
                                    ) .. "%"
                                end
                            end
                        )
                        task.spawn(function()
                            local didTriggerGrab = false

                            pcall(function()
                                if not result9.useFallback and #result9.hold > 0 then
                                    for _, value4 in ipairs(result9.hold) do
                                        task.spawn(function()
                                            pcall(value4)
                                        end)
                                    end

                                    task.wait(scriptState.stealingSettings.StealDuration)

                                    for _, value5 in ipairs(result9.trigger) do
                                        do
                                            task.spawn(function()
                                                pcall(value5)
                                            end)

                                            continue
                                        end
                                    end

                                    didTriggerGrab = true
                                end
                            end)

                            if not didTriggerGrab and type(fireproximityprompt) == "function" then
                                pcall(function()
                                    fireproximityprompt(prompt2)
                                    didTriggerGrab = true
                                end)

                                if didTriggerGrab then
                                    task.wait(scriptState.stealingSettings.StealDuration)
                                end
                            end

                            if not didTriggerGrab then
                                pcall(function()
                                    prompt2:InputHoldBegin()
                                    task.wait(scriptState.stealingSettings.StealDuration)
                                    prompt2:InputHoldEnd()
                                end)
                            end

                            task.wait(scriptState.stealingSettings.StealDuration * 0.3)

                            if scriptState.featureConnections.progress then
                                scriptState.featureConnections.progress:Disconnect()
                                scriptState.featureConnections.progress = nil
                            end

                            scriptState.resetStealProgressDisplay()
                            task.wait(0.05)
                            result9.ready = true
                            scriptState.isStealInProgress = false
                        end)
                    end

                    scriptState.startAutomaticSteal = function()
                        if scriptState.featureConnections.autoSteal then
                            return
                        end

                        if scriptState.stealingSettings.StealMode == "Semi" then
                            if scriptState.startSemiAutomaticSteal then
                                scriptState.startSemiAutomaticSteal()
                            end

                            return
                        end

                        if scriptState.stopSemiAutomaticSteal then
                            scriptState.stopSemiAutomaticSteal(false)
                        end

                        scriptState.featureConnections.autoSteal = scriptState.RunService.Heartbeat:Connect(
                            function()
                                if
                                    not scriptState.stealingSettings.AutoStealEnabled
                                    or scriptState.isStealInProgress
                                then
                                    return
                                end

                                local nearestStealPrompt2 = scriptState.findNearestStealPrompt()

                                if nearestStealPrompt2 then
                                    triggerStealPrompt(nearestStealPrompt2)
                                end
                            end
                        )
                    end
                end

                scriptState.stopAutomaticSteal = function()
                    if scriptState.featureConnections.autoSteal then
                        scriptState.featureConnections.autoSteal:Disconnect()
                        scriptState.featureConnections.autoSteal = nil
                    end

                    if scriptState.stopSemiAutomaticSteal then
                        scriptState.stopSemiAutomaticSteal(false)
                    end

                    if scriptState.featureConnections.progress then
                        scriptState.featureConnections.progress:Disconnect()
                        scriptState.featureConnections.progress = nil
                    end

                    scriptState.isStealInProgress = false
                    lastGrabTime = 0
                    scriptState.stealingSettings.plotCache = {}
                    scriptState.stealingSettings.plotCacheTime = {}
                    scriptState.stealingSettings.cachedPrompts = {}
                    scriptState.resetStealProgressDisplay()
                end
            end
        end
    end

do
    local semiGrabHoldDuration = 1.3
    local semiGrabTimeout = 2.6
    local semiStealResultDelay = 0.3

    do
        local semiStealScanRadius = 80
        local semiGrabUpdateInterval = 0.05

        local function getPromptDistance(prompt3)
            local Character = scriptState.LocalPlayer.Character
            local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")

            if not HumanoidRootPart or not prompt3 or not prompt3.Parent then
                return math.huge
            end

            local Parent = prompt3.Parent
            local WorldPosition

            if Parent:IsA("Attachment") then
                WorldPosition = Parent.WorldPosition
            elseif Parent:IsA("BasePart") then
                WorldPosition = Parent.Position
            else
                local BasePart = Parent:FindFirstAncestorWhichIsA("BasePart")

                if BasePart then
                    WorldPosition = BasePart.Position
                end
            end

            return WorldPosition and (HumanoidRootPart.Position - WorldPosition).Magnitude
                or math.huge
        end
        local function getOrCachePromptCallbacks(prompt4)
            local result10 = scriptState.stealingSettings.Data[prompt4]

            if result10 then
                return result10
            end

            local promptCallbacks = {
                hold = {},
                trigger = {},
                ready = true,
            }

            scriptState.stealingSettings.Data[prompt4] = promptCallbacks
            pcall(function()
                if getconnections then
                    for _, callback4 in ipairs(getconnections(prompt4.PromptButtonHoldBegan)) do
                        if not callback4.Function then
                            continue
                        end

                        do
                            table.insert(promptCallbacks.hold, callback4.Function)

                            continue
                        end
                    end

                    for _, callback5 in ipairs(getconnections(prompt4.Triggered)) do
                        if callback5.Function then
                            table.insert(promptCallbacks.trigger, callback5.Function)
                        end
                    end
                end
            end)

            return promptCallbacks
        end
        local function executeSemiAutomaticGrab(prompt5)
            local orCachePromptCallbacks = getOrCachePromptCallbacks(prompt5)

            if
                not orCachePromptCallbacks.ready
                or #orCachePromptCallbacks.hold == 0
                or #orCachePromptCallbacks.trigger == 0
            then
                return
            end

            orCachePromptCallbacks.ready = false
            scriptState.semiStealState.active = true
            scriptState.semiStealState.startTime = tick()
            scriptState.semiStealState.phase = "holding"
            task.spawn(function()
                for _, value6 in ipairs(orCachePromptCallbacks.hold) do
                    do
                        task.spawn(function()
                            pcall(value6)
                        end)

                        continue
                    end
                end

                task.wait(semiGrabHoldDuration)
                scriptState.semiStealState.phase = "waitingRange"

                local isMatching = getPromptDistance(prompt5)
                    <= scriptState.stealingSettings.SemiRadius
                local success = false
                local exitTo

                while true do
                    if
                        scriptState.stealingSettings.StealMode == "Semi"
                        and scriptState.stealingSettings.AutoStealEnabled
                        and tick() - scriptState.semiStealState.startTime <= semiGrabTimeout
                        and prompt5.Parent
                    then
                        if
                            not (
                                getPromptDistance(prompt5) <= scriptState.stealingSettings.SemiRadius
                            )
                        then
                            task.wait()

                            continue
                        end

                        break
                    end

                    exitTo = 1

                    break
                end

                if exitTo ~= 1 then
                    if not isMatching then
                        task.wait(semiStealResultDelay)
                    end

                    for _, value7 in ipairs(orCachePromptCallbacks.trigger) do
                        task.spawn(function()
                            pcall(value7)
                        end)
                    end

                    success = true
                end

                scriptState.semiStealState.success = success
                scriptState.semiStealState.phase = success and "success" or "failed"
                scriptState.semiStealState.active = false
                scriptState.semiStealState.lastResultTime = tick()
                task.wait(semiGrabUpdateInterval)
                orCachePromptCallbacks.ready = true
            end)
        end

        do
            local function updateSemiAutomaticGrab(dt)
                if
                    scriptState.stealingSettings.StealMode ~= "Semi"
                    or not scriptState.stealProgressFill
                    or not scriptState.stealProgressLabel
                then
                    return
                end

                local isMatching2 = scriptState.semiStealState.lastResultTime > 0
                    and tick() - scriptState.semiStealState.lastResultTime < 0.75
                local targetStealProgress = 0
                local backgroundColor3 = scriptState.accentColor

                if scriptState.semiStealState.active then
                    targetStealProgress = math.clamp(
                        (tick() - scriptState.semiStealState.startTime) / semiGrabHoldDuration,
                        0,
                        1
                    )
                elseif isMatching2 then
                    targetStealProgress = 1
                end

                scriptState.semiStealState.fill += (targetStealProgress - scriptState.semiStealState.fill) * math.min(
                    (dt or 0.016) * 14,
                    1
                )

                local clampedValue2 = math.clamp(scriptState.semiStealState.fill, 0, 1)

                scriptState.stealProgressFill.Size = UDim2.new(clampedValue2, 0, 1, 0)
                scriptState.stealProgressFill.BackgroundColor3 = backgroundColor3

                local UIGradient =
                    scriptState.stealProgressFill:FindFirstChildWhichIsA("UIGradient")

                if UIGradient then
                    UIGradient.Color = ColorSequence.new(
                        backgroundColor3:Lerp(scriptState.stealProgressColors.WHITE, 0.18),
                        backgroundColor3:Lerp(Color3.new(0, 0, 0), 0.38)
                    )
                end

                scriptState.stealProgressLabel.Text = tostring(
                    math.floor(clampedValue2 * 100 + 0.5)
                ) .. "%"
                scriptState.stealProgressLabel.TextColor3 = scriptState.stealProgressColors.WHITE
            end

            do
                scriptState.startSemiAutomaticSteal = function()
                    if scriptState.featureConnections.autoSteal then
                        scriptState.featureConnections.autoSteal:Disconnect()
                        scriptState.featureConnections.autoSteal = nil
                    end

                    if scriptState.semiStealUpdateConnection then
                        return
                    end

                    scriptState.semiStealState.fill = 0
                    scriptState.resetStealProgressDisplay()
                    scriptState.signalConnection =
                        scriptState.RunService.RenderStepped:Connect(updateSemiAutomaticGrab)
                    scriptState.semiStealUpdateConnection = scriptState.RunService.Heartbeat:Connect(
                        function()
                            if scriptState.stealingSettings.StealMode == "Semi" then
                                if
                                    scriptState.stealingSettings.AutoStealEnabled
                                    and not scriptState.semiStealState.active
                                then
                                    local nearestStealPrompt3 =
                                        scriptState.findNearestStealPrompt(semiStealScanRadius)

                                    if nearestStealPrompt3 then
                                        executeSemiAutomaticGrab(nearestStealPrompt3)
                                    end

                                    return
                                end
                            end
                        end
                    )
                end
            end
        end

        scriptState.stopSemiAutomaticSteal = function(value33)
            if scriptState.semiStealUpdateConnection then
                scriptState.semiStealUpdateConnection:Disconnect()
                scriptState.semiStealUpdateConnection = nil
            end

            if scriptState.signalConnection then
                scriptState.signalConnection:Disconnect()
                scriptState.signalConnection = nil
            end

            scriptState.semiStealState.active = false
            scriptState.semiStealState.phase = "idle"
            scriptState.semiStealState.fill = 0

            if value33 ~= false then
                scriptState.resetStealProgressDisplay()
            end

            if scriptState.textLabel then
                scriptState.textLabel.TextColor3 = Color3.fromRGB(190, 190, 190)
            end
        end
        scriptState.RunService.Stepped:Connect(function()
            for _, player in ipairs(scriptState.Players:GetPlayers()) do
                if player ~= scriptState.LocalPlayer and player.Character then
                    for _, descendant in ipairs(player.Character:GetDescendants()) do
                        if descendant:IsA("BasePart") then
                            descendant.CanCollide = false
                        end
                    end
                end
            end
        end)
        scriptState.RunService.RenderStepped:Connect(function()
            local Character = scriptState.LocalPlayer.Character

            if not Character then
                return
            end

            local Humanoid = Character:FindFirstChildOfClass("Humanoid")
            local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

            if not Humanoid or not HumanoidRootPart then
                return
            end

            if scriptState.isAutomaticMovementAllowed(Humanoid) then
                scriptState.vector6 = Vector3.new(0, 0, 0)

                return
            end

            if
                not scriptState.autoBat
                and not scriptState.isAutomaticLeftEnabled
                and not scriptState.isAutomaticRightEnabled
            then
                local MoveDirection = Humanoid.MoveDirection
                local activeMovementSpeed = scriptState.getActiveMovementSpeed()

                if MoveDirection.Magnitude > 0 then
                    scriptState.vector6 = MoveDirection
                    HumanoidRootPart.Velocity = Vector3.new(
                        MoveDirection.X * activeMovementSpeed,
                        HumanoidRootPart.Velocity.Y,
                        MoveDirection.Z * activeMovementSpeed
                    )
                elseif
                    scriptState.isRagdollRecoveryEnabled and scriptState.vector6.Magnitude > 0
                then
                    local isEnabled17 = false

                    for key in pairs(scriptState.movementKeyCodes) do
                        if scriptState.UserInputService:IsKeyDown(key) then
                            isEnabled17 = true

                            break
                        end
                    end

                    if isEnabled17 then
                        HumanoidRootPart.Velocity = Vector3.new(
                            scriptState.vector6.X * activeMovementSpeed,
                            HumanoidRootPart.Velocity.Y,
                            scriptState.vector6.Z * activeMovementSpeed
                        )
                    end
                end
            end

            if scriptState.textLabel2 then
                scriptState.textLabel2.Text = string.format(
                    "Speed: %.1f",
                    Vector3.new(HumanoidRootPart.Velocity.X, 0, HumanoidRootPart.Velocity.Z).Magnitude
                )
            end
        end)
        scriptState.automaticLeftConnection = nil
        scriptState.automaticRightConnection = nil

        do
            local automaticLeftPhase = 1
            local automaticRightPhase = 1

            scriptState.stopAutomaticLeftMovement = function()
                if scriptState.automaticLeftConnection then
                    scriptState.automaticLeftConnection:Disconnect()
                    scriptState.automaticLeftConnection = nil
                end

                automaticLeftPhase = 1

                local Character = scriptState.LocalPlayer.Character

                if Character then
                    local Humanoid = Character:FindFirstChildOfClass("Humanoid")

                    if Humanoid then
                        Humanoid:Move(Vector3.zero, false)
                    end
                end

                if scriptState.updateAutomaticLeftToggle then
                    scriptState.updateAutomaticLeftToggle(false)
                end
            end
            scriptState.stopAutomaticRightMovement = function()
                if scriptState.automaticRightConnection then
                    scriptState.automaticRightConnection:Disconnect()
                    scriptState.automaticRightConnection = nil
                end

                automaticRightPhase = 1

                local Character = scriptState.LocalPlayer.Character

                if Character then
                    local Humanoid = Character:FindFirstChildOfClass("Humanoid")

                    if Humanoid then
                        Humanoid:Move(Vector3.zero, false)
                    end
                end

                if scriptState.updateAutomaticRightToggle then
                    scriptState.updateAutomaticRightToggle(false)
                end
            end
            scriptState.startAutomaticLeftMovement = function()
                if scriptState.automaticLeftConnection then
                    scriptState.automaticLeftConnection:Disconnect()
                end

                automaticLeftPhase = 1
                scriptState.automaticLeftConnection = scriptState.RunService.Heartbeat:Connect(
                    function()
                        if not scriptState.isAutomaticLeftEnabled then
                            return
                        end

                        local Character = scriptState.LocalPlayer.Character

                        if not Character then
                            return
                        end

                        local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

                        end

                                    scriptState.vector.X,
                                    HumanoidRootPart.Position.Y,
                                    scriptState.vector.Z
                                ) - HumanoidRootPart.Position).Magnitude
                                < 1
                            then
                                automaticLeftPhase = 2

                            HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(

                        elseif automaticLeftPhase == 2 then

                                    scriptState.vector2.X,

                                    scriptState.vector2.Z

                )

            scriptState.startAutomaticRightMovement = function()
                if scriptState.automaticRightConnection then
                    scriptState.automaticRightConnection:Disconnect()

                automaticRightPhase = 1
                scriptState.automaticRightConnection = scriptState.RunService.Heartbeat:Connect(

                        end

                        end

                        end

                            Humanoid:Move(Vector3.zero, false)

                        end

                        local normalModeMovementSpeed2 = scriptState.getNormalModeMovementSpeed()

                        if automaticRightPhase == 1 then
                            if
                                (Vector3.new(
                                    scriptState.vector4.X,
                                    HumanoidRootPart.Position.Y,
                                    scriptState.vector4.Z
                                ) - HumanoidRootPart.Position).Magnitude
                                < 1
                            then
                                automaticRightPhase = 2

                                local offset4 = scriptState.vector5 - HumanoidRootPart.Position
                                local Unit = Vector3.new(offset4.X, 0, offset4.Z).Unit

                                Humanoid:Move(Unit, false)
                                HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(
                                    Unit.X * normalModeMovementSpeed2,
                                    HumanoidRootPart.AssemblyLinearVelocity.Y,
                                    Unit.Z * normalModeMovementSpeed2
                                )

                                return
                            end

                            local offset5 = scriptState.vector4 - HumanoidRootPart.Position
                            local Unit = Vector3.new(offset5.X, 0, offset5.Z).Unit

                            Humanoid:Move(Unit, false)
                            HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(
                                Unit.X * normalModeMovementSpeed2,
                                HumanoidRootPart.AssemblyLinearVelocity.Y,
                                Unit.Z * normalModeMovementSpeed2
                            )
                        elseif automaticRightPhase == 2 then
                            if
                                (Vector3.new(
                                    scriptState.vector5.X,
                                    HumanoidRootPart.Position.Y,
                                    scriptState.vector5.Z
                                ) - HumanoidRootPart.Position).Magnitude
                                < 1
                            then
                                Humanoid:Move(Vector3.zero, false)
                                HumanoidRootPart.AssemblyLinearVelocity = Vector3.zero
                                scriptState.isAutomaticRightEnabled = false

                                if scriptState.automaticRightConnection then

                            local offset6 = scriptState.vector5 - HumanoidRootPart.Position
                            local Unit = Vector3.new(offset6.X, 0, offset6.Z).Unit

                            Humanoid:Move(Unit, false)
                            HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(
                                Unit.X * normalModeMovementSpeed2,
                                HumanoidRootPart.AssemblyLinearVelocity.Y,
                                Unit.Z * normalModeMovementSpeed2
                            )
                        end
                    end
                )
            end
        end

do
    local function performInfiniteJump(jumpVelocity)
        if not scriptState.isInfiniteJumpEnabled then
            return
        end

        local Character = scriptState.LocalPlayer.Character

        if Character then
            local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

            if HumanoidRootPart then
                HumanoidRootPart.Velocity = Vector3.new(
                    HumanoidRootPart.Velocity.X,
                    jumpVelocity,
                    HumanoidRootPart.Velocity.Z
                )
            end

            return
        end
    end

    scriptState.UserInputService.JumpRequest:Connect(function()
        performInfiniteJump(50)
    end)
    scriptState.UserInputService.InputBegan:Connect(function(input)
        if
            not (
                input.UserInputType ~= Enum.UserInputType.Keyboard
                or input.KeyCode ~= Enum.KeyCode.Space
                or scriptState.UserInputService:GetFocusedTextBox()
            )
        then
            scriptState.isKeyboardJumpHeld = true
            task.delay(0.12, function()
                if scriptState.isKeyboardJumpHeld then
                    scriptState.isKeyboardJumpBoostActive = true
                    performInfiniteJump(50)
                end
            end)
        end
    end)
    scriptState.UserInputService.InputEnded:Connect(function(input)
        if
            input.UserInputType == Enum.UserInputType.Keyboard
            and input.KeyCode == Enum.KeyCode.Space
        then
            scriptState.isKeyboardJumpHeld = false
            scriptState.isKeyboardJumpBoostActive = false
        end
    end)

    local mobileJumpConnections = {}

    local function connectMobileJumpButton(jumpButton)
        if jumpButton and jumpButton.Name == "JumpButton" and jumpButton:IsA("GuiButton") then
            if not mobileJumpConnections[jumpButton] then
                mobileJumpConnections[jumpButton] = true
                jumpButton.InputBegan:Connect(function(input)
                    if
                        input.UserInputType ~= Enum.UserInputType.Touch
                        or not scriptState.isInfiniteJumpEnabled
                    then
                        return
                    end

                    scriptState.isTouchJumpHeld = true
                    task.delay(0.12, function()
                        if scriptState.isTouchJumpHeld and scriptState.isInfiniteJumpEnabled then
                            scriptState.isTouchJumpBoostActive = true
                            performInfiniteJump(50)
                        end
                    end)
                end)
                jumpButton.InputEnded:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.Touch then
                        scriptState.isTouchJumpHeld = false
                        scriptState.isTouchJumpBoostActive = false
                    end
                end)
                jumpButton.AncestryChanged:Connect(function(child, parent)
                    if not parent then
                        mobileJumpConnections[jumpButton] = nil
                        scriptState.isTouchJumpHeld = false
                        scriptState.isTouchJumpBoostActive = false
                    end
                end)

                return
            end
        end
    end

    do
        local PlayerGui = scriptState.LocalPlayer:WaitForChild("PlayerGui")

        for _, descendant in ipairs(PlayerGui:GetDescendants()) do
            connectMobileJumpButton(descendant)
        end

        PlayerGui.DescendantAdded:Connect(function(descendant)
            task.defer(connectMobileJumpButton, descendant)
        end)
    end

    scriptState.RunService.Heartbeat:Connect(function()
        if scriptState.isKeyboardJumpBoostActive or scriptState.isTouchJumpBoostActive then
            performInfiniteJump(50)
        end
    end)
end

scriptState.startFloating = function()
    if scriptState.isFloatActive then
        return
    end

    if scriptState.autoBat then
        scriptState.autoBat = false

        if scriptState.updateAimTarget then
            scriptState.updateAimTarget()
        end

        if scriptState.updateAutomaticBatToggle then
            scriptState.updateAutomaticBatToggle(false)
        end
    end

    local Character = scriptState.LocalPlayer.Character

    if not Character then
        return
    end

    if not Character:FindFirstChild("HumanoidRootPart") then
        return
    end

    scriptState.isFloatActive = true

    local now = tick()
    local semiStealUpdateConnection3

    semiStealUpdateConnection3 = scriptState.RunService.Heartbeat:Connect(function()
        local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")

        if not HumanoidRootPart then
            semiStealUpdateConnection3:Disconnect()
            scriptState.isFloatActive = false

            return
        end

        if not (scriptState.floatBoostDuration <= tick() - now) then
            HumanoidRootPart.Velocity = Vector3.new(
                HumanoidRootPart.Velocity.X,
                scriptState.floatBoostVelocity,
                HumanoidRootPart.Velocity.Z
            )

            return
        end

        semiStealUpdateConnection3:Disconnect()

        local raycastParams = RaycastParams.new()

        raycastParams.FilterDescendantsInstances = { Character }
        raycastParams.FilterType = Enum.RaycastFilterType.Exclude

        local hit =
            workspace:Raycast(HumanoidRootPart.Position, Vector3.new(0, -2000, 0), raycastParams)

        if hit then
            local Humanoid = Character:FindFirstChildOfClass("Humanoid")

            HumanoidRootPart.CFrame = CFrame.new(
                HumanoidRootPart.Position.X,
                hit.Position.Y
                    + ((Humanoid and Humanoid.HipHeight or 2) + HumanoidRootPart.Size.Y / 2),
                HumanoidRootPart.Position.Z
            )
            HumanoidRootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end

        scriptState.isFloatActive = false
    end)
end
scriptState.teleportBelowMap = function(forceTeleport)
    local Character = scriptState.LocalPlayer.Character

    if not Character then
        return
    end

    local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

    if not HumanoidRootPart then
        return
    end

    local Humanoid = Character:FindFirstChildOfClass("Humanoid")

    if not Humanoid then
        return
    end

    if not forceTeleport then
        if Humanoid.FloorMaterial ~= Enum.Material.Air then
            return
        end

        if HumanoidRootPart.Position.Y < scriptState.autoTPHeight then
            return
        end
    end

    HumanoidRootPart.CFrame = CFrame.new(
        HumanoidRootPart.Position.X,
        -7,
        HumanoidRootPart.Position.Z
    ) * CFrame.Angles(0, select(2, HumanoidRootPart.CFrame:ToEulerAnglesYXZ()), 0)
    HumanoidRootPart.AssemblyLinearVelocity = Vector3.zero
end

                local function findNearestAimTarget()
                    local HumanoidRootPart = scriptState.LocalPlayer.Character
                        and scriptState.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

                    if not HumanoidRootPart then
                        return nil, nil, math.huge
                    end

                    local humanoidRootPart2
                    local player6
                    local huge2 = math.huge

                    for _, player in ipairs(scriptState.Players:GetPlayers()) do
                        if player ~= scriptState.LocalPlayer and player.Character then
                            local HumanoidRootPart2 =
                                player.Character:FindFirstChild("HumanoidRootPart")
                            local Humanoid = player.Character:FindFirstChildOfClass("Humanoid")

                            if HumanoidRootPart2 and Humanoid and Humanoid.Health > 0 then
                                local Magnitude = (
                                    HumanoidRootPart2.Position - HumanoidRootPart.Position
                                ).Magnitude

                                if Magnitude < huge2 then
                                    huge2 = Magnitude
                                    humanoidRootPart2 = HumanoidRootPart2
                                    player6 = player
                                end
                            end
                        end
                    end

                    return humanoidRootPart2, player6, huge2
                end
                local function updateAutomaticSwing()
                    if isAutomaticSwingRunning or not scriptState.autoSwing then
                        return
                    end

                    isAutomaticSwingRunning = true
                    pcall(function()
                        local Character = scriptState.LocalPlayer.Character

                        if not Character then
                            return
                        end

                        local meleeTool = findMeleeTool()

                        if meleeTool then
                            if meleeTool.Parent ~= Character then
                                local Humanoid = Character:FindFirstChildOfClass("Humanoid")

                                if Humanoid then
                                    pcall(function()
                                        Humanoid:EquipTool(meleeTool)
                                    end)
                                end
                            end

                            pcall(function()
                                meleeTool:Activate()
                            end)

                            return
                        end
                    end)
                    task.delay(automaticSwingCooldown, function()
                        isAutomaticSwingRunning = false
                    end)
                end

                enableMeleeAimbot = function()
                    if scriptState.featureConnections.aimbot then
                        scriptState.featureConnections.aimbot:Disconnect()
                    end

                    if scriptState.isAutomaticLeftEnabled then

                        and scriptState.LocalPlayer.Character:FindFirstChildOfClass("Humanoid")

                    if Humanoid then
                        humanoid = Humanoid
                        AutoRotate = Humanoid.AutoRotate
                        Humanoid.AutoRotate = false
                    end

                    scriptState.featureConnections.aimbot = scriptState.RunService.RenderStepped:Connect(
                        function()
                            if not scriptState.autoBat then
                                return
                            end

                            local Character = scriptState.LocalPlayer.Character

                            if not Character then
                                return
                            end

                            local HumanoidRootPart = Character:FindFirstChild("HumanoidRootPart")

                            if not HumanoidRootPart then
                                return
                            end

                            local Humanoid2 = Character:FindFirstChildOfClass("Humanoid")

                            if not Humanoid2 then
                                return
                            end

                            if Humanoid2 ~= humanoid then
                                humanoid = Humanoid2
                                AutoRotate = Humanoid2.AutoRotate
                            end

                            Humanoid2.AutoRotate = false

                            if not Character:FindFirstChildOfClass("Tool") then
                                local meleeTool2 = findMeleeTool()

                                if meleeTool2 then
                                    pcall(function()
                                        Humanoid2:EquipTool(meleeTool2)
                                    end)
                                end
                            end

                            local nearestAimTarget, result14, result15 = findNearestAimTarget()

                            if not nearestAimTarget then
                                nearestAimTarget2 = nil
                                isActive = nil

                                return
                            end

                            nearestAimTarget2 = nearestAimTarget
                            isActive = result14

                            local AssemblyLinearVelocity = nearestAimTarget.AssemblyLinearVelocity
                            local Position = HumanoidRootPart.Position
                            local Position2 = nearestAimTarget.Position
                            local offset9 = Position2
                                + AssemblyLinearVelocity * 0.14
                                + nearestAimTarget.CFrame.LookVector * 0.3
                                - Position
                            local vector7 = Vector3.new(offset9.X, 0, offset9.Z)
                            local Unit

                            if vector7.Magnitude > 0 then
                                Unit = vector7.Unit
                            else
                                Unit = Vector3.new(0, 0, 0)
                            end

                            local activeAimbotSpeed = getActiveAimbotSpeed()
                            local combinedValue3 = (Position2.Y + 3.7 - Position.Y) * 19.5
                                + AssemblyLinearVelocity.Y * 0.8

                            if Humanoid2.FloorMaterial ~= Enum.Material.Air then
                                combinedValue3 = math.max(combinedValue3, 13)
                            end

                            HumanoidRootPart.AssemblyLinearVelocity =
                                HumanoidRootPart.AssemblyLinearVelocity:Lerp(
                                    Vector3.new(
                                        Unit.X * activeAimbotSpeed,
                                        math.clamp(combinedValue3, -70, 110),
                                        Unit.Z * activeAimbotSpeed
                                    ),
                                    0.8
                                )

                            local combinedValue4 = Position2
                                + AssemblyLinearVelocity
                                    * math.clamp(AssemblyLinearVelocity.Magnitude / 150, 0.05, 0.2)

                            if (combinedValue4 - Position).Magnitude > 0.1 then
                                local cFrame = CFrame.lookAt(Position, combinedValue4)
                                local result16, result17, result18 = (
                                    HumanoidRootPart.CFrame:Inverse() * cFrame
                                ):ToEulerAnglesXYZ()

                                HumanoidRootPart.AssemblyAngularVelocity =
                                    HumanoidRootPart.CFrame:VectorToWorldSpace(
                                        Vector3.new(
                                            math.clamp(result16, -2.5, 2.5) * 42,
                                            math.clamp(result17, -2.5, 2.5) * 42,
                                            math.clamp(result18, -2.5, 2.5) * 42
                                        )
                                    )
                            end

                            if scriptState.autoSwing and result15 <= automaticSwingRange then
                                updateAutomaticSwing()
                            end
                        end
                    )
                end
            end

scriptState.colorFromRgbArray = function(rgb)
    return Color3.fromRGB(rgb[1], rgb[2], rgb[3])
end
scriptState.clearCustomSky = function()
    for _, child in ipairs(scriptState.Lighting:GetChildren()) do
        if child:GetAttribute("_AceDuelsSky") then
            pcall(function()
                child:Destroy()
            end)
        end
    end

    local Terrain = workspace:FindFirstChildOfClass("Terrain")

    if Terrain then
        for _, child in ipairs(Terrain:GetChildren()) do
            if child:GetAttribute("_AceDuelsSky") then
                pcall(function()
                    child:Destroy()
                end)
            end
        end
    end
end
scriptState.applySkyTheme = function(themeName)
    scriptState.clearCustomSky()

    local result19 = scriptState.skyThemes[themeName]

    if not result19 or result19.kind == "off" then
        scriptState.Lighting.FogEnd = 100000
        scriptState.Lighting.FogStart = 0
        scriptState.Lighting.FogColor = Color3.fromRGB(192, 192, 192)
        scriptState.Lighting.Brightness = 2
        scriptState.Lighting.ClockTime = 14
        scriptState.Lighting.GlobalShadows = true
        scriptState.appearanceSettings.skyTheme = "Off"

        return
    end

    scriptState.Lighting.FogEnd = 100000
    scriptState.Lighting.FogStart = 0
    scriptState.Lighting.FogColor = Color3.fromRGB(200, 200, 200)
    scriptState.Lighting.GlobalShadows = true
    scriptState.Lighting.ClockTime = result19.clock or 14
    scriptState.Lighting.Brightness = result19.brightness or 2

    if result19.outAmb then
        scriptState.Lighting.OutdoorAmbient = scriptState.colorFromRgbArray(result19.outAmb)
    end

    if result19.ambient then
        scriptState.Lighting.Ambient = scriptState.colorFromRgbArray(result19.ambient)
    end

    if result19.sky then
        local Sky = Instance.new("Sky")

        Sky:SetAttribute("_AceDuelsSky", true)

        if result19.sky.stars then
            Sky.StarCount = result19.sky.stars
        end

        if result19.sky.moon then
            Sky.MoonAngularSize = result19.sky.moon
        end

        if result19.sky.sun then
            Sky.SunAngularSize = result19.sky.sun
        end

        if result19.sky.moonTex then
            Sky.MoonTextureId = "rbxasset://sky/moon.jpg"
        end

        Sky.Parent = scriptState.Lighting
    end

    if result19.atm then
        local Atmosphere = Instance.new("Atmosphere")

        Atmosphere:SetAttribute("_AceDuelsSky", true)
        Atmosphere.Density = result19.atm.dens or 0.3
        Atmosphere.Color = scriptState.colorFromRgbArray(result19.atm.color)
        Atmosphere.Decay = scriptState.colorFromRgbArray(result19.atm.decay)
        Atmosphere.Glare = result19.atm.glare or 1
        Atmosphere.Haze = result19.atm.haze or 1
        Atmosphere.Parent = scriptState.Lighting
    end

    local Terrain = workspace:FindFirstChildOfClass("Terrain")

    if result19.clouds and Terrain then
        local Clouds = Instance.new("Clouds")

        Clouds:SetAttribute("_AceDuelsSky", true)
        Clouds.Cover = result19.clouds.cover or 0.5
        Clouds.Density = result19.clouds.dens or 0.5
        Clouds.Color = scriptState.colorFromRgbArray(result19.clouds.color)
        Clouds.Parent = Terrain
    end

    scriptState.appearanceSettings.skyTheme = themeName
end
scriptState.enableUltraGraphicsOptimization = function()
    scriptState.appearanceSettings.ultraModeEnabled = true

    if not scriptState.antiLag then
        scriptState.enableAntiLagGraphics()
    end

    pcall(function()
        scriptState.Lighting.GlobalShadows = false
        scriptState.Lighting.FogEnd = 1e10
        scriptState.Lighting.Brightness = 1
        scriptState.Lighting.EnvironmentDiffuseScale = 0
        scriptState.Lighting.EnvironmentSpecularScale = 0

        local Terrain = workspace:FindFirstChildOfClass("Terrain")

        if Terrain then
            Terrain.WaterWaveSize = 0
            Terrain.WaterWaveSpeed = 0
            Terrain.WaterReflectance = 0
            Terrain.WaterTransparency = 1
            Terrain.Decoration = false
        end
    end)

    for _, descendant in ipairs(workspace:GetDescendants()) do
        pcall(function()
            if descendant:IsA("BasePart") then
                descendant.Material = Enum.Material.SmoothPlastic
                descendant.Reflectance = 0
                descendant.CastShadow = false
            elseif
                descendant:IsA("ParticleEmitter")
                or descendant:IsA("Trail")
                or descendant:IsA("Beam")
                or descendant:IsA("Smoke")
                or descendant:IsA("Fire")
                or descendant:IsA("Sparkles")
            then
                descendant.Enabled = false
            elseif
                descendant:IsA("PointLight")
                or descendant:IsA("SpotLight")
                or descendant:IsA("SurfaceLight")
            then
                descendant.Enabled = false
            elseif descendant:IsA("Decal") or descendant:IsA("Texture") then
                descendant.Transparency = 1
            end
        end)
    end
end

scriptState.enableAccessoryRemoval = function()
    scriptState.appearanceSettings.removeAccessoriesEnabledSep = true

    local function removeCharacterAccessories(character7)
        for _, descendant in ipairs(character7:GetDescendants()) do
            if descendant:IsA("Accessory") or descendant:IsA("Hat") then
                pcall(function()
                    descendant:Destroy()
                end)
            end
        end
    end

    for _, player in ipairs(scriptState.Players:GetPlayers()) do
        if player.Character then
            removeCharacterAccessories(player.Character)
        end
    end

    if scriptState.appearanceSettings.removeAccConn then
        scriptState.appearanceSettings.removeAccConn:Disconnect()
    end

    scriptState.appearanceSettings.removeAccConn = workspace.DescendantAdded:Connect(
        function(descendant)
            if
                scriptState.appearanceSettings.removeAccessoriesEnabledSep
                and (descendant:IsA("Accessory") or descendant:IsA("Hat"))
            then
                pcall(function()
                    descendant:Destroy()
                end)
            end
        end
    )
end

            local function isIconTextElement(textElement)
                if
                    not textElement:IsA("TextLabel")
                    and not textElement:IsA("TextButton")
                    and not textElement:IsA("TextBox")
                then
                    return true
                end

                if textElement.TextStrokeTransparency ~= 1 then
                    return false
                end

                local textValue2 = tostring(textElement.FontFace)

                return textValue2 == textValue or string.find(textValue2, "BuilderIcons")
            end
            local function applyCustomFontToText(textElement2)
                if
                    (
                        textElement2:IsA("TextLabel")
                        or textElement2:IsA("TextButton")
                        or textElement2:IsA("TextBox")
                    ) and not isIconTextElement(textElement2)
                then
                    if not originalFontFaces[textElement2] then
                        originalFontFaces[textElement2] = textElement2.FontFace
                    end

                    pcall(function()
                        textElement2.FontFace = font
                    end)
                end
            end
            local function loadCustomFontAsset()
                if font then
                    return true
                end

                return pcall(function()
                    if isfile and writefile and getcustomasset then
                        if not isfile("v4mp_starborn.ttf") then
                            writefile(
                                "v4mp_starborn.ttf",
                                game:HttpGet(
                                    "https://granny.anondrop.net/uploads/6c2505542959f371/Starborn.ttf"
                                )
                            )
                        end

                        writefile(
                            "v4mp_starborn.json",
                            scriptState.HttpService:JSONEncode({
                                name = "Starborn",
                                faces = {
                                    {
                                        name = "Regular",
                                        weight = 400,
                                        style = "normal",
                                        assetId = getcustomasset("v4mp_starborn.ttf"),
                                    },
                                },
                            })
                        )
                        font = Font.new(getcustomasset("v4mp_starborn.json"))
                        textValue = tostring(
                            Font.new(
                                "rbxasset://LuaPackages/Packages/_Index/BuilderIcons/BuilderIcons/BuilderIcons.json"
                            )
                        )
                    end
                end) and font ~= nil
            end

                    return
                end

                if not loadCustomFontAsset() then
                    return
