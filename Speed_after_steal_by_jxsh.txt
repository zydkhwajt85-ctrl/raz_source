local character, humanoid, humanoidRootPart = nil, nil, nil
local function onCharacterAdded(newChar)
    character        = newChar
    humanoid         = character:WaitForChild("Humanoid")
    humanoidRootPart = character:WaitForChild("HumanoidRootPart")
end
if player.Character then onCharacterAdded(player.Character) end
player.CharacterAdded:Connect(onCharacterAdded)
local speedActive = false
local speedConnection
local function startSpeed()
    if speedConnection then speedConnection:Disconnect() end
    speedConnection = RunService.Heartbeat:Connect(function()
        if not humanoid or not humanoidRootPart then return end
        local moveDirection = humanoid.MoveDirection
        if moveDirection.Magnitude == 0 then return end
        local currentSpeed
        if SpeedState.isLaggerMode then
            -- In lagger mode: detect steal the same way as normal (WalkSpeed < 25)
            local isSteal = humanoid.WalkSpeed < 25
            currentSpeed = isSteal and SpeedState.laggerSteal or SpeedState.laggerSpeed
        else
            -- Normal mode: detect steal via WalkSpeed
            local isSteal = humanoid.WalkSpeed < 25
            currentSpeed = isSteal and SpeedState.carrySpeed or SpeedState.normalSpeed
        end
        humanoidRootPart.Velocity = Vector3.new(
            moveDirection.X * currentSpeed,
            humanoidRootPart.Velocity.Y,
            moveDirection.Z * currentSpeed
        )
    end)
end
local function stopSpeed()
    if speedConnection then
        speedConnection:Disconnect()
        speedConnection = nil
    end
end

local function makeTab(label, order, isActive)
    local btn = Instance.new("TextButton")
    btn.Size             = UDim2.new(0.5, 0, 1, 0)
    btn.BackgroundColor3 = isActive and Color3.fromRGB(0, 120, 255) or Color3.fromRGB(15, 15, 15)
    btn.BorderSizePixel  = 0
    btn.Text             = label
    btn.Font             = Enum.Font.GothamBold
    btn.TextSize         = 12
    btn.TextColor3       = isActive and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(160, 160, 160)
    btn.LayoutOrder      = order
    btn.Parent           = tabContainer
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)
    return btn
end

local laggerBox,      laggerLabel      = createRow(laggerPanel, "Lagger Boost", 0,  Config["Lagger Speed Value"])
local laggerStealBox, laggerStealLabel = createRow(laggerPanel, "Lagger Steal", 32, Config["Lagger Steal Value"])
local function applyInput(box, minV, maxV, default, configKey, stateKey)
    box.FocusLost:Connect(function()
        local text = box.Text:gsub("[^%d%.]", "")
        local num  = tonumber(text) or default
        num = math.clamp(num, minV, maxV)
        box.Text       = tostring(num)
        Config[configKey] = num
        SpeedState[stateKey] = num
    end)
end
applyInput(speedBox,      15, 200, 53,   "Speed Value",        "normalSpeed")
applyInput(stealBox,      15, 200, 29,   "Steal Speed Value",  "carrySpeed")
applyInput(laggerBox,     1,  200, 10.1, "Lagger Speed Value", "laggerSpeed")
applyInput(laggerStealBox,1,  200, 8,    "Lagger Steal Value", "laggerSteal")
local function switchTab(tab)
    activeTab = tab
    if tab == "Normal" then
        -- Visual
        normalTab.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
        normalTab.TextColor3       = Color3.fromRGB(255, 255, 255)
        laggerTab.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
        laggerTab.TextColor3       = Color3.fromRGB(160, 160, 160)
        -- Panels
        normalPanel.Visible = true
        laggerPanel.Visible = false
        -- Mode
        SpeedState.isLaggerMode = false
    else
        -- Visual
        laggerTab.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
        laggerTab.TextColor3       = Color3.fromRGB(255, 255, 255)
        normalTab.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
        normalTab.TextColor3       = Color3.fromRGB(160, 160, 160)
        -- Panels
        normalPanel.Visible = false
        laggerPanel.Visible = true
        -- Mode
        SpeedState.isLaggerMode = true
    end
end
normalTab.MouseButton1Click:Connect(function() switchTab("Normal") end)
laggerTab.MouseButton1Click:Connect(function() switchTab("Lagger") end)
activate.MouseButton1Click:Connect(function()
    speedActive = not speedActive
    if speedActive then
        activate.Text             = "ON"
        activate.BackgroundColor3 = Color3.fromRGB(0, 120, 255)
        startSpeed()
    else
        activate.Text             = "OFF"
        activate.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
        stopSpeed()
    end
end)
local FULL_SIZE = UDim2.new(0, 200, 0, 185)
local MINI_SIZE = UDim2.new(0, 200, 0, 70)
local elementsToHide = { tabContainer, normalPanel, laggerPanel }
if Config["Speed Minimized"] then
    minimized = true
    minimizeButton.Text = "▼"
    for _, v in pairs(elementsToHide) do v.Visible = false end
    main.Size = MINI_SIZE
end
minimizeButton.MouseButton1Click:Connect(function()
    minimized = not minimized
    Config["Speed Minimized"] = minimized
    if minimized then
        minimizeButton.Text = "▼"
        for _, v in pairs(elementsToHide) do v.Visible = false end
        TweenService:Create(main,
            TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {Size = MINI_SIZE}
        ):Play()
    else
        minimizeButton.Text = "▲"
        local tween = TweenService:Create(main,
            TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
            {Size = FULL_SIZE}
        )
        tween:Play()
        tween.Completed:Connect(function()
            tabContainer.Visible = true
            -- Restore the correct panel for whichever tab is active
            normalPanel.Visible = activeTab == "Normal"
            laggerPanel.Visible = activeTab == "Lagger"
        end)
    end
end)
