--  getValue4()
--
--  Calls processItems6() (via v2 alias), takes the first return
--  value, adds the constant 463989418072344 mod 2^32, then XORs

--  This token is used as the cipher key in processItems7.
-- ================================================================

local function getValue4()
    local a = table.pack(v2())
    return bit32.bxor(
        (463989418072344 + a[1] % 4294967296) % 4294967296,
        3999492378

end


-- ================================================================
--  processItems7(a, b)   — XOR stream cipher / payload decryptor
--
--  XORs each byte of string `a` against successive low-bytes of
--  integer key `b`, shifting b right 8 bits each step.
--  Loop exits when i+1 > #a (i.e. processes bytes 1 through #a-1).
-- ================================================================
local function processItems7(a, b)
    local c = b
    local d = 1
    local e = #a
    local f = {}
    if 1 <= e then
        while true do
            f[d] = string.char(bit32.bxor(a:byte(d), c % 256))
            c = (c - c % 256) / 256
            d = d + 1
            if d + 1 > e then break end
        end
    end
    return table.concat(f)
end
