local function getPlayerPlots()
    local plots = {}
    pcall(function()
        if Workspace:FindFirstChild("Plots") then
            for _, plot in pairs(Workspace.Plots:GetChildren()) do
                if plot:FindFirstChild("Owner") then
                    local owner = plot.Owner.Value
                    if owner == player.Name or owner == player.UserId then
                        table.insert(plots, plot)
                    end
                elseif plot:FindFirstChild("PlotOwner") then
                    local owner = plot.PlotOwner.Value
                    if owner == player then
                        table.insert(plots, plot)
                    end
                end
            end
        end
    end)
    return plots
end
local function harvestCrop(plot)
    pcall(function()
        if not plot then return end
        -- Try UnlockPrompt
        local unlock = plot:FindFirstChild("Unlock")
        if unlock then
            for _, item in pairs(unlock:GetChildren()) do
                if item:IsA("Model") or item:IsA("Part") then
                    for _, prompt in pairs(item:GetDescendants()) do
                        if prompt:IsA("ProximityPrompt") then
                            fireproximityprompt(prompt)
                            wait(0.1)
                        end
                    end
                end
            end
        end
        -- Try Harvest directly
        local harvest = plot:FindFirstChild("Harvest")
        if harvest then
            for _, prompt in pairs(harvest:GetDescendants()) do
                if prompt:IsA("ProximityPrompt") then
                    fireproximityprompt(prompt)
                    wait(0.1)
                end
            end
        end
    end)
end
local function sellPlot(plot)

end

local function buySeed()
    pcall(function()
        local store = Workspace:FindFirstChild("Store") or Workspace:FindFirstChild("Shop")
        if store then
            for _, item in pairs(store:GetDescendants()) do
                if item:IsA("ProximityPrompt") then
                    if string.find(item.Parent.Name:lower(), selectedSeed:lower()) then
                        fireproximityprompt(item)
                        wait(0.2)
                        break
                    end
                end
            end
        end
    end)
end
local function clearESP()
    for _, drawing in pairs(espDrawings) do
        pcall(function()
            drawing:Remove()
        end)
    end
    espDrawings = {}
end
local function createESP()
    clearESP()
    -- ESP Players
    if FEATURES.espPlayers then
        for _, otherPlayer in pairs(Players:GetPlayers()) do
            if otherPlayer ~= player and otherPlayer.Character then
                local humanoidRootPart = otherPlayer.Character:FindFirstChild("HumanoidRootPart")
                if humanoidRootPart then
                    pcall(function()
                        local nameLabel = Drawing.new("Text")
                        nameLabel.Text = "👤 " .. otherPlayer.Name
                        nameLabel.Size = 15
                        nameLabel.Color = Color3.fromRGB(0, 255, 100)
                        nameLabel.Outline = true
                        nameLabel.Visible = true
                        table.insert(espDrawings, nameLabel)
                    end)
                end
            end
        end
    end
    -- ESP Fruits
    if FEATURES.espFruits then
        if Workspace:FindFirstChild("Fruits") then
            for _, fruit in pairs(Workspace.Fruits:GetChildren()) do
                if fruit:FindFirstChild("Humanoid") or fruit:IsA("Model") then
                    pcall(function()
                        local label = Drawing.new("Text")
                        label.Text = "🍎 " .. fruit.Name
                        label.Size = 13
                        label.Color = Color3.fromRGB(255, 150, 0)
                        label.Outline = true
                        label.Visible = true
                        table.insert(espDrawings, label)
                    end)
                end
            end
        end
    end
    -- ESP Pets
    if FEATURES.espPets then
        if Workspace:FindFirstChild("Pets") then
            for _, pet in pairs(Workspace.Pets:GetChildren()) do
                pcall(function()
                    local label = Drawing.new("Text")
                    label.Text = "🐱 " .. pet.Name
                    label.Size = 13
                    label.Color = Color3.fromRGB(255, 100, 255)
                    label.Outline = true
                    label.Visible = true
                    table.insert(espDrawings, label)
                end)
            end
        end
    end
end
