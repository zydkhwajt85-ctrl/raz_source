local function corner(o, r)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, r)
	c.Parent = o
end
local function stroke(o, col, th, tr)
	local s = Instance.new("UIStroke")
	s.Color = col
	s.Thickness = th or 1.5
	s.Transparency = tr or 0.2
	s.Parent = o
	return s
end
local function playSound(id, vol, speed)
	pcall(function()
		local s = Instance.new("Sound")
		s.SoundId = id
		s.Volume = vol or 0.6
		s.PlaybackSpeed = speed or 1
		s.Parent = SoundService
		s:Play()
		task.delay(2, function() pcall(function() s:Destroy() end) end)
	end)
end

local function setDice(i, name)
	selected[i] = name
	local b = diceBtns[i]
	if not b then return end
	if name == "?" then
		b.BackgroundColor3 = theme.dice
		if diceGlows[i] then
			diceGlows[i].BackgroundColor3 = theme.accent
			diceGlows[i].BackgroundTransparency = 0.85
		end
		if diceStrokes[i] then
			diceStrokes[i].Color = Color3.fromRGB(255, 255, 255)
			diceStrokes[i].Transparency = 0.35
			diceStrokes[i].Thickness = 2
		end
	else
		b.BackgroundColor3 = COLORS[name]
		if diceGlows[i] then
			diceGlows[i].BackgroundColor3 = COLORS[name]
			diceGlows[i].BackgroundTransparency = 0.7
		end
		if diceStrokes[i] then
			diceStrokes[i].Color = Color3.fromRGB(255, 255, 255)
			diceStrokes[i].Transparency = 0.1
			diceStrokes[i].Thickness = 2.4
		end
		b.Size = UDim2.fromOffset(36, 36)
		b.Position = UDim2.fromOffset(diceXs[i] + 5, 51)
		TweenService:Create(b, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
			Size = UDim2.fromOffset(46, 46),
			Position = UDim2.fromOffset(diceXs[i], 46),
		}):Play()
	end
end

			d.BackgroundColor3 = Color3.fromRGB(255, 255, 255)

			corner(d, 3)
			x = x + 26

	end

local function predictColor()
	for i = 1, 4 do
		if selected[i] == "?" then return nil end
	end
	local freq = {}
	for _, n in ipairs(ORDER) do freq[n] = 0 end
	for _, n in ipairs(selected) do freq[n] = freq[n] + 1.4 end
	local hStart = math.max(1, #history - 20)
	for i = hStart, #history do
		local n = history[i]
		if freq[n] then freq[n] = freq[n] + (1 + (i - hStart) / 20) end
	end
	local trans = {}
	for _, n in ipairs(ORDER) do trans[n] = {} end
	for i = hStart, #history - 1 do
		local a, b = history[i], history[i + 1]
		if a and b and trans[a] then trans[a][b] = (trans[a][b] or 0) + 1 end
	end
	for i = 1, 3 do
		local a, b = selected[i], selected[i + 1]
		if a ~= "?" and b ~= "?" and trans[a] then trans[a][b] = (trans[a][b] or 0) + 1.5 end
	end
	local last = selected[4]
	local scores = {}
	for _, n in ipairs(ORDER) do
		local s = -freq[n] * 0.85
		if trans[last] and trans[last][n] then s = s + trans[last][n] * 1.6 end
		if n == last then
			if selected[3] == last and selected[2] == last then s = s + 0.3 else s = s - 1.4 end
		end
		if selected[3] == selected[4] and n ~= last then s = s + 1.25 end
		if selected[1] == selected[3] and selected[2] == selected[4] and n == selected[1] then s = s + 1.1 end
		if mode == "v2" then
			s = s + (n ~= last and 0.7 or -0.9)
			if trans[last] and trans[last][n] then s = s + trans[last][n] * 0.8 end
		else
			s = s - freq[n] * 0.25
		end
		s = s + (math.random() - 0.5) * 0.35
		scores[n] = s
	end
	local best, bestS = ORDER[1], -1e9
	for _, n in ipairs(ORDER) do
		if scores[n] > bestS then bestS = scores[n]; best = n end
	end
	return best
end

predictBtn.MouseButton1Click:Connect(function()
	if predicting then return end
	playPredict()
	popup.Visible = false
	local color = predictColor()
	if not color then
		rName.Text = "Need 4 colors"
		rName.TextColor3 = Color3.fromRGB(255, 120, 130)
		return
	end
	predicting = true
	predictBtn.Text = "..."
	rName.Text = "Reading..."
	rName.TextColor3 = theme.muted
	task.spawn(function()
		for _ = 1, 8 do
			local r = ORDER[math.random(1, #ORDER)]
			rDot.BackgroundColor3 = COLORS[r]
			task.wait(0.03)
		end
		rDot.BackgroundColor3 = COLORS[color]
		rName.Text = color
		rName.TextColor3 = COLORS[color]
		rDot.Size = UDim2.fromOffset(28, 28)
		rDot.Position = UDim2.fromOffset(14, 13)
		TweenService:Create(rDot, TweenInfo.new(0.2, Enum.EasingStyle.Back), {
			Size = UDim2.fromOffset(36, 36),
			Position = UDim2.fromOffset(10, 9),
		}):Play()
		table.insert(history, color)
		refreshHistory()
		for i = 1, 4 do setDice(i, "?") end
		predictBtn.Text = "PREDICT"
		predicting = false
	end)
end)

-- neon pulse
task.spawn(function()
	while main.Parent do
		local t = tick()
		local pulse = 0.5 + 0.5 * math.sin(t * 2.6)
		mainStroke.Transparency = 0.03 + 0.18 * pulse
		mainStroke.Color = Color3.fromRGB(
			theme.stroke.R * 255,
			theme.stroke.G * 255,
			theme.stroke.B * 255
		)
		aura.BackgroundTransparency = 0.78 + 0.12 * pulse
		aura.BackgroundColor3 = theme.accent
		predStroke.Transparency = 0.1 + 0.2 * pulse
		for i = 1, 4 do
			local p2 = 0.5 + 0.5 * math.sin(t * 3.0 + i * 0.8)
			local filled = selected[i] and selected[i] ~= "?"
			if diceGlows[i] then
				diceGlows[i].BackgroundTransparency = (filled and 0.65 or 0.82) + 0.12 * p2
			end
			if diceStrokes[i] then
				-- keep white border clean like reference
				diceStrokes[i].Color = Color3.fromRGB(255, 255, 255)
				diceStrokes[i].Transparency = (filled and 0.08 or 0.3) + 0.08 * p2
				diceStrokes[i].Thickness = filled and 2.4 or 2
			end
		end
		task.wait(0.04)
	end
end)
