-- Variáveis do sistema POWER
local PowerActive = false
local CurrentSpeed = 0
local PowerThreads = {}

-- Função para atualizar label de velocidade
local function UpdateSpeedLabel(speed)
    if speed == 0 then
        SpeedLabel.Text = "LAG POWER: 0% (LAG OTHERS EXTREME) 💀"
        SpeedLabel.TextColor3 = Color3.fromRGB(255, 0, 0)
    elseif speed <= 20 then
        SpeedLabel.Text = "LAG POWER: "..speed.."% (OTHERS MEGA LAG) 🔥"
        SpeedLabel.TextColor3 = Color3.fromRGB(255, 50, 0)
    elseif speed <= 40 then
        SpeedLabel.Text = "LAG POWER: "..speed.."% (OTHERS SUPER LAG) ⚡"
        SpeedLabel.TextColor3 = Color3.fromRGB(255, 100, 0)
    elseif speed <= 60 then
        SpeedLabel.Text = "LAG POWER: "..speed.."% (OTHERS LAG MÉDIO) 💫"
        SpeedLabel.TextColor3 = Color3.fromRGB(255, 200, 0)
    elseif speed <= 80 then
        SpeedLabel.Text = "LAG POWER: "..speed.."% (OTHERS LAG LEVE) ✨"
        SpeedLabel.TextColor3 = Color3.fromRGB(200, 255, 0)
    else
        SpeedLabel.Text = "LAG POWER: "..speed.."% (SEM EFEITO) ✅"
        SpeedLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
    end
end

-- Atualizar velocidade quando input mudar
SpeedInput.FocusLost:Connect(function()
    local value = tonumber(SpeedInput.Text)
    if value and value >= 0 and value <= 100 then
        CurrentSpeed = value
        UpdateSpeedLabel(CurrentSpeed)
    else
        SpeedInput.Text = tostring(CurrentSpeed)
        StatusLabel.Text = "Status: Valor inválido! Use 0-100"
        StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
        task.wait(2)
        StatusLabel.Text = "Status: Aguardando..."
        StatusLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    end
end)

-- Função POWER EXTREMO (causar lag nos OUTROS jogadores SEM lagar você)
local function StartPowerMode()
    local Character = LocalPlayer.Character
    if not Character then
        StatusLabel.Text = "Status: Personagem não encontrado!"
        return false
    end
    
    local Humanoid = Character:FindFirstChildOfClass("Humanoid")
    if not Humanoid then
        StatusLabel.Text = "Status: Humanoide não encontrado!"
        return false
    end
    
    local Quantum_Cloner = Backpack:FindFirstChild("Quantum Cloner")
    if not Quantum_Cloner then
        StatusLabel.Text = "Status: Quantum Cloner não encontrado!"
        return false
    end
    
    -- Equipar Quantum Cloner
    Humanoid:EquipTool(Quantum_Cloner)
    task.wait(0.1)
    
    -- Calcular delay e ativações baseado na velocidade
    -- SEMPRE 12 THREADS independente da porcentagem
    -- 0% = 12 threads com delay ULTRA mínimo, MUITAS ativações
    -- 100% = 12 threads com delay alto, poucas ativações
    local threadCount = 16 -- FIXO em 12 threads sempre
    local baseDelay = 0.0001 + (CurrentSpeed / 100) * 0.4999 -- 0.0001s (0.1ms) a 0.5s
    local activations = math.floor(50 - (CurrentSpeed / 100) * 48) -- 50 a 2 ativações
    
    StatusLabel.Text = "Status: 16 THREADS ATIVAS! "..activations.."x 🔥💀⚡"
    StatusLabel.TextColor3 = Color3.fromRGB(0, 255, 100)
    
    -- Criar 12 threads (SEMPRE)
    for i = 1, threadCount do
        local thread = task.spawn(function()
            while PowerActive do
                local Character = LocalPlayer.Character
                if Character and Character:FindFirstChildOfClass("Humanoid") then
                    -- Pegar todos os tools do backpack
                    local tools = {}
                    for _, v in ipairs(Backpack:GetChildren()) do
                        if v:IsA("Tool") and PowerActive then
                            table.insert(tools, v)
                        end
                    end
                    
                    -- Loop rápido de equipar/desequipar (replica para servidor)
                    for _, tool in ipairs(tools) do
                        if PowerActive and tool then
                            -- Equipar
                            pcall(function()
                                tool.Parent = Character
                            end)
                            task.wait(baseDelay)
                            
                            -- Desequipar
                            pcall(function()
                                tool.Parent = Backpack
                            end)
                            task.wait(baseDelay)
                            
                            -- Equipar novamente
                            pcall(function()
                                tool.Parent = Character
                            end)
                            task.wait(baseDelay)
                        end
                    end
                    
                    -- ATIVAR QUANTUM CLONER MÚLTIPLAS VEZES ULTRA RÁPIDO (isso DESTROI os outros!)
                    for _ = 1, activations do
                        if Quantum_Cloner and Quantum_Cloner.Parent == Character and PowerActive then
                            pcall(function()
                                Quantum_Cloner:Activate()
                            end)
                            -- Delay MÍNIMO para não travar você
                            if baseDelay > 0.001 then
                                task.wait(baseDelay)
                            end
                        end
                    end
                    
                    -- Desequipar todos
                    for _, tool in ipairs(tools) do
                        if PowerActive and tool then
                            pcall(function()
                                tool.Parent = Backpack
                            end)
                        end
                    end
                    
                    -- Delay mínimo entre ciclos
                    task.wait(baseDelay * 2)
                else
                    break
                end
            end
        end)
        
        table.insert(PowerThreads, thread)
    end
    
    return true
end

-- Função para parar POWER
local function StopPowerMode()
    PowerActive = false
    
    -- Cancelar todos os threads
    for _, thread in ipairs(PowerThreads) do
        task.cancel(thread)
    end
    PowerThreads = {}
    
    StatusLabel.Text = "Status: POWER MODE DESATIVADO"
    StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 100)
    task.wait(2)
    StatusLabel.Text = "Status: Aguardando..."
    StatusLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
end

-- Botão POWER
PowerButton.MouseButton1Click:Connect(function()
    if not PowerActive then
        PowerActive = true
        PowerButton.Text = "⏸ STOP LAG"
        PowerButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50)
        
        if not StartPowerMode() then
            PowerActive = false
            PowerButton.Text = "▶ LAG OTHERS"
            PowerButton.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
        end
    else
        StopPowerMode()
        PowerButton.Text = "▶ LAG OTHERS"
        PowerButton.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
    end
end)
