local VERSION        = "V2"
local BURST          = 500
local WINDOW         = 0.3
local FLING          = 1e9
local COOLDOWN       = 1.0

local _busy      = false
local _gaveUp    = false
local _lastReset = 0
local _camConn   = nil
local _camSaved  = nil

local function lockCamera()
    local cam = workspace.CurrentCamera
    if not cam then return end
    if _camConn then pcall(function() _camConn:Disconnect() end) end
    _camConn = nil
    _camSaved = { cframe = cam.CFrame, type = cam.CameraType, subject = cam.CameraSubject }
    cam.CameraType = Enum.CameraType.Scriptable
    _camConn = RunService.RenderStepped:Connect(function()
        local c = workspace.CurrentCamera
        if not c or not _camSaved then return end
        if c.CameraType ~= Enum.CameraType.Scriptable then
            c.CameraType = Enum.CameraType.Scriptable
        end
        c.CFrame = _camSaved.cframe
    end)
end

local function unlockCamera()
    if _camConn then pcall(function() _camConn:Disconnect() end); _camConn = nil end
    local saved = _camSaved; _camSaved = nil
    if not saved then return end
    local cam = workspace.CurrentCamera
    if not cam then return end
    pcall(function()
        local char = LP.Character
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        cam.CameraType = (saved.type == Enum.CameraType.Scriptable)
            and Enum.CameraType.Custom or saved.type
        cam.CameraSubject = hum or saved.subject
    end)
end

local function doInstantResetV2()
    _gaveUp = false
    local character = LP.Character
    if not character then _gaveUp = true return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root     = character:FindFirstChild("HumanoidRootPart")
    if not humanoid or not root then _gaveUp = true return end

    local savedPlatform, savedBreak = humanoid.PlatformStand, humanoid.BreakJointsOnDeath

    local function applyFling(hum, r)
        hum.BreakJointsOnDeath = true
        hum.PlatformStand = true
        hum:ChangeState(Enum.HumanoidStateType.Physics)
        r.AssemblyAngularVelocity = Vector3.zero
        r.AssemblyLinearVelocity = Vector3.new(0, FLING, 0)
    end

    local function attempt()
        local ch = LP.Character
        if ch ~= character or not character.Parent then return false end
        local hum = ch:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return false end
        local r = ch:FindFirstChild("HumanoidRootPart")
        if r then pcall(applyFling, hum, r) end
        return true
    end

    local t0 = os.clock()
    for _ = 1, BURST do
        if not attempt() then return end
    end
    while os.clock() - t0 < WINDOW do
        RunService.Heartbeat:Wait()
        if not attempt() then return end
    end

    if not character.Parent or LP.Character ~= character or humanoid.Health <= 0 then
        return
    end

    _gaveUp = true
    pcall(function()
        humanoid.PlatformStand = savedPlatform
        humanoid.BreakJointsOnDeath = savedBreak
        humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        root.AssemblyLinearVelocity = Vector3.zero
    end)
end

local function runInstantReset()
    if _busy then return end
    if os.clock() - _lastReset < COOLDOWN then return end
    _lastReset = os.clock()
    _busy  = true
    _gaveUp = false

    local wasAntiDie = false
    pcall(function()
        if _G.__CrystalAntiDieIsEnabled then
            wasAntiDie = (_G.__CrystalAntiDieIsEnabled() == true)
        end
    end)
    if wasAntiDie then
        if _G.__CrystalAntiDieStop then
            pcall(_G.__CrystalAntiDieStop)
        elseif AntiDieModule and AntiDieModule.stop then
            pcall(function() AntiDieModule.stop() end)
        end
    end

    local oldChar = LP.Character
    lockCamera()

    task.spawn(function()
        local ok, err = pcall(doInstantResetV2)
        if not ok then
            warn("[Instant Reset] " .. tostring(err))
            _gaveUp = true
        end
    end)

    task.spawn(function()
        local deadline = os.clock() + 25
        while os.clock() < deadline do
            if _gaveUp then break end
            local ch = LP.Character
            if ch and ch.Parent and ch ~= oldChar then
                local hum  = ch:FindFirstChildOfClass("Humanoid")
                local root = ch:FindFirstChild("HumanoidRootPart")
                if hum and root and hum.Health > 0 then break end
            end
            task.wait(0.05)
        end
        unlockCamera()

        if wasAntiDie then
            local aliveDeadline = os.clock() + 20
            while os.clock() < aliveDeadline do
                local ch  = LP.Character
                local hum = ch and ch:FindFirstChildOfClass("Humanoid")
                if ch and ch.Parent and hum and hum.Health > 0 then break end
                task.wait(0.1)
            end
            if AntiDieModule and AntiDieModule.start then
                pcall(function() AntiDieModule.start() end)
            elseif _G.__CrystalAntiDieSet then
                pcall(_G.__CrystalAntiDieSet, true)
            end
        end
        _busy = false
    end)
end

_G.__InstantResetRun = runInstantReset

                startPos.Y.Scale, startPos.Y.Offset + delta.Y
