-- // 1 (Services) //
local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local TweenService = game:GetService("TweenService")
local UserInputService = game:GetService("UserInputService")
local CoreGui = game:GetService("CoreGui")
local HttpService = game:GetService("HttpService")

-- // 2 (Player + state) //
local lp = Players.LocalPlayer
local activated = false
local keybind = Enum.KeyCode.E
local waitingForKey = false
local power = 97000
local lagAmount = 0.12
local lagConn = nil

-- // 3 (Boost state – always enabled) //
local boostEnabled = true
local boostSpeed = 34
local boostConn = nil

-- // 4 (Map power to lag amount) //
local function applyPower(val)
power = math.clamp(val, 10000, 500000)
local t = (power - 10000) / 490000
lagAmount = t * 0.2
end
applyPower(power)

-- // 5 (Boost logic) //
local function startBoost()
if boostConn then boostConn:Disconnect() end
boostConn = RunService.RenderStepped:Connect(function()
if not boostEnabled then return end
local char = lp.Character
if not char then return end
local hum = char:FindFirstChildOfClass("Humanoid")
local hrp = char:FindFirstChild("HumanoidRootPart")
if not hum or not hrp then return end

local st = hum:GetState()
if st == Enum.HumanoidStateType.Physics or
st == Enum.HumanoidStateType.Ragdoll or
st == Enum.HumanoidStateType.FallingDown then
return
end

local md = hum.MoveDirection
if md.Magnitude > 0 then
hrp.Velocity = Vector3.new(md.X * boostSpeed, hrp.Velocity.Y, md.Z * boostSpeed)
end
end)
end

-- // 6 (Apply boost speed) //
local function applyBoostSpeed(val)
boostSpeed = math.clamp(val, 1, 500)
saveConfig()
end

-- // 7 (Start / stop lag) //
local function startLag()
if lagConn then lagConn:Disconnect() end
lagConn = RunService.RenderStepped:Connect(function()
if not activated then return end
if lagAmount > 0 then
local t = tick()
while tick() - t < lagAmount do end
end
end)
end
local function stopLag()
activated = false
if lagConn then lagConn:Disconnect(); lagConn = nil end
end

-- // 8 (Config save/load) //
local CONFIG_FILE = "EctralysiumConfig.json"

local function saveConfig()
if not writefile then return end
local config = {
boostSpeed = boostSpeed,
power = power,
keybind = keybind.Name,
lagEnabled = activated
}
local json = HttpService:JSONEncode(config)
pcall(function() writefile(CONFIG_FILE, json) end)
end

local function loadConfig()
if not readfile then return end
local success, data = pcall(readfile, CONFIG_FILE)
if not success then return end
local config
pcall(function() config = HttpService:JSONDecode(data) end)
if not config then return end

if config.boostSpeed then
boostSpeed = math.clamp(config.boostSpeed, 1, 500)
if boostBox then boostBox.Text = tostring(boostSpeed) end
end
if config.power then
applyPower(config.power)
if powerBox then powerBox.Text = tostring(power) end
end
if config.keybind then
local k = Enum.KeyCode[config.keybind]
if k then
keybind = k
if kbBtn then kbBtn.Text = keybind.Name end
end
end
if config.lagEnabled then
task.defer(function()
if not activated then
toggleLag()
end
end)
end
end

-- // 9 (Toggle lag) //
local function toggleLag()
if not activated then
activated = true
if enableBtn then
enableBtn.Text = "Disable"
enableBtn.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
end
startLag()
else
stopLag()
if enableBtn then
enableBtn.Text = "Enable"
enableBtn.BackgroundColor3 = Color3.fromRGB(100, 100, 100)
end
end
saveConfig()
end

-- // 14 (Minimize Button) //

minimizeBtn.Font = Enum.Font.GothamBold

minimizeBtn.AutoButtonColor = false
minimizeBtn.ZIndex = 10
minimizeBtn.Parent = main
Instance.new("UICorner", minimizeBtn).CornerRadius = UDim.new(0, 4)

minimizeBtn.Modal = true

local minimized = false
local isAnimating = false

local function getControls()
local controls = {}
for _, child in ipairs(main:GetChildren()) do
if child:IsA("GuiObject") and child ~= title and child ~= minimizeBtn then
table.insert(controls, child)
end
end
return controls
end

local function toggleMinimize()
if isAnimating then return end
isAnimating = true

local controls = getControls()
local oldYOff = main.Position.Y.Offset
local oldHeight = main.Size.Y.Offset

minimized = not minimized
local targetHeight = minimized and MIN_HEIGHT or HEIGHT
local targetYOff = oldYOff - (oldHeight - targetHeight) / 2

local sizeTween = TweenService:Create(main, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
Size = UDim2.new(0, WIDTH, 0, targetHeight)
})
local posTween = TweenService:Create(main, TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
Position = UDim2.new(0.5, main.Position.X.Offset, 0.5, targetYOff)
})

if minimized then
sizeTween.Completed:Connect(function()
for _, child in ipairs(controls) do
child.Visible = false
end
minimizeBtn.Text = "+"
isAnimating = false
end)
else
for _, child in ipairs(controls) do
child.Visible = true
end
sizeTween.Completed:Connect(function()
minimizeBtn.Text = "-"
isAnimating = false
end)
end

sizeTween:Play()
posTween:Play()

task.delay(0.5, function()
if isAnimating then
main.Size = UDim2.new(0, WIDTH, 0, targetHeight)
main.Position = UDim2.new(0.5, main.Position.X.Offset, 0.5, targetYOff)
if minimized then
for _, child in ipairs(controls) do
child.Visible = false
end
minimizeBtn.Text = "+"
else
for _, child in ipairs(controls) do
child.Visible = true
end
minimizeBtn.Text = "-"
end
isAnimating = false
end
end)
end
minimizeBtn.MouseButton1Click:Connect(toggleMinimize)

if waitingForKey then
if input.UserInputType == Enum.UserInputType.Keyboard then
keybind = input.KeyCode
kbBtn.Text = keybind.Name
kbBtn.BackgroundColor3 = Color3.fromRGB(100, 100, 100)
waitingForKey = false
saveConfig()
end
return
end
if input.KeyCode == keybind then
toggleLag()
end
end)
